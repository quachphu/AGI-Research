To solve this problem, we need to find the shortest path from the starting point 'S' to the terminating point 'T' on a grid, while adhering to the constraint that the path can visit no more than `k` different types of cells (excluding 'S' and 'T'). Among all possible shortest paths, we must choose the lexicographically smallest one.

### Approach
1. **Breadth-First Search (BFS)**: We use BFS to explore the grid level by level, ensuring that we find the shortest path first.
2. **State Tracking**: Each state in our BFS is represented by the current position on the grid, the mask of visited cell types (represented as a bitmask), the number of steps taken, and the path taken so far.
3. **Lexicographical Order**: To ensure the lexicographically smallest path, we process neighboring cells in lex order when expanding each node in the BFS.
4. **Visited States**: We maintain a dictionary to track visited states to avoid reprocessing the same state with a non-optimal path. Each entry in the dictionary keeps track of the minimum steps required to reach that state and the lexicographically smallest path for those steps.

import sys
from collections import deque

def main():
    n, m, k = map(int, sys.stdin.readline().split())
    grid = [sys.stdin.readline().strip() for _ in range(n)]
    
    s_pos = None
    t_pos = None
    for i in range(n):
        for j in range(m):
            if grid[i][j] == 'S':
                s_pos = (i, j)
            elif grid[i][j] == 'T':
                t_pos = (i, j)
    s_x, s_y = s_pos
    t_x, t_y = t_pos
    
    visited = {}
    queue = deque()
    initial_mask = 0
    queue.append((s_x, s_y, initial_mask, 0, ""))
    visited_key = (s_x, s_y, initial_mask)
    visited[visited_key] = (0, "")
    
    t_candidates = []
    
    directions = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    
    while queue:
        x, y, mask, steps, path = queue.popleft()
        
        if (x, y) == (t_x, t_y):
            if bin(mask).count('1') <= k:
                t_candidates.append((steps, path))
            continue
        
        neighbors = []
        for dx, dy in directions:
            nx = x + dx
            ny = y + dy
            if 0 <= nx < n and 0 <= ny < m:
                cell = grid[nx][ny]
                if cell == 'S':
                    continue
                neighbors.append((nx, ny, cell))
        
        neighbors.sort(key=lambda x: x[2])
        
        for nx, ny, cell in neighbors:
            if cell == 'T':
                new_mask = mask
                new_steps = steps + 1
                new_path = path
                if bin(new_mask).count('1') > k:
                    continue
                key = (nx, ny, new_mask)
                if key not in visited:
                    visited[key] = (new_steps, new_path)
                    queue.append((nx, ny, new_mask, new_steps, new_path))
                else:
                    existing_steps, existing_path = visited[key]
                    if new_steps < existing_steps or (new_steps == existing_steps and new_path < existing_path):
                        visited[key] = (new_steps, new_path)
                        queue.append((nx, ny, new_mask, new_steps, new_path))
            else:
                c = cell
                new_mask = mask
                if (mask & (1 << (ord(c) - ord('a')))) == 0:
                    new_mask = mask | (1 << (ord(c) - ord('a')))
                    if bin(new_mask).count('1') > k:
                        continue
                new_steps = steps + 1
                new_path = path + c
                key = (nx, ny, new_mask)
                if key not in visited:
                    visited[key] = (new_steps, new_path)
                    queue.append((nx, ny, new_mask, new_steps, new_path))
                else:
                    existing_steps, existing_path = visited[key]
                    if new_steps < existing_steps or (new_steps == existing_steps and new_path < existing_path):
                        visited[key] = (new_steps, new_path)
                        queue.append((nx, ny, new_mask, new_steps, new_path))
    
    if t_candidates:
        min_steps = min(steps for steps, _ in t_candidates)
        candidates = [path for steps, path in t_candidates if steps == min_steps]
        if candidates:
            print(min(candidates))
        else:
            print(-1)
    else:
        print(-1)

if __name__ == "__main__":
    main()
