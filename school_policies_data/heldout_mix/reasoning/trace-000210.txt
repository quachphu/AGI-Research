To solve this problem, we need to determine the maximum amount of money Yu-kun can collect while navigating a grid, considering the costs of breaking small rocks. The solution involves exploring all possible paths using a priority queue to track the most promising paths first, ensuring we maximize the collected treasure while minimizing the costs incurred from breaking rocks.

### Approach
1. **Input Parsing**: Read the grid dimensions, number of treasures, and rock breaking cost. Parse the grid to identify the starting position, treasure locations, and small rock locations.
2. **State Representation**: Use a state tuple (i, j, treasure_mask, rocks_mask) where (i, j) is the current position, `treasure_mask` is a bitmask representing collected treasures, and `rocks_mask` is a bitmask representing broken rocks.
3. **Priority Queue**: Utilize a max-heap (priority queue) to explore paths starting from the highest current sum to ensure we process the most promising paths first.
4. **Movement and State Transitions**: For each move, check adjacent cells. Handle rock breaking and treasure collection by updating the respective masks and sums. Track the maximum sum for each state to avoid redundant processing of less optimal paths.
5. **Result Calculation**: After processing all possible states, the maximum valid sum (non-negative) is the answer.

### Solution Code
```python
import heapq

def main():
    h, w, n, r = map(int, input().split())
    grid = []
    start = None
    asterisk_indices = {}  # Maps (i,j) to their index in asterisks list
    asterisks = []  # List of (i,j) positions of *

    # Read grid and find start position and asterisks
    for i in range(h):
        row = input().strip()
        grid.append(row)
        for j in range(w):
            c = row[j]
            if c == '@':
                start = (i, j)
            elif c == '*':
                if (i, j) not in asterisk_indices:
                    idx = len(asterisks)
                    asterisks.append((i, j))
                    asterisk_indices[(i, j)] = idx

    # Read treasures and map their characters to values
    treasure_info = {}
    for _ in range(n):
        m, v = input().split()
        treasure_info[m] = int(v)

    # Map treasure positions to their index and value
    treasure_positions = {}
    treasure_index = 0
    for i in range(h):
        for j in range(w):
            c = grid[i][j]
            if c in treasure_info:
                treasure_positions[(i, j)] = (treasure_index, treasure_info[c])
                treasure_index += 1

    # Number of asterisks
    m_asterisks = len(asterisks)

    # Directions: up, down, left, right
    dirs = [(-1, 0), (1, 0), (0, -1), (0, 1)]

    # Priority queue (max-heap using negative sum)
    heap = []
    # Dictionary to keep track of the maximum sum for each state
    max_sum = {}

    # Initial state: (i, j, treasure_mask, rocks_mask)
    initial_i, initial_j = start
    initial_t_mask = 0
    initial_r_mask = 0
    initial_sum = 0

    max_sum_key = (initial_i, initial_j, initial_t_mask, initial_r_mask)
    max_sum[max_sum_key] = initial_sum
    heapq.heappush(heap, (-initial_sum, initial_i, initial_j, initial_t_mask, initial_r_mask))

    best = 0  # The best valid sum (>=0)

    while heap:
        current_neg_sum, i, j, t_mask, r_mask = heapq.heappop(heap)
        current_sum = -current_neg_sum

        # If this state is no longer the best known, skip
        if max_sum.get((i, j, t_mask, r_mask), -float('inf')) > current_sum:
            continue

        # Update best if current_sum is valid
        if current_sum >= 0 and current_sum > best:
            best = current_sum

        # Explore all directions
        for di, dj in dirs:
            ni, nj = i + di, j + dj
            if 0 <= ni < h and 0 <= nj < w:
                cell = grid[ni][nj]
                if cell == '#':
                    continue  # Cannot pass

                # Initialize new state variables
                new_r_mask = r_mask
                new_sum = current_sum
                new_t_mask = t_mask

                # Check if cell is a * that needs breaking
                if cell == '*':
                    # Check if this asterisk is in the list
                    if (ni, nj) in asterisk_indices:
                        asterisk_idx = asterisk_indices[(ni, nj)]
                        if not (new_r_mask & (1 << asterisk_idx)):
                            # Break the rock
                            new_r_mask |= (1 << asterisk_idx)
                            new_sum -= r

                # Check if cell is a treasure and not yet collected
                if (ni, nj) in treasure_positions:
                    t_idx, t_value = treasure_positions[(ni, nj)]
                    if not (new_t_mask & (1 << t_idx)):
                        new_t_mask |= (1 << t_idx)
                        new_sum += t_value

                # Now, check if the cell is passable
                # (cell is not #, and if it's a *, it's broken now)
                # Create new state
                new_state = (ni, nj, new_t_mask, new_r_mask)
                state_key = new_state

                # Update max_sum and push to heap if this is a better path
                if new_sum > max_sum.get(state_key, -float('inf')):
                    max_sum[state_key] = new_sum
                    heapq.heappush(heap, (-new_sum, ni, nj, new_t_mask, new_r_mask))

    print(best)

if __name__ == "__main__":
    main()
```
