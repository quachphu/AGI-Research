class SegmentTree:
    def __init__(self, size):
        self.n = size
        self.tree = [ {'left': 0, 'right':0, 'min':0, 'max':0, 'sum':0, 'count':0, 'lazy':0} for _ in range(4 * self.n) ]
        self.build(0, 0, self.n -1)
    
    def build(self, node, l, r):
        self.tree[node]['left'] = l
        self.tree[node]['right'] = r
        self.tree[node]['lazy'] = 0
        if l == r:
            self.tree[node]['min'] = 0
            self.tree[node]['max'] = 0
            self.tree[node]['sum'] = 0
            self.tree[node]['count'] = 1
            return
        mid = (l + r) // 2
        left_child = 2*node +1
        right_child = 2*node +2
        self.build(left_child, l, mid)
        self.build(right_child, mid+1, r)
        self.push_up(node)
    
    def push_up(self, node):
        left_child = 2*node +1
        right_child = 2*node +2
        self.tree[node]['min'] = min( self.tree[left_child]['min'], self.tree[right_child]['min'] )
        self.tree[node]['max'] = max( self.tree[left_child]['max'], self.tree[right_child]['max'] )
        self.tree[node]['sum'] = self.tree[left_child]['sum'] + self.tree[right_child]['sum']
        self.tree[node]['count'] = self.tree[left_child]['count'] + self.tree[right_child]['count']
    
    def apply(self, node, value):
        self.tree[node]['min'] += value
        self.tree[node]['max'] += value
        self.tree[node]['sum'] += value * self.tree[node]['count']
        self.tree[node]['lazy'] += value
    
    def push_down(self, node):
        if self.tree[node]['lazy'] !=0:
            left_child = 2*node +1
            right_child = 2*node +2
            self.apply(left_child, self.tree[node]['lazy'])
            self.apply(right_child, self.tree[node]['lazy'])
            self.tree[node]['lazy'] =0
    
    def range_add(self, node, l, r, value):
        if self.tree[node]['right'] < l or self.tree[node]['left'] > r:

            right_child = 2 * node + 2
            self.build(left_child, l, mid)
            self.build(right_child, mid + 1, r)

            self.tree[node]['count'] = self.tree[left_child]['count'] + self.tree[right_child]['count']
