To solve this problem, we need to simulate the movement of a bicycle on a series of slopes, taking into account jumps that occur when the slope becomes steeper. The goal is to compute the total distance the bicycle travels on the ground, considering both normal movement and jumps.

### Approach
1. **Identify Takeoff Points**: Determine the points where the bicycle will jump. This happens when the slope of the current segment is steeper than the previous segment.
2. **Simulate Movement**: Track the bicycle's movement from the start point to each takeoff point, accumulating the distance traveled on the ground.
3. **Calculate Jumps**: For each takeoff point, compute the parabolic trajectory and determine the landing point on subsequent segments.
4. **Update Position**: After each jump, update the bicycle's position to the landing point and continue simulating movement from there.

import sys
import math
from bisect import bisect_left

def main():
    input = sys.stdin.read().split()
    idx = 0
    n = int(input[idx])
    idx += 1
    v = int(input[idx])
    idx +=1
    points = []
    for _ in range(n):
        x = int(input[idx])
        y = int(input[idx+1])
        points.append((x, y))
        idx +=2
    
    # Precompute takeoff points (0-based indices)
    takeoff = []
    for i in range(1, n-1):
        x_prev, y_prev = points[i-1]
        x_i, y_i = points[i]
        x_next, y_next = points[i+1]
        dx_prev = x_i - x_prev
        dy_prev = y_i - y_prev
        dx_curr = x_next - x_i
        dy_curr = y_next - y_i
        
        # Handle vertical segments
        if dx_prev == 0:
            slope_prev = math.inf if dy_prev > 0 else -math.inf
        else:
            slope_prev = dy_prev / dx_prev
        if dx_curr == 0:
            slope_curr = math.inf if dy_curr > 0 else -math.inf
        else:
            slope_curr = dy_curr / dx_curr
        
        if slope_curr < slope_prev:
            takeoff.append(i)
    
    total = 0.0
    current_pos = points[0]
    t_idx = 0  # index into takeoff list
    len_points = len(points)
    
    # Binary search to find the segment for a given x
    def find_seg(x):
        left = 0
        right = len_points - 2  # last possible segment index
        while left <= right:
            mid = (left + right) // 2
            if points[mid][0] <= x <= points[mid+1][0]:
                return mid
            elif x < points[mid][0]:
                right = mid -1
            else:
                left = mid +1
        return -1
    
    while True:
        # Check if reached the end
        if current_pos == points[-1]:
            break
        # Check if no more takeoff points, move to end
        if t_idx >= len(takeoff):
            dx = points[-1][0] - current_pos[0]
            dy = points[-1][1] - current_pos[1]
            total += math.hypot(dx, dy)
            break
        
        next_t_i = takeoff[t_idx]
        # Move from current_pos to points[next_t_i]
        # Find current segment
        current_seg = find_seg(current_pos[0])
        if current_seg == -1:
            current_seg = 0
        
        # Accumulate distance from current_pos to points[next_t_i]
        while current_seg < next_t_i:
            # Move to end of current_seg
            x_end, y_end = points[current_seg +1]
            dx = x_end - current_pos[0]
            dy = y_end - current_pos[1]
            total += math.hypot(dx, dy)
            current_pos = (x_end, y_end)
            current_seg +=1
        
        # Now at points[next_t_i], process jump
        # Compute velocity based on previous segment
        prev_seg = next_t_i -1
        x0, y0 = current_pos
        x_prev, y_prev = points[prev_seg]
        dx_prev = x0 - x_prev
        dy_prev = y0 - y_prev
        L_prev = math.hypot(dx_prev, dy_prev)
        if L_prev == 0:
            vx = 0.0
            vy = 0.0
        else:
            vx = v * dx_prev / L_prev
            vy = v * dy_prev / L_prev
        
        # Find the landing segment and position
        min_t = math.inf
        landing_seg = None
        x_land = y_land = 0.0
        for j in range(next_t_i, len_points -1):
            x_a, y_a = points[j]
            x_b, y_b = points[j+1]
            dx_seg = x_b - x_a
            if dx_seg == 0:
                continue  # skip vertical segments (invalid as per problem statement)
            a = -4.9 * dx_seg
            b = vy * dx_seg - vx * (y_b - y_a)
            c = (x0 - x_a)*(y_b - y_a) + (y_a - y0)*dx_seg
            
            discriminant = b*b -4*a*c
            if discriminant < 0:
                continue
            sqrt_d = math.sqrt(discriminant)
            t1 = (-b + sqrt_d) / (2*a)
            t2 = (-b - sqrt_d) / (2*a)
            
            for t in [t1, t2]:
                if t < 1e-8:
                    continue
                x_t = x0 + vx * t
                if not (x_a - 1e-8 <= x_t <= x_b + 1e-8):
                    continue
                y_t = y0 + vy * t - 4.9 * t**2
                s = (x_t - x_a) / dx_seg
                if -1e-8 <= s <= 1 + 1e-8:
                    if t < min_t:
                        min_t = t
                        landing_seg = j
                        x_land = x_t
                        y_land = y_t
        
        # Update current_pos to landing point
        current_pos = (x_land, y_land)
        # Find the segment of the landing point
        seg_land = find_seg(x_land)
        if seg_land == -1:
            seg_land = len_points -2  # last segment
        
        # Add distance from landing point to end of segment
        x_a, y_a = points[seg_land]
        x_b, y_b = points[seg_land +1]
        dx = x_b - x_land
        dy = y_b - y_land
        total += math.hypot(dx, dy)
        current_pos = (x_b, y_b)
        
        # Update t_idx to next possible takeoff after seg_land +1
        t_idx = bisect_left(takeoff, seg_land +1)
    
    print("{0:.8f}".format(total))

if __name__ == "__main__":
    main()
