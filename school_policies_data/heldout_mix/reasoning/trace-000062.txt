        valid = True
        for i in range(n):
            block_row = (i // x) * x
            row = matrix[i]
            for j in range(n):
                block_col = (j // x) * x
                if row[j] != matrix[block_row][block_col]:
                    valid = False
                    break
            if not valid:

        if valid:

            return
