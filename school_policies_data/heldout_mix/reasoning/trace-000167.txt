Implementing a Fenwick Tree (BIT) for this purpose.

The BIT can be represented as an array, where each index i represents the presence of the number i (1 if available, 0 if used).

The operations needed are:

- Update: set a position to 0 (subtract 1 from the initial 1).

- Query the prefix sum up to x.

Implementing the BIT:

class FenwickTree:

    def __init__(self, size):

        self.n = size

        self.tree = [0]*(self.n +1)

    def update(self, index, delta):

        # index is 1-based

        while index <= self.n:

            self.tree[index] += delta

            index += index & -index

    def query(self, index):

        res =0

        while index >0:

            res += self.tree[index]

            index -= index & -index

        return res

So initially, the BIT is filled with 1s for all positions 1..n.

import sys

class FenwickTree:
    def __init__(self, size):
        self.n = size
        self.tree = [0] * (self.n + 2)  # 1-based indexing

    def update(self, index, delta):
        while index <= self.n:
            self.tree[index] += delta
            index += index & -index

    def query(self, index):
        res = 0
        while index > 0:
            res += self.tree[index]
            index -= index & -index
        return res

    def range_query(self, l, r):
        return self.query(r) - self.query(l - 1)

def find_min_available(ft, a, b):
    if ft.range_query(a, b) == 0:
        return -1  # should not happen per problem statement
    low = a
    high = b
    ans = b
    while low <= high:
        mid = (low + high) // 2
        if ft.range_query(a, mid) >= 1:
            ans = mid
            high = mid - 1
        else:
            low = mid + 1
    return ans

def find_max_available(ft, a, b):
    if ft.range_query(a, b) == 0:
        return -1  # should not happen per problem statement
    low = a
    high = b
    ans = a
    while low <= high:
        mid = (low + high) // 2
        if ft.range_query(mid, b) >= 1:
            ans = mid
            low = mid + 1
        else:
            high = mid - 1
    return ans

def generate_lex_smallest(n, intervals):
    ft = FenwickTree(n)
    for i in range(1, n+1):
        ft.update(i, 1)
    perm = []
    for a, b in intervals:
        x = find_min_available(ft, a, b)
        perm.append(x)
        ft.update(x, -1)
    return perm
