import itertools

def is_square(p0, p1, p2, p3):
    points = [p0, p1, p2, p3]
    dists = []
    for i in range(4):
        for j in range(i + 1, 4):
            dx = points[i][0] - points[j][0]
            dy = points[i][1] - points[j][1]
            dist_sq = dx * dx + dy * dy
            dists.append(dist_sq)
    unique = list(set(dists))
    if len(unique) != 2:
        return False
    unique.sort()
    d1, d2 = unique
    if d2 != 2 * d1:
        return False
    if d1 == 0:
        return False
    count_d1 = dists.count(d1)
    count_d2 = dists.count(d2)
    return count_d1 == 4 and count_d2 == 2

def main():
    n = int(input())
    for _ in range(n):
        moles = []
        for _ in range(4):
            x, y, a, b = map(int, input().split())
            moles.append((x, y, a, b))
        options = []
        for x, y, a, b in moles:
            dx = x - a
            dy = y - b
            rot = [
                (a + dx, b + dy, 0),
                (a - dy, b + dx, 1),
                (a - dx, b - dy, 2),
                (a + dy, b - dx, 3),
            ]
            options.append(rot)
        min_total = None
        for comb in itertools.product(*options):
            points = [(c[0], c[1]) for c in comb]
            total = sum(c[2] for c in comb)
            if is_square(*points):
