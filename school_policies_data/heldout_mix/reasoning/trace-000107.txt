        self.size = 1

        while self.size < self.n:

class FenwickTree:

    def __init__(self, size):

        self.n = size

        self.tree = [0]*(self.n +1)

    def update(self, idx, delta):

        while idx <= self.n:

            self.tree[idx] += delta

            idx += idx & -idx

    def query(self, idx):

        res =0

        while idx >0:

            res += self.tree[idx]

            idx -= idx & -idx

        return res

def find_kth(fenwick, k):

    # find the smallest x where query(x) >= k

    low =1

    high = fenwick.n

    res = high

    while low <= high:

        mid = (low + high) //2

        s = fenwick.query(mid)

        if s >=k:

            res = mid

            high = mid -1

        else:

            low = mid +1

    return res

        while idx <= self.n:

            self.tree[idx] += delta

            idx += idx & -idx

        res =0

        while idx >0:

            res += self.tree[idx]

            idx -= idx & -idx

def compute_inversions(arr):

2. **Inversion Count**: The number of swaps required to sort an array is equivalent to the number of inversions in the array. If the original array has more inversions than K, we need to reduce the number of inversions to K using the minimum number of swaps.
3. **Lexicographical Order**: Among all possible arrays with at most K inversions, we need the lexicographically smallest one. This requires a greedy approach where we select the smallest possible element at each step while ensuring the remaining elements can form the required number of inversions.

import sys

class FenwickTree:
    def __init__(self, size):
        self.n = size
        self.tree = [0] * (self.n + 2)  # 1-based indexing

    def update(self, idx, delta):
        while idx <= self.n:
            self.tree[idx] += delta
            idx += idx & -idx

    def query(self, idx):
        res = 0
        while idx > 0:
            res += self.tree[idx]
            idx -= idx & -idx
        return res

def compute_inversions(arr):
    max_val = max(arr)
    ft = FenwickTree(max_val)
    inversions = 0
    for x in reversed(arr):
        inversions += ft.query(x - 1)
        ft.update(x, 1)
    return inversions

def find_kth(fenwick, k):
    low = 1
    high = fenwick.n
    res = high
    while low <= high:
        mid = (low + high) // 2
        s = fenwick.query(mid)
        if s >= k:
            res = mid
            high = mid - 1
        else:
            low = mid + 1
    return res
