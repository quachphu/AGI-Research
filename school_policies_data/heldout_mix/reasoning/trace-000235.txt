    if new_L > new_R:
        return False
    # Check if there exists a pair in states[node] that dominates new_L and new_R
    for (L, R) in states[node]:
        if L <= new_L and R >= new_R:
            return False  # new is dominated, do not add

    to_remove = []

        if L >= new_L and R <= new_R:

    # Remove them in reverse order to avoid index shifting
    for i in reversed(to_remove):
        del states[node][i]
    # Add the new pair
    states[node].append( (new_L, new_R) )
    return True

To solve this problem, we need to determine the maximum loyalty of a path from node 1 to node n in an undirected graph where each edge has a range of integers. The loyalty of a path is defined as the number of integers x that can traverse the entire path, which is determined by the intersection of the ranges of all edges in the path.

### Approach
1. **Graph Representation**: Represent the graph using an adjacency list where each node points to its connected edges along with their respective ranges.
2. **Breadth-First Search (BFS) with State Tracking**: Use BFS to explore paths from node 1 to node n. Track the current maximum lower bound (L) and minimum upper bound (R) of the ranges encountered along each path.
3. **Dominance Pruning**: For each node, maintain a list of non-dominated (L, R) pairs. A pair (L1, R1) is dominated by (L2, R2) if L2 ≤ L1 and R2 ≥ R1. This helps in efficiently managing the states and avoiding redundant checks.
4. **State Management**: For each node, add new states only if they are not dominated by existing states and remove any existing states that are dominated by the new state.

### Solution Code

def add_state(states, node, new_L, new_R):
    if new_L > new_R:
        return False
    for (L, R) in states[node]:
        if L <= new_L and R >= new_R:
            return False
    new_states = []
    for (L, R) in states[node]:
        if not (L >= new_L and R <= new_R):
            new_states.append((L, R))
    new_states.append((new_L, new_R))
    states[node] = new_states
    return True

def main():
    import sys
    from collections import deque

    input = sys.stdin.read().split()
    idx = 0
    n = int(input[idx])
    idx += 1
    m = int(input[idx])
    idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        a = int(input[idx])
        idx += 1
        b = int(input[idx])
        idx += 1
        l = int(input[idx])
        idx += 1
        r = int(input[idx])
        idx += 1
        adj[a].append((b, l, r))
        adj[b].append((a, l, r))

    states = [[] for _ in range(n + 1)]
    queue = deque()

    for edge in adj[1]:
        v, l, r = edge
        if add_state(states, v, l, r):
            queue.append((v, l, r))

    while queue:
        u, current_L, current_R = queue.popleft()
        for edge in adj[u]:
            v, l, r = edge
            new_L = max(current_L, l)
            new_R = min(current_R, r)
            if add_state(states, v, new_L, new_R):
                queue.append((v, new_L, new_R))
