1. **Preprocessing Movement Paths**: For each cell in the grid, precompute the path Wilbur would follow starting from that cell. This path is determined by the movement vectors of the digits in the cells. Track the first occurrence of each digit along these paths.
2. **Check Reachable Digits**: For each cell, determine which digits can be encountered along the path starting from that cell. This helps quickly verify if a digit required by the query string can be found.
3. **Query Processing**: For each query string, check if all required digits are present in the grid. Then, simulate Wilbur's movement step-by-step, using the preprocessed information to determine if each subsequent digit can be reached.

### Solution Code
```python
import sys

def main():
    sys.setrecursionlimit(1 << 25)
    n, m, q = map(int, sys.stdin.readline().split())
    grid = []
    for _ in range(n):
        line = sys.stdin.readline().strip()
        grid.append([int(c) for c in line])
    # Read the 10 movement vectors (a_0 to a_9, b_0 to b_9)
    movement = []
    for _ in range(10):
        a, b = map(int, sys.stdin.readline().split())
        movement.append((a, b))
    
    # Precompute for each cell (a, b) the next cell when moving according to its digit
    next_cell = [[(0, 0) for _ in range(m+1)] for __ in range(n+1)]
    for x in range(1, n+1):
        for y in range(1, m+1):
            d = grid[x-1][y-1]
            a, b = movement[d]
            nx = x + a
            ny = y + b
            if 1 <= nx <= n and 1 <= ny <= m:
                next_cell[x][y] = (nx, ny)
            else:
                next_cell[x][y] = (x, y)
    
    # Precompute first_occurrence and digits_reachable for each cell
    first_occurrence = [[{} for _ in range(m+1)] for __ in range(n+1)]
    digits_reachable = [[set() for _ in range(m+1)] for __ in range(n+1)]
    
    for x in range(1, n+1):
        for y in range(1, m+1):
            current = (x, y)
            visited = {}
            path = []
            fo = {}
            cell = current
            steps = 0
            while True:
                if cell in visited:
                    idx = visited[cell]
                    cycle = path[idx:]
                    # Process cycle to fill remaining digits
                    for (cx, cy) in cycle:
                        d = grid[cx-1][cy-1]
                        if d not in fo:
                            fo[d] = (cx, cy)
                    break
                visited[cell] = len(path)
                cx, cy = cell
                d = grid[cx-1][cy-1]
                if d not in fo:
                    fo[d] = cell
                path.append(cell)
                # Move to next cell
                cell = next_cell[cx][cy]
                if cell in visited:
                    # Check if we need to process the cycle
                    idx = visited[cell]
                    cycle = path[idx:]
                    for (cx_cyc, cy_cyc) in cycle:
                        d_cyc = grid[cx_cyc-1][cy_cyc-1]
                        if d_cyc not in fo:
                            fo[d_cyc] = (cx_cyc, cy_cyc)
                    break
            # Update first_occurrence and digits_reachable
            first_occurrence[x][y] = fo
            digits_reachable[x][y] = set(fo.keys())
    
    # Precompute all digits present in the grid
    grid_digits = set()
    for row in grid:
        for d in row:
            grid_digits.add(d)
    
    for _ in range(q):
        s = sys.stdin.readline().strip()
        # Check if all characters are present in the grid
        valid = True
        for c in s:
            if int(c) not in grid_digits:
                valid = False
                break
        if not valid:
            print("NO")
            continue
        if len(s) == 0:
            print("YES")
            continue
        # Initialize current cells
        current = set()
        target = int(s[0])
        for x in range(1, n+1):
            for y in range(1, m+1):
                if grid[x-1][y-1] == target:
                    current.add((x, y))
        if not current:
            print("NO")
            continue
        # Process each character in s
        ok = True
        for i in range(len(s)-1):
            next_current = set()
            current_d = int(s[i])
            next_d = int(s[i+1])
            for (x, y) in current:
                # Move according to current_d's vector
                a, b = movement[current_d]
                nx = x + a
                ny = y + b
                if nx < 1 or nx > n or ny < 1 or ny > m:
                    nx, ny = x, y
                # Check if next_d is reachable from (nx, ny)
                if next_d not in digits_reachable[nx][ny]:
                    continue
                # Get the first occurrence of next_d in the path from (nx, ny)
                fo = first_occurrence[nx][ny].get(next_d, None)
                if fo is not None:
                    next_current.add(fo)
            current = next_current
            if not current:
                ok = False
                break
        print("YES" if ok else "NO")

if __name__ == "__main__":
    main()
```

2. **Query Handling**: For each query, check if all required digits are present in the grid. Then, simulate the movement step-by-step, using the preprocessed data to efficiently determine if each subsequent digit can be reached from the current position.

This approach ensures that each query is processed efficiently by leveraging precomputed information about the possible paths and reachable digits from each cell.
