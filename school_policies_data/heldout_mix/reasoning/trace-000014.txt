import sys
from collections import defaultdict

def find_apocalypse_date():
    s = sys.stdin.readline().strip()
    freq = defaultdict(int)
    max_days = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
    n = len(s)
    
    for i in range(n - 9):
        sub = s[i:i+10]
        if sub[2] != '-' or sub[5] != '-':
            continue
        valid = True
        for pos in [0, 1, 3, 4, 6, 7, 8, 9]:
            if not sub[pos].isdigit():
                valid = False
                break
        if not valid:
            continue
        
        day_part = sub[:2]
        month_part = sub[3:5]
        year_part = sub[6:10]
        
        year = int(year_part)
        if year not in {2013, 2014, 2015}:
            continue
        
        month = int(month_part)
        if month < 1 or month > 12:
            continue
        
        day = int(day_part)
        if day < 1:
            continue
        
        if day > max_days[month - 1]:
            continue
        
        freq[sub] += 1
    
    max_count = -1
    result = ""
    for date, cnt in freq.items():
        if cnt > max_count:
            max_count = cnt
            result = date
    print(result)

find_apocalypse_date()
