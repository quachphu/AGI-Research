import sys

def main():
    s = list(sys.stdin.readline().strip())
    q = int(sys.stdin.readline())
    for _ in range(q):
        parts = sys.stdin.readline().split()
        if not parts:
            continue  # handle empty lines if any
        if parts[0] == '1':
            # Update query: 1 i c
            i = int(parts[1]) - 1
            c = parts[2]
            s[i] = c
        else:
            # Query: 2 l r y
            l = int(parts[1]) - 1
            r = int(parts[2]) - 1
            y = parts[3]
            m = len(y)
            if m == 0 or m > (r - l + 1):
                print(0)
                continue
            # Compute failure array for KMP
            failure = [0] * m
            j = 0
            for i in range(1, m):
                while j > 0 and y[i] != y[j]:
                    j = failure[j - 1]
                if y[i] == y[j]:
                    j += 1
                    failure[i] = j
                else:
                    failure[i] = 0
            # Perform KMP search in the range [l, r]
            j = 0
            count = 0
            for i in range(l, r + 1):
                current_char = s[i]
                while j > 0 and current_char != y[j]:
                    j = failure[j - 1]
                if current_char == y[j]:
                    j += 1
                if j == m:
                    start_pos = i - m + 1
                    if start_pos >= l and start_pos <= r - m + 1:
                        count += 1
                    j = failure[j - 1]
            print(count)
