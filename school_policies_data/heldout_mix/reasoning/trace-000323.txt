class Node:

    def __init__(self, total, prefix_max, suffix_max, max_subarray):

        self.total = total

        self.prefix_max = prefix_max

        self.suffix_max = suffix_max

        self.max_subarray = max_subarray

def merge(left, right):

    total = left.total + right.total

    prefix_max = max(left.prefix_max, left.total + right.prefix_max)

    suffix_max = max(right.suffix_max, right.total + left.suffix_max)

    max_subarray = max(left.max_subarray, right.max_subarray, left.suffix_max + right.prefix_max)

    return Node(total, prefix_max, suffix_max, max_subarray)

class SegmentTree:

    def __init__(self, data):

        self.n = len(data)

        self.size = 1

To solve this problem, we need to determine the maximum possible score Alice can achieve after Bob removes a card from a segment she chooses. The score is the sum of the remaining cards in the segment after Bob removes the card with the maximum value. Alice aims to maximize this score, while Bob aims to minimize it.

### Approach
1. **Identify Previous and Next Greater Elements**: For each element, determine the nearest greater element to the left (L) and the nearest greater element to the right (R). These boundaries help identify the segments where the current element is the maximum.
2. **Segment Tree for Maximum Subarray Sum**: Use a segment tree to efficiently query the maximum subarray sum within any given range. This helps in determining the maximum sum of subarrays where the current element is the maximum.
3. **Calculate Contributions**: For each element, calculate the contribution to the score by finding the maximum subarray sum within the boundaries defined by L and R, then subtract the element's value. The maximum of these contributions across all elements gives the result.

class Node:
    def __init__(self, total, prefix_max, suffix_max, max_subarray):
        self.total = total
        self.prefix_max = prefix_max
        self.suffix_max = suffix_max
        self.max_subarray = max_subarray

def merge(left, right):
    if left.max_subarray == -float('inf'):
        return right
    if right.max_subarray == -float('inf'):
        return left
    total = left.total + right.total
    prefix_max = max(left.prefix_max, left.total + right.prefix_max)
    suffix_max = max(right.suffix_max, right.total + left.suffix_max)
    max_sub = max(left.max_subarray, right.max_subarray, left.suffix_max + right.prefix_max)
    return Node(total, prefix_max, suffix_max, max_sub)

class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.tree = [Node(0, 0, 0, 0) for _ in range(2 * self.size)]
        for i in range(self.n):
            val = data[i]
            self.tree[self.size + i] = Node(val, val, val, val)
        for i in range(self.size - 1, 0, -1):
            self.tree[i] = merge(self.tree[2 * i], self.tree[2 * i + 1])
    
    def query_range(self, l, r):
        res_left = Node(-float('inf'), -float('inf'), -float('inf'), -float('inf'))
        res_right = Node(-float('inf'), -float('inf'), -float('inf'), -float('inf'))
        l += self.size
        r += self.size
        while l <= r:
            if l % 2 == 1:
                res_left = merge(res_left, self.tree[l])
                l += 1
            if r % 2 == 0:
                res_right = merge(self.tree[r], res_right)
                r -= 1
            l >>= 1
            r >>= 1
        merged = merge(res_left, res_right)
        return merged.max_subarray if merged.max_subarray != -float('inf') else 0

n = int(input())
a = list(map(int, input().split()))
n = len(a)

# Compute L[i]: previous greater element's index
L = [-1] * n
stack = []
for i in range(n):
    while stack and a[stack[-1]] <= a[i]:
        stack.pop()
    if stack:
        L[i] = stack[-1]
    else:
        L[i] = -1
    stack.append(i)

# Compute R[i]: next greater element's index
R = [n] * n
stack = []
for i in range(n-1, -1, -1):
    while stack and a[stack[-1]] <= a[i]:
        stack.pop()
    if stack:
        R[i] = stack[-1]
    else:
        R[i] = n
    stack.append(i)

st = SegmentTree(a)
max_contribution = 0

for i in range(n):
    window_start = L[i] + 1
    window_end = R[i] - 1
    if window_start > window_end:
        continue
    max_sum = st.query_range(window_start, window_end)
    contribution = max_sum - a[i]
    if contribution > max_contribution:
        max_contribution = contribution

answer = max(max_contribution, 0)
print(answer)

### Explanation
1. **Previous and Next Greater Elements**: Using a monotonic stack, we efficiently determine the boundaries (L and R) for each element where it is the maximum in the segment.
2. **Segment Tree**: The segment tree is built to support range queries for maximum subarray sums, allowing efficient calculation of the maximum possible sum within any segment.
3. **Contribution Calculation**: For each element, the maximum subarray sum within its valid segment is queried using the segment tree. The contribution of this element to the score is calculated and the maximum contribution is tracked.

This approach ensures that we efficiently handle the constraints and find the optimal solution in O(n log n) time complexity.
