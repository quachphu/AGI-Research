import sys

def left_rotate(s):
    return s[1:] + s[0] if s else s

    return s[-1] + s[:-1] if s else s

    n = len(s)
    if n == 0:
        return s
    if n % 2 == 0:
        mid = n // 2
        return s[mid:] + s[:mid]
    else:
        mid = n // 2
        return s[mid+1:] + s[mid] + s[:mid]

def reverse_string(s):
    return s[::-1]

    res = []
    for c in s:

            res.append(str((int(c) + 1) % 10))
        else:
            res.append(c)

        if c.isdigit():
            res.append(str((int(c) - 1) % 10))
        else:
            res.append(c)
    return ''.join(res)

def main():

    for _ in range(n):
