from itertools import combinations
from collections import defaultdict
import sys

def main():
    likes = defaultdict(set)
    n = int(sys.stdin.readline())
    for _ in range(n):
        line = sys.stdin.readline().strip()
        p, q = line.split(' likes ')
        likes[p].add(q)
    a, b, c = map(int, sys.stdin.readline().split())
    
    heroes = ['Anka', 'Chapay', 'Cleo', 'Troll', 'Dracul', 'Snowy', 'Hexadecimal']
    
    min_diff = float('inf')
    optimal_triplets = []
    
    # Generate all possible ordered triplets (sa, sb, sc)
    for sa in range(1, 6):
        max_sb = 7 - sa - 1
        for sb in range(1, max_sb + 1):
            sc = 7 - sa - sb
            if sc < 1:
                continue
            exp_a = a // sa
            exp_b = b // sb
            exp_c = c // sc
            current_max = max(exp_a, exp_b, exp_c)
            current_min = min(exp_a, exp_b, exp_c)
            diff = current_max - current_min
            if diff < min_diff:
                min_diff = diff
                optimal_triplets = [(sa, sb, sc)]
            elif diff == min_diff:
                optimal_triplets.append((sa, sb, sc))
    
    max_likes = 0
    for triplet in optimal_triplets:
        sa, sb, sc = triplet
        for team_a in combinations(heroes, sa):
            remaining_after_a = [h for h in heroes if h not in team_a]
            for team_b in combinations(remaining_after_a, sb):
                team_c = [h for h in remaining_after_a if h not in team_b]
                total = 0
                # Check likes in team_a
                for p in team_a:
                    for q in team_a:
                        if p != q and q in likes[p]:
                            total += 1
                # Check likes in team_b
                for p in team_b:
                    for q in team_b:
                        if p != q and q in likes[p]:
                            total += 1
                # Check likes in team_c
                for p in team_c:
                    for q in team_c:
                        if p != q and q in likes[p]:
                            total += 1
                if total > max_likes:
                    max_likes = total
    print(min_diff, max_likes)

if __name__ == "__main__":
