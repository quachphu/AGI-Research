def compute_next_pos(s):
    n = len(s)
    next_pos = [{'0': n, '1': n} for _ in range(n + 1)]
    for i in range(n - 1, -1, -1):
        current_char = s[i]
        next_pos[i]['0'] = i if current_char == '0' else next_pos[i + 1]['0']
        next_pos[i]['1'] = i if current_char == '1' else next_pos[i + 1]['1']
    return next_pos

def is_subsequence(s, next_pos, length):
    pos = 0
    for c in s:
        next_p = next_pos[pos][c]
        if next_p >= length:
            return False
        pos = next_p + 1
    return True

def find_shortest_non_subsequence(s1, s2):
    next1 = compute_next_pos(s1)
    next2 = compute_next_pos(s2)
    len1, len2 = len(s1), len(s2)
    
    k = 1
    while True:
        max_num = 1 << k  # 2^k
        for num in range(max_num):
            # Generate binary string with leading zeros of length k
            candidate = bin(num)[2:].zfill(k)
            # Check against s1
            if is_subsequence(candidate, next1, len1):
                continue
            # Check against s2
            if is_subsequence(candidate, next2, len2):
                continue
            return candidate
        k += 1
