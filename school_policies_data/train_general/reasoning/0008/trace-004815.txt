def main():
    import sys
    from sympy import nextprime

    n = int(sys.stdin.readline())
    if n == 0:
        m = 0
    else:
        m = int('2' * n)
    p = nextprime(m)
    k = p - 1
    mod = 9 * p
    a = pow(10, k, mod)
    remainder = (a - 1) // 9 % p
    print(remainder)

main()
