def lca(u, v, depth, up, max_level):

    # Bring u and v to the same depth.

    if depth[u] < depth[v]:

        u, v = v, u

    # Bring u up to depth of v.

    for k in reversed(range(max_level)):

    for k in reversed(range(max_level)):

        if up[k][u] != up[k][v]:

            u = up[k][u]

            v = up[k][v]

    return up[0][u]

        w2 = int(input[ptr + 3])
        ptr += 4

        u_o = 2 * x - 1
        v_o = 2 * y - 1
        adj_o[u_o].append((v_o, w1))
        adj_o[v_o].append((u_o, w1))

        u_e = 2 * x
        v_e = 2 * y
        adj_e[u_e].append((v_e, w2))
        adj_e[v_e].append((u_e, w2))

    max_level = 20

    # Preprocess O tree
    root_o = 1
    parent_o = [-1] * max_node
    depth_o = [0] * max_node
    distance_o = [0] * max_node
    q = deque([root_o])
    parent_o[root_o] = -1
    while q:
        u = q.popleft()
        for v, w in adj_o[u]:
            if parent_o[v] == -1 and v != root_o:
                parent_o[v] = u
                depth_o[v] = depth_o[u] + 1
                distance_o[v] = distance_o[u] + w
                q.append(v)

    up_o = [[-1] * max_node for _ in range(max_level)]
    for i in range(max_node):
        up_o[0][i] = parent_o[i]
    for k in range(1, max_level):
        for i in range(max_node):
            if up_o[k-1][i] != -1:
                up_o[k][i] = up_o[k-1][up_o[k-1][i]]

    # Preprocess E tree
    root_e = 2
    parent_e = [-1] * max_node
    depth_e = [0] * max_node
    distance_e = [0] * max_node
    q = deque([root_e])
    parent_e[root_e] = -1
    while q:
        u = q.popleft()
        for v, w in adj_e[u]:
            if parent_e[v] == -1 and v != root_e:
                parent_e[v] = u
                depth_e[v] = depth_e[u] + 1
                distance_e[v] = distance_e[u] + w
                q.append(v)

    up_e = [[-1] * max_node for _ in range(max_level)]
    for i in range(max_node):
        up_e[0][i] = parent_e[i]
    for k in range(1, max_level):
        for i in range(max_node):
            if up_e[k-1][i] != -1:
                up_e[k][i] = up_e[k-1][up_e[k-1][i]]

    def lca(u, v, depth, up):
        if depth[u] < depth[v]:
            u, v = v, u
        for k in reversed(range(max_level)):
            if depth[u] - (1 << k) >= depth[v]:
                u = up[k][u]
        if u == v:
            return u
        for k in reversed(range(max_level)):
            if up[k][u] != up[k][v] and up[k][u] != -1 and up[k][v] != -1:
                u = up[k][u]
                v = up[k][v]
        return up[0][u]

    def get_distance(u, v, depth, up, distance):
        ancestor = lca(u, v, depth, up)
        if ancestor == -1:
            return 0
        return distance[u] + distance[v] - 2 * distance[ancestor]

    q_count = int(input[ptr])
    ptr += 1
    output = []
    for _ in range(q_count):
        u = int(input[ptr])
        v = int(input[ptr + 1])
        ptr += 2

        if u % 2 == 1 and v % 2 == 1:
            d_o = get_distance(u, v, depth_o, up_o, distance_o)
            e_u = u + 1
            e_v = v + 1
            d_e = get_distance(e_u, e_v, depth_e, up_e, distance_e)
            wu = w_oe[(u - 1) // 2]
            wv = w_oe[(v - 1) // 2]
            ans = min(d_o, wu + d_e + wv)
        elif u % 2 == 0 and v % 2 == 0:
            d_e = get_distance(u, v, depth_e, up_e, distance_e)
            o_u = u - 1
            o_v = v - 1
            d_o = get_distance(o_u, o_v, depth_o, up_o, distance_o)
            wu = w_oe[(o_u - 1) // 2]
            wv = w_oe[(o_v - 1) // 2]
            ans = min(d_e, wu + d_o + wv)
        elif u % 2 == 1 and v % 2 == 0:
            wu = w_oe[(u - 1) // 2]
            e_u_plus = u + 1
            d_e = get_distance(e_u_plus, v, depth_e, up_e, distance_e)
            option1 = wu + d_e
            o_v_minus = v - 1
            d_o = get_distance(u, o_v_minus, depth_o, up_o, distance_o)
            option2 = d_o + w_oe[(o_v_minus - 1) // 2]
            ans = min(option1, option2)
        else:
            o_u_minus = u - 1
            option1 = w_oe[(o_u_minus - 1) // 2] + get_distance(o_u_minus, v, depth_o, up_o, distance_o)
            e_v_plus = v + 1
            d_e = get_distance(u, e_v_plus, depth_e, up_e, distance_e)
            option2 = d_e + w_oe[(v - 1) // 2]
            ans = min(option1, option2)
        output.append(str(ans))
    
    print('\n'.join(output))

2. **LCA and Distance Preprocessing**: Using BFS, we compute the parent, depth, and distance arrays for each node from the root. Binary lifting tables are constructed to enable efficient LCA queries.
3. **Query Handling**: For each query, based on the parity of the nodes, we compute the shortest path by considering both direct paths within the same tree and paths that switch between trees using the connecting edges. The LCA and precomputed distances allow us to efficiently determine the shortest path.

This approach ensures that each query is processed in O(log n) time, making the solution efficient even for large input sizes.
