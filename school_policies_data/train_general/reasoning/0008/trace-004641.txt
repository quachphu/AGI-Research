To solve this problem, we need to efficiently answer multiple queries about finding the minimal element in a subarray that occurs more than a certain threshold number of times. Given the constraints, a direct approach would be too slow, so we use a segment tree combined with frequency tracking to handle the queries efficiently.

### Approach
1. **Preprocess Occurrences**: For each element in the array, maintain a list of indices where the element occurs. This allows quick frequency checks using binary search.
2. **Segment Tree Construction**: Build a segment tree where each node stores up to 4 candidate elements that are frequent in the interval covered by the node. This helps in quickly narrowing down potential candidates for each query.
3. **Query Processing**: For each query, use the segment tree to collect candidate elements. Merge these candidates and check their actual frequencies in the queried interval using binary search on the preprocessed occurrence lists.

        size <<= 1
    tree = [[] for _ in range(2 * size)]

    # Fill leaves (0-based array)
    for i in range(n):
        tree[size + i] = [(a[i], 1)]

    # Build internal nodes
    for i in range(size - 1, 0, -1):
        left = tree[2 * i]
        right = tree[2 * i + 1]
        freq = defaultdict(int)
        for val, cnt in left:
            freq[val] += cnt
        for val, cnt in right:
            freq[val] += cnt
        merged = sorted(freq.items(), key=lambda x: (-x[1], x[0]))
        tree[i] = merged[:4]

    # Function to query the segment tree for range [l, r] (0-based)
    def query_segment_tree(l, r):
        res = []
        l += size
        r += size
        while l <= r:
            if l % 2 == 1:
                res.extend(tree[l])
                l += 1
            if r % 2 == 0:
                res.extend(tree[r])
                r -= 1
            l >>= 1
            r >>= 1
        # Merge all candidates
        freq = defaultdict(int)
        for val, cnt in res:
            freq[val] += cnt
        merged = sorted(freq.items(), key=lambda x: (-x[1], x[0]))
        return merged

    # Process each query
    output = []
    for _ in range(q):
        l_input = int(input[ptr])
        r_input = int(input[ptr+1])
        k = int(input[ptr+2])
        ptr += 3
        l = l_input - 1  # Convert to 0-based
        r = r_input - 1
        s = r - l + 1
        t = s // k
        required = t + 1  # Because we need count > s/k

        # Query the segment tree to get candidates
        candidates = query_segment_tree(l, r)
        candidates = candidates[:k-1]  # Take up to k-1 candidates

        # Check each candidate in sorted order of value
        ans = -1
        for val, _ in sorted(candidates, key=lambda x: x[0]):
            lst = occ.get(val, [])
            left_pos = bisect_left(lst, l_input)  # l_input is 1-based
            right_pos = bisect_right(lst, r_input)
            cnt = right_pos - left_pos
            if cnt >= required:
                ans = val
                break
        output.append(str(ans))

    print('\n'.join(output))

if __name__ == "__main__":
    main()
