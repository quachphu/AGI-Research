To solve this problem, we need to determine the number of valid n-length Martian DNA chains given certain forbidden consecutive nucleotide pairs. The solution involves using matrix exponentiation to efficiently compute the result even for very large values of n.

### Approach
1. **Matrix Representation**: Represent the allowed transitions between nucleotides using a matrix where each entry (i, j) indicates if transitioning from nucleotide i to j is allowed.
2. **Matrix Exponentiation**: Use matrix exponentiation to compute the number of valid transitions for large values of n efficiently. This reduces the time complexity from O(n) to O(log n) using exponentiation by squaring.
3. **Modulo Operation**: Since the result can be very large, use modulo 10^9 + 7 to keep the numbers manageable during computations.

```python
MOD = 10**9 + 7

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1
    k = int(input[ptr])
    ptr += 1
    
    if n == 1:
        print(m % MOD)
        return
    
    # Initialize transition matrix with all 1s
    transition = [[1 for _ in range(m)] for _ in range(m)]
    
    for _ in range(k):
        s = input[ptr]
        ptr += 1
        first = s[0]
        second = s[1]
        
        # Convert characters to their nucleotide numbers
        if first.islower():
            i = ord(first) - ord('a') + 1
        else:
            i = 26 + ord(first) - ord('A') + 1
        if second.islower():
            j = ord(second) - ord('a') + 1
        else:
            j = 26 + ord(second) - ord('A') + 1
        
        # Update the transition matrix
        transition[i-1][j-1] = 0
    
    def matrix_mult(a, b, mod):
        size = len(a)
        result = [[0]*size for _ in range(size)]
        for i in range(size):
            for k in range(size):
                if a[i][k] == 0:
                    continue
                for j in range(size):
                    result[i][j] = (result[i][j] + a[i][k] * b[k][j]) % mod
        return result
    
    def matrix_power(mat, power, mod):
        size = len(mat)
        result = [[0]*size for _ in range(size)]
        # Initialize result as identity matrix
        for i in range(size):
            result[i][i] = 1
        while power > 0:
            if power % 2 == 1:
                result = matrix_mult(result, mat, mod)
            mat = matrix_mult(mat, mat, mod)
            power //= 2
        return result
    
    powered = matrix_power(transition, n-1, MOD)
    total = 0
    for row in powered:
        total = (total + sum(row)) % MOD
    print(total % MOD)

if __name__ == "__main__":
    main()
```
