```python
import sys
from math import gcd
from collections import deque

def main():
    input = sys.stdin.read().split()
    ptr = 0
    while True:
        m = int(input[ptr])
        n = int(input[ptr+1])
        ptr += 2
        if m == 0 and n == 0:
            break
        blue = list(map(int, input[ptr:ptr+m]))
        ptr += m
        red = list(map(int, input[ptr:ptr+n]))
        ptr += n
        
        # Build adjacency list
        adj = [[] for _ in range(m)]
        for i in range(m):
            b = blue[i]
            for j in range(n):
                r = red[j]
                if gcd(b, r) > 1:
                    adj[i].append(j)
        
        # Hopcroft-Karp algorithm implementation
        def hopcroft_karp():
            pair_u = [-1] * m
            pair_v = [-1] * n
            dist = [0] * m
            result = 0
            
            def bfs():
                queue = deque()
                for u in range(m):
                    if pair_u[u] == -1:
                        dist[u] = 0
                        queue.append(u)
                    else:
                        dist[u] = float('inf')
                dist_null = float('inf')
                while queue:
                    u = queue.popleft()
                    if dist[u] < dist_null:
                        for v in adj[u]:
                            if pair_v[v] == -1:
                                dist_null = dist[u] + 1
                            elif dist[pair_v[v]] == float('inf'):
                                dist[pair_v[v]] = dist[u] + 1
                                queue.append(pair_v[v])
                return dist_null != float('inf')
            
            def dfs(u):
                for v in adj[u]:
                    if pair_v[v] == -1 or (dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v])):
                        pair_u[u] = v
                        pair_v[v] = u
                        return True
                dist[u] = float('inf')
                return False
            
            while bfs():
                for u in range(m):
                    if pair_u[u] == -1:
                        if dfs(u):
                            result += 1
            return result
        
        print(hopcroft_karp())

if __name__ == '__main__':
    main()
```

### Explanation
1. **Reading Input**: The input is read in one go for efficiency, and then parsed dataset by dataset.
2. **Graph Construction**: For each dataset, an adjacency list is constructed where each blue card is connected to all red cards it can pair with based on their common divisor.
3. **Hopcroft-Karp Algorithm**:
   - **BFS**: This initializes the layers of nodes starting from unmatched blue nodes and checks for potential augmenting paths.
   - **DFS**: This searches for augmenting paths using the layers defined by BFS, updating the matching pairs when a path is found.
4. **Result Calculation**: The algorithm iterates until no more augmenting paths are found, and the total number of pairs is printed for each dataset.

This approach efficiently finds the maximum number of valid pairs using the Hopcroft-Karp algorithm, ensuring optimal performance even for the upper constraint limits.
