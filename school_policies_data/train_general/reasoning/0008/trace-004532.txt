To solve this problem, we need to determine the R-th digit in the decimal expansion of a proper fraction N/D. The solution involves understanding whether the decimal expansion terminates or repeats, and efficiently computing the desired digit without simulating each step up to R, especially for large values of R.

### Approach
1. **Check for Terminating Decimal**: A fraction terminates if the denominator (after removing all factors of 2 and 5) is 1. The number of digits before termination is determined by the maximum exponent of 2 or 5 in the denominator.
2. **Non-Terminating Decimal**: If the decimal expansion does not terminate, it will eventually repeat. The length of the non-repeating part is determined by the maximum exponent of 2 or 5. The repeating cycle's length is found using the multiplicative order of 10 modulo the remaining part of the denominator.
3. **Efficient Calculation**: For non-terminating decimals, use modular arithmetic to directly compute the R-th digit in the repeating cycle without generating all preceding digits.

import sys
from math import gcd
from collections import defaultdict

def factorize(n):
    factors = []
    while n % 2 == 0:
        factors.append(2)
        n = n // 2
    i = 3
    while i * i <= n:
        while n % i == 0:
            factors.append(i)
            n = n // i
        i += 2
    if n > 1:
        factors.append(n)
    return factors

def euler_phi(n):
    if n == 1:
        return 1
    factors = factorize(n)
    unique_factors = {}
    for p in factors:
        unique_factors[p] = unique_factors.get(p, 0) + 1
    result = 1
    for p in unique_factors:
        exp = unique_factors[p]
        result *= (p - 1) * (p ** (exp - 1))
    return result

def get_divisors(m):
    if m == 0:
        return []
    factors = factorize(m)
    factor_counts = defaultdict(int)
    for p in factors:
        factor_counts[p] += 1
    primes = list(factor_counts.keys())
    exponents = [factor_counts[p] for p in primes]

    divisors = set()
    def backtrack(index, current):
        if index == len(primes):
            divisors.add(current)
            return
        p = primes[index]
        max_exp = exponents[index]
        for e in range(max_exp + 1):
            backtrack(index + 1, current * (p ** e))
    backtrack(0, 1)
    return sorted(divisors)

def multiplicative_order(a, n):
    if gcd(a, n) != 1:
        return -1
    phi_val = euler_phi(n)
    divisors = get_divisors(phi_val)
    for d in divisors:
        if pow(a, d, n) == 1:
            return d
    return -1

def get_exponent(n, p):
    count = 0
    while n % p == 0:
        count += 1
        n = n // p
    return count

def solve():
    input = sys.stdin.read().split()
    idx = 0
    T = int(input[idx])
    idx += 1
    for _ in range(T):
        N = int(input[idx])
        D = int(input[idx+1])
        R = int(input[idx+2])
        idx += 3
        
        a = get_exponent(D, 2)
        b = get_exponent(D, 5)
        remaining_D = D // ((2**a) * (5**b))
        
        if remaining_D == 1:
            K = max(a, b)
            if R > K:
                print(0)
            else:
                current_remainder = N
                digit = 0
                for _ in range(R):
                    current_remainder *= 10
                    digit = current_remainder // D
                    current_remainder %= D
                print(digit)
        else:
            K = max(a, b)
            if R <= K:
                current_remainder = N
                digit = 0
                for _ in range(R):
                    current_remainder *= 10
                    digit = current_remainder // D
                    current_remainder %= D
                print(digit)
            else:
                S = R - K
                M = remaining_D
                exponent_2 = K - a
                exponent_5 = K - b
                factor_2 = pow(2, exponent_2, M)
                factor_5 = pow(5, exponent_5, M)
                r = (N % M) * factor_2 % M
                r = r * factor_5 % M

                t = (S - 1) % L
