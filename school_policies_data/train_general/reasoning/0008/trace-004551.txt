   - **Priority 3**: If stopping before the line is impossible, ensure the spaceship's velocity becomes negative after passing the line.
   - **Priority 4**: If all else fails, the spaceship crashes.

3. **Algorithm**: Use a Breadth-First Search (BFS) approach to simulate all possible sequences of deceleration modes and track the best outcomes. import sys
from collections import deque

def main():
    input = sys.stdin.read().split()
    idx = 0
    c = int(input[idx])
    idx += 1
    first_case = True
    for _ in range(c):
        if not first_case:
            print()
        first_case = False
        n = int(input[idx])
        idx += 1
        a_list = list(map(int, input[idx:idx+n]))
        idx += n
        q = int(input[idx])
        idx += 1
        queries = []
        for __ in range(q):
            x, v = map(int, input[idx:idx+2])
            idx += 2
            queries.append((x, v))
        
        a_list = sorted(a_list, reverse=True)
        
        for x, v in queries:
            has_perfect = False
            best_good = None
            has_try_again = False
            visited = set()
            queue = deque()
            initial_p = x
            initial_v = v
            queue.append((initial_p, initial_v))
            visited.add((initial_p, initial_v))
            
            while queue:
                p, current_v = queue.popleft()
                
                if p == 0.0:
                    has_perfect = True
                    break
                elif p > 0:
                    if best_good is None or p < best_good:
                        best_good = p
                else:
                    pass
                
                for a in a_list:
                    new_v = current_v - a
                    s = current_v * 1.0 - 0.5 * a * 1.0 ** 2
                    new_p = p - s
                    
                    if new_p < 0 and new_v < 0:
                        has_try_again = True
                    
                    key = (new_p, new_v)
                    if key not in visited:
                        visited.add(key)
                        queue.append(key)
                
            if has_perfect:
                print("perfect")
            elif best_good is not None:
                print(f"good {best_good:.3f}")
            elif has_try_again:
                print("try again")
            else:
                print("crash")

if __name__ == "__main__":
    main()
