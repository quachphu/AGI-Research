class DSU:

    def __init__(self, size):

        self.parent = list(range(size +1))  # nodes are 1-based to size

        self.rank = [1]*(size +1)

    def find(self, x):

        if self.parent[x] != x:

            self.parent[x] = self.find(self.parent[x])

        return self.parent[x]

    def union(self, x, y):

        x_root = self.find(x)

        y_root = self.find(y)

        if x_root == y_root:

            return

        if self.rank[x_root] < self.rank[y_root]:

            self.parent[x_root] = y_root

        else:

            self.parent[y_root] = x_root

            if self.rank[x_root] == self.rank[y_root]:

                self.rank[x_root] +=1

2. **Preserved Cells Check**: Process the preserved cells to ensure they do not violate the product constraints. If any row or column with all cells preserved does not meet the required product, the answer is 0.

3. **Union-Find Data Structure**: Use a Disjoint Set Union (DSU) to model the bipartite graph of rows and columns connected by unpreserved cells. This helps in identifying connected components.

4. **Component Consistency Check**: For each connected component in the bipartite graph, check if the sum of row constraints matches the sum of column constraints modulo 2. If any component fails this check, the answer is 0.

5. **Calculate Solutions**: Compute the number of valid configurations using the number of connected components and other derived values, applying modulo arithmetic to handle large numbers.

class DSU:
    def __init__(self, size):
        self.parent = list(range(size + 1))  # 1-based indexing
        self.rank = [1] * (size + 1)
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return
        if self.rank[x_root] < self.rank[y_root]:
            self.parent[x_root] = y_root
        else:
            self.parent[y_root] = x_root
            if self.rank[x_root] == self.rank[y_root]:
                self.rank[x_root] += 1

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1
    
    if (n - m) % 2 != 0:
        print(0)
        return
    
    k = int(input[ptr])
    ptr += 1
    
    preserved = set()
    count_rows = [0] * (n + 1)
    count_cols = [0] * (m + 1)
    s = [0] * (n + 1)  # sum of -1s in row i (mod 2)
    t = [0] * (m + 1)  # sum of -1s in column j (mod 2)
    
    for _ in range(k):
        a = int(input[ptr])
        ptr += 1
        b = int(input[ptr])
        ptr += 1
        c = int(input[ptr])
        ptr += 1
        preserved.add((a, b))
        count_rows[a] += 1
        count_cols[b] += 1
        if c == -1:
            s[a] = (s[a] + 1) % 2
            t[b] = (t[b] + 1) % 2
    
    # Check rows with all cells preserved
    for i in range(1, n + 1):
        if count_rows[i] == m and s[i] % 2 != 1:
            print(0)
            return
    
    # Check columns with all cells preserved
    for j in range(1, m + 1):
        if count_cols[j] == n and t[j] % 2 != 1:
            print(0)
            return
    
    dsu = DSU(n + m)
    
    # Add edges for all unpreserved cells
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if (i, j) not in preserved:
                u = i
                v = n + j
                dsu.union(u, v)
    
    active_rows = [i for i in range(1, n + 1) if count_rows[i] < m]
    active_cols = [j for j in range(1, m + 1) if count_cols[j] < n]
    
    components = {}
    
    for i in active_rows:
        root = dsu.find(i)
        if root not in components:
            components[root] = {'rows': [], 'cols': []}
        components[root]['rows'].append(i)
    
    for j in active_cols:
        col_node = n + j
        root = dsu.find(col_node)
        if root not in components:
            components[root] = {'rows': [], 'cols': []}
        components[root]['cols'].append(j)
    
    # Check each component's sum_r and sum_c
    for comp in components.values():
        sum_r = 0
        for row in comp['rows']:
            sum_r = (sum_r + (1 - s[row])) % 2
        sum_c = 0
        for col in comp['cols']:
            sum_c = (sum_c + (1 - t[col])) % 2
        if sum_r != sum_c:
            print(0)
            return
    
    R = len(active_rows)
    C = len(active_cols)
    K = len(components)
    V = n * m - k
    exponent = V - R - C + K
    
    p = int(input[ptr])
    print(pow(2, exponent, p))

if __name__ == "__main__":
    main()

### Explanation
1. **Parity Check**: Ensures the problem is solvable by checking if rows and columns have the same parity.
2. **Preserved Cells Check**: Validates that preserved cells in fully filled rows/columns meet the product constraints.
3. **Union-Find (DSU)**: Constructs a bipartite graph of rows and columns connected by unpreserved cells, helping identify connected components.
4. **Component Consistency**: Each component's row and column sums are checked for consistency.
5. **Solution Calculation**: Uses the number of connected components and other derived values to compute the number of valid configurations modulo the given value.
