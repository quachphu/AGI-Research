Now, implementing the DSU:

We need a DSU with path compression and union by rank to ensure efficient operations.

The DSU class can be a list of parents and a list of ranks.

So, the code structure:

class DSU:

    def __init__(self, size):

        self.parent = list(range(size))

        self.rank = [1] * size

    def find(self, x):

        if self.parent[x] != x:

            self.parent[x] = self.find(self.parent[x])

        return self.parent[x]

    def union(self, x, y):

        x_root = self.find(x)

        y_root = self.find(y)

        if x_root == y_root:

            return

        if self.rank[x_root] < self.rank[y_root]:

            self.parent[x_root] = y_root

        else:

            self.parent[y_root] = x_root

            if self.rank[x_root] == self.rank[y_root]:

                self.rank[x_root] += 1

To solve this problem, we need to determine the maximum distance the ship's route can maintain from all pirate hideouts while traveling through a strait from the bottom edge (y=0) to the top edge (y=H). The solution involves using a binary search approach combined with a union-find data structure to check for horizontal and vertical barriers formed by the pirate hideouts.

### Approach
1. **Binary Search**: We perform a binary search on the possible distance values (D) to find the maximum feasible distance that allows a path from the bottom to the top edge of the strait without entering any pirate hideout's forbidden region (circle of radius D).

2. **Union-Find Data Structure**: This data structure helps efficiently check if the pirate hideouts form horizontal or vertical barriers that block all possible paths from the bottom to the top edge.

3. **Horizontal and Vertical Barrier Checks**:
   - **Horizontal Barrier Check**: Determine if there exists a connected component of pirate hideouts that spans from the left wall to the right wall of the strait.
   - **Vertical Barrier Check**: Determine if there exists a connected component of pirate hideouts that spans from the bottom edge to the top edge of the strait.

### Solution Code
```python
import sys

class DSU:
    def __init__(self, size):
        self.parent = list(range(size))
        self.rank = [1] * size

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return
        if self.rank[x_root] < self.rank[y_root]:
            self.parent[x_root] = y_root
        else:
            self.parent[y_root] = x_root
            if self.rank[x_root] == self.rank[y_root]:
                self.rank[x_root] += 1

def main():
    input = sys.stdin.read().split()
    idx = 0
    W = int(input[idx]); idx +=1
    H = int(input[idx]); idx +=1
    N = int(input[idx]); idx +=1
    pirates = []
    for _ in range(N):
        x = int(input[idx]); idx +=1
        y = int(input[idx]); idx +=1
        pirates.append( (x, y) )
    
    # Binary search
    low = 0.0
    high = 10**18  # Sufficiently large upper bound
    for _ in range(100):
        mid = (low + high) / 2.0

        # Check horizontal barrier
        horizontal = False
        dsu_h = DSU(N + 2)
        left = N
        right = N + 1
        for i in range(N):
            xi, yi = pirates[i]
            if xi < mid:  # open circle, so xi - mid < 0
                dsu_h.union(i, left)
            if xi + mid > W:  # xi + mid > W
                dsu_h.union(i, right)
        # Check all pairs
        for i in range(N):
            xi1, yi1 = pirates[i]
            for j in range(i+1, N):
                xi2, yi2 = pirates[j]
                dx = xi1 - xi2
                dy = yi1 - yi2
                if dx*dx + dy*dy < 4 * mid * mid:  # open circle, distance < 2*mid
                    dsu_h.union(i, j)
        if dsu_h.find(left) == dsu_h.find(right):
            horizontal = True

        # Check vertical barrier
        vertical = False
        dsu_v = DSU(N + 2)
        bottom = N
        top = N + 1
        for i in range(N):
            xi, yi = pirates[i]
            if yi < mid:  # open circle, yi - mid < 0
                dsu_v.union(i, bottom)
            if yi + mid > H:  # yi + mid > H
                dsu_v.union(i, top)
        # Check all pairs
        for i in range(N):
            xi1, yi1 = pirates[i]
            for j in range(i+1, N):
                xi2, yi2 = pirates[j]
                dx = xi1 - xi2
                dy = yi1 - yi2
                if dx*dx + dy*dy < 4 * mid * mid:  # open circle, distance < 2*mid
                    dsu_v.union(i, j)
        if dsu_v.find(bottom) == dsu_v.find(top):
            vertical = True

        if horizontal or vertical:
            high = mid
        else:
            low = mid

    print("{0:.10f}".format(low))

if __name__ == "__main__":
    main()
```

3. **Horizontal and Vertical Checks**: For each midpoint (D) in the binary search:
   - **Horizontal Check**: Determine if pirate hideouts form a connected component from the left to the right wall using the union-find structure.
   - **Vertical Check**: Determine if pirate hideouts form a connected component from the bottom to the top edge.
