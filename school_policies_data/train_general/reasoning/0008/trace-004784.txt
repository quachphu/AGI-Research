3. **Avoid Duplicates**: Using a set to store generated numbers ensures that each number is counted only once, even if it can be formed in multiple ways.

def main():
    import sys
    l, r = map(int, sys.stdin.readline().split())
    periodic_numbers = set()

    max_kt = 60  # since 2^60 is about 1e18

    for k in range(1, max_kt + 1):
        t_max = max_kt // k
        for t in range(2, t_max + 1):
            m = k * t
            if m > max_kt:
                continue
            denominator = (1 << k) - 1
            numerator = (1 << m) - 1
            S = numerator // denominator
            if S == 0:
                continue
            B_min = 1 << (k - 1)
            B_max = (1 << k) - 1
            if B_min > B_max:
                continue
            # Compute B_low and B_high
            B_low = max(B_min, (l + S - 1) // S)  # ceil(l/S)
            B_high = min(B_max, r // S)
            if B_low > B_high:
                continue
            # Iterate through B from B_low to B_high
            # But if the range is large, generate them directly
            for B in range(B_low, B_high + 1):
                num = B * S
                if l <= num <= r:
                    periodic_numbers.add(num)
    print(len(periodic_numbers))
