2. **Dynamic Programming with Matrix Exponentiation**:

   - **Transition Matrix**: Construct a transition matrix where each entry indicates whether moving from one column state to another is valid (i.e., does not form a 2x2 monochromatic square).
   - **Matrix Exponentiation**: Use matrix exponentiation to efficiently compute the number of valid configurations for large values of `b`.

MOD = 10**9 + 7

def main():
    import sys
    a, b = map(int, sys.stdin.readline().split())
    if a == 1:
        print(pow(2, b, MOD))
        return
    
    n = 2 ** a
    # Build the transition matrix
    trans = [[0] * n for _ in range(n)]
    for prev in range(n):
        for curr in range(n):
            allowed = True
            for i in range(a - 1):
                prev_i = (prev >> (a - 1 - i)) & 1
                prev_j = (prev >> (a - 1 - (i + 1))) & 1
                curr_i = (curr >> (a - 1 - i)) & 1
                curr_j = (curr >> (a - 1 - (i + 1))) & 1
                if prev_i == prev_j and curr_i == prev_i and curr_j == prev_j:
                    allowed = False
                    break
            trans[prev][curr] = 1 if allowed else 0
    
    if b == 1:
        print(pow(2, a, MOD))
        return
    
    # Matrix exponentiation functions
    def mat_mult(a_mat, b_mat, mod):
        size = len(a_mat)
        result = [[0] * size for _ in range(size)]
        for i in range(size):
            for k in range(size):
                if a_mat[i][k]:
                    for j in range(size):
                        result[i][j] = (result[i][j] + a_mat[i][k] * b_mat[k][j]) % mod
        return result

        size = len(matrix)
        result = [[0] * size for _ in range(size)]
        for i in range(size):
            result[i][i] = 1  # Identity matrix
        current = [row[:] for row in matrix]
        while power > 0:
            if power % 2 == 1:
                result = mat_mult(result, current, mod)
            current = mat_mult(current, current, mod)
            power //= 2
        return result

    total = 0

    print(total)
