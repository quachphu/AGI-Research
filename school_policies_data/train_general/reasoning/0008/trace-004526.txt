import sys
from heapq import heappush, heappop

class MinCostFlow:
    def __init__(self, N):
        self.N = N
        self.graph = [[] for _ in range(N)]
    
    def add_edge(self, fr, to, cap, cost):
        forward = [to, cap, cost, None]
        backward = [fr, 0, -cost, forward]
        forward[3] = backward
        self.graph[fr].append(forward)
        self.graph[to].append(backward)
    
    def flow(self, s, t, maxf):
        N = self.N
        res = 0
        h = [0] * N  # Potential
        prv_v = [0] * N
        prv_e = [None] * N
        INF = float('inf')
        while maxf >0:
            dist = [INF] * N
            dist[s] = 0
            heap = []
            heappush(heap, (0, s))
            while heap:
                d, v = heappop(heap)
                if dist[v] < d:
                    continue
                for e in self.graph[v]:
                    if e[1] >0 and dist[e[0]] > d + e[2] + h[v] - h[e[0]]:
                        dist[e[0]] = d + e[2] + h[v] - h[e[0]]
                        prv_v[e[0]] = v
                        prv_e[e[0]] = e
                        heappush(heap, (dist[e[0]], e[0]))
            if dist[t] == INF:
                return -1
            for v in range(N):
                h[v] += dist[v] if dist[v] < INF else 0
            d = maxf
            v = t
            while v != s:
                d = min(d, prv_e[v][1])
                v = prv_v[v]
            maxf -= d
            res += d * h[t]
            v = t
            while v != s:
                e = prv_e[v]
                e[1] -= d
                e[3][1] += d
                v = prv_v[v]
        return res
