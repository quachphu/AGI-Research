Given the constraints, the graph is almost a tree with a small number of additional edges, making the problem manageable by leveraging the properties of spanning trees and fundamental cycles.

### Approach
1. **Spanning Tree Construction**: Construct a spanning tree of the graph using a union-find data structure. This helps identify the non-tree edges, which are crucial for forming cycles.
2. **Fundamental Cycles Identification**: For each non-tree edge, compute the fundamental cycle formed by the edge and the unique path between its endpoints in the spanning tree.
3. **Subset Enumeration**: Enumerate all possible subsets of the non-tree edges. Each subset represents a potential combination of fundamental cycles.
4. **Cycle Validation**: For each subset, compute the combined edges (symmetric difference of edges from the fundamental cycles) and check if they form a single cycle. This involves ensuring all nodes in the combined edges have a degree of 2, the edges form a connected graph, and the number of edges equals the number of nodes.

import sys
from sys import stdin
from collections import defaultdict, deque

def main():
    n, m = map(int, stdin.readline().split())
    edges = [tuple(map(int, stdin.readline().split())) for _ in range(m)]

    parent = list(range(n + 1))
    def find(u):
        while parent[u] != u:
            parent[u] = parent[parent[u]]
            u = parent[u]
        return u

    def union(u, v):
        u_root = find(u)
        v_root = find(v)
        if u_root == v_root:
            return False
        parent[v_root] = u_root
        return True

    tree_edges = []
    non_tree_edges = []
    for u, v in edges:
        if union(u, v):
            tree_edges.append((u, v))
        else:
            non_tree_edges.append((u, v))

    tree_adj = defaultdict(list)
    for u, v in tree_edges:
        tree_adj[u].append(v)
        tree_adj[v].append(u)

    parent_info = [0] * (n + 1)
    depth = [0] * (n + 1)
    visited = [False] * (n + 1)
    q = deque()
    q.append(1)
    visited[1] = True
    while q:
        u = q.popleft()
        for v in tree_adj[u]:
            if not visited[v]:
                visited[v] = True
                parent_info[v] = u
                depth[v] = depth[u] + 1
                q.append(v)

    def get_path_edges(u, v):
        path = set()
        while u != v:
            if depth[u] > depth[v]:
                path.add((u, parent_info[u]))
                u = parent_info[u]
            else:
                path.add((v, parent_info[v]))
                v = parent_info[v]
        return path

    fundamental_cycles = []
    for u, v in non_tree_edges:
        path = get_path_edges(u, v)
        edges_in_path = set()
        for a, b in path:
            if a > b:
                a, b = b, a
            edges_in_path.add((a, b))
        fundamental_cycles.append(edges_in_path)

    c = len(non_tree_edges)
    count = 0

    for mask in range(1, 1 << c):
        combined_tree_edges = set()
        combined_non_tree = set()
        for i in range(c):
            if mask & (1 << i):
                u, v = non_tree_edges[i]
                if u > v:
                    u, v = v, u
                combined_non_tree.add((u, v))
                for edge in fundamental_cycles[i]:
                    if edge in combined_tree_edges:
                        combined_tree_edges.remove(edge)
                    else:
                        combined_tree_edges.add(edge)
        combined_edges = combined_non_tree.union(combined_tree_edges)
        if not combined_edges:
            continue

        degrees = defaultdict(int)
        nodes = set()
        for u, v in combined_edges:
            nodes.add(u)
            nodes.add(v)
            degrees[u] += 1
            degrees[v] += 1

        valid = True
        for node in nodes:
            if degrees[node] != 2:
                valid = False
                break
        if not valid:
            continue

        adj = defaultdict(list)
        for u, v in combined_edges:
            adj[u].append(v)
            adj[v].append(u)

        start_node = next(iter(nodes))
        visited_nodes = set()
        stack = [start_node]
        visited_nodes.add(start_node)
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if v not in visited_nodes:
                    visited_nodes.add(v)
                    stack.append(v)
        if visited_nodes != nodes:
            continue

        if len(combined_edges) != len(nodes):
            continue

        count += 1

    print(count)

if __name__ == "__main__":
    main()
