### Approach
The key insight is to generate the next permutation in-place using the standard algorithm for generating the next lexicographical permutation. This algorithm involves finding the rightmost element that can be swapped to form a larger permutation, swapping it with the smallest larger element from the suffix, and then reversing the suffix to get the smallest possible lexicographical order.

For each query of type 2, we apply the next permutation algorithm x times. For each query of type 1, we compute the sum of the elements in the specified range using a prefix sum array to ensure efficient range sum queries.

import sys

def main():
    n, q = map(int, sys.stdin.readline().split())
    a = list(range(1, n+1))
    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i+1] = prefix[i] + a[i]
    current = a.copy()

    def next_permutation(arr):
        # Find the largest index k such that arr[k] < arr[k+1]
        k = len(arr) - 2
        while k >= 0 and arr[k] >= arr[k+1]:
            k -= 1
        if k == -1:
            return False
        # Find the largest index l > k such that arr[k] < arr[l]
        l = len(arr) - 1
        while arr[l] <= arr[k]:
            l -= 1
        # Swap arr[k] and arr[l]
        arr[k], arr[l] = arr[l], arr[k]
        # Reverse the suffix starting from k+1
        arr[k+1:] = arr[k+1:][::-1]
        return True

    for _ in range(q):
        query = sys.stdin.readline().split()
        if not query:
            continue
        if query[0] == '1':
            l = int(query[1])
            r = int(query[2])
            print(prefix[r] - prefix[l-1])
        else:
            x = int(query[1])
            for _ in range(x):
                if not next_permutation(current):
                    break
            # Update prefix sum array
            for i in range(n):
                prefix[i+1] = prefix[i] + current[i]

    main()
