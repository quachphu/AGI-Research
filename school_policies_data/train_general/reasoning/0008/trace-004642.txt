To solve this problem, we need to compute the sizes of widgets created using a script in VasyaScript. The widgets can be of type Widget, HBox, or VBox. HBox and VBox are containers that dynamically adjust their sizes based on their children, border, and spacing parameters. The solution involves parsing the script, building a dependency graph, and computing the sizes in topological order to ensure all dependencies are resolved.

### Approach
1. **Parse the Input**: Read and parse the script to create widget objects with their respective properties.
2. **Build Dependency Graph**: Construct a graph where each container (HBox or VBox) depends on its children. This helps in determining the order of processing to ensure children are processed before their parents.
3. **Topological Sort**: Use Kahn's algorithm to perform a topological sort on the dependency graph. This ensures that each widget is processed after all its dependencies (children) have been processed.
4. **Compute Sizes**: Process each widget in topological order to compute their sizes. Base widgets (Widget) have fixed sizes, while HBox and VBox sizes are computed based on their children, border, and spacing parameters.

### Solution Code
```python
import re
from collections import deque

def main():
    n = int(input())
    widgets = {}

    for _ in range(n):
        line = input().strip()
        if line.startswith('Widget'):
            m = re.match(r'^Widget (\w+)\((\d+),(\d+)\)$', line)
            name = m.group(1)
            x = int(m.group(2))
            y = int(m.group(3))
            widgets[name] = {
                'type': 'Widget',
                'width': x,
                'height': y
            }
        elif line.startswith('HBox'):
            m = re.match(r'^HBox (\w+)$', line)
            name = m.group(1)
            widgets[name] = {
                'type': 'HBox',
                'children': [],
                'border': 0,
                'spacing': 0
            }
        elif line.startswith('VBox'):
            m = re.match(r'^VBox (\w+)$', line)
            name = m.group(1)
            widgets[name] = {
                'type': 'VBox',
                'children': [],
                'border': 0,
                'spacing': 0
            }
        elif '.pack(' in line:
            m = re.match(r'^(\w+)\.pack\((\w+)\)$', line)
            parent = m.group(1)
            child = m.group(2)
            widgets[parent]['children'].append(child)
        elif '.set_border(' in line or '.set_spacing(' in line:
            m = re.match(r'^(\w+)\.set_(border|spacing)\((\d+)\)$', line)
            name = m.group(1)
            param = m.group(2)
            value = int(m.group(3))
            widgets[name][param] = value

    # Build adjacency list and in_degree for topological sort
    adj = {name: [] for name in widgets}
    in_degree = {name: 0 for name in widgets}

    for name in widgets:
        widget = widgets[name]
        if widget['type'] in ['HBox', 'VBox']:
            for child_name in widget['children']:
                adj[child_name].append(name)
                in_degree[name] += 1

    # Kahn's algorithm
    queue = deque()
    for name in in_degree:
        if in_degree[name] == 0:
            queue.append(name)

    top_order = []
    while queue:
        u = queue.popleft()
        top_order.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)

    # Compute sizes in topological order
    for name in top_order:
        widget = widgets[name]
        if widget['type'] in ['HBox', 'VBox']:
            children = widget['children']
            border = widget['border']
            spacing = widget['spacing']
            if not children:
                widget['width'] = 0
                widget['height'] = 0
            else:
                if widget['type'] == 'HBox':
                    sum_width = 0
                    max_height = 0
                    for child_name in children:
                        child = widgets[child_name]
                        sum_width += child['width']
                        if child['height'] > max_height:
                            max_height = child['height']
                    total_spacing = spacing * (len(children) - 1)
                    total_width = 2 * border + sum_width + total_spacing
                    total_height = 2 * border + max_height
                    widget['width'] = total_width
                    widget['height'] = total_height
                else:  # VBox
                    sum_height = 0
                    max_width = 0
                    for child_name in children:
                        child = widgets[child_name]
                        sum_height += child['height']
                        if child['width'] > max_width:
                            max_width = child['width']
                    total_spacing = spacing * (len(children) - 1)
                    total_height = 2 * border + sum_height + total_spacing
                    total_width = 2 * border + max_width
                    widget['width'] = total_width
                    widget['height'] = total_height

    # Output sorted by lex order
    for name in sorted(widgets.keys()):
        widget = widgets[name]
        print(f"{name} {widget['width']} {widget['height']}")

if __name__ == "__main__":
    main()
```

3. **Topological Sort**: Using Kahn's algorithm, the widgets are processed in an order that ensures all children are processed before their parent containers.
4. **Size Calculation**: Containers compute their sizes based on their children's sizes, border, and spacing. This is done in the topological order to ensure all dependencies are resolved before computation.
