def indent_ruby_code(code)
  lines = code.split("\n")
  current_indent = 0
  result = []
  lines.each do |line|
    stripped_line = line.strip
    # Determine the indent for this line
    indent = ' ' * (current_indent * 2)
    result << "#{indent}#{stripped_line}"
    # Update current_indent for next line
    if stripped_line.start_with?('end')
      current_indent -= 1
    elsif stripped_line.start_with?('class', 'def', 'if', 'unless', 'case', 'while', 'until', 'for', 'begin', 'do', 'else', 'elsif', 'rescue', 'ensure', 'when', 'catch')
      current_indent += 1
    end
    # Ensure current_indent is not negative
    current_indent = [current_indent, 0].max
  end
  result.join("\n")
end
