To solve this problem, we need to simulate the life cycle of a numoeba, which is a tree structure of cells, each containing a numbosome value that changes every biological clock. The numoeba evolves through several steps each clock: updating numbosome values, handling cell deaths, adding bonus leaves, and changing the leader. The goal is to compute the lifetime (when it dies) and the maximum number of cells during its life for each initial seed.

### Approach

1. **Numbosome Update**: For each cell, compute the next value based on the current value using the given rules.
2. **Death Processing**: Check if cells die based on their new values, handle the removal of dead cells, and restructure the tree if necessary.
3. **Bonus Leaves**: Surviving leaves that increased their value spawn a new child.
4. **Leader Selection**: The cell with the maximum numbosome value becomes the new leader, and the tree is restructured to make it the root. A new bonus child is added if the leader is unique.

import sys
from collections import deque

class Cell:
    def __init__(self, value):
        self.value = value
        self.new_value = None
        self.old_value = None
        self.parent = None
        self.children = []

def compute_next_value(n):
    next_val = 3 * n + 1
    while next_val % 2 == 0:
        next_val //= 2
    if next_val > 12345678:
        next_val -= 12345678
    return next_val

def get_all_cells(root):
    if not root:
        return []
    cells = []
    queue = deque([root])
    while queue:
        cell = queue.popleft()
        cells.append(cell)
        for child in cell.children:
            queue.append(child)
    return cells

def process_deaths(root):
    dead_cells = set()
    queue = deque([root])
    
    if root.new_value == 1:
        return True, None
    
    while queue:
        current = queue.popleft()
        if current in dead_cells:
            continue
        if current.new_value != 1:
            for child in current.children:
                queue.append(child)
            continue
        
        parent = current.parent
        if parent is None:
            continue  # root is already handled
        
        num_children = len(current.children)
        if num_children == 1:
            child = current.children[0]
            parent.children.remove(current)
            parent.children.append(child)
            child.parent = parent
            dead_cells.add(current)
            queue.append(child)
        else:
            to_remove = []
            stack = [current]
            while stack:
                cell = stack.pop()
                if cell in dead_cells:
                    continue
                to_remove.append(cell)
                dead_cells.add(cell)
                for ch in cell.children:
                    stack.append(ch)
            parent.children.remove(current)
    
    alive_cells = []
    visited = set()
    queue = deque([root])
    while queue:
        cell = queue.popleft()
        if cell in dead_cells or cell in visited:
            continue
        visited.add(cell)
        alive_cells.append(cell)
        for child in cell.children:
            queue.append(child)
    
    if not alive_cells:
        return True, None
    return (root in dead_cells), root if root not in dead_cells else None

def reroot(new_leader):
    current = new_leader
    while current.parent is not None:
        parent = current.parent
        if current in parent.children:
            parent.children.remove(current)
        current.children.append(parent)
        parent.parent = current
        current = parent
    new_leader.parent = None
    return new_leader

def simulate(initial_value):
    root = Cell(initial_value)
    lifetime = 0
    max_cells = 1
    current_root = root
    
    while True:
        all_cells = get_all_cells(current_root)
        for cell in all_cells:
            cell.old_value = cell.value
            cell.new_value = compute_next_value(cell.value)
        
        is_dead, new_root = process_deaths(current_root)
        if is_dead:
            lifetime += 1
            break
        if new_root is None:
            lifetime += 1
            break
        
        surviving_cells = get_all_cells(new_root)
        for cell in surviving_cells:
            cell.value = cell.new_value
        
        all_cells = get_all_cells(new_root)
        to_add = []
        for cell in all_cells:
            if not cell.children and cell.value > cell.old_value:
                x = (cell.value + 1) // 2
                new_child_value = x if x % 2 == 1 else x + 1
                to_add.append((cell, new_child_value))
        
        for cell, val in to_add:
            new_child = Cell(val)
            new_child.parent = cell
            cell.children.append(new_child)
        
        all_cells = get_all_cells(new_root)
        if not all_cells:
            lifetime += 1
            break
        max_val = max(c.value for c in all_cells)
        max_cells_list = [c for c in all_cells if c.value == max_val]
        
        if len(max_cells_list) == 1:
            new_leader_candidate = max_cells_list[0]
            if new_leader_candidate != new_root:
                new_root = reroot(new_leader_candidate)
                current_root = new_root
            m = new_leader_candidate.value
            x = (m + 1) // 2
            leader_child_value = x if x % 2 == 1 else x - 1
            new_child = Cell(leader_child_value)
            new_child.parent = new_leader_candidate
            new_leader_candidate.children.append(new_child)
        
        current_count = len(get_all_cells(new_root))
