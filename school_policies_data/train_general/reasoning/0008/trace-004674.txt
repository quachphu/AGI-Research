        factor = (3 * power) // 2
        part = multiply_powers_of_2(a_div, b_div)
        return multiply(factor, part)
    elif a >= power:
        part = multiply_powers_of_2(a_div, b)
        return part * power
    else:
        part = multiply_powers_of_2(a, b_div)
        return part * power

def multiply(a, b):
    res = 0
    a_copy = a
    while a_copy:
        lsb_a = a_copy & -a_copy
        a_copy ^= lsb_a
        temp_b = b
        while temp_b:
            lsb_b = temp_b & -temp_b
            temp_b ^= lsb_b
            term = multiply_powers_of_2(lsb_a, lsb_b)
            res ^= term
    return res

def power_nimber(a, exponent):
    result = 1
    current = a
    while exponent > 0:
        if exponent % 2 == 1:
            result = multiply(result, current)
        current = multiply(current, current)
        exponent = exponent // 2
    return result

def baby_step_giant_step(gamma, beta, p):
    if gamma == 1:
        if beta == 1:
            return 0
        else:
            return None
    m = int(math.isqrt(p)) + 1
    table = {}
    current = 1
    for j in range(m):
        if current not in table:
            table[current] = j
        current = multiply(current, gamma)
    inv_gamma_m = power_nimber(gamma, p - 1 - m)
    current = beta
    for i in range(m):
        if current in table:
            return i * m + table[current]
        current = multiply(current, inv_gamma_m)
    return None
