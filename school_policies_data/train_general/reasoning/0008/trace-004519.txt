import sys
import bisect
import math

from collections import defaultdict

def normalize_line(A, B, C):
    # Compute GCD of A, B, C
    g = math.gcd(math.gcd(abs(A), abs(B)), abs(C))
    if g == 0:
        return (0, 0, 0)
    A //= g
    B //= g
    C //= g

    # Find the first non-zero coefficient
    first_non_zero = 0
    while first_non_zero < 3:
        if first_non_zero ==0 and A !=0:
            break
        elif first_non_zero ==1 and B !=0:
            break
        elif first_non_zero ==2 and C !=0:
            break
        first_non_zero +=1

    if first_non_zero >=3:
        # All zeros, but this is impossible for a valid line
        return (0, 0, 0)

    # Check the sign of the first non-zero coefficient
    if first_non_zero ==0:
        if A <0:
            A = -A
            B = -B
            C = -C
    elif first_non_zero ==1:
        if B <0:
            A = -A
            B = -B
            C = -C
    else:
        if C <0:

### Approach
1. **Preprocessing Segments**: 
   - For each segment, compute its line equation and normalize it to handle different representations of the same line.
   - Store segments in a hash map grouped by their normalized line equations. For each line, store intervals (ranges) along the line to quickly check if a point lies on any segment of that line.

2. **Checking Circle Pairs**:
   - For each pair of circles, check if they have the same radius and are non-overlapping.
   - Compute the perpendicular bisector line of the segment connecting their centers.
   - Check if there are any segments on this bisector line and if the midpoint of the segment connecting the circles' centers lies on any of these segments.

import sys
import bisect
import math
from collections import defaultdict

def normalize_line(A, B, C):
    g = math.gcd(math.gcd(abs(A), abs(B)), abs(C))
    if g == 0:
        return (0, 0, 0)
    A //= g
    B //= g
    C //= g

    first_non_zero = 0
    while first_non_zero < 3:
        if first_non_zero == 0 and A != 0:
            break
        elif first_non_zero == 1 and B != 0:
            break
        elif first_non_zero == 2 and C != 0:
            break
        first_non_zero += 1

    if first_non_zero >= 3:
        return (0, 0, 0)

    if first_non_zero == 0:
        if A < 0:
            A, B, C = -A, -B, -C
    elif first_non_zero == 1:
        if B < 0:
            A, B, C = -A, -B, -C
    else:
        if C < 0:
            A, B, C = -A, -B, -C

    return (A, B, C)
