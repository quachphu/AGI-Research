class SegmentTree:

    def __init__(self, data):

        self.n = len(data)

        self.size = 1

        while self.size < self.n:

            self.size <<= 1

        self.tree = [1] * (2 * self.size)

        for i in range(self.n):

            self.tree[self.size + i] = min(data[i], 1e18 + 1)

        for i in range(self.size - 1, 0, -1):

            self.tree[i] = min(self.tree[2*i] * self.tree[2*i+1], 1e18 + 1)

        pos += self.size

        self.tree[pos] = min(value, 1e18 + 1)

        pos >>= 1

### Approach
1. **Tree Representation and LCA Computation**: Represent the tree using adjacency lists and compute the Lowest Common Ancestor (LCA) using binary lifting to efficiently split paths between two nodes.
2. **Heavy-Light Decomposition (HLD)**: Decompose the tree into chains to allow efficient path queries. Each chain is managed using a segment tree.
3. **Segment Trees**: Each chain's segment tree stores the product of edge values, capped at \(10^{18} + 1\) to handle large values efficiently and avoid overflow.

from collections import deque

    n, m = int(input[ptr]), int(input[ptr+1])
    ptr += 2

    edges = [[] for _ in range(n+1)]  # 1-based
    edge_list = []
    for i in range(n-1):
        u = int(input[ptr])
        v = int(input[ptr+1])
        x = int(input[ptr+2])
        ptr +=3
        edges[u].append((v, x, i+1))
        edges[v].append((u, x, i+1))
        edge_list.append((u, v, x))

    parent = [0]*(n+1)
    depth = [0]*(n+1)
    edge_x = [0]*(n+1)
    original_edge_index = [0]*(n+1)
    visited = [False]*(n+1)
    q = deque([1])
    visited[1] = True
    parent[1] = 0

    while q:
        u = q.popleft()
        for v, x, idx in edges[u]:
            if not visited[v] and v != parent[u]:
                visited[v] = True
                parent[v] = u
                edge_x[v] = x
                original_edge_index[v] = idx
                depth[v] = depth[u] + 1
                q.append(v)

    LOG = ceil(log2(n)) if n > 1 else 1
    up = [[0]*(n+1) for _ in range(LOG)]
    up[0] = parent
    for k in range(1, LOG):
        for v in range(1, n+1):
            up[k][v] = up[k-1][up[k-1][v]]

    def lca(u, v):
        if depth[u] < depth[v]:
            u, v = v, u
        for k in range(LOG-1, -1, -1):
            if depth[u] - (1<<k) >= depth[v]:
                u = up[k][u]
        if u == v:
            return u
        for k in range(LOG-1, -1, -1):
            if up[k][u] != up[k][v]:
                u = up[k][u]
                v = up[k][v]
        return parent[u]

    size = [0]*(n+1)
    head = [0]*(n+1)
    pos_in_chain = [0]*(n+1)

        size[u] = 1
        for v, x, idx in edges[u]:
            if v != parent[u]:
                dfs_size(v)
                size[u] += size[v]

            chain_edges.append([])

            for v, x, idx in edges[u]:
                if v != parent[u] and v != heavy_child:
                    chain_id +=1
                    hld(v, v)

    chain_id +=1

    chain_segment_trees = []
    for i in range(chain_id):

        for j in range(len(ce)):
            st[size_st + j] = min(ce[j], 10**18 +1)
        for j in range(size_st -1, 0, -1):
            st[j] = min(st[2*j] * st[2*j+1], 10**18 +1)
        chain_segment_trees.append( (st, size_st) )

        pos >>=1
        while pos >=1:
            new_v = min(st[2*pos] * st[2*pos+1], 10**18 +1)
            if st[pos] == new_v:
                break

            return 1

                start = 0

                q = query_chain_segment(chain_idx, start, end)
                product *= q
