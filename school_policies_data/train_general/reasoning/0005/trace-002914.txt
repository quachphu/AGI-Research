class SegmentTree:
    def __init__(self, size):
        self.n = 1
        while self.n < size:
            self.n <<=1
        self.size = size
        self.tree = [float('inf')]*(2*self.n)
        self.lazy = [float('inf')]*(2*self.n)
    
    def push(self, node, l, r):
        if self.lazy[node] == float('inf'):
            return
        self.tree[node] = min(self.tree[node], self.lazy[node])
        if l != r:
            self.lazy[2*node] = min(self.lazy[2*node], self.lazy[node])
            self.lazy[2*node+1] = min(self.lazy[2*node+1], self.lazy[node])
        self.lazy[node] = float('inf')
    
    def range_min_update(self, a, b, val, node=1, l=0, r=None):
        if r is None:
            r = self.n -1
        self.push(node, l, r)
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.lazy[node] = min(self.lazy[node], val)
            self.push(node, l, r)
            return
        mid = (l +r)//2
        self.range_min_update(a, b, val, 2*node, l, mid)
        self.range_min_update(a, b, val, 2*node+1, mid+1, r)
        self.tree[node] = min(self.tree[2*node], self.tree[2*node+1])
    
    def query(self, idx, node=1, l=0, r=None):
        if r is None:
            r = self.n -1
        self.push(node, l, r)
        if l == r:
            return self.tree[node]
        mid = (l +r)//2
        if idx <= mid:
            return self.query(idx, 2*node, l, mid)
        else:
            return self.query(idx, 2*node+1, mid+1, r)

To solve this problem, we need to compute the cost of the Minimum Spanning Tree (MST) for each graph obtained by removing one edge at a time from the original graph. If the graph becomes disconnected after removing an edge, we return -1 for that edge.

### Approach
1. **Check Connectivity**: First, we check if the original graph is connected using Kruskal's algorithm to form the MST. If the MST has fewer than \(n-1\) edges, the graph is disconnected, and all outputs are -1.
2. **Construct MST**: Using Kruskal's algorithm, we construct the MST and track which edges are part of it.
3. **Tree Representation**: Convert the MST into a tree structure with parent pointers and depths.
4. **Heavy-Light Decomposition (HLD)**: Decompose the tree into chains to efficiently handle path queries and updates.
5. **Binary Lifting for LCA**: Preprocess the tree to compute the Lowest Common Ancestor (LCA) efficiently.
6. **Segment Tree for Range Updates**: Use a segment tree with lazy propagation to maintain the minimum replacement edge weights for each tree edge.
7. **Process Non-MST Edges**: For each non-MST edge, update the segment tree with the minimum weight for edges along the path between its endpoints in the MST.
8. **Query Results**: For each edge in the original graph, determine if removing it affects the MST and compute the result using the segment tree.

```python
import sys
from collections import deque

def main():
    sys.setrecursionlimit(1 << 25)
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    for i in range(m):
        a, b, w = map(int, sys.stdin.readline().split())
        a -= 1
        b -= 1
        edges.append((a, b, w, i))
    
    sorted_edges = sorted(edges, key=lambda x: x[2])
    parent = list(range(n))
    rank = [1] * n
    edge_in_mst = [False] * m
    mst_edges = []
    original_cost = 0

    def find(u):
        while parent[u] != u:
            parent[u] = parent[parent[u]]
            u = parent[u]
        return u

    for a, b, w, idx in sorted_edges:
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            if rank[root_a] < rank[root_b]:
                parent[root_a] = root_b
                rank[root_b] += rank[root_a]
            else:
                parent[root_b] = root_a
                rank[root_a] += rank[root_b]
            mst_edges.append((a, b, w, idx))
            edge_in_mst[idx] = True
            original_cost += w
            if len(mst_edges) == n - 1:
                break

    if len(mst_edges) != n - 1:
        print('\n'.join(['-1'] * m))
        return

    adj = [[] for _ in range(n)]
    for a, b, w, idx in mst_edges:
        adj[a].append((b, w))
        adj[b].append((a, w))

    root = 0
    tree_parent = [-1] * n
    depth = [0] * n
    children = [[] for _ in range(n)]
    q = deque()
    q.append(root)
    tree_parent[root] = -1

    while q:
        u = q.popleft()
        for v, w in adj[u]:
            if tree_parent[v] == -1 and v != tree_parent[u]:
                tree_parent[v] = u
                depth[v] = depth[u] + 1
                children[u].append(v)
                q.append(v)

    LOG = 20
    up = [[-1] * n for _ in range(LOG)]
    up[0] = tree_parent.copy()
    for k in range(1, LOG):
        for v in range(n):
            if up[k-1][v] != -1:
                up[k][v] = up[k-1][up[k-1][v]]

    def lca(u, v):
        if depth[u] < depth[v]:
            u, v = v, u
        for k in range(LOG-1, -1, -1):
            if depth[u] - (1 << k) >= depth[v]:
                u = up[k][u]
        if u == v:
            return u
        for k in range(LOG-1, -1, -1):
            if up[k][u] != -1 and up[k][u] != up[k][v]:
                u = up[k][u]
                v = up[k][v]
        return up[0][u]

    size = [1] * n
    stack = [(root, False)]
    while stack:
        node, visited = stack.pop()
        if visited:
            for child in children[node]:
                size[node] += size[child]
            children[node].sort(key=lambda x: -size[x])
        else:
            stack.append((node, True))
            for child in reversed(children[node]):
                stack.append((child, False))

    chain = [-1] * n
    chain_head = [-1] * (n + 1)
    chain_pos = [0] * n
    chain_idx = 0
    q = deque()
    q.append(root)
    current_pos = 0
    edge_pos = [-1] * n

    while q:
        node = q.popleft()
        if chain[node] == -1:
            chain_idx += 1
            current_node = node
            chain_head[chain_idx] = current_node
            pos_in_chain = 0
            while True:
                chain[current_node] = chain_idx
                chain_pos[current_node] = pos_in_chain
                if current_node != root:
                    edge_pos[current_node] = current_pos
                    current_pos += 1
                if not children[current_node]:
                    break
                next_node = children[current_node][0]
                for child in children[current_node][1:]:
                    q.append(child)
                current_node = next_node
                pos_in_chain += 1

    class SegmentTree:
        def __init__(self, size):
            self.n = 1
            while self.n < size:
                self.n <<= 1
            self.size = size
            self.tree = [float('inf')] * (2 * self.n)
            self.lazy = [float('inf')] * (2 * self.n)
        
        def push(self, node, l, r):
            if self.lazy[node] == float('inf'):
                return
            self.tree[node] = min(self.tree[node], self.lazy[node])
            if l != r:
                left = 2 * node
                right = 2 * node + 1
                self.lazy[left] = min(self.lazy[left], self.lazy[node])
                self.lazy[right] = min(self.lazy[right], self.lazy[node])
            self.lazy[node] = float('inf')
        
        def range_min_update(self, a, b, val, node=1, l=0, r=None):
            if r is None:
                r = self.n - 1
            self.push(node, l, r)
            if a > r or b < l:
                return
            if a <= l and r <= b:
                self.lazy[node] = min(self.lazy[node], val)
                self.push(node, l, r)
                return
            mid = (l + r) // 2
            self.range_min_update(a, b, val, 2 * node, l, mid)
            self.range_min_update(a, b, val, 2 * node + 1, mid + 1, r)
            self.tree[node] = min(self.tree[2 * node], self.tree[2 * node + 1])
        
        def query(self, idx):
            node = 1
            l = 0
            r = self.n - 1
            while True:
                self.push(node, l, r)
                if l == r:
                    return self.tree[node]
                mid = (l + r) // 2
                if idx <= mid:
                    node = 2 * node
                    r = mid
                else:
                    node = 2 * node + 1
                    l = mid + 1

    st_size = current_pos
    st = SegmentTree(st_size)

    edge_info = [None] * m
    for a, b, w, idx in mst_edges:
        if tree_parent[b] == a:
            child_node = b
        else:
            child_node = a
        edge_info[idx] = (child_node, edge_pos[child_node], w)

    non_mst_edges = []
    for i in range(m):
        if not edge_in_mst[i]:
            a, b, w, idx = edges[i]
            non_mst_edges.append((a, b, w))

    def update_path(u, v, w):
        ancestor = lca(u, v)
        while u != ancestor:
            if chain[u] == chain[ancestor]:
                pos_u = edge_pos[u]
                if pos_u == -1:
                    u = tree_parent[u]
                    continue
                st.range_min_update(pos_u, pos_u, w)
                u = tree_parent[u]
            else:
                head = chain_head[chain[u]]
                start = edge_pos[head]
                end = edge_pos[u]
                if start <= end:
                    st.range_min_update(start, end, w)
                u = tree_parent[head]
        while v != ancestor:
            if chain[v] == chain[ancestor]:
                pos_v = edge_pos[v]
                if pos_v == -1:
                    v = tree_parent[v]
                    continue
                st.range_min_update(pos_v, pos_v, w)
                v = tree_parent[v]
            else:
                head = chain_head[chain[v]]
                start = edge_pos[head]
                end = edge_pos[v]
                if start <= end:
                    st.range_min_update(start, end, w)
                v = tree_parent[head]

    for a, b, w in non_mst_edges:
        update_path(a, b, w)

    result = []
    for i in range(m):
        if not edge_in_mst[i]:
            result.append(str(original_cost))
        else:
            child_node, pos, w = edge_info[i]
            replacement = st.query(pos)
            if replacement == float('inf'):
                result.append("-1")
            else:
                new_cost = original_cost - w + replacement
                result.append(str(new_cost))
    
    print('\n'.join(result))

if __name__ == "__main__":
    main()
```
