class FenwickTree:

    def __init__(self, size):

        self.n = size

        self.tree = [0] * (self.n + 1)  # 1-based indexing

    def update(self, idx, delta):

        # idx is 1-based.

        while idx <= self.n:

            self.tree[idx] += delta

            idx += idx & -idx

    def query(self, idx):

        # sum from 1 to idx (1-based)

        res = 0

        while idx > 0:

            res += self.tree[idx]

            idx -= idx & -idx

        return res

2. **Dynamic Programming with Fenwick Trees**: Use Fenwick Trees to maintain counts of subsequences of various lengths efficiently. For each element, track the number of increasing subsequences ending at that element for each possible length up to \( k+1 \).
