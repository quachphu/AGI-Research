import sys
from collections import deque

def find_centers(adj):
    n = len(adj)
    if n == 1:
        return [0]
    degrees = [len(neighbors) for neighbors in adj]
    leaves = [i for i in range(n) if degrees[i] == 1]
    layer = leaves
    remaining = n
    while remaining > 2:
        next_layer = []
        for u in layer:
            for v in adj[u]:
                degrees[v] -= 1
                if degrees[v] == 1:
                    next_layer.append(v)
        remaining -= len(layer)
        layer = next_layer
    return layer

def root_tree(adj, root):
    n = len(adj)
    parent = [-1] * n
    children = [[] for _ in range(n)]
    queue = deque([root])
    parent[root] = -1
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if parent[v] == -1 and v != parent[u]:
                parent[v] = u
                children[u].append(v)
                queue.append(v)
    return children

def get_post_order(children, root):
    stack = [(root, False)]
    post_order = []
    while stack:
        node, visited = stack.pop()
        if visited:
            post_order.append(node)
        else:
            stack.append( (node, True) )
            for child in reversed(children[node]):
                stack.append( (child, False) )
    return post_order

def compute_canonical_form(adj, global_cache):
    centers = find_centers(adj)
    min_label = None

        children = root_tree(adj, center)
        post_order = get_post_order(children, center)

                labels[node] = global_cache[key]
        current_label = labels[center]
        if min_label is None or current_label < min_label:
            min_label = current_label
    return min_label

2. **Component Extraction**: Split G1 into its connected components using BFS.
3. **Tree Isomorphism Check**:
   - **Canonical Form Calculation**: For each tree (both G2 and components of G1), compute a canonical form using a combination of BFS for center detection, tree rooting, and post-order traversal with memoization to ensure isomorphic trees have the same canonical form.
   - **Comparison**: Compare the canonical form of each component of G1 with that of G2 to count matches.

import sys
from collections import deque

def find_centers(adj):
    n = len(adj)
    if n == 1:
        return [0]
    degrees = [len(neighbors) for neighbors in adj]
    leaves = [i for i in range(n) if degrees[i] == 1]
    layer = leaves
    remaining = n
    while remaining > 2:
        next_layer = []
        for u in layer:
            for v in adj[u]:
                degrees[v] -= 1
                if degrees[v] == 1:
                    next_layer.append(v)
        remaining -= len(layer)
        layer = next_layer
    return layer

def root_tree(adj, root):
    n = len(adj)
    parent = [-1] * n
    children = [[] for _ in range(n)]
    queue = deque([root])
    parent[root] = -1
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if parent[v] == -1 and v != parent[u]:
                parent[v] = u
                children[u].append(v)
                queue.append(v)
    return children

def get_post_order(children, root):
    stack = [(root, False)]
    post_order = []
    while stack:
        node, visited = stack.pop()
        if visited:
            post_order.append(node)
        else:
            stack.append((node, True))
            for child in reversed(children[node]):
                stack.append((child, False))
    return post_order

def compute_canonical_form(adj, global_cache):
    centers = find_centers(adj)
    min_label = None
    for center in centers:
        children = root_tree(adj, center)
        post_order = get_post_order(children, center)
        labels = {}
        for node in post_order:
            if not children[node]:
                labels[node] = 0
            else:
                child_labels = [labels[child] for child in children[node]]
                child_labels.sort()
                key = tuple(child_labels)
                if key not in global_cache:
                    global_cache[key] = len(global_cache)
                labels[node] = global_cache[key]
        current_label = labels[center]
        if (min_label is None) or (current_label < min_label):
            min_label = current_label
    return min_label

def main():
    input = sys.stdin.read().split()
    ptr = 0
    N1 = int(input[ptr])
    ptr += 1
    M1 = int(input[ptr])
    ptr += 1
    adj_g1 = [[] for _ in range(N1 + 1)]
    for _ in range(M1):
        u = int(input[ptr])
        ptr += 1
        v = int(input[ptr])
        ptr += 1
        adj_g1[u].append(v)
        adj_g1[v].append(u)
    N2 = int(input[ptr])
    ptr += 1
    adj_g2 = [[] for _ in range(N2 + 1)]
    for _ in range(N2 - 1):
        w = int(input[ptr])
        ptr += 1
        x = int(input[ptr])
        ptr += 1
        adj_g2[w].append(x)
        adj_g2[x].append(w)
    original_g2_nodes = list(range(1, N2 + 1))
    remap_g2 = {node: idx for idx, node in enumerate(original_g2_nodes)}
    adj_g2_0 = [[] for _ in range(N2)]
    for w in range(1, N2 + 1):
        for x in adj_g2[w]:
            if x > w:
                u = remap_g2[w]
                v = remap_g2[x]
                adj_g2_0[u].append(v)
                adj_g2_0[v].append(u)
    global_cache = {}
    g2_label = compute_canonical_form(adj_g2_0, global_cache)
    visited = [False] * (N1 + 1)
    count = 0
    for start in range(1, N1 + 1):
        if not visited[start]:
            component = []
            queue = deque([start])
            visited[start] = True
            while queue:
                u = queue.popleft()
                component.append(u)
                for v in adj_g1[u]:
                    if not visited[v]:
                        visited[v] = True
                        queue.append(v)
            K = len(component)
            if K != N2:
                continue
            component_set = set(component)
            remap = {node: idx for idx, node in enumerate(sorted(component))}
            adj = [[] for _ in range(K)]
            for u in component:
                u_idx = remap[u]
                for v in adj_g1[u]:
                    if v in component_set:
                        v_idx = remap[v]
                        adj[u_idx].append(v_idx)
            component_label = compute_canonical_form(adj, global_cache)
            if component_label == g2_label:
                count += 1
    print(count)

if __name__ == "__main__":
