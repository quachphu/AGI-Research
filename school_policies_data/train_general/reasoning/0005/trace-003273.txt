To solve this problem, we need to determine the number of "lucky" bus tickets in Berland. A ticket is considered lucky if the sum of the first half of its digits is equal to the sum of the second half. Given the constraints, a direct approach is infeasible, so we use a dynamic programming approach combined with polynomial exponentiation using Number Theoretic Transform (NTT) for efficient computation.

### Approach
1. **Dynamic Programming with Polynomials**: The key insight is to model the problem using generating functions. Each digit allowed in the ticket number can be represented as a polynomial term. The sum of these terms raised to the power of half the ticket length (m) gives the generating function for all possible sums of m digits.
2. **Fast Polynomial Exponentiation**: Using NTT, we efficiently compute the polynomial raised to the power m, which represents the number of ways to form each possible sum with m digits.
3. **Sum of Squares**: The result is the sum of the squares of the coefficients of the resulting polynomial, which gives the number of valid lucky tickets.

### Solution Code

import sys
MOD = 998244353
ROOT = 3

def ntt(a, inv=False):
    n = len(a)
    rev = [0] * n
    for i in range(n):
        rev[i] = rev[i >> 1] >> 1
        if i & 1:
            rev[i] |= n >> 1
        if i < rev[i]:
            a[i], a[rev[i]] = a[rev[i]], a[i]
    logn = (n).bit_length() - 1
    for i in range(logn):
        m = 1 << i
        m2 = m << 1
        w = pow(ROOT, (MOD - 1) // m2, MOD)
        if inv:
            w = pow(w, MOD - 2, MOD)
        for j in range(0, n, m2):
            wk = 1
            for k in range(m):
                x = a[j + k]
                y = a[j + k + m] * wk
                a[j + k] = (x + y) % MOD
                a[j + k + m] = (x - y) % MOD
                wk = wk * w % MOD
    if inv:
        inv_n = pow(n, MOD - 2, MOD)
        for i in range(n):
            a[i] = a[i] * inv_n % MOD
    return a

def multiply(a, b):
    len_a = len(a)
    len_b = len(b)
    max_len = len_a + len_b - 1
    size = 1
    while size < max_len:
        size <<= 1
    a_padded = a + [0] * (size - len_a)
    b_padded = b + [0] * (size - len_b)
    a_ntt = ntt(a_padded.copy())
    b_ntt = ntt(b_padded.copy())
    c_ntt = [(x * y) % MOD for x, y in zip(a_ntt, b_ntt)]
    c = ntt(c_ntt, inv=True)
    c = c[:max_len]
    for i in range(max_len):
        c[i] = c[i] % MOD
    return c

def poly_pow(p, exponent):
    result = [1]
    current_p = p.copy()
    while exponent > 0:
        if exponent % 2:
            result = multiply(result, current_p)
        current_p = multiply(current_p, current_p)
        exponent = exponent // 2
    return result

def main():
    n, k = map(int, sys.stdin.readline().split())
    digits = list(map(int, sys.stdin.readline().split()))
    m = n // 2
    max_d = max(digits) if digits else 0
    initial_p = [0] * (10)
    for d in digits:
        initial_p[d] += 1
    poly = initial_p
    res_poly = poly_pow(poly, m)
    total = 0
    for coeff in res_poly:
        total = (total + coeff * coeff) % MOD
    print(total % MOD)

if __name__ == '__main__':
    main()
