To solve this problem, we need to determine if there exists a path from the start coordinates to the goal coordinates on a board using blocks of the same color. The blocks can be placed either horizontally or vertically and must form a connected path without overlapping.

### Approach
1. **Input Parsing**: Read the board dimensions, start and goal coordinates, and the list of blocks.
2. **Grid Construction**: Create a grid to track which block covers each cell on the board.
3. **Block Coverage**: For each block, mark the cells it covers in the grid.
4. **Validation**: Check if the start and goal coordinates are covered by blocks and if those blocks are of the same color.
5. **Graph Construction**: Build a graph where nodes represent blocks and edges exist between blocks of the same color that are adjacent.
6. **Path Finding**: Use BFS to check if there is a connected path from the start block to the goal block in the graph.

### Solution Code
```python
import sys
from collections import deque

def main():
    while True:
        w, h = map(int, sys.stdin.readline().split())
        if w == 0 and h == 0:
            break
        xs, ys = map(int, sys.stdin.readline().split())
        xg, yg = map(int, sys.stdin.readline().split())
        n = int(sys.stdin.readline())
        blocks = []
        grid = [[-1 for _ in range(h + 2)] for __ in range(w + 2)]  # 1-based indexing

        for block_idx in range(n):
            c, d, xi, yi = map(int, sys.stdin.readline().split())
            x_coords = []
            y_coords = []
            if d == 0:
                x_start = xi
                x_end = xi + 3
                y_start = yi
                y_end = yi + 1
                x_coords = list(range(x_start, x_end + 1))
                y_coords = list(range(y_start, y_end + 1))
            else:
                x_start = xi
                x_end = xi + 1
                y_start = yi
                y_end = yi + 3
                x_coords = list(range(x_start, x_end + 1))
                y_coords = list(range(y_start, y_end + 1))
            for x in x_coords:
                for y in y_coords:
                    grid[x][y] = block_idx
            blocks.append((c, d, xi, yi))
        
        start_block = grid[xs][ys]
        goal_block = grid[xg][yg]

        if start_block == -1 or goal_block == -1:
            print("NG")
            continue
        
        if blocks[start_block][0] != blocks[goal_block][0]:
            print("NG")
            continue
        
        if start_block == goal_block:
            print("OK")
            continue
        
        edges = set()
        graph = {}
        for x in range(1, w + 1):
            for y in range(1, h + 1):
                current_block = grid[x][y]
                if current_block == -1:
                    continue
                current_color = blocks[current_block][0]
                for dx, dy in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
                    nx = x + dx
                    ny = y + dy
                    if 1 <= nx <= w and 1 <= ny <= h:
                        neighbor_block = grid[nx][ny]
                        if neighbor_block == -1 or neighbor_block == current_block:
                            continue
                        neighbor_color = blocks[neighbor_block][0]
                        if neighbor_color == current_color:
                            a = current_block
                            b = neighbor_block
                            if a > b:
                                a, b = b, a
                            if (a, b) not in edges:
                                edges.add((a, b))
                                if a not in graph:
                                    graph[a] = []
                                graph[a].append(b)
                                if b not in graph:
                                    graph[b] = []
                                graph[b].append(a)
        
        visited = set()
        queue = deque([start_block])
        visited.add(start_block)
        found = False
        while queue:
            node = queue.popleft()
            if node == goal_block:
                found = True
                break
            if node in graph:
                for neighbor in graph[node]:
                    if neighbor not in visited:
                        visited.add(neighbor)
                        queue.append(neighbor)
        
        print("OK" if found else "NG")

if __name__ == "__main__":
    main()
```

3. **Validation**: Checks if the start and goal cells are covered by blocks and if those blocks are the same color.
4. **Graph Construction**: Adjacent blocks of the same color are connected in a graph using a set to avoid duplicate edges.
5. **Path Finding**: BFS is used to determine if there is a path from the start block to the goal block in the graph, ensuring connectivity through same-colored blocks.
