4. **Check Validity**: For each permutation, check if the sums of rows, columns, and diagonals modulo the value in cell J are congruent.

### Solution Code
```python
import sys
from itertools import permutations

def main():
    for line in sys.stdin:
        parts = list(map(int, line.strip().split()))
        if parts == [-1] * 10:
            break
        
        # Check for duplicates in non-zero cells
        filled = [x for x in parts if x != 0]
        if len(filled) != len(set(filled)):
            print(0)
            continue
        
        # Determine missing numbers and check zero count
        present = set(filled)
        missing = [x for x in range(1, 11) if x not in present]
        zero_count = parts.count(0)
        if zero_count != len(missing):
            print(0)
            continue
        
        # Get positions of zeros
        zero_positions = [i for i, val in enumerate(parts) if val == 0]
        
        count = 0
        for perm in permutations(missing):
            # Create a filled square
            square = parts.copy()
            for idx, pos in enumerate(zero_positions):
                square[pos] = perm[idx]
            
            # Extract J and check modulus
            J = square[9]
            sums = [
                # Rows
                square[0] + square[1] + square[2],
                square[3] + square[4] + square[5],
                square[6] + square[7] + square[8],
                # Columns
                square[0] + square[3] + square[6],
                square[1] + square[4] + square[7],
                square[2] + square[5] + square[8],
                # Diagonals
                square[0] + square[4] + square[8],
                square[2] + square[4] + square[6]
            ]
            
            # Calculate all mod values
            mod = sums[0] % J
            valid = True
            for s in sums[1:]:
                if s % J != mod:
                    valid = False
                    break
            if valid:
                count += 1
        
        print(count)

if __name__ == "__main__":
    main()
```
