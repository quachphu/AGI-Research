class SegmentTree:

    def __init__(self, data):

        self.n = len(data) -1  # data is 1-based

        self.size = 1

        while self.size < self.n:

            self.size <<=1

        self.sum_tree = [0]*(2*self.size)

        self.min_val = [0]*(2*self.size)

        self.max_val = [0]*(2*self.size)

        self.assign_lazy = [None]*(2*self.size)

        # Fill the leaves

        for i in range(1, self.n+1):

            self.sum_tree[self.size + i -1] = data[i]

            self.min_val[self.size + i -1] = data[i]

            self.max_val[self.size + i -1] = data[i]

        # Build the internal nodes

        for i in range(self.size-1, 0, -1):

            self.sum_tree[i] = self.sum_tree[2*i] + self.sum_tree[2*i+1]

            self.min_val[i] = min(self.min_val[2*i], self.min_val[2*i+1])

            self.max_val[i] = max(self.max_val[2*i], self.max_val[2*i+1])

    def push(self, node, l, r):

        if self.assign_lazy[node] is not None:

            mid = (l + r) //2

            left = 2*node

            right = 2*node+1

            # Assign to left child

            self.sum_tree[left] = self.assign_lazy[node] * (mid -l +1)

            self.min_val[left] = self.assign_lazy[node]

            self.max_val[left] = self.assign_lazy[node]

            self.assign_lazy[left] = self.assign_lazy[node]

            # Assign to right child

            self.sum_tree[right] = self.assign_lazy[node] * (r - mid)

            self.min_val[right] = self.assign_lazy[node]

            self.max_val[right] = self.assign_lazy[node]

            self.assign_lazy[right] = self.assign_lazy[node]

            # Clear the lazy value for current node

            self.assign_lazy[node] = None

    def range_assign(self, a, b, val, node, l, r):

        if a > r or b < l:

            return

        if a <= l and r <= b:

            self.sum_tree[node] = val * (r -l +1)

            self.min_val[node] = val

            self.max_val[node] = val

            self.assign_lazy[node] = val

            return

        self.push(node, l, r)

        mid = (l + r) //2

        self.range_assign(a, b, val, 2*node, l, mid)

        self.range_assign(a, b, val, 2*node+1, mid+1, r)

        self.sum_tree[node] = self.sum_tree[2*node] + self.sum_tree[2*node+1]

        self.min_val[node] = min(self.min_val[2*node], self.min_val[2*node+1])

        self.max_val[node] = max(self.max_val[2*node], self.max_val[2*node+1])

    def assign(self, a, b, val):

        self.range_assign(a, b, val, 1, 1, self.size)

    def point_add(self, pos, val):

        # Update the leaf node

        node = 1

        l =1

        r = self.size

        stack = []

        while l != r:

            stack.append(node)

            mid = (l +r) //2

            if pos <= mid:

                r = mid

                node = 2*node

            else:

                l = mid+1

                node = 2*node+1

        # Now at leaf node

        self.sum_tree[node] += val

        self.min_val[node] += val

        self.max_val[node] += val

        # Bubble up

        for node in reversed(stack):

To solve this problem, we need to efficiently handle two types of queries on an array `a` after applying certain constraints using another array `k`. The first type of query updates elements in `a` and propagates changes to maintain a non-decreasing property, while the second type calculates the sum of a subarray. The solution must handle up to 100,000 operations efficiently.

### Approach
1. **Initial Setup**: We start by converting the array `a` into a base array and a delta array. The base array is constructed using the initial values of `a` and the constraints from `k`, ensuring that each element `a_i` is at least the sum of previous elements and their corresponding `k` values. The delta array captures the difference between the actual values in `a` and the computed base values.

2. **Segment Tree**: We use a segment tree to manage the delta array efficiently. The segment tree supports:
   - Point updates to increment a specific element.
   - Range assignments to propagate constraints.
   - Sum queries to compute the sum of any subarray.
   - Binary search to find the first position where a value meets a threshold, ensuring the non-decreasing property.

3. **Handling Queries**:
   - **Update Query**: When updating an element, we adjust the delta array and propagate changes to maintain the non-decreasing property using binary search and range assignments.
   - **Sum Query**: Compute the sum using precomputed base array values and the segment tree for delta values.

import sys

class SegmentTree:
    def __init__(self, data):
        self.n = len(data) - 1  # data is 1-based
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.sum_tree = [0] * (2 * self.size)
        self.min_val = [0] * (2 * self.size)
        self.max_val = [0] * (2 * self.size)
        self.assign_lazy = [None] * (2 * self.size)

        for i in range(1, self.n + 1):
            pos = self.size + i - 1
            self.sum_tree[pos] = data[i]
            self.min_val[pos] = data[i]
            self.max_val[pos] = data[i]
        for i in range(self.size - 1, 0, -1):
            self.sum_tree[i] = self.sum_tree[2 * i] + self.sum_tree[2 * i + 1]
            self.min_val[i] = min(self.min_val[2 * i], self.min_val[2 * i + 1])
            self.max_val[i] = max(self.max_val[2 * i], self.max_val[2 * i + 1])

    def push(self, node, l, r):
        if self.assign_lazy[node] is not None:
            mid = (l + r) // 2
            left = 2 * node
            right_node = 2 * node + 1

            self.sum_tree[left] = self.assign_lazy[node] * (mid - l + 1)
            self.min_val[left] = self.assign_lazy[node]
            self.max_val[left] = self.assign_lazy[node]
            self.assign_lazy[left] = self.assign_lazy[node]

            self.sum_tree[right_node] = self.assign_lazy[node] * (r - mid)
            self.min_val[right_node] = self.assign_lazy[node]
            self.max_val[right_node] = self.assign_lazy[node]
            self.assign_lazy[right_node] = self.assign_lazy[node]

            self.assign_lazy[node] = None

    def range_assign(self, a, b, val, node, l, r):
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.sum_tree[node] = val * (r - l + 1)
            self.min_val[node] = val
            self.max_val[node] = val
            self.assign_lazy[node] = val
            return
        self.push(node, l, r)
        mid = (l + r) // 2
        self.range_assign(a, b, val, 2 * node, l, mid)
        self.range_assign(a, b, val, 2 * node + 1, mid + 1, r)
        self.sum_tree[node] = self.sum_tree[2 * node] + self.sum_tree[2 * node + 1]
        self.min_val[node] = min(self.min_val[2 * node], self.min_val[2 * node + 1])
        self.max_val[node] = max(self.max_val[2 * node], self.max_val[2 * node + 1])

    def assign(self, a, b, val):
        self.range_assign(a, b, val, 1, 1, self.size)

    def point_add(self, pos, val):
        node = 1
        l = 1
        r = self.size
        stack = []
        while l != r:
            stack.append(node)
            self.push(node, l, r)
            mid = (l + r) // 2
            if pos <= mid:
                r = mid
                node = 2 * node
            else:
                l = mid + 1
                node = 2 * node + 1
        self.sum_tree[node] += val
        self.min_val[node] += val
        self.max_val[node] += val
        for n in reversed(stack):
            left = 2 * n
            right = 2 * n + 1
            self.sum_tree[n] = self.sum_tree[left] + self.sum_tree[right]
            self.min_val[n] = min(self.min_val[left], self.min_val[right])
            self.max_val[n] = max(self.max_val[left], self.max_val[right])

    def query_sum(self, a, b):
        return self._range_sum(a, b, 1, 1, self.size)

    def _range_sum(self, a, b, node, l, r):
        if a > r or b < l:
            return 0
        if a <= l and r <= b:
            return self.sum_tree[node]
        self.push(node, l, r)
        mid = (l + r) // 2
        return self._range_sum(a, b, 2 * node, l, mid) + self._range_sum(a, b, 2 * node + 1, mid + 1, r)

    def find_first_ge(self, start, target):
        res = self._find_first_ge(1, 1, self.size, start, target)
        return res if res is not None else (self.n + 1)

    def _find_first_ge(self, node, node_l, node_r, start, target):
        if node_r < start:
            return None
        if self.max_val[node] < target:
            return None
        if node_l == node_r:
            return node_l if node_l >= start else None
        self.push(node, node_l, node_r)
        mid = (node_l + node_r) // 2
        left_res = self._find_first_ge(2 * node, node_l, mid, start, target)
        if left_res is not None:
            return left_res
        return self._find_first_ge(2 * node + 1, mid + 1, node_r, start, target)
