For each round, transitions to new states are computed by considering all possible card pairs that can be played, updating the scores and remaining cards accordingly.

3. **State Representation**: Each state is represented by tuples of remaining cards (sorted for consistency) and the current scores. This ensures that states with the same remaining cards and scores are merged, reducing redundant calculations.

4. **Transition Probabilities**: For each state, compute the probability of transitioning to a new state by considering all possible card pairs. The probability of each transition is determined by the uniform random selection of cards from each player's remaining cards.

5. **Final Calculation**: After all rounds, sum the probabilities of all states where Gates' score is higher than Jackie's and vice versa to determine the winning probabilities.

import sys
from collections import defaultdict

def main():
    input = sys.stdin.read().split()
    idx = 0
    T = int(input[idx])
    idx += 1
    for _ in range(T):
        gates_hand = list(map(int, input[idx:idx+9]))
        idx +=9
        jackie_hand = list(map(int, input[idx:idx+9]))
        idx +=9
        
        gates_initial = tuple(sorted(gates_hand))
        jackie_initial = tuple(sorted(jackie_hand))
        
        current_dp = defaultdict(float)
        current_dp[(gates_initial, jackie_initial, 0, 0)] = 1.0
        
        for _ in range(9):
            next_dp = defaultdict(float)
            for state, prob in current_dp.items():
                g_rem, j_rem, g_sc, j_sc = state
                g_len = len(g_rem)
                j_len = len(j_rem)
                for g in g_rem:
                    new_g_rem = list(g_rem)
                    new_g_rem.remove(g)
                    new_g_rem_tuple = tuple(sorted(new_g_rem))
                    for j in j_rem:
                        if g > j:
                            new_g = g_sc + g + j
                            new_j = j_sc
                        else:
                            new_g = g_sc
                            new_j = j_sc + g + j
                        new_j_rem = list(j_rem)
                        new_j_rem.remove(j)
                        new_j_rem_tuple = tuple(sorted(new_j_rem))
                        trans_prob = prob * (1.0 / g_len) * (1.0 / j_len)
                        next_dp[(new_g_rem_tuple, new_j_rem_tuple, new_g, new_j)] += trans_prob
            current_dp = next_dp
        
        gates_win = 0.0
        jackie_win = 0.0
        for state, prob in current_dp.items():
            g_sc = state[2]
            j_sc = state[3]
            if g_sc > j_sc:
                gates_win += prob
            elif j_sc > g_sc:
                jackie_win += prob
        
        print("{0:.5f} {1:.5f}".format(gates_win, jackie_win))

if __name__ == "__main__":
    main()
