The graph's structure under the chosen edges must form cycles that cover all possible starting nodes.

### Approach
1. **Graph Representation and Edge Sorting**: Read the graph and sort the edges for each node by their weights. This helps in efficiently determining the c_i-th smallest edge for each node.
2. **Valid c_i Determination**: For each out-degree i, determine all valid c_i values such that the targets (nodes pointed to by the c_i-th edges) are unique among nodes with the same out-degree.
3. **Compatibility Check**: Precompute compatibility matrices to check if the target sets for different out-degrees are disjoint. This helps in ensuring that the combined targets form a permutation of all nodes.
4. **Backtracking with Pruning**: Use backtracking to generate all valid tuples of c_i values, checking compatibility at each step to ensure the combined targets form a valid permutation.

import sys
from collections import defaultdict

def main():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = [[] for _ in range(n + 1)]  # 1-based indexing

    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges[u].append((w, v))
    
    # Sort each node's edges by weight
    for u in range(1, n + 1):
        edges[u].sort()
    
    # Group nodes by their out-degree
    nodes_by_degree = defaultdict(list)
    for u in range(1, n + 1):
        d = len(edges[u])
        nodes_by_degree[d].append(u)
    
    # Precompute valid_c for each i and their target sets
    valid_c = defaultdict(list)
    target_sets = defaultdict(dict)  # target_sets[i][c] = set of targets

    for i in range(1, k + 1):
        nodes = nodes_by_degree.get(i, [])
        if nodes:
            # Check each possible c_i (1..i)
            for c in range(1, i + 1):
                targets = []
                for u in nodes:
                    # edges are sorted, so c-th is index c-1
                    v = edges[u][c - 1][1]
                    targets.append(v)
                # Check uniqueness
                if len(set(targets)) == len(targets):
                    valid_c[i].append(c)
                    target_sets[i][c] = set(targets)
        else:
            # If no nodes have out-degree i, all c_i are valid
            valid_c[i] = list(range(1, i + 1))
    
    # Precompute compatibility matrix for each pair i < j
    compatibility = defaultdict(lambda: defaultdict(dict))  # compatibility[i][j][ci][cj] = True/False

    for i in range(1, k + 1):
        for j in range(i + 1, k + 1):
            # Check if either i or j has no nodes
            has_i = len(nodes_by_degree.get(i, [])) > 0
            has_j = len(nodes_by_degree.get(j, [])) > 0
            
            if not has_i or not has_j:
                # All pairs are compatible
                valid_ci = valid_c[i]
                valid_cj = valid_c[j]
                compatibility[i][j] = defaultdict(dict)
                for ci in valid_ci:
                    compatibility[i][j][ci] = {}
                    for cj in valid_cj:
                        compatibility[i][j][ci][cj] = True
            else:
                # Both have nodes, compute compatibility
                valid_ci = valid_c[i]
                valid_cj = valid_c[j]
                compatibility[i][j] = defaultdict(dict)
                for ci in valid_ci:
                    si = target_sets[i][ci]
                    compatibility[i][j][ci] = {}
                    for cj in valid_cj:
                        sj = target_sets[j][cj]
                        compatibility[i][j][ci][cj] = si.isdisjoint(sj)
    
    # Backtracking to count valid combinations
    count = 0
    current = [None] * (k + 1)  # current[i] is the chosen c_i for out-degree i

    def backtrack(i):
        nonlocal count
        if i > k:
            count += 1
            return
        # Get valid c_i for current i
        for c_i in valid_c[i]:
            # Check compatibility with all previous j < i
            compatible = True
            for j in range(1, i):
                # j < i
                c_j = current[j]
                # Check if compatibility[j][i][c_j][c_i] is True
                if j not in compatibility or i not in compatibility[j]:
                    # If compatibility wasn't precomputed, then it's automatically compatible
                    continue
                # Get compatibility
                if c_j not in compatibility[j][i]:
                    compatible = False
                    break
                if not compatibility[j][i][c_j].get(c_i, False):
                    compatible = False
                    break
            if compatible:
                current[i] = c_i
                backtrack(i + 1)
                current[i] = None
    
    backtrack(1)
    print(count)

if __name__ == "__main__":
    main()
