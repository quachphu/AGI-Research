MOD = 998244353

def main():
    import sys
    N, M = map(int, sys.stdin.readline().split())
    
    max_n = N
    # Precompute factorial and inverse factorial
    fact = [1] * (max_n + 1)
    for i in range(1, max_n + 1):
        fact[i] = fact[i-1] * i % MOD
    inv_fact = [1] * (max_n + 1)
    inv_fact[max_n] = pow(fact[max_n], MOD-2, MOD)
    for i in range(max_n-1, -1, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD
    
    def comb(n, k):
        if k < 0 or k > n:
            return 0
        return fact[n] * inv_fact[k] % MOD * inv_fact[n - k] % MOD
    
    # Initialize DP
    dp_prev = [0] * (N + 1)
    dp_prev[0] = 1
    
    for j in range(1, M + 1):
        dp_current = [0] * (N + 1)
        for k_prev in range(N + 1):
            if dp_prev[k_prev] == 0:
                continue
            n_available = N - k_prev
            for delta in range(n_available + 1):
                new_k = k_prev + delta
                if new_k == 0:
                    contrib = 1
                else:
                    contrib = (pow(2, new_k, MOD) - 1) % MOD
                ways = comb(n_available, delta) * contrib % MOD
                dp_current[new_k] = (dp_current[new_k] + dp_prev[k_prev] * ways) % MOD
        dp_prev = dp_current
