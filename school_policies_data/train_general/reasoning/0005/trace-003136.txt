### Approach
1. **Rolling Hash**: Precompute hash values for all possible substrings of each string using a rolling hash technique. This allows efficient computation of hash values for any substring of a given length.
2. **Frequency Counting**: For each length l, compute the frequency of each hash value in each string. This helps in quickly determining how many times each substring of length l appears in each string.
3. **Intersection of Hash Values**: For each length l, find the intersection of hash values from all three strings. This intersection represents the common substrings of length l across all three strings.
4. **Product of Frequencies**: For each common hash value, compute the product of its frequencies in the three strings. Sum these products to get the result for the given length l.

import sys
from collections import defaultdict

MOD = 10**9 + 7
BASE = 911382629

def precompute_rolling(s):
    n = len(s)
    prefix = [0] * (n + 1)
    power = [1] * (n + 1)
    for i in range(n):
        prefix[i+1] = (prefix[i] * BASE + ord(s[i])) % MOD
        power[i+1] = (power[i] * BASE) % MOD
    return prefix, power

def main():
    s1 = sys.stdin.readline().strip()
    s2 = sys.stdin.readline().strip()
    s3 = sys.stdin.readline().strip()

    prefix1, power1 = precompute_rolling(s1)
    prefix2, power2 = precompute_rolling(s2)
    prefix3, power3 = precompute_rolling(s3)

    min_len = min(len(s1), len(s2), len(s3))
    ans = [0] * (min_len + 1)  # 1-based indexing

    for l in range(1, min_len + 1):
        def compute_hashes(prefix, power, s_len):
            cnt = defaultdict(int)
            max_i = s_len - l
            if max_i < 0:
                return cnt
            pl = power[l]
            for i in range(max_i + 1):
                # Compute hash for substring starting at i, length l
                current_hash = (prefix[i + l] - prefix[i] * pl) % MOD
                cnt[current_hash] += 1
            return cnt

        cnt1 = compute_hashes(prefix1, power1, len(s1))
        cnt2 = compute_hashes(prefix2, power2, len(s2))
        cnt3 = compute_hashes(prefix3, power3, len(s3))

        # Find common keys
        common = cnt1.keys() & cnt2.keys() & cnt3.keys()
        total = 0
        for key in common:
            total = (total + cnt1[key] * cnt2[key] * cnt3[key]) % MOD
        ans[l] = total

    output = [str(ans[l]) for l in range(1, min_len + 1)]
    print(' '.join(output))

if __name__ == "__main__":
    main()
