The game ends immediately if a knight reaches its target without being under attack or captures the opponent's knight.

2. **Key Insight**: The solution involves calculating the minimal steps required for each knight to reach their target using Breadth-First Search (BFS). This helps determine which knight can reach its target faster, considering the game's turn order.

3. **Algorithm Selection**: BFS is used to find the shortest path for each knight to their target. The knight with the shorter path is chosen to ensure Alice can reach the target first, leading to a win.

4. **Complexity Analysis**: BFS ensures that we efficiently find the shortest path in O(N*M) time complexity for each knight, where N and M are the dimensions of the chessboard. This is efficient given the constraints (up to 1000x1000 board).

### Solution Code
```python
import sys
from collections import deque

def bfs(n, m, start, end):
    directions = [(2, 1), (2, -1), (-2, 1), (-2, -1),
                  (1, 2), (1, -2), (-1, 2), (-1, -2)]
    visited = [[None for _ in range(m + 1)] for _ in range(n + 1)]
    q = deque()
    q.append(start)
    visited[start[0]][start[1]] = (None, None)
    while q:
        x, y = q.popleft()
        if (x, y) == end:
            path = []
            current = (x, y)
            while current != start:
                px, py = visited[current[0]][current[1]]
                path.append(current)
                current = (px, py)
            path.reverse()
            return path
        for dx, dy in directions:
            nx = x + dx
            ny = y + dy
            if 1 <= nx <= n and 1 <= ny <= m and visited[nx][ny] is None:
                visited[nx][ny] = (x, y)
                q.append((nx, ny))
    return []

def is_attacking(pos, target):
    x, y = pos
    tx, ty = target
    directions = [(2, 1), (2, -1), (-2, 1), (-2, -1),
                  (1, 2), (1, -2), (-1, 2), (-1, -2)]
    for dx, dy in directions:
        nx = x + dx
        ny = y + dy
        if (nx, ny) == (tx, ty):
            return True
    return False

n, m = map(int, sys.stdin.readline().split())
x1, y1, x2, y2 = map(int, sys.stdin.readline().split())

white_target = (n // 2, m // 2)
black_target = (n // 2 + 1, m // 2)

path_white = bfs(n, m, (x1, y1), white_target)
path_white_capture = bfs(n, m, (x1, y1), (x2, y2))
d_white = len(path_white)
d_white_capture = len(path_white_capture)

path_black = bfs(n, m, (x2, y2), black_target)
path_black_capture = bfs(n, m, (x2, y2), (x1, y1))
d_black = len(path_black)
d_black_capture = len(path_black_capture)

white_min = min(d_white, d_white_capture)
black_min = min(d_black, d_black_capture)

if white_min <= black_min:
    print("WHITE")
    sys.stdout.flush()
    current_white = (x1, y1)
    current_black = (x2, y2)
    if d_white <= d_white_capture:
        path = path_white
    else:
        path = path_white_capture
    for move in path:
        print(move[0], move[1])
        sys.stdout.flush()
        if move == current_black:
            sys.exit(0)
        if move == white_target and not is_attacking(current_black, move):
            sys.exit(0)
        bx, by = map(int, sys.stdin.readline().split())
        if bx == -1 and by == -1:
            sys.exit(0)
        current_black = (bx, by)
        if current_black == move:
            sys.exit(0)
else:
    print("BLACK")
    sys.stdout.flush()
    current_white_pos = (x1, y1)
    current_black = (x2, y2)
    if d_black <= d_black_capture:
        path = path_black
    else:
        path = path_black_capture
    bx, by = map(int, sys.stdin.readline().split())
    if bx == -1 and by == -1:
        sys.exit(0)
    current_white_pos = (bx, by)
    for move in path:
        print(move[0], move[1])
        sys.stdout.flush()
        if move == current_white_pos:
            sys.exit(0)
        if move == black_target and not is_attacking(current_white_pos, move):
            sys.exit(0)
        wx, wy = map(int, sys.stdin.readline().split())
        if wx == -1 and wy == -1:
            sys.exit(0)
        current_white_pos = (wx, wy)
        if current_white_pos == move:
            sys.exit(0)
```

### Explanation
1. **Reading Input**: The code reads the dimensions of the chessboard and the initial positions of the knights.
2. **BFS Calculation**: Using BFS, the code calculates the shortest path for each knight to reach their target and capture the opponent.
3. **Decision Making**: The code compares the minimal steps required for each knight to determine the optimal choice (white or black).
4. **Game Simulation**: The chosen knight follows the precomputed path, checking for winning conditions (capturing the opponent or reaching the target safely) after each move. The opponent's moves are read and tracked to ensure valid game progression.

This approach ensures that Alice can always choose a knight with a winning strategy, either by reaching the target first or capturing the opponent's knight.
