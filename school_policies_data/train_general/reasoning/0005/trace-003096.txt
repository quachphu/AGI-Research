1. **Precompute Fibonacci Numbers**: Generate a list of Fibonacci numbers up to a sufficiently large value (beyond 10^20) to ensure coverage for all possible inputs.
2. **Compute Lower Bounds**: For each Fibonacci number, compute the lower bound of the interval where the Nth non-Fibonacci number might lie.
3. **Binary Search for Interval**: For each input N, use binary search to find the interval where the Nth non-Fibonacci number lies based on precomputed lower bounds.
4. **Adjust for Fibonacci Check**: After determining the candidate number using the interval, check if the candidate is a Fibonacci number and adjust if necessary.

    next_fib = fibs[-1] + fibs[-2]
    fibs.append(next_fib)
    if next_fib > 1e20:

for i in range(len(fibs)):
    m = i + 1  # fibs[i] corresponds to m = i + 1
    lb = fibs[i] - m
    lower_bounds.append(lb)

def is_fib(x):
    """Check if x is a Fibonacci number using binary search."""
    left, right = 0, len(fibs) - 1
    while left <= right:
        mid = (left + right) // 2
        if fibs[mid] == x:
            return True
        elif fibs[mid] < x:
            left = mid + 1
        else:
            right = mid - 1
    return False

def nth_non_fib(n):
    """Find the nth non-Fibonacci number."""
    # Binary search to find the largest m where lower_bounds[m] <= n
    left, right = 0, len(lower_bounds) - 1
    best = 0
    while left <= right:
        mid = (left + right) // 2
        if lower_bounds[mid] <= n:
            best = mid
            left = mid + 1
        else:
            right = mid - 1
    m = best + 1  # m is (best index + 1)
    X = n + m
    # Check if X is a Fibonacci number and adjust if necessary
    return X - 1 if is_fib(X) else X
