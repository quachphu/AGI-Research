2. **Interval Overlap Calculation**: For each lucky number `m`, determine the interval `[prev + 1, m]` where `prev` is the previous lucky number. This interval represents all numbers for which `m` is the smallest lucky number greater than or equal to them.

import itertools

def generate_lucky_numbers():
    lucky = []
    for length in range(1, 11):  # Generate for lengths 1 to 10
        for digits in itertools.product('47', repeat=length):
            num = int(''.join(digits))
            lucky.append(num)
    return lucky

lucky_numbers = generate_lucky_numbers()

def main():
    import sys
    l, r = map(int, sys.stdin.readline().split())
    sum_total = 0
    prev = 0
    for m in lucky_numbers:
        start = prev + 1
        end = m
        current_start = max(start, l)
        current_end = min(end, r)
        if current_start <= current_end:
            count = current_end - current_start + 1
            sum_total += m * count
        prev = m
    print(sum_total)

if __name__ == "__main__":
    main()

3. **Overlap and Sum Calculation**: For each interval, we determine the overlap with the input range `[l, r]`. The sum of the contributions of each overlapping interval is computed by multiplying the count of numbers in the overlap by the lucky number `m`.
