def is_point_on_segment(p, seg):
    p0, p1 = seg
    x0, y0 = p0
    x1, y1 = p1
    px, py = p

    cross = (x1 - x0) * (py - y0) - (y1 - y0) * (px - x0)
    if cross != 0:
        return False

    min_x = min(x0, x1)
    max_x = max(x0, x1)
    min_y = min(y0, y1)
    max_y = max(y0, y1)
    if not (px >= min_x - 1e-8 and px <= max_x + 1e-8 and py >= min_y - 1e-8 and py <= max_y + 1e-8):
        return False

    dx = x1 - x0
    dy = y1 - y0

    if dx == 0 and dy == 0:
        return False

    if abs(dx) > abs(dy):
        t = (px - x0) / dx if dx != 0 else 0.0
    else:
        t = (py - y0) / dy if dy != 0 else 0.0

    return -1e-8 <= t <= 1 + 1e-8
