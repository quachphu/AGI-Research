def main():
    import sys
    p = sys.stdin.read().strip()
    n = len(p)
    memo = {}

    def compare(a_start, a_end, b_start, b_end):
        len_a = a_end - a_start + 1
        len_b = b_end - b_start + 1
        if len_a != len_b:
            return len_a - len_b
        for i in range(len_a):
            a_char = p[a_start + i]
            b_char = p[b_start + i]
            if a_char != b_char:
                return ord(a_char) - ord(b_char)
        return 0

    def helper(start, end):
        if (start, end) in memo:
            return memo[(start, end)]
        s_length = end - start + 1
        if s_length == 1:
            memo[(start, end)] = 1
            return 1
        max_val = 1
        for split in range(1, s_length):
            a_start = start
            a_end = start + split - 1
            b_start = start + split
            b_end = end
            
            if p[a_start] == '0' or p[b_start] == '0':
                continue
            
            cmp = compare(a_start, a_end, b_start, b_end)
            if cmp >= 0:
                current = helper(a_start, a_end) + helper(b_start, b_end)
                if current > max_val:
                    max_val = current
        memo[(start, end)] = max_val
        return max_val
