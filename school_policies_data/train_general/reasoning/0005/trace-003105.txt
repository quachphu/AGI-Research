To solve this problem, we need to determine the maximum number of fixed points (elements in their original positions) in a permutation after performing any number of allowed swaps. The key insight is to use the connected components of the swap graph to determine which positions can be permuted among each other, and then count how many elements can be fixed within each component.

### Approach
1. **Graph Representation**: Represent the swap pairs as an undirected graph where each node is a position in the permutation.
2. **Union-Find Data Structure**: Use the Union-Find (Disjoint Set Union, DSU) data structure to efficiently find and union connected components of the graph.
3. **Component Analysis**: For each connected component, determine how many elements in the component's positions can be fixed by checking if the element's value is within the component itself.

### Solution Code
```python
import sys
from collections import defaultdict

class DSU:
    def __init__(self, size):
        self.parent = list(range(size + 1))  # 1-based indexing
        self.rank = [1] * (size + 1)
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return
        if self.rank[x_root] < self.rank[y_root]:
            self.parent[x_root] = y_root
        else:
            self.parent[y_root] = x_root
            if self.rank[x_root] == self.rank[y_root]:
                self.rank[x_root] += 1

def main():
    input = sys.stdin.read().split()
    ptr = 0
    N = int(input[ptr])
    ptr += 1
    M = int(input[ptr])
    ptr +=1
    p = list(map(int, input[ptr:ptr+N]))
    ptr += N
    dsu = DSU(N)
    for _ in range(M):
        x = int(input[ptr])
        ptr +=1
        y = int(input[ptr])
        ptr +=1
        dsu.union(x, y)
    
    components = defaultdict(list)
    for i in range(1, N+1):
        root = dsu.find(i)
        components[root].append(i)
    
    ans = 0
    for root in components:
        positions = components[root]
        s = set(positions)
        cnt = 0
        for pos in positions:
            e = p[pos-1]
            if e in s:
                cnt +=1
        ans += cnt
    print(ans)

if __name__ == "__main__":
    main()
```

### Explanation
1. **Graph Representation**: The swap pairs are treated as edges in an undirected graph where nodes are positions in the permutation.
