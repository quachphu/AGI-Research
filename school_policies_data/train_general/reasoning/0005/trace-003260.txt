def main():
    import sys
    input = sys.stdin.read().split()
    n = int(input[0])
    a = list(map(int, input[1:n+1]))
    
    total = sum(a)
    s = n * (n + 1) // 2
    if total % s != 0:
        print("NO")
        return
    k = total // s
    
    for i in range(n):
        current = a[i]
        next_val = a[(i + 1) % n]
        delta = current - next_val
        temp = k + delta
        if temp % n != 0:
            print("NO")
            return
        x = temp // n
        if x < 0:
            print("NO")
            return
    print("YES")

if __name__ == '__main__':
