def main():
    import sys
    input = sys.stdin.read().split()
    n = int(input[0])
    a = list(map(int, input[1:n+1]))

    # Coordinate compression for prefix and suffix
    def compress(arr):
        sorted_unique = sorted(set(arr))
        return {v: i+1 for i, v in enumerate(sorted_unique)}, len(sorted_unique)

    # For prefix, we need to handle elements in their original order
    # For suffix, we need to handle elements in reverse order but also compress based on reverse sorted
    # But since we need to compare for greater in suffix, we can compress based on the same sorted values
    # So compress once
    compress_map, size = compress(a)
    compressed = [compress_map[x] for x in a]

    # Compute prefix_inversions
    class FenwickTree:
        def __init__(self, size):
            self.size = size
            self.tree = [0]*(self.size +2)
        def update(self, idx, delta=1):
            while idx <= self.size:
                self.tree[idx] += delta
                idx += idx & -idx
        def query(self, idx):
            res = 0
            while idx >0:
                res += self.tree[idx]
                idx -= idx & -idx
            return res

    prefix_inversions = [0]*(n+1)
    ft_prefix = FenwickTree(size)
    for i in range(n):
        current = compressed[i]
        # Number of elements greater than current
        greater = ft_prefix.query(size) - ft_prefix.query(current)
        prefix_inversions[i+1] = prefix_inversions[i] + greater
        ft_prefix.update(current)

    # Compute suffix_inversions
    suffix_inversions = [0]*(n+1)
    ft_suffix = FenwickTree(size)
    for i in reversed(range(n)):
        current = compressed[i]
        # Number of elements greater than current (since we are sorting in non-increasing)
        # So when processing from the end, elements added are from i+1 to end.
        # For a[i], the number of elements greater than it in the suffix is the number of elements already in the Fenwick tree that are greater than current.
        greater = ft_suffix.query(size) - ft_suffix.query(current)
        suffix_inversions[i] = suffix_inversions[i+1] + greater
        ft_suffix.update(current)

    min_swaps = float('inf')
    for k in range(n+1):
        total = prefix_inversions[k] + suffix_inversions[k]
        if total < min_swaps:
            min_swaps = total
    print(min_swaps)

if __name__ == "__main__":
    main()
