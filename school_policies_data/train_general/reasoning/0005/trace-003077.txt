import sys

def sieve(n):
    is_prime = [True] * (n + 1)
    is_prime[0] = is_prime[1] = False
    for i in range(2, int(n ** 0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, n + 1, i):
                is_prime[j] = False
    primes = [i for i, val in enumerate(is_prime) if val]
    return primes

# Precompute primes up to sqrt(2e7) ~ 4472
primes = sieve(4472)

def factor(n):
    factors = {}
    while n % 2 == 0:
        factors[2] = factors.get(2, 0) + 1
        n = n // 2
    i = 3
    while i * i <= n:
        while n % i == 0:
            factors[i] = factors.get(i, 0) + 1
            n = n // i
        i += 2
    if n > 1:
        factors[n] = 1
    return factors

def generate_divisors(factors):
    divisors = [1]
    for p, exp in factors.items():
        temp = []
        for d in divisors:
            current = d
            for _ in range(exp + 1):
                temp.append(current)
                current *= p
        divisors = temp
    return divisors

def count_distinct_primes(k):
    if k == 1:
        return 0
    s = 0
    remaining = k
    for p in primes:
        if p * p > remaining:
            break
        if remaining % p == 0:
            s += 1
            while remaining % p == 0:
                remaining //= p
    if remaining > 1:
        s += 1
    return s

def solve():

    t = int(input[0])

    for _ in range(t):

        divisors = generate_divisors(factors_x)
