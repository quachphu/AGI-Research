class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __sub__(self, other):
        return Point(self.x - other.x, self.y - other.y)

    def cross(self, other):
        return self.x * other.y - self.y * other.x

def convex_hull(points):
    points = sorted(points, key=lambda p: (p.x, p.y))
    lower = []
    for p in points:
        while len(lower) >= 2 and (lower[-1] - lower[-2]).cross(p - lower[-1]) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(points):
        while len(upper) >= 2 and (upper[-1] - upper[-2]).cross(p - upper[-1]) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]

### Approach

2. **Compute Convex Hull**: For each combination, compute the convex hull to check if it forms a convex polygon with exactly K vertices.
3. **Check Validity**: Verify that no other points lie inside the convex hull of the selected K points.
4. **Calculate Area**: Compute the area of valid convex polygons and track the maximum area found.

import sys
import itertools

class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __sub__(self, other):
        return Point(self.x - other.x, self.y - other.y)

    def cross(self, other):
        return self.x * other.y - self.y * other.x

    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

    def __hash__(self):
        return hash((self.x, self.y))

def convex_hull(points):
    points = sorted(points, key=lambda p: (p.x, p.y))
    lower = []
    for p in points:
        while len(lower) >= 2 and (lower[-1] - lower[-2]).cross(p - lower[-1]) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(points):
        while len(upper) >= 2 and (upper[-1] - upper[-2]).cross(p - upper[-1]) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]

def polygon_area(polygon):
    area = 0.0
    n = len(polygon)
    for i in range(n):
        j = (i + 1) % n
        area += polygon[i].x * polygon[j].y - polygon[i].y * polygon[j].x
    return abs(area) / 2.0

def is_point_inside_convex(point, polygon):
    prev_cross = None
    n = len(polygon)
    for i in range(n):
        a = polygon[i]
        b = polygon[(i+1) % n]
        cross = (b.x - a.x) * (point.y - a.y) - (b.y - a.y) * (point.x - a.x)
        if prev_cross is None:
            prev_cross = cross
        else:
            if (prev_cross < 0 and cross > 0) or (prev_cross > 0 and cross < 0):
                return False
        if cross != 0:
            prev_cross = cross
    return True
