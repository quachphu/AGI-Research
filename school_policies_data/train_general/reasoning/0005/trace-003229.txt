import bisect
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n, m = int(input[ptr]), int(input[ptr+1])
    ptr +=2
    a = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    b = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    c = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    
    # Initialize B_pos and C_pos
    B_pos = defaultdict(list)
    C_pos = defaultdict(list)
    for i in range(n):
        a_i = a[i]
        B_val = b[a_i-1]
        bisect.insort(B_pos[B_val], i+1)  # 1-based index
        C_val = c[a_i-1]
        bisect.insort(C_pos[C_val], i+1)
    
    for _ in range(m):
        op = input[ptr]
        if op == '1':
            k = int(input[ptr+1]) -1  # 0-based index
            x = int(input[ptr+2])
            ptr +=3
            old_v = a[k]
            new_v = x
            
            # Remove old_v's entries from B_pos and C_pos
            B_old = b[old_v-1]
            pos_list = B_pos[B_old]
            idx = bisect.bisect_left(pos_list, k+1)
            if idx < len(pos_list) and pos_list[idx] == k+1:
                pos_list.pop(idx)
            
            C_old = c[old_v-1]
            pos_list = C_pos[C_old]
            idx = bisect.bisect_left(pos_list, k+1)
            if idx < len(pos_list) and pos_list[idx] == k+1:
                pos_list.pop(idx)
            
            # Add new_v's entries to B_pos and C_pos
            a[k] = new_v
            B_new = b[new_v-1]
            bisect.insort(B_pos[B_new], k+1)
            
            C_new = c[new_v-1]
            bisect.insort(C_pos[C_new], k+1)
        else:
            r = int(input[ptr+1])
            ptr +=2
            ans = 0
            for j in range(1, r):  # j is 1-based, from 1 to r-1
                current_a = a[j-1]  # since j is 1-based, a[j-1] is current a_j
                # Compute left: number of B_pos[current_a] < j
                B_list = B_pos.get(current_a, [])
                left = bisect.bisect_left(B_list, j)
