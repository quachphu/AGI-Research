To solve this problem, we need to determine the states in which Vasya's value is considered "interesting" based on specific criteria. A state's value is interesting if there exists a valid path starting from an assignment state, passing through the state, and ending at a usage state without any intermediate assignments.

### Approach
1. **Identify Reachable States from Assignments**: Use BFS to determine all states reachable from any assignment state (f=1) without encountering another assignment.
2. **Identify Reachable Usage States**: Use reverse BFS to determine all states that can reach a usage state (f=2) through non-assignment states (f=0 or f=2).
3. **Check Valid Paths**: For each state, check if it is reachable from an assignment and can reach a usage state under the given constraints.

### Solution Code
```python
import sys
from collections import deque

def main():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1
    f = list(map(int, input[ptr:ptr + n]))
    ptr += n

    adj = [[] for _ in range(n)]
    for _ in range(m):
        a = int(input[ptr]) - 1
        ptr += 1
        b = int(input[ptr]) - 1
        ptr += 1
        adj[a].append(b)

    # Compute from_assign: states reachable from any assignment (f=1) without encountering another assignment
    from_assign = [False] * n
    for u in range(n):
        if f[u] == 1 and not from_assign[u]:
            q = deque([u])
            from_assign[u] = True
            while q:
                current = q.popleft()
                for v in adj[current]:
                    if f[v] != 1 and not from_assign[v]:
                        from_assign[v] = True
                        q.append(v)

    # Build reversed subgraph adjacency for non-assignment nodes (f != 1)
    rev_sub_adj = [[] for _ in range(n)]
    for u in range(n):
        if f[u] != 1:
            for v in adj[u]:
                if f[v] != 1:
                    rev_sub_adj[v].append(u)

    # Compute can_reach_use_non_assign using reverse BFS from use nodes (f=2)
    use_nodes = [u for u in range(n) if f[u] == 2]
    can_reach_use_non_assign = [False] * n
    for t in use_nodes:
        if not can_reach_use_non_assign[t]:
            q = deque([t])
            can_reach_use_non_assign[t] = True
            while q:
                u = q.popleft()
                for v in rev_sub_adj[u]:
                    if not can_reach_use_non_assign[v]:
                        can_reach_use_non_assign[v] = True
                        q.append(v)

    # Compute can_reach_use_assign for assignment nodes (f=1)
    can_reach_use_assign = [False] * n
    for u in range(n):
        if f[u] == 1:
            for v in adj[u]:
                if f[v] != 1 and can_reach_use_non_assign[v]:
                    can_reach_use_assign[u] = True
                    break

    # Generate the result
    result = []
    for u in range(n):
        if from_assign[u]:
            if f[u] != 1:
                if can_reach_use_non_assign[u]:
                    result.append(1)
                else:
                    result.append(0)
            else:
                if can_reach_use_assign[u]:
                    result.append(1)
                else:
                    result.append(0)
        else:
            result.append(0)

    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```

### Explanation
1. **Reachable States from Assignments**: Using BFS, we start from each assignment state and mark all reachable states that do not encounter another assignment.
2. **Reverse BFS for Usage States**: By reversing the graph and considering only non-assignment states, we determine which states can reach a usage state.
3. **Check Validity**: For each state, we check if it is reachable from an assignment and can reach a usage state under the constraints. This ensures that the state is part of a valid path, making Vasya's value interesting.
