def simulate(board, scores):

    total = 0

    bonus = 1

    while True:

        marked = [[False for _ in range(5)] for _ in range(5)]

        # Check horizontal

        for i in range(5):

            j = 0

            while j <5:

                if board[i][j] ==0:

                    j +=1

                    continue

                current = board[i][j]

                run_length =1

                k = j+1

                while k <5 and board[i][k] == current:

                    run_length +=1

                    k +=1

                if run_length >=3:

                    for l in range(j, k):

                        marked[i][l] = True

                j = k

        # Check vertical

        for j in range(5):

            i =0

            while i <5:

                if board[i][j] ==0:

                    i +=1

                    continue

                current = board[i][j]

                run_length =1

                k = i+1

                while k <5 and board[k][j] == current:

                    run_length +=1

                    k +=1

                if run_length >=3:

                    for l in range(i, k):

                        marked[l][j] = True

                i = k

        # Check if any marked

        any_marked = any( any(row) for row in marked )

        if not any_marked:

            break

        # Compute the score

        current_score =0

        for i in range(5):

            for j in range(5):

                if marked[i][j]:

                    current_type = board[i][j]

                    current_score += scores[current_type-1]

        total += current_score * bonus

        # Remove marked blocks

        for i in range(5):

            for j in range(5):

                if marked[i][j]:

                    board[i][j] =0

        # Process falling

        for j in range(5):

            column = [board[i][j] for i in range(5)]

            non_zero = [x for x in column if x !=0]

            new_column = [0]*(5 - len(non_zero)) + non_zero

            for i in range(5):

                board[i][j] = new_column[i]

        # Increase bonus

        bonus +=1

    return total

2. **Simulation of Moves**: Generate all possible sequences of swaps (up to `n` moves) starting from each possible initial block position. Each sequence of moves will generate a new board configuration.
3. **Score Calculation**: For each generated board configuration, simulate the removal of blocks, calculate the score, and handle the falling of blocks. This process is repeated until no more blocks can be removed.
4. **Max Score Determination**: Track the maximum score obtained from all possible board configurations generated by valid swap sequences.

import sys

def simulate(board, scores):
    total = 0
    bonus = 1
    while True:
        marked = [[False for _ in range(5)] for _ in range(5)]
        # Check horizontal
        for i in range(5):
            j = 0
            while j < 5:
                if board[i][j] == 0:
                    j += 1
                    continue
                current = board[i][j]
                run_length = 1
                k = j + 1
                while k < 5 and board[i][k] == current:
                    run_length += 1
                    k += 1
                if run_length >= 3:
                    for l in range(j, k):
                        marked[i][l] = True
                j = k
        # Check vertical
        for j in range(5):
            i = 0
            while i < 5:
                if board[i][j] == 0:
                    i += 1
                    continue
                current = board[i][j]
                run_length = 1
                k = i + 1
                while k < 5 and board[k][j] == current:
                    run_length += 1
                    k += 1
                if run_length >= 3:
                    for l in range(i, k):
                        marked[l][j] = True
                i = k
        # Check if any marked
        any_marked = any(any(row) for row in marked)
        if not any_marked:
            break
        # Compute the score
        current_score = 0
        for i in range(5):
            for j in range(5):
                if marked[i][j]:
                    current_type = board[i][j]
                    current_score += scores[current_type - 1]
        total += current_score * bonus
        # Remove marked blocks
        for i in range(5):
            for j in range(5):
                if marked[i][j]:
                    board[i][j] = 0
        # Process falling
        for j in range(5):
            column = [board[i][j] for i in range(5)]
            non_zero = [x for x in column if x != 0]
            new_column = [0] * (5 - len(non_zero)) + non_zero
            for i in range(5):
                board[i][j] = new_column[i]
        # Increase bonus
        bonus += 1
    return total
