### Solution Code

import sys
from collections import defaultdict

def main():
    n = int(sys.stdin.readline())
    max_pos = 0
    chars = dict()
    for _ in range(n):
        parts = sys.stdin.readline().split()
        ti = parts[0]
        ki = int(parts[1])
        positions = list(map(int, parts[2:2+ki]))
        m = len(ti)
        for x in positions:
            end = x + m - 1  # 1-based end position
            if end > max_pos:
                max_pos = end
            for i in range(m):
                pos = x - 1 + i  # convert to 0-based
                c = ti[i]
                if pos not in chars:
                    chars[pos] = c
    # Now build the string up to max_pos (1-based, so length is max_pos)
    s = []
    for i in range(max_pos):
        s.append(chars.get(i, 'a'))
    print(''.join(s))

if __name__ == "__main__":
    main()
