import sys

class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None

def insert(root, key):
    new_node = Node(key)
    if root is None:
        return new_node
    current = root
    while True:
        if key < current.key:
            if current.left is None:
                current.left = new_node
                break
            else:
                current = current.left
        else:
            if current.right is None:
                current.right = new_node
                break
            else:
                current = current.right
    return root

def find(root, key):
    current = root
    while current is not None:
        if current.key == key:
            return True
        elif key < current.key:
            current = current.left
        else:
            current = current.right
    return False

def inorder(root):
    res = []
    stack = []
    current = root
    while True:
        while current is not None:
            stack.append(current)
            current = current.left
        if not stack:
            break
        current = stack.pop()
        res.append(current.key)
        current = current.right
    return res

def preorder(root):
    res = []
    stack = []
    if root is not None:
        stack.append(root)
    while stack:
        current = stack.pop()
        res.append(current.key)
        if current.right is not None:
            stack.append(current.right)
        if current.left is not None:
            stack.append(current.left)
    return res

def main():
    input = sys.stdin.read().splitlines()
    m = int(input[0])
    root = None
    for line in input[1:m+1]:
        parts = line.split()
        if parts[0] == 'insert':
            k = int(parts[1])
            root = insert(root, k)
        elif parts[0] == 'find':
            k = int(parts[1])
            print('yes' if find(root, k) else 'no')
        elif parts[0] == 'print':
            in_order = inorder(root)
            pre_order = preorder(root)
            print(' ' + ' '.join(map(str, in_order)))
            print(' ' + ' '.join(map(str, pre_order)))

if __name__ == '__main__':
    main()
