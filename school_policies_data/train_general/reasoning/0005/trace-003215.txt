To solve this problem, we need to determine the minimum number of moves required for a robot to clean all dirty tiles in a room, starting from a given initial position and avoiding obstacles. The solution involves a combination of graph traversal and dynamic programming to handle the pathfinding and optimization aspects.

### Approach
1. **Input Parsing**: Read the grid dimensions and the grid itself, identifying the robot's starting position and the positions of all dirty tiles.
2. **Precompute Distances**: Use Breadth-First Search (BFS) to compute the shortest path between the robot's starting position and each dirty tile, as well as between each pair of dirty tiles.
3. **Check Reachability**: Ensure all dirty tiles are reachable from the starting position and from each other. If any tile is unreachable, output `-1`.
4. **Dynamic Programming for TSP**: Use dynamic programming with a bitmask to solve the Traveling Salesman Problem (TSP), where the goal is to visit all dirty tiles in the minimum number of moves.

import sys
from collections import deque

def main():
    while True:
        w, h = map(int, sys.stdin.readline().split())
        if w == 0 and h == 0:
            break
        grid = []
        start = None
        dirty = []
        for y in range(h):
            line = sys.stdin.readline().strip()
            grid.append(line)
            for x, c in enumerate(line):
                if c == 'o':
                    start = (x, y)
                elif c == '*':
                    dirty.append((x, y))
        m = len(dirty)
        if m == 0:
            print(0)
            continue
        nodes = [start] + dirty
        distance = [[0] * len(nodes) for _ in range(len(nodes))]
        ok = True
        for i in range(len(nodes)):
            sx, sy = nodes[i]
            visited = [[-1] * w for _ in range(h)]
            q = deque([(sx, sy)])
            visited[sy][sx] = 0
            dirs = [(-1, 0), (1, 0), (0, -1), (0, 1)]
            while q:
                x, y = q.popleft()
                for dx, dy in dirs:
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < w and 0 <= ny < h:
                        if grid[ny][nx] != 'x' and visited[ny][nx] == -1:
                            visited[ny][nx] = visited[y][x] + 1
                            q.append((nx, ny))
            for j in range(len(nodes)):
                if i == j:
                    distance[i][j] = 0
                    continue
                tx, ty = nodes[j]
                if visited[ty][tx] == -1:
                    ok = False
                distance[i][j] = visited[ty][tx] if visited[ty][tx] != -1 else float('inf')
            if not ok:
                break
        if not ok:
            print(-1)
            continue
        # Solve TSP
        INF = float('inf')
        dp = [[INF] * m for _ in range(1 << m)]
        for i in range(m):
            dp[1 << i][i] = distance[0][i + 1]
        for mask in range(1, 1 << m):
            for u in range(m):
                if not (mask & (1 << u)) or dp[mask][u] == INF:
                    continue
                for v in range(m):
                    if mask & (1 << v):
                        continue
                    new_mask = mask | (1 << v)
                    new_dist = dp[mask][u] + distance[u + 1][v + 1]
                    if new_dist < dp[new_mask][v]:
                        dp[new_mask][v] = new_dist
        full_mask = (1 << m) - 1
        min_moves = min(dp[full_mask][v] for v in range(m))
        print(min_moves if min_moves != INF else -1)
