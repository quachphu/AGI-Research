2. **Expected Cost Calculation**: The expected cost from each node can be computed using a dynamic programming approach. For each node, the expected cost is the average of the costs to travel to each neighbor plus the expected cost from that neighbor.
3. **Modular Arithmetic**: Since the result needs to be a fraction modulo \(10^9 + 7\), we use modular inverses to handle divisions.

```python
import sys
from collections import deque

MOD = 10**9 + 7

def main():
    sys.setrecursionlimit(1 << 25)
    n = int(sys.stdin.readline())
    adj = [[] for _ in range(n)]
    for _ in range(n-1):
        u, v, c = map(int, sys.stdin.readline().split())
        adj[u].append((v, c))
        adj[v].append((u, c))
    
    # Compute sum_c for each node
    sum_c = [0] * n
    for u in range(n):
        s = 0
        for (v, c) in adj[u]:
            s = (s + c) % MOD
        sum_c[u] = s
    
    # Root the tree at 0 and build parent and children
    parent = [-1] * n
    children = [[] for _ in range(n)]
    q = deque([0])
    parent[0] = -1
    while q:
        u = q.popleft()
        for (v, c) in adj[u]:
            if parent[v] == -1 and v != parent[u]:
                parent[v] = u
                children[u].append(v)
                q.append(v)
    
    # Post-order traversal
    post_order = []
    stack = [(0, False)]
    while stack:
        u, processed = stack.pop()
        if processed:
            post_order.append(u)
            continue
        stack.append((u, True))
        # Push children in reverse order to process in correct order
        for v in reversed(children[u]):
            stack.append((v, False))
    
    a = [0] * n
    b = [0] * n
    
    for u in post_order:
        if u == 0:
            continue  # root is handled later
        if len(adj[u]) == 1:
            a[u] = 0
            b[u] = 0
        else:
            # Collect children (parent is excluded)
            c_children = children[u]
            sum_a = 0
            sum_b = 0
            for v in c_children:
                sum_a = (sum_a + a[v]) % MOD
                sum_b = (sum_b + b[v]) % MOD
            d = len(adj[u])
            denominator = (d - sum_b) % MOD
            inv_denominator = pow(denominator, MOD-2, MOD)
            a_u = ((sum_c[u] + sum_a) % MOD) * inv_denominator % MOD
            b_u = 1 * inv_denominator % MOD
            a[u] = a_u
            b[u] = b_u
    
    # Handle root
    root = 0
    c_children = children[root]
    sum_a = 0
    sum_b = 0
    for v in c_children:
        sum_a = (sum_a + a[v]) % MOD
        sum_b = (sum_b + b[v]) % MOD
    d_root = len(adj[root])
    denominator = (d_root - sum_b) % MOD
    if denominator == 0:
        print(0)
        return
    inv_denominator = pow(denominator, MOD-2, MOD)
    E0 = ((sum_c[root] + sum_a) % MOD) * inv_denominator % MOD
    print(E0)

if __name__ == "__main__":
```
