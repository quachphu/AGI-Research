3. **Sum of Divisors**: Compute the sum of divisors of the remaining part of \( N \) (after removing the highest power of 2 if \( N \) is even). This sum is adjusted to exclude divisors divisible by 4.

### Solution Code
```python
def max_power_of_2(n):
    k = 0
    while n % 2 == 0:
        k += 1
        n = n // 2
    return k, n

def factorize(m):
    factors = {}
    while m % 2 == 0:
        factors[2] = factors.get(2, 0) + 1
        m = m // 2
    i = 3
    while i * i <= m:
        while m % i == 0:
            factors[i] = factors.get(i, 0) + 1
            m = m // i
        i += 2
    if m > 1:
        factors[m] = 1
    return factors

def sum_of_divisors(factors):
    sum_div = 1
    for p, exp in factors.items():
        sum_p = (pow(p, exp + 1) - 1) // (p - 1)
        sum_div *= sum_p
    return sum_div

def compute_ways(n):
    if n % 2 == 0:
        k, m = max_power_of_2(n)
        factors = factorize(m)
        sigma = sum_of_divisors(factors)
        s = 3 * sigma
    else:
        factors = factorize(n)
        sigma = sum_of_divisors(factors)
        s = sigma
    return 8 * s

import sys

def main():
    input = sys.stdin.read().split()
    T = int(input[0])
    for i in range(1, T + 1):
        N = int(input[i])
        print(compute_ways(N))

if __name__ == "__main__":
    main()
```
