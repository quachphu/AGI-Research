To solve this problem, we need to simulate the operations of a commodity exchange market where dealers place buy or sell orders. The goal is to process these orders according to specific rules and generate reports on commodity deals and dealer transactions.

### Approach
1. **Order Processing**: For each new order, check if there is a matching order in the active list (orders that haven't been matched yet) according to the rules. If a match is found, process the deal and update the relevant data structures. If no match is found, add the order to the active list.
2. **Commodity Report**: For each commodity, calculate the lowest, average (floor division), and highest prices of successful deals.
3. **Dealer Report**: For each dealer, calculate the total amount paid and received from their transactions.

### Solution Code
```python
import sys
from collections import defaultdict

def process_dataset(orders):
    active_orders = []
    commodities = defaultdict(list)
    dealers_paid = defaultdict(int)
    dealers_received = defaultdict(int)
    dealers_set = set()

    for i, order in enumerate(orders):
        dealer, typ, comm, price = order
        dealers_set.add(dealer)
        price = int(price)

        if typ == 'SELL':
            candidates = [o for o in active_orders if o['type'] == 'BUY' and o['commodity'] == comm and o['price'] >= price and o['dealer'] != dealer]
            if candidates:
                max_price = max(o['price'] for o in candidates)
                best_candidates = [o for o in candidates if o['price'] == max_price]
                best = min(best_candidates, key=lambda x: x['index'])
                active_orders.remove(best)
                deal_price = (best['price'] + price) // 2
                commodities[comm].append(deal_price)
                dealers_received[dealer] += deal_price
                dealers_paid[best['dealer']] += deal_price
            else:
                active_orders.append({'dealer': dealer, 'type': typ, 'commodity': comm, 'price': price, 'index': i})
        else:  # BUY
            candidates = [o for o in active_orders if o['type'] == 'SELL' and o['commodity'] == comm and o['price'] <= price and o['dealer'] != dealer]
            if candidates:
                min_price = min(o['price'] for o in candidates)
                best_candidates = [o for o in candidates if o['price'] == min_price]
                best = min(best_candidates, key=lambda x: x['index'])
                active_orders.remove(best)
                deal_price = (price + best['price']) // 2
                commodities[comm].append(deal_price)
                dealers_received[best['dealer']] += deal_price
                dealers_paid[dealer] += deal_price
            else:
                active_orders.append({'dealer': dealer, 'type': typ, 'commodity': comm, 'price': price, 'index': i})

    # Generate output
    output = []
    # Commodity part
    comm_list = sorted(commodities.keys())
    for comm in comm_list:
        prices = commodities[comm]
        low = min(prices)
        high = max(prices)
        avg = sum(prices) // len(prices)
        output.append(f"{comm} {low} {avg} {high}")
    output.append("--")

    # Dealers part
    dealers_sorted = sorted(dealers_set)
    for dealer in dealers_sorted:
        paid = dealers_paid.get(dealer, 0)
        received = dealers_received.get(dealer, 0)
        output.append(f"{dealer} {paid} {received}")

    # Add the ten hyphens
    output.append('-' * 10)
    return '\n'.join(output)

def main():
    data = sys.stdin.read().split('\n')
    idx = 0
    while True:
        line = data[idx].strip()
        if not line:
            idx += 1
            continue
        n = int(line)
        if n == 0:
            break
        idx += 1
        orders = []
        for _ in range(n):
            if idx >= len(data):
                break
            current_line = data[idx].strip()
            if not current_line:
                idx += 1
                continue
            parts = current_line.split()
            orders.append((parts[0], parts[1], parts[2], parts[3]))
            idx += 1
        result = process_dataset(orders)
        print(result)

if __name__ == "__main__":
    main()
```

### Explanation
1. **Reading Input**: The input is read line by line, and each dataset is processed individually.
2. **Active Orders**: Maintain a list of active orders that have not yet been matched.
3. **Matching Orders**: For each new order, check if there is a matching order in the active list. The matching criteria depend on the order type (buy/sell) and exclude orders from the same dealer.
4. **Deal Calculation**: When a deal is made, the price is calculated as the floor of the average of the buy and sell prices. The dealers involved are updated with their respective amounts paid or received.
5. **Generating Reports**: After processing all orders in a dataset, generate the commodity report (sorted alphabetically) and the dealer report (sorted lexicographically).

This approach ensures that all orders are processed efficiently, and the required reports are generated accurately.
