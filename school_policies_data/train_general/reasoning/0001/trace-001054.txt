class SegmentTreeNode:
    def __init__(self, l, r):
        self.l = l
        self.r = r
        self.left_child = None
        self.right_child = None
        self.min_sj = 0
        self.max_fj = 0

def build_segment_tree(buses, l, r):
    node = SegmentTreeNode(l, r)
    if l == r:
        node.min_sj = buses[l][0]
        node.max_fj = buses[l][1]
    else:
        mid = (l + r) // 2
        node.left_child = build_segment_tree(buses, l, mid)
        node.right_child = build_segment_tree(buses, mid+1, r)
        node.min_sj = min(node.left_child.min_sj, node.right_child.min_sj)
        node.max_fj = max(node.left_child.max_fj, node.right_child.max_fj)
    return node

### Approach
1. **Sort Buses by Departure Time**: First, we sort all buses based on their departure time to facilitate efficient range queries.
2. **Segment Tree Construction**: We build a segment tree where each node stores the minimum start stop and maximum end stop for the buses in its range. This helps in quickly determining if any bus in a given range can cover a citizen's route.
3. **Binary Search for Departure Time**: For each citizen query, we use binary search to find the earliest bus that departs after the citizen's arrival time.
4. **Segment Tree Query**: Using the segment tree, we check the valid buses starting from the found index to determine the earliest bus that covers the citizen's route.

import bisect

class SegmentTreeNode:
    def __init__(self, l, r):
        self.l = l
        self.r = r
        self.left_child = None
        self.right_child = None
        self.min_sj = 0
        self.max_fj = 0

def build_segment_tree(buses, l, r):
    node = SegmentTreeNode(l, r)
    if l == r:
        node.min_sj = buses[l][0]
        node.max_fj = buses[l][1]
    else:
        mid = (l + r) // 2
        node.left_child = build_segment_tree(buses, l, mid)
        node.right_child = build_segment_tree(buses, mid + 1, r)
        node.min_sj = min(node.left_child.min_sj, node.right_child.min_sj)
        node.max_fj = max(node.left_child.max_fj, node.right_child.max_fj)
    return node

def query_segment_tree(node, start, li, ri):
    if node.r < start:
        return -1
    if node.min_sj > li:
        return -1
    if node.max_fj < ri:
        return -1
    if node.l == node.r:
        return node.l if (node.min_sj <= li and node.max_fj >= ri) else -1
    left_res = query_segment_tree(node.left_child, start, li, ri)
    if left_res != -1:
        return left_res
    right_res = query_segment_tree(node.right_child, start, li, ri)
    return right_res if right_res != -1 else -1

n, m = map(int, input().split())
buses = []
for i in range(n):
    s, f, t = map(int, input().split())
    buses.append((t, s, f, i + 1))

# Sort buses by departure time (t)
buses.sort()
sorted_tj = [bus[0] for bus in buses]
sorted_buses = [(bus[1], bus[2], bus[3]) for bus in buses]  # (sj, fj, original_index)

# Build segment tree
if n == 0:
    root = None
else:
    root = build_segment_tree(sorted_buses, 0, n - 1)

output = []
for _ in range(m):
    l, r, b = map(int, input().split())
    # Find the first index where tj >= b
    start = bisect.bisect_left(sorted_tj, b)
    if start >= n:
        output.append(-1)
        continue
    if root is None:
        output.append(-1)

    idx = query_segment_tree(root, start, l, r)
    if idx == -1:
        output.append(-1)
    else:
        original_index = sorted_buses[idx][2]
        output.append(original_index)
