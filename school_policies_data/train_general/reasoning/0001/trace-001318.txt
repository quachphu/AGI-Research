n, k, A, B = map(int, input().split())
positions = list(map(int, input().split()))
positions.sort()

def compute_min(start, end):

    right_idx = bisect.bisect_right(positions, end)
    count = right_idx - left
    
    if count == 0:
        return A
    
    current_length = end - start + 1
    burn = B * count * current_length
    
    if current_length == 1:

total_length = 2 ** n
print(compute_min(1, total_length))
