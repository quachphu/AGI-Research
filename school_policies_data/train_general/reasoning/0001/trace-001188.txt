    if n == 0:
        return 0
    # Compute k1 for case1 (start with 1)
    k1 = math.ceil((math.sqrt(1 + 8 * n) - 1) / 2)
    valid1 = (k1 * (k1 + 1) // 2 >= n) and (n >= k1)

    k2 = math.ceil((math.sqrt(9 + 8 * n) - 3) / 2)

    if valid1:
