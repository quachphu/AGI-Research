    import sys
    input = sys.stdin.read().split()
    ptr = 0
    original = input[ptr]
    ptr += 1
    k = int(input[ptr])
    ptr += 1
    n = int(input[ptr])
    ptr += 1
    mutations = []
    for _ in range(n):
        li = int(input[ptr])
        ri = int(input[ptr + 1])
        mutations.append((li, ri))
        ptr += 2
    
    # Calculate initial length after all mutations
    initial_length = len(original)
    for li, ri in mutations:
        initial_length += (ri - li + 1)

    # Precompute all mutations in reverse order
    reversed_mutations = reversed(mutations)
    # Process each position from 1 to k
    for pos in range(1, k + 1):
        current_k = pos
        current_length = initial_length
        for li, ri in reversed_mutations:
            m = ri - li + 1
            current_length_before = current_length - m
            inserted_copy_start = ri + 1
            inserted_copy_end = ri + m
            if current_k > inserted_copy_end:
                current_k -= m
