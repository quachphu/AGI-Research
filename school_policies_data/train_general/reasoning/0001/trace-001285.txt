        stack.append((node, True))
        # Push children in reverse to process them left to right in post-order
        for child in reversed(children[node]):
            stack.append((child, False))

    f = [0] * (n + 1)
    for node in post_order:
        if not children[node]:
            f[node] = 1
        else:
            groups = defaultdict(list)
            for child in children[node]:
                groups[s[child - 1]].append(f[child])
            sum_max = 0
            for key in groups:
                sum_max += max(groups[key])
            f[node] = 1 + sum_max

    max_val = -1
    count = 0
    for i in range(1, n + 1):
        current = f[i] + c[i - 1]
        if current > max_val:
            max_val = current
            count = 1
        elif current == max_val:
            count += 1

    print(max_val)
    print(count)
