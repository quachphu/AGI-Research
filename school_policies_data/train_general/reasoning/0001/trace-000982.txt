class DSU:

    def __init__(self, size):

        self.parent = list(range(size+1))  # 1-based

        self.rank = [1]*(size+1)

    def find(self, x):

        if self.parent[x] != x:

            self.parent[x] = self.find(self.parent[x])

        return self.parent[x]

    def union(self, x, y):

        x_root = self.find(x)

        y_root = self.find(y)

        if x_root == y_root:

            return False

        if self.rank[x_root] < self.rank[y_root]:

            self.parent[x_root] = y_root

        else:

            self.parent[y_root] = x_root

            if self.rank[x_root] == self.rank[y_root]:

                self.rank[x_root] +=1

        return True

To solve this problem, we need to determine the minimum possible total weight of edges chosen by Takahashi in each query such that every vertex in the graph is reachable from either of the two specified vertices. The solution involves leveraging the Minimum Spanning Tree (MST) and efficiently answering queries using binary lifting to find the maximum edge on the path between two nodes.

### Approach
1. **Compute the MST**: Using Kruskal's algorithm, we first compute the MST of the given graph. This helps us determine the minimum total weight required to keep the graph connected.
2. **Binary Lifting for LCA**: Preprocess the MST to support efficient queries for the maximum edge weight on the path between any two nodes. This is done using binary lifting, which allows us to compute the maximum edge in logarithmic time.
3. **Answer Queries**: For each query, determine the maximum edge weight on the path between the two specified vertices in the MST using the preprocessed data. Subtract this maximum edge weight from the total weight of the MST to get the result for the query.

import sys
from collections import deque

class DSU:
    def __init__(self, size):
        self.parent = list(range(size + 1))  # 1-based indexing
        self.rank = [1] * (size + 1)
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return False
        if self.rank[x_root] < self.rank[y_root]:
            self.parent[x_root] = y_root
        else:
            self.parent[y_root] = x_root
            if self.rank[x_root] == self.rank[y_root]:
                self.rank[x_root] += 1
        return True

def main():

    adj = [[] for _ in range(n + 1)]
    total_weight = 0
    for c, a, b in edges:
        if dsu.find(a) != dsu.find(b):
            dsu.union(a, b)
            adj[a].append((b, c))
            adj[b].append((a, c))
            total_weight += c
    
    # Build parent, depth, edge_to_parent arrays
    parent = [0] * (n + 1)
    depth = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    root = 1
    parent[root] = -1
    depth[root] = 0
    q = deque()
    q.append(root)
    while q:
        u = q.popleft()
        for v, c in adj[u]:
            if parent[v] == 0 and v != root:
                parent[v] = u
                depth[v] = depth[u] + 1
                edge_to_parent[v] = c
                q.append(v)
    
    # Prepare binary lifting tables
    log_max = 20
    up = [[-1] * (n + 1) for _ in range(log_max)]

    for u in range(1, n + 1):

        max_edge[0][u] = edge_to_parent[u] if parent[u] != -1 else 0
    
    for k in range(1, log_max):
        for u in range(1, n + 1):
            if up[k-1][u] != -1:
                up[k][u] = up[k-1][up[k-1][u]]
                max_edge[k][u] = max(max_edge[k-1][u], max_edge[k-1][up[k-1][u]])
            else:
                up[k][u] = -1
                max_edge[k][u] = 0

    def get_max_edge(u, v):
        max_val = 0
        if depth[u] < depth[v]:
            u, v = v, u
        # Bring u up to depth of v
        for k in reversed(range(log_max)):
            if depth[u] - (1 << k) >= depth[v]:
                max_val = max(max_val, max_edge[k][u])
                u = up[k][u]

            return max_val
