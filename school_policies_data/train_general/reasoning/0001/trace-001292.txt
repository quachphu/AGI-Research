import bisect
import sys

    input = sys.stdin.read().split()
    ptr = 0

    ptr +=2
    t = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    a = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    q = int(input[ptr])
    ptr +=1
    queries = []
    for _ in range(q):
        l, r = int(input[ptr]), int(input[ptr+1])
        queries.append( (l, r) )
        ptr +=2
    
    # Compute prefix sums
    s = [0]*(n+1)
    for i in range(1, n+1):
        if t[i-1] == 1:
            s[i] = s[i-1] + a[i-1]
        else:
            s[i] = s[i-1] - a[i-1]
    
    # Build pos dictionary
    from collections import defaultdict
    pos = defaultdict(list)
    for idx in range(n+1):
        pos[s[idx]].append(idx)
    
    # Process queries
    for l, r in queries:
        a_val = l -1

        res = 0
        for j in range(l, r+1):

            lst = pos.get(x, [])
            # Find number of elements in lst >= a_val and <= j-1
            left = bisect.bisect_left(lst, a_val)

            res += (right - left)
