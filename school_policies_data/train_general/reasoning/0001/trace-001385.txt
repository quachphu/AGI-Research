To solve this problem, we need to determine the minimum number of subsequences required to partition a given range of an array such that each subsequence meets the condition that no element occurs more than the ceiling of half the length of the subsequence. The solution involves efficiently finding the maximum frequency of any element in the given range for multiple queries.

### Approach
1. **Mo's Algorithm**: This algorithm is used to efficiently answer range queries by sorting the queries in a way that minimizes the number of element additions and removals when moving from one query to the next. This reduces the overall time complexity.
2. **Frequency Tracking**: For each query, maintain the frequency of elements in the current window and track the maximum frequency encountered. This helps in determining the minimum number of subsequences required.
3. **Dynamic Adjustment**: As elements are added or removed from the current window, dynamically update the frequency counts and adjust the maximum frequency to ensure accurate results.

### Solution Code
```python
import sys
import bisect
from random import randint

def main():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    q = int(input[ptr])
    ptr += 1
    a = list(map(int, input[ptr:ptr + n]))
    ptr += n

    queries = []
    for idx in range(q):
        l = int(input[ptr]) - 1  # converting to 0-based
        ptr += 1
        r = int(input[ptr]) - 1
        ptr += 1
        queries.append((l, r, idx))

    # Mo's algorithm setup
    block_size = int(n**0.5) + 1

    # Sort queries
    def mo_cmp(query):
        l, r, idx = query
        block = l // block_size
        return (block, r if block % 2 == 0 else -r)

    queries.sort(key=mo_cmp)

    max_a = max(a) if a else 0
    freq = [0] * (n + 2)  # Assuming elements are 1-based, but in the input they are up to n
    cnt_freq = [0] * (n + 2)
    current_max = 0
    current_l = 0
    current_r = -1  # initial window is empty
    answers = [0] * q

    for l, r, idx in queries:
        # Expand to l and r
        while current_l > l:
            current_l -= 1
            x = a[current_l]
            prev = freq[x]
            cnt_freq[prev] -= 1
            freq[x] += 1
            cnt_freq[prev + 1] += 1
            if prev + 1 > current_max:
                current_max = prev + 1
        while current_r < r:
            current_r += 1
            x = a[current_r]
            prev = freq[x]
            cnt_freq[prev] -= 1
            freq[x] += 1
            cnt_freq[prev + 1] += 1
            if prev + 1 > current_max:
                current_max = prev + 1
        while current_l < l:
            x = a[current_l]
            prev = freq[x]
            cnt_freq[prev] -= 1
            freq[x] -= 1
            cnt_freq[prev - 1] += 1
            if prev == current_max and cnt_freq[prev] == 0:
                current_max -= 1
            current_l += 1
        while current_r > r:
            x = a[current_r]
            prev = freq[x]
            cnt_freq[prev] -= 1
            freq[x] -= 1
            cnt_freq[prev - 1] += 1
            if prev == current_max and cnt_freq[prev] == 0:
                current_max -= 1
            current_r -= 1

        len_sub = r - l + 1
        res = max(1, 2 * current_max - len_sub)
        answers[idx] = res

    for ans in answers:
        print(ans)

if __name__ == "__main__":
    main()
```
