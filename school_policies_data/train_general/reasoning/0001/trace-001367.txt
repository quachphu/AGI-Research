def find_digits(n, k):

    remaining = n

    d = 1

    numbers_in_d = 9

    digits_per_number = d

    while remaining > numbers_in_d * digits_per_number:

        remaining -= numbers_in_d * digits_per_number

        d +=1

        numbers_in_d = 9 * (10 ** (d-1))

        digits_per_number = d

    start_number = 10 ** (d-1) + (remaining -1) // d

    digit_offset = (remaining -1) % d

    result = []

    current_number = start_number

    current_pos = digit_offset

    remaining_k = k

    while remaining_k >0:

        s = str(current_number)

        available = len(s) - current_pos

        if available >= remaining_k:

            result.append(s[current_pos : current_pos + remaining_k])

            remaining_k =0

        else:

            result.append(s[current_pos:])

            remaining_k -= available

            current_number +=1

            current_pos =0

    return ''.join(result)

3. **Collect K Digits**: Starting from the identified position, collect K digits by iterating through the numbers as necessary, which may span multiple numbers if K is large enough.
