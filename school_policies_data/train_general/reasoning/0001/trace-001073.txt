import itertools

n = int(input())
nums = [int(input()) for _ in range(n)]

min_l1 = float('inf')

    if 0 not in assign or 1 not in assign or 2 not in assign:

    sum_a = sum(nums[i] for i in range(n) if assign[i] == 0)
    sum_b = sum(nums[i] for i in range(n) if assign[i] == 1)

print(min_l1)
