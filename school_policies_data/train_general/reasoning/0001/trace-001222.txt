2. **Factorize**: Compute m = n / 3. We need to find all possible factor triplets (x, y, z) of m where each factor is at least 2.
3. **Generate Triplets**: Iterate through possible values of x, y, and z such that their product is m, ensuring they form valid dimensions for a cube.
4. **Check Constraints**: For each valid triplet, check if they satisfy the triangle inequality and if their sum is even.
5. **Count Permutations**: Count the number of valid permutations of each triplet based on their values.

import math

def main():
    import sys
    n = int(sys.stdin.readline())
    if n % 3 != 0:
        print(0)
        return
    m = n // 3
    # Early exit if the product cannot be achieved with a, b, c >=2
    if m < 8:
        print(0)
        return
    total = 0
    max_a = int(m ** (1/3)) + 2  # Adding 2 to handle potential floating point inaccuracies
    for a in range(2, max_a + 1):
        if m % a != 0:
            continue
        m_rest = m // a
        if m_rest < 2 * 2:  # b and c need to be at least 2 each
            continue
        max_b = int(math.isqrt(m_rest)) + 2
        for b in range(a, max_b + 1):
            if m_rest % b != 0:
                continue
            c = m_rest // b
            if c < b:
                continue
            # Check if a + b > c (since a <= b <= c)
            if a + b <= c:
                continue
            sum_abc = a + b + c
            if sum_abc % 2 != 0:
                continue
            # Calculate the number of permutations
            if a == b:
                if b == c:
                    count = 1
                else:
                    count = 3
            elif b == c:
                count = 3
            else:
                count = 6
            total += count
    print(total)

if __name__ == "__main__":
