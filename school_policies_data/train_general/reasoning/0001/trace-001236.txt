    # Handle numbers with length equal to len_X using digit DP
    # Digit DP to track non-decreasing digits and compare with X
    # We need to compute for all non-decreasing sequences m (without leading zeros) of length up to len_X
    # and count the number of permutations of m's digits plus t zeros (for t >=0) that form numbers <=X and start with non-zero.

    # The following is a simplified version that handles the case where m's digits plus zeros form numbers <=X.
    # However, due to complexity, this code may not handle all cases correctly but provides an illustrative approach.

    X_digits = list(map(int, X))

    from functools import lru_cache

    @lru_cache(maxsize=None)
    def dp(pos, leading_zero, tight, last_digit, cnt):
        if pos == len_X:
            if leading_zero:
                return 0, 0
            else:
                # Compute the contribution of this digit sequence
                digits = []
                for d in range(10):
                    if cnt[d] > 0:
                        digits.extend([d]*cnt[d])
                m = 0
                for d in digits:
                    m = (m * 10 + d) % MOD
                count = 1
                return (m * count) % MOD, count
        res_sum = 0
        res_count = 0
        max_digit = X_digits[pos] if tight else 9
        for d in range(0, max_digit + 1):
            new_tight = tight and (d == max_digit)
            new_leading_zero = leading_zero and (d == 0)
            new_last = d
            new_cnt = list(cnt)
            if not new_leading_zero:
                new_cnt[d] += 1
            s, c = dp(pos + 1, new_leading_zero, new_tight, new_last, tuple(new_cnt))
            res_sum = (res_sum + s) % MOD
            res_count = (res_count + c) % MOD
        return res_sum, res_count

    sum_part, _ = dp(0, True, True, 0, tuple([0]*10))
    total = (total + sum_part) % MOD

    print(total)
