```python
import sys
from collections import deque

sys.setrecursionlimit(1 << 25)

def main():
    n, l = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    pickup_lines = [sys.stdin.readline().strip() for _ in range(n)]
    
    # Build the Aho-Corasick automaton
    nodes = [{'children': {}, 'failure': 0, 'output': 0}]
    for i in range(n):
        s = pickup_lines[i]
        current = 0
        for c in s:
            if c not in nodes[current]['children']:
                nodes.append({'children': {}, 'failure': 0, 'output': 0})
                nodes[current]['children'][c] = len(nodes) - 1
            current = nodes[current]['children'][c]
        nodes[current]['output'] += a[i]
    
    # Build failure links using BFS
    queue = deque()
    for c in nodes[0]['children']:
        child = nodes[0]['children'][c]
        nodes[child]['failure'] = 0
        queue.append(child)
    
    while queue:
        current_idx = queue.popleft()
        current_node = nodes[current_idx]
        for c, child_idx in current_node['children'].items():
            fail = current_node['failure']
            while fail != 0 and c not in nodes[fail]['children']:
                fail = nodes[fail]['failure']
            child_failure = nodes[fail]['children'].get(c, 0)
            nodes[child_idx]['failure'] = child_failure
            nodes[child_idx]['output'] += nodes[child_failure]['output']
            queue.append(child_idx)
    
    # Precompute transitions for all nodes and characters
    transitions = [{} for _ in range(len(nodes))]
    for u in range(len(nodes)):
        for c in 'abcdefghijklmnopqrstuvwxyz':
            v = u
            while True:
                if c in nodes[v]['children']:
                    transitions[u][c] = nodes[v]['children'][c]
                    break
                elif v == 0:
                    transitions[u][c] = 0
                    break
                else:
                    v = nodes[v]['failure']
    
    # Build transition matrix M
    m = len(nodes)
    M = [[-float('inf')] * m for _ in range(m)]
    for u in range(m):
        for c in 'abcdefghijklmnopqrstuvwxyz':
            v = transitions[u][c]
            M[u][v] = max(M[u][v], nodes[v]['output'])
    
    # Matrix exponentiation functions
    def multiply(A, B):
        res = [[-float('inf')] * m for _ in range(m)]
        for i in range(m):
            for k in range(m):
                if A[i][k] == -float('inf'):
                    continue
                for j in range(m):
                    if B[k][j] == -float('inf'):
                        continue
                    res[i][j] = max(res[i][j], A[i][k] + B[k][j])
        return res
    
    def matrix_pow(mat, power):
        result = [[-float('inf')] * m for _ in range(m)]
        for i in range(m):
            result[i][i] = 0  # Identity for max operation
        while power > 0:
            if power % 2 == 1:
                result = multiply(result, mat)
            mat = multiply(mat, mat)
            power //= 2
        return result
    
    # Compute M^l and find the maximum value in the initial state's row
    mat_l = matrix_pow(M, l)
    max_happiness = max(mat_l[0])
    print(max_happiness)

if __name__ == "__main__":
    main()
```

### Explanation
1. **Automaton Construction**: We build a trie where each node represents a state corresponding to a prefix of a pickup line. Each node tracks transitions to other nodes, failure links, and the accumulated happiness from pickup lines ending at that node.
2. **Failure Links**: These links help the automaton efficiently backtrack to the longest valid suffix when a character mismatch occurs, ensuring all possible pickup lines are considered.
3. **Transition Matrix**: This matrix captures the maximum happiness gain achievable by transitioning between states using any character.
4. **Matrix Exponentiation**: By exponentiating the transition matrix, we efficiently compute the maximum happiness over L steps using properties of the (max, +) semiring, which is crucial given the large possible value of L. This approach ensures we handle up to \(10^{14}\) steps efficiently.
