class DSU:

    def __init__(self, size):

        self.parent = list(range(size+1))  # assuming 1-based indexing

        self.rank = [0]*(size+1)

        self.parity = [0]*(size+1)  # parity from node to parent

    def find(self, x):

        orig = x

        path = []

        # Find the root

        while self.parent[x] != x:

            path.append(x)

            x = self.parent[x]

        root = x

        # Now, compute the parity from orig to root

        p = 0

        for node in path:

            p ^= self.parity[node]

        # Now, path contains [orig, ...], parents up to root.

        # Compress the path by pointing all nodes to root and updating their parity.

        current_parity = 0

        for node in reversed(path):

            parent_node = self.parent[node]

            self.parent[node] = root

            old_parity = self.parity[node]

            # The parity from node to root is old_parity (node to parent_node) plus current_parity (parent_node to root)

            self.parity[node] = current_parity ^ old_parity

            current_parity = self.parity[node]

        return root, p

    def union(self, x, y, z):

        rx, px = self.find(x)

        ry, py = self.find(y)

        if rx == ry:

            return

        if self.rank[rx] < self.rank[ry]:

            rx, ry = ry, rx

            px, py = py, px

            z ^= 1  # since the required relation is px ^ py ^ new_parity == z

        self.parent[ry] = rx

        required_parity = (z - px - py) % 2

        self.parity[ry] = required_parity

        if self.rank[rx] == self.rank[ry]:

            self.rank[rx] += 1

### Approach
1. **Model the Problem with Parity Constraints**: Each constraint can be translated into a parity relationship between two cards. If the sum of two cards and a given integer is even, the parity (even or odd nature) of the two cards can be determined relative to each other.
2. **Union-Find Data Structure with Parity Tracking**: We use a Disjoint Set Union (DSU) structure with an added feature to track the parity of each node relative to its parent. This helps in efficiently merging sets and checking parity constraints.
3. **Path Compression and Union by Rank**: These optimizations ensure that the DSU operations are efficient, allowing the solution to handle large input sizes up to the given constraints.
4. **Count Connected Components**: After processing all constraints, the number of connected components in the DSU structure gives the minimum number of cards we need to check. Each component can be resolved by checking one card, as the constraints determine the rest.

import sys

def main():
    sys.setrecursionlimit(1 << 25)
    N, M = map(int, sys.stdin.readline().split())
    parent = list(range(N + 1))  # 1-based indexing
    rank = [0] * (N + 1)
    parity = [0] * (N + 1)  # Parity from node to parent

    def find(x):
        orig = x
        path = []
        # Find the root and collect the path
        while parent[x] != x:
            path.append(x)
            x = parent[x]
        root = x
        # Calculate the parity from orig to root
        p = 0
        for node in path:
            p ^= parity[node]
        # Compress the path and update parities
        current_parity = 0
        for node in reversed(path):
            orig_parent = parent[node]
            parent[node] = root
            old_parity = parity[node]
            new_parity = old_parity ^ current_parity
            parity[node] = new_parity
            current_parity = new_parity
        return root, p

    for _ in range(M):
        X, Y, Z = map(int, sys.stdin.readline().split())
        required = Z % 2
        rx, px = find(X)
        ry, py = find(Y)
        if rx != ry:
            # Calculate the required parity between roots
            req_par = (required - px - py) % 2
            # Union by rank
            if rank[rx] < rank[ry]:
                rx, ry = ry, rx
                px, py = py, px
                req_par = (required - px - py) % 2  # Recalculate if swapped
            parent[ry] = rx
            parity[ry] = req_par
            if rank[rx] == rank[ry]:
                rank[rx] += 1

    # Count the number of connected components
    count = 0
    for i in range(1, N + 1):
        if parent[i] == i:
            count += 1
    print(count)

if __name__ == '__main__':
    main()

3. **Union Operation**: When merging two sets, the union operation checks the parity constraints and adjusts the parent and parity values accordingly to maintain consistency.
4. **Counting Components**: After processing all constraints, the number of unique roots in the DSU structure is counted. Each root represents a connected component, and each component requires checking one card to determine all values.
