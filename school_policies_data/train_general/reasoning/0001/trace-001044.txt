### Approach

Specifically, the number of configurations for N units, S(N), can be expressed as S(N) = S(N-1) + S(N-M), with base cases S(0) = 1 and S(n) = 0 for n < 0.
2. **Matrix Exponentiation**: To efficiently compute the result for large N (up to 10^18), we use matrix exponentiation. This allows us to compute the Nth term of the recurrence relation in O(log N) time by representing the recurrence as a matrix multiplication problem.
3. **Transition Matrix**: Construct a transition matrix that represents the recurrence relation. This matrix is used to compute the Nth term through exponentiation.

MOD = 10**9 + 7

def create_transition_matrix(M):
    mat = [[0] * M for _ in range(M)]
    mat[0][0] = 1
    mat[0][M-1] = 1
    for i in range(1, M):
        mat[i][i-1] = 1
    return mat

def multiply(A, B, mod):
    M = len(A)
    result = [[0]*M for _ in range(M)]
    for i in range(M):
        for k in range(M):
            if A[i][k]:
                for j in range(M):
                    result[i][j] = (result[i][j] + A[i][k] * B[k][j]) % mod
    return result

def matrix_power(mat, power, mod):
    M = len(mat)
    result = [[0]*M for _ in range(M)]
    for i in range(M):
        result[i][i] = 1
    while power > 0:
        if power % 2 == 1:
            result = multiply(result, mat, mod)
        mat = multiply(mat, mat, mod)
        power = power // 2
    return result

def multiply_vector(vec, mat, mod):

    result = [0] * M
    for i in range(M):
        for k in range(M):
            result[i] = (result[i] + vec[k] * mat[k][i]) % mod
    return result
