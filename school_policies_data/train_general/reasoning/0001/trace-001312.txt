class SegmentTree:
    def __init__(self, size):
        self.n = size
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.data = [0] * (2 * self.size)
        self.lazy = [0] * (2 * self.size)
    
    def push(self, node, l, r):
        if self.lazy[node] !=0:
            self.data[node] += self.lazy[node]
            if l != r:
                self.lazy[2*node] += self.lazy[node]
                self.lazy[2*node+1] += self.lazy[node]
            self.lazy[node] =0
    
    def range_add(self, a, b, val, node=1, l=0, r=None):
        if r is None:
            r = self.size -1
        self.push(node, l, r)
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.lazy[node] += val
            self.push(node, l, r)
            return
        mid = (l +r)//2
        self.range_add(a, b, val, 2*node, l, mid)
        self.range_add(a, b, val, 2*node+1, mid+1, r)
        self.data[node] = max(self.data[2*node], self.data[2*node+1])
    
    def range_max(self, a, b, node=1, l=0, r=None):
        if r is None:
            r = self.size -1
        self.push(node, l, r)
        if a > r or b < l:
            return -float('inf')
        if a <= l and r <= b:
            return self.data[node]
        mid = (l +r)//2
        left = self.range_max(a, b, 2*node, l, mid)
        right = self.range_max(a, b, 2*node+1, mid+1, r)
        return max(left, right)

class SegmentTree:
    def __init__(self, size):
        self.n = size
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.data = [0] * (2 * self.size)
        self.lazy = [0] * (2 * self.size)
    
    def push(self, node, l, r):
        if self.lazy[node] != 0:
            self.data[node] += self.lazy[node]
            if l != r:
                self.lazy[2*node] += self.lazy[node]
                self.lazy[2*node+1] += self.lazy[node]
            self.lazy[node] = 0
    
    def range_add(self, a, b, val, node=1, l=0, r=None):
        if r is None:
            r = self.size - 1
        self.push(node, l, r)
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.lazy[node] += val
            self.push(node, l, r)
            return
        mid = (l + r) // 2
        self.range_add(a, b, val, 2*node, l, mid)
        self.range_add(a, b, val, 2*node+1, mid+1, r)
        self.data[node] = max(self.data[2*node], self.data[2*node+1])

To solve this problem, we need to select a subset of sweets such that none of Chisaki's friends cry, while maximizing the total deliciousness of the selected sweets. Each friend has a threshold price and a count limit for sweets priced at or above that threshold. The goal is to ensure that no friend's count limit is exceeded.

### Approach
1. **Problem Analysis**:
   - Each friend will cry if the number of sweets priced at or above their threshold exceeds their count limit.
   - The optimal subset of sweets should maximize the total deliciousness without violating any friend's constraints.

2. **Key Insights**:
   - **Sorting**: Both sweets and friends are sorted by their price thresholds. This allows efficient range queries and updates.
   - **Prefix Minimum Calculation**: For each friend, compute the minimum allowed count (threshold - 1) considering all previous friends. This helps in determining the most restrictive constraint for any given price threshold.

3. **Algorithm**:
   - **Segment Tree**: Use a segment tree to efficiently track and update the count of sweets in specific price ranges. This helps in maintaining the constraints dynamically as we process each sweet.
   - **Greedy Selection**: Process sweets in descending order of their deliciousness to maximize the total sum while adhering to constraints.

import bisect

class SegmentTree:
    def __init__(self, size):
        self.n = size
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.data = [0] * (2 * self.size)
        self.lazy = [0] * (2 * self.size)
    
    def push(self, node, l, r):
        if self.lazy[node] != 0:
            self.data[node] += self.lazy[node]
            if l != r:
                self.lazy[2*node] += self.lazy[node]
                self.lazy[2*node+1] += self.lazy[node]
            self.lazy[node] = 0
    
    def range_add(self, a, b, val, node=1, l=0, r=None):
        if r is None:
            r = self.size - 1
        self.push(node, l, r)
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.lazy[node] += val
            self.push(node, l, r)
            return
        mid = (l + r) // 2
        self.range_add(a, b, val, 2*node, l, mid)
        self.range_add(a, b, val, 2*node+1, mid+1, r)
        self.data[node] = max(self.data[2*node], self.data[2*node+1])
    
    def range_max(self, a, b, node=1, l=0, r=None):
        if r is None:
            r = self.size - 1
        self.push(node, l, r)
        if a > r or b < l:
            return -float('inf')
        if a <= l and r <= b:
            return self.data[node]
        mid = (l + r) // 2
        left = self.range_max(a, b, 2*node, l, mid)
        right = self.range_max(a, b, 2*node+1, mid+1, r)
        return max(left, right)

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    N, M = int(input[ptr]), int(input[ptr+1])
    ptr += 2
    
    sweets = []
    for _ in range(N):
        A = int(input[ptr])
        B = int(input[ptr+1])
        sweets.append((A, B))
        ptr += 2
    
    friends = []
    for _ in range(M):
        C = int(input[ptr])
        D = int(input[ptr+1])
        friends.append((C, D))
        ptr += 2
    
    if M == 0:
        print(sum(b for a, b in sweets))
        return
    
    C_list = [c for c, d in friends]
    D_list = [d for c, d in friends]
    
    # Compute prefix minimum of D_j - 1
    q = []
    current_min = D_list[0] - 1
    q.append(current_min)
    for j in range(1, M):
        current_min = min(current_min, D_list[j] - 1)
        q.append(current_min)
    
    # Sort sweets in descending order of B_i
    sweets.sort(key=lambda x: -x[1])
    
    # Build segment tree for friends
    st = SegmentTree(M)
    
    total = 0
    
    for A, B in sweets:
        j_max = bisect.bisect_right(C_list, A) - 1
        if j_max < 0:
            total += B
        else:
            current_max = st.range_max(0, j_max)
            new_max = current_max + 1
            if new_max <= q[j_max]:
                total += B
                st.range_add(0, j_max, 1)
    
    print(total)

if __name__ == "__main__":
    main()

3. **Prefix Minimum Calculation**: For each friend, compute the minimum allowed count considering all previous friends to determine the most restrictive constraint.
4. **Segment Tree**: A segment tree is used to dynamically track the count of sweets in specific price ranges and efficiently check constraints.
5. **Greedy Selection**: Sweets are processed in descending order of deliciousness. For each sweet, the maximum allowed count is checked using the segment tree, and the sweet is included if it does not violate any constraints.

This approach ensures that we efficiently check and update constraints while maximizing the total deliciousness of the selected sweets.
