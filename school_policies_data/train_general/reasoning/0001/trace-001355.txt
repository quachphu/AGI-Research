def main():
    import sys
    p, *rest = list(map(int, sys.stdin.read().split()))
    a = rest[:p]
    
    # Initialize coefficients of x^p - x mod p
    coeffs = [0] * (p + 1)
    coeffs[0] = 1  # x^p term
    coeffs[p-1] = (p - 1) % p  # x term (since -1 mod p = p-1)
    coeffs[p] = 0  # constant term
    
    sum_coeffs = [0] * p
    
    def synthetic_division(c, i, prime):
        n = len(c) - 1  # degree of dividend
        q = [0] * n
        for k in range(n):
            if k == 0:
                q[k] = c[k] % prime
            else:
                q[k] = (c[k] + q[k-1] * i) % prime
        return q
    
    for i in range(p):
        if a[i] == 0:
            continue
        # Compute quotient coefficients for (x^p - x) divided by (x - i)

    print(' '.join(map(str, sum_coeffs)))
