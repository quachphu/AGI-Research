2. **Extract Bracket Sequence**: For each command, extract the sequence of brackets from the text.
3. **Check Validity**: Check if the extracted bracket sequence forms a valid balanced sequence using a stack-based approach.
4. **Compute Maximum Depth**: If the sequence is valid, compute the maximum nesting depth which determines the minimal number of colors.

import sys

def main():
    n = int(sys.stdin.readline())
    s = sys.stdin.readline().strip()
    
    text = {}
    cursor = 0
    brackets = []
    
    output = []
    
    for cmd in s:
        if cmd == 'L':
            cursor = max(0, cursor - 1)
        elif cmd == 'R':
            cursor += 1
        else:
            prev_char = text.get(cursor, ' ')
            if prev_char in '()':
                left = 0
                right = len(brackets) - 1
                pos_to_remove = -1
                while left <= right:
                    mid = (left + right) // 2
                    if brackets[mid][0] == cursor:
                        pos_to_remove = mid
                        break
                    elif brackets[mid][0] < cursor:
                        left = mid + 1
                    else:
                        right = mid - 1
                if pos_to_remove != -1:
                    brackets.pop(pos_to_remove)
            if cmd in '()':
                left = 0
                right = len(brackets) - 1
                insert_pos = 0
                while left <= right:
                    mid = (left + right) // 2
                    if brackets[mid][0] < cursor:
                        left = mid + 1
                    else:
                        right = mid - 1
                insert_pos = left
                brackets.insert(insert_pos, (cursor, cmd))
            text[cursor] = cmd
        
        valid = True

        current_depth = 0
        total = 0
        for pos, char in brackets:
            if char == '(':
                current_depth += 1
                total += 1
            else:
                current_depth -= 1
                total -= 1
            if current_depth < 0:
                valid = False
