To solve this problem, we need to determine the optimal road to add between two special fields such that the shortest path from field 1 to field n is maximized. The solution involves calculating the shortest paths from both ends of the graph and leveraging these distances to find the optimal road.

### Approach
1. **Compute Shortest Paths**: Use Breadth-First Search (BFS) to compute the shortest distances from field 1 (d1) and from field n (dn) to all other fields.
2. **Collect Special Fields**: Extract the special fields and their respective shortest distances from both ends.
3. **Sort Special Fields**: Sort the special fields based on the difference between their shortest distances from field 1 and field n (d1 - dn).
4. **Precompute Maximum Values**: Precompute suffix and prefix arrays to efficiently find the maximum possible values of the shortest paths when adding a new road.
5. **Evaluate Pairs**: For each special field, determine the best possible pair to form a new road by evaluating the precomputed maximum values.

### Solution Code
```python
import sys
from collections import deque

def main():
    n, m, k = map(int, sys.stdin.readline().split())
    special = list(map(int, sys.stdin.readline().split()))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        x, y = map(int, sys.stdin.readline().split())
        adj[x].append(y)
        adj[y].append(x)
    
    # Compute d1 (distance from 1)
    d1 = [-1] * (n + 1)
    d1[1] = 0
    q = deque([1])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if d1[v] == -1:
                d1[v] = d1[u] + 1
                q.append(v)
    
    # Compute dn (distance from n)
    dn = [-1] * (n + 1)
    dn[n] = 0
    q = deque([n])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if dn[v] == -1:
                dn[v] = dn[u] + 1
                q.append(v)
    
    # Process special nodes
    specs = []
    for a in special:
        specs.append((d1[a], dn[a]))
    
    # Sort by c_i = a_i - b_i
    specs_sorted = sorted(specs, key=lambda x: x[0] - x[1])
    k_sorted = len(specs_sorted)
    if k_sorted < 2:
        print(d1[n])
        return
    
    # Precompute suffix_max_b and suffix_second_max_b
    suffix_max_b = [-float('inf')] * k_sorted
    suffix_second_max_b = [-float('inf')] * k_sorted
    current_max_b = -float('inf')
    current_second_b = -float('inf')
    for i in range(k_sorted - 1, -1, -1):
        b = specs_sorted[i][1]
        if b > current_max_b:
            current_second_b = current_max_b
            current_max_b = b
        elif b > current_second_b:
            current_second_b = b
        suffix_max_b[i] = current_max_b
        suffix_second_max_b[i] = current_second_b
    
    # Precompute prefix_max_a and prefix_second_max_a
    prefix_max_a = [-float('inf')] * k_sorted
    prefix_second_max_a = [-float('inf')] * k_sorted
    current_max_a = -float('inf')
    current_second_a = -float('inf')
    for i in range(k_sorted):
        a = specs_sorted[i][0]
        if a > current_max_a:
            current_second_a = current_max_a
            current_max_a = a
        elif a > current_second_a:
            current_second_a = a
        prefix_max_a[i] = current_max_a
        prefix_second_max_a[i] = current_second_a
    
    global_max = -float('inf')
    for i in range(k_sorted):
        a_u, b_u = specs_sorted[i]
        c_u = a_u - b_u
        
        # Binary search for the first index j where c_v >= c_u
        left, right = 0, k_sorted
        while left < right:
            mid = (left + right) // 2
            a_mid, b_mid = specs_sorted[mid]
            c_mid = a_mid - b_mid
            if c_mid >= c_u:
                right = mid
            else:
                left = mid + 1
        j = left
        
        # Case 1: v in suffix [j ... k_sorted-1]
        case1 = -float('inf')
        if j <= k_sorted - 1:
            max_b = suffix_max_b[j]
            if i >= j:
                if max_b == b_u:
                    current_b = suffix_second_max_b[j] if suffix_second_max_b[j] != -float('inf') else -float('inf')
                else:
                    current_b = max_b
            else:
                current_b = max_b
            if current_b != -float('inf'):
                case1 = a_u + current_b
        
        # Case 2: v in prefix [0 ... j-1]
        case2 = -float('inf')
        if j >= 1:
            max_a = prefix_max_a[j-1]
            if i < j:
                if max_a == a_u:
                    current_a = prefix_second_max_a[j-1] if (j-1 >=0 and prefix_second_max_a[j-1] != -float('inf')) else -float('inf')
                else:
                    current_a = max_a
            else:
                current_a = max_a
            if current_a != -float('inf'):
                case2 = current_a + b_u
        
        current_max = max(case1, case2)
        if current_max > global_max:
            global_max = current_max
    
    original_S = d1[n]
    candidate = global_max + 1
    answer = min(original_S, candidate)
```
