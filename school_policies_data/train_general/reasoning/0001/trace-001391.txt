To solve this problem, we need to determine the cheapest way to replace each '?' in a given bracket sequence with either '(' or ')', such that the resulting sequence is a regular bracket sequence. A regular bracket sequence is one that can be balanced, meaning every opening bracket has a corresponding closing bracket in the correct order.

### Approach
1. **Input Parsing and Initial Checks**: Read the input pattern and the costs for replacing each '?'. Check if the length of the pattern is even and compute the required number of '(' replacements (x) needed to balance the sequence.
2. **Greedy Replacement with Priority Queue**: Use a priority queue (min-heap) to efficiently determine the cheapest replacements. Initially, replace all '?' with ')', and track the balance of the sequence. If the balance becomes negative, flip the cheapest possible previous '?' from ')' to '(' to maintain a non-negative balance.
3. **Adjust for Required Replacements**: After ensuring the sequence is valid, adjust the number of '(' replacements to meet the exact requirement (x) by flipping the remaining cheapest possible '?'s.
4. **Construct the Result**: Generate the final sequence based on the replacements made and output the total cost and the sequence.

### Solution Code
```python
import heapq

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    pattern = input[ptr]
    ptr += 1
    m = pattern.count('?')
    q = []
    for _ in range(m):
        a = int(input[ptr])
        b = int(input[ptr + 1])
        ptr += 2
        q.append((a, b))
    
    fixed_open = pattern.count('(')
    fixed_close = pattern.count(')')
    total_m = m
    sum_diff = fixed_close - fixed_open + total_m
    if sum_diff % 2 != 0:
        print(-1)
        return
    x = sum_diff // 2
    if x < 0 or x > total_m:
        print(-1)
        return
    
    heap = []
    flipped = [False] * m
    total_cost = 0
    cnt_flips = 0
    balance = 0
    current_q_idx = 0
    q_ptr = 0  # Tracks the index in q for each '?' encountered
    
    for c in pattern:
        if c == '(':
            balance += 1
        elif c == ')':
            balance -= 1
        else:
            a_i, b_i = q[q_ptr]
            delta = a_i - b_i
            heapq.heappush(heap, (delta, q_ptr))
            total_cost += b_i
            balance -= 1
            q_ptr += 1
        
        # Fix balance if it goes negative
        while balance < 0:
            if not heap:
                print(-1)
                return
            delta, idx = heapq.heappop(heap)
            total_cost += delta
            cnt_flips += 1
            flipped[idx] = True
            balance += 2
    
    if balance < 0:
        print(-1)
        return
    
    if cnt_flips > x:
        print(-1)
        return
    
    # Flip remaining required
    k = x - cnt_flips
    if k > 0:
        if len(heap) < k:
            print(-1)
            return
        for _ in range(k):
            delta, idx = heapq.heappop(heap)
            total_cost += delta
            flipped[idx] = True
            cnt_flips += 1
    
    # Construct the result
    res = []
    q_ptr = 0
    for c in pattern:
        if c != '?':
            res.append(c)
        else:
            if flipped[q_ptr]:
                res.append('(')
            else:
                res.append(')')
            q_ptr += 1
    
    print(total_cost)
    print(''.join(res))

if __name__ == "__main__":
    main()
```
