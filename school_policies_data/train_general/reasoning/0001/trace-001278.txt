def build_tree(adj, root):
    parent = [0] * (n + 1)  # assuming nodes are 1-based
    children = [[] for _ in range(n + 1)]
    visited = [False] * (n + 1)
    stack = [root]
    visited[root] = True
    while stack:
        u = stack.pop()
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                children[u].append(v)
                stack.append(v)
    return children

    n = int(sys.stdin.readline())
    adj = [[] for _ in range(n+1)]

        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    input_points = [tuple(map(int, sys.stdin.readline().split())) for _ in range(n)]
    leftmost_point = min(input_points, key=lambda p: (p[0], p[1]))
    leftmost_idx = input_points.index(leftmost_point)

    for root_node in range(1, n+1):

        children = [[] for _ in range(n+1)]
        visited = [False]*(n+1)

        parent = [0]*(n+1)
        while stack:
            u = stack.pop()
            for v in adj[u]:

                    parent[v] = u
                    children[u].append(v)
                    stack.append(v)

                return

To solve this problem, we need to embed a tree structure onto given points on a plane such that edges do not cross. The key challenge is to map each tree node to a point such that edges between nodes do not intersect, except at their common vertices.

### Approach
1. **Identify the Leftmost Point**: Start by finding the leftmost point among the given points. This point will be used as the root of our tree to ensure a stable starting position for embedding.
2. **Tree Construction**: For each possible root node (candidate), construct the tree and compute the subtree sizes using Depth-First Search (DFS).
3. **Recursive Assignment**: Recursively assign points to each subtree. For each node, select the leftmost point from the available points, sort the remaining points in counter-clockwise (CCW) order around this point, and distribute the points among the node's children based on their subtree sizes.

### Solution Code
```python
import sys
import math

def main():
    sys.setrecursionlimit(1 << 25)
    n = int(sys.stdin.readline())
    adj = [[] for _ in range(n+1)]  # Using 1-based indexing
    for _ in range(n-1):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    input_points = [tuple(map(int, sys.stdin.readline().split())) for _ in range(n)]
    
    leftmost_point = min(input_points, key=lambda p: (p[0], p[1]))
    leftmost_idx = input_points.index(leftmost_point)

    # Iterate over all possible root nodes to find a valid assignment
    for root_node in range(1, n+1):
        # Build the tree with root_node as root
        children = [[] for _ in range(n+1)]
        visited = [False] * (n+1)
        stack = [root_node]
        visited[root_node] = True
        parent = [0] * (n+1)
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parent[v] = u
                    children[u].append(v)
                    stack.append(v)
        
        # Compute subtree sizes
        subtree_sizes = [1] * (n+1)
        valid_dfs = True
        try:
            def dfs(u):
                for v in children[u]:
                    dfs(v)
                    subtree_sizes[u] += subtree_sizes[v]
            dfs(root_node)
        except:
            valid_dfs = False
        
        if not valid_dfs:
            continue
        
        # Attempt to assign points
        assignment = {}
        valid_assignment = [True]

        def assign(u, points):
            if not valid_assignment[0]:
                return
            if not points:
                valid_assignment[0] = False
                return
            # Find the leftmost point in the current points
            leftmost = min(points, key=lambda p: (p[0], p[1]))
            idx = points.index(leftmost)
            current_point = points.pop(idx)
            assignment[u] = current_point
            # Sort remaining points in CCW order around current_point
            cx, cy = current_point
            remaining = sorted(points, key=lambda p: (math.atan2(p[1]-cy, p[0]-cx)))
            # Process each child
            start = 0
            for v in children[u]:
                size = subtree_sizes[v]
                if start + size > len(remaining):
                    valid_assignment[0] = False
                    return
                child_points = remaining[start:start+size]
                start += size
                assign(v, child_points)
                if not valid_assignment[0]:
                    return
        
        all_points = input_points.copy()
        assign(root_node, all_points)
        if valid_assignment[0] and len(assignment) == n:
            # Build the output
            point_to_node = {point: node for node, point in assignment.items()}
            output = []
            for p in input_points:
                output.append(str(point_to_node[p]))
            print(' '.join(output))
            return
    
    # If no solution found (which shouldn't happen per problem statement)
    print(" ".join(['1'] * n))

if __name__ == "__main__":
    main()
```

3. **Recursive Assignment**: Starting from the root node, we recursively assign the leftmost point to the current node, sort the remaining points in CCW order, and distribute these points to the node's children based on their subtree sizes. This ensures that each subtree is placed in a region that avoids edge crossings.

This approach guarantees a valid embedding by leveraging geometric sorting and subtree size partitioning, ensuring that edges do not intersect except at their common vertices.
