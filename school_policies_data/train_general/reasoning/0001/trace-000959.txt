import sys

def main():
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1
    points = []
    for _ in range(m):
        x = int(input[ptr])
        ptr += 1
        y = int(input[ptr])
        ptr += 1
        points.append((x, y))
    
    max_count = [1] * (n + 1)  # 1-based indexing
    
    processed = set()
    
    def gcd(a, b):
        while b:
            a, b = b, a % b
        return a
    
    for i in range(m):
        x1, y1 = points[i]
        for j in range(i + 1, m):
            x2, y2 = points[j]
            dx = x2 - x1
            dy = y2 - y1
            A = dy
            B = -dx
            C = dx * y1 - dy * x1
            
            # Compute GCD of A, B, C
            g = gcd(gcd(abs(A), abs(B)), abs(C))
            if g == 0:
                g = 1
            A_div = A // g
            B_div = B // g
            C_div = C // g
            
            # Determine the sign based on the first non-zero coefficient
            first_non_zero = 0
            while first_non_zero < 2:
                if first_non_zero == 0 and A_div != 0:
                    if A_div < 0:
                        A_div = -A_div
                        B_div = -B_div
                        C_div = -C_div
                    break
                elif first_non_zero == 1 and B_div != 0:
                    if B_div < 0:
                        B_div = -B_div
                        C_div = -C_div
                    break
                first_non_zero += 1
            
            line_key = (A_div, B_div, C_div)
            if line_key in processed:
                continue
            processed.add(line_key)
            
            # Count the number of points on this line
            t = 0
            for (x, y) in points:
                if A_div * x + B_div * y + C_div == 0:
                    t += 1
            if t < 2:
                continue
            
            # Compute x-intercept k
            if B_div == 0:
                # Vertical line, A_div is 1 after normalization
                k = -C_div // A_div
            else:
                # Check if -C_div is divisible by A_div
                if A_div == 0:
                    continue  # should not happen as B_div is non-zero
                numerator = -C_div
                denominator = A_div
                if denominator == 0:
                    continue
                if numerator % denominator != 0:
                    continue
                k = numerator // denominator
            
            if 1 <= k <= n:
                if max_count[k] < t:
                    max_count[k] = t
    
    total = sum(max_count[1:n+1])
    print(total)

if __name__ == "__main__":
    main()
