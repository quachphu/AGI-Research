def min_cost_flow(graph, s, t, maxf):
    n = len(graph)
    res = 0
    h = [0] * n  # potential
    prevv = [0] * n
    preve = [0] * n
    INF = float('inf')
    flow = 0
    cost = 0
    while flow < maxf:
        # Dijkstra to find shortest path in residual graph
        dist = [INF] * n
        dist[s] = 0
        from heapq import heappush, heappop
        heap = []
        heappush(heap, (0, s))
        while heap:
            d, v = heappop(heap)
            if d > dist[v]:
                continue
            for i, e in enumerate(graph[v]):
                if e.capacity > 0 and dist[e.to] > d + e.cost + h[v] - h[e.to]:
                    dist[e.to] = d + e.cost + h[v] - h[e.to]
                    prevv[e.to] = v
                    preve[e.to] = i
                    heappush(heap, (dist[e.to], e.to))
        if dist[t] == INF:
            return -1  # no more augmenting paths
        # update potential
        for v in range(n):
            h[v] += dist[v] if dist[v] < INF else 0
        # find minimum capacity on the path
        d = maxf - flow
        v = t
        while v != s:
            d = min(d, graph[prevv[v]][preve[v]].capacity)
            v = prevv[v]
        flow += d
        cost += d * h[t]
        # update residual capacities
        v = t
        while v != s:
            e = graph[prevv[v]][preve[v]]
            e.capacity -= d
            graph[v][e.rev].capacity += d
            v = prevv[v]
    return cost

def add_edge(graph, fr, to, capacity, cost):
    graph[fr].append(Edge(to, len(graph[to]), capacity, cost))
    graph[to].append(Edge(fr, len(graph[fr])-1, 0, -cost))

def min_cost_flow(graph, s, t, maxf):
    n = len(graph)
    res = 0
    h = [0] * n  # potential
    prevv = [0] * n
    preve = [0] * n
    INF = float('inf')
    flow = 0
    cost = 0
    while flow < maxf:
        dist = [INF] * n
        dist[s] = 0
        heap = []
        heapq.heappush(heap, (0, s))
        while heap:
            d, v = heapq.heappop(heap)
            if d > dist[v]:
                continue
            for i, e in enumerate(graph[v]):
                if e.capacity > 0 and dist[e.to] > d + e.cost + h[v] - h[e.to]:
                    dist[e.to] = d + e.cost + h[v] - h[e.to]
                    prevv[e.to] = v
                    preve[e.to] = i
                    heapq.heappush(heap, (dist[e.to], e.to))
        if dist[t] == INF:
            return -1
        for v in range(n):
            if dist[v] < INF:
                h[v] += dist[v]
        d = maxf - flow
        v = t
        while v != s:
            d = min(d, graph[prevv[v]][preve[v]].capacity)
            v = prevv[v]
        flow += d
        cost += d * h[t]
        v = t
        while v != s:
            e = graph[prevv[v]][preve[v]]
            e.capacity -= d
            graph[v][e.rev].capacity += d
            v = prevv[v]
    return cost if flow == maxf else -1

def generate_buys(remaining, L):
    K = len(remaining)
    buys = []
    stack = [(0, [], 0)]
    while stack:
        idx, current, sum_so_far = stack.pop()
        if idx == K:
            if sum_so_far <= L:
                buys.append(tuple(current))
            continue
        max_buy = min(remaining[idx], L - sum_so_far)
        for buy in range(max_buy + 1):
            new_current = current + [buy]
            new_sum = sum_so_far + buy
            stack.append((idx+1, new_current, new_sum))
    return buys
