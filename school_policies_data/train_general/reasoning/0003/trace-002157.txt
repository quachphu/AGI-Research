    # Generate the permutation P after one swap
    indices = list(range(L))
    m = (L - 1) // 2
    for i in range(m, 0, -1):
        pos = L - i
        c = indices[pos]
        # Remove the character at pos
        indices = indices[:pos] + indices[pos+1:]
