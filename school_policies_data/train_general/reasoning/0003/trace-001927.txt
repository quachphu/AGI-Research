import sys

def main():
    n = int(sys.stdin.readline())
    intervals = []
    for _ in range(n):
        sa, ea, sb, eb = map(int, sys.stdin.readline().split())
        intervals.append(((sa, ea), (sb, eb)))

    # Check case 1: overlapping in A but not in B
    # Sort by A's start time
    sorted_A = sorted(intervals, key=lambda x: x[0][0])
    active = []
    for a, b in sorted_A:
        sa_a, ea_a = a
        sb_i, eb_i = b
        # Remove intervals that end before the current start
        while active and active[0][0] < sa_a:
            active.pop(0)
        # Check all active intervals
        for (ea_j_prev, (sb_j, eb_j)) in active:
            if max(sb_i, sb_j) > min(eb_i, eb_j):
                print("NO")
                return
        # Add current interval's A end and B interval
        active.append((ea_a, (sb_i, eb_i)))
        # Keep active sorted by ea_a for efficient removal
        active.sort()

    # Check case 2: overlapping in B but not in A
    # Sort by B's start time
    sorted_B = sorted(intervals, key=lambda x: x[1][0])
    active = []
    for a, b in sorted_B:
        sb_b, eb_b = b

            active.pop(0)
