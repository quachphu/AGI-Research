        if remaining == 1:

import sys

def generate_factors(n):
    result = []
    def backtrack(remaining, max_factor, current):
        if remaining == 1:
            result.append(current.copy())
            return
        start = min(remaining, max_factor)
        for d in range(start, 1, -1):
            if remaining % d == 0:
                current.append(d)
                backtrack(remaining // d, d, current)
                current.pop()
    backtrack(n, n, [])
    return result

def get_primes(k):
    primes = []
    num = 2
    while len(primes) < k:
        if all(num % p != 0 for p in primes):
            primes.append(num)
        num += 1
    return primes

def find_min_number(n):
    if n == 1:
        return 1
    factor_sequences = generate_factors(n)
    min_num = float('inf')
    for seq in factor_sequences:
        exponents = [d - 1 for d in seq]
        k = len(exponents)
        primes = get_primes(k)
        product = 1
        valid = True
        for i in range(k):
            product *= primes[i] ** exponents[i]
            if product > 1e18:
                valid = False
                break
        if valid and product < min_num:
            min_num = product
    return min_num

n = int(sys.stdin.readline())
print(find_min_number(n))
