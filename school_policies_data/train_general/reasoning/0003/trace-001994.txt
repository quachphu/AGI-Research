class DSU:

    def __init__(self, n):

        self.parent = list(range(n+1))  # nodes are 1-based?

    def find(self, x):

        if self.parent[x] != x:

            self.parent[x] = self.find(self.parent[x])

        return self.parent[x]

    def union(self, x, y):

        x_root = self.find(x)

        y_root = self.find(y)

To solve this problem, we need to maximize the total beauty of Hoses Mehrdad can invite to the party without exceeding the weight capacity of the amphitheater. The constraint is that from each friendship group, either all members must be invited or at most one member can be invited.

### Approach
1. **Identify Friendship Groups**: Use Disjoint Set Union (DSU) to group Hoses into connected components based on their friendships.
2. **Dynamic Programming (DP) Setup**: Use a DP array to track the maximum beauty achievable for each possible weight up to the given limit.
3. **Process Each Group**:
   - **Option 1**: Consider inviting the entire group if the total weight does not exceed the capacity.
   - **Option 2**: Consider inviting at most one member from the group, which involves checking each member individually to maximize beauty without exceeding the weight limit.

### Solution Code

class DSU:
    def __init__(self, n):
        self.parent = list(range(n + 1))  # 1-based indexing

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root != y_root:
            self.parent[y_root] = x_root

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1
    w = int(input[ptr])
    ptr += 1

    weights = list(map(int, input[ptr:ptr+n]))
    ptr += n
    beauties = list(map(int, input[ptr:ptr+n]))
    ptr += n

    dsu = DSU(n)
    for _ in range(m):
        x = int(input[ptr])
        ptr += 1
        y = int(input[ptr])
        ptr += 1
        dsu.union(x, y)

    groups = {}
    for i in range(1, n+1):
        root = dsu.find(i)
        if root not in groups:
            groups[root] = []
        groups[root].append(i)

    groups_list = []
    for group in groups.values():
        sum_w = 0
        sum_b = 0
        members = []
        for idx in group:
            sum_w += weights[idx-1]
            sum_b += beauties[idx-1]
            members.append((weights[idx-1], beauties[idx-1]))
        groups_list.append((sum_w, sum_b, members))

    dp = [-float('inf')] * (w + 1)
    dp[0] = 0

    for sum_w_group, sum_b_group, members in groups_list:
        prev_dp = dp.copy()
        # Option 1: take the entire group
        for j in range(w + 1):
            if prev_dp[j] != -float('inf'):
                new_j = j + sum_w_group
                if new_j <= w and dp[new_j] < prev_dp[j] + sum_b_group:
                    dp[new_j] = prev_dp[j] + sum_b_group
        # Option 2: take at most one member
        temp_dp = [-float('inf')] * (w + 1)
        for j in range(w + 1):
            current_max = prev_dp[j]
            for wi, bi in members:
                if j >= wi and prev_dp[j - wi] + bi > current_max:

            temp_dp[j] = current_max
        # Update dp to the maximum between existing and temp_dp
        for j in range(w + 1):

                dp[j] = temp_dp[j]
