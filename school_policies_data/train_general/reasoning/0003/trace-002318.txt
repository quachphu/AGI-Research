def main():
    import sys
    input = sys.stdin.read().split()
    T = int(input[0])
    cases = input[1:T+1]

            continue

                    if total > n - 1:
                        continue
                    part_a = s[:a]
                    part_b = s[a:a+b]
                    part_c = s[a+b:a+b+c]
                    part_d = s[a+b+c:]
                    if not part_d:
                        continue
                    if (part_a != part_b and part_a != part_c and part_a != part_d and
                        part_b != part_c and part_b != part_d and
                        part_c != part_d):
                        possible = True
                        break
                if possible:
                    break
            if possible:
                break
        print("YES" if possible else "NO")

if __name__ == "__main__":
    main()
