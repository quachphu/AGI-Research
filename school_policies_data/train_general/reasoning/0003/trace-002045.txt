def matrix_mult(a, b, M, mod):
    result = [[0]*M for _ in range(M)]
    for i in range(M):
        for k in range(M):
            if a[i][k] ==0:
                continue
            for j in range(M):
                result[i][j] = (result[i][j] + a[i][k] * b[k][j]) % mod
    return result

def matrix_power(mat, power, M, mod):
    # Initialize result as identity matrix
    result = [[0]*M for _ in range(M)]
    for i in range(M):
        result[i][i] = 1
    while power >0:
        if power %2 ==1:
            result = matrix_mult(result, mat, M, mod)
        mat = matrix_mult(mat, mat, M, mod)
        power = power //2
    return result

def build_transition_matrix(freq, M):

    for i in range(M):

            T[i][j] = freq[delta]

def matrix_mult(a, b, M, mod):
    result = [[0] * M for _ in range(M)]
    for i in range(M):
        for k in range(M):
            if a[i][k]:
                for j in range(M):
                    result[i][j] = (result[i][j] + a[i][k] * b[k][j]) % mod
    return result

def matrix_power(mat, power, M, mod):
    result = [[0] * M for _ in range(M)]
    for i in range(M):
        result[i][i] = 1
    while power > 0:
        if power % 2 == 1:
            result = matrix_mult(result, mat, M, mod)
        mat = matrix_mult(mat, mat, M, mod)
        power //= 2
    return result

def multiply_matrix_vector(mat, vec, M, mod):

            new_vec[j] = (new_vec[j] + vec[i] * mat[i][j]) % mod
