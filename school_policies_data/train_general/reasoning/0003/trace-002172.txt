To solve this problem, we need to efficiently compute the sum of D(N) for all numbers N in the range [A, B], where D(N) is the last digit of the sum of the odd digits of N plus twice the sum of the even digits of N. Given the constraints, a brute-force approach is infeasible, so we use a digit dynamic programming (DP) approach to optimize the solution.

### Approach
1. **Understanding D(N)**: For a number N, D(N) is the last digit of S(N), where S(N) is the sum of the odd digits plus twice the sum of the even digits. This can be computed as `(sum_odd + 2 * sum_even) % 10`.

2. **Digit DP**: We use digit DP to efficiently compute the sum of D(N) for all numbers up to a given number X. This involves breaking down the problem into smaller subproblems based on the digits of X.

3. **Summing Over Ranges**: For each test case, compute the sum from 0 to B and subtract the sum from 0 to A-1. This requires two main functions: one to compute the sum for numbers with exactly m digits and another to handle numbers with the same number of digits as X but less than or equal to X.

            contribution = d % 10 if d % 2 else (2 * d) % 10
            prev_dp[contribution] += 1
    else:
        for d in range(1, 10):
            contribution = d % 10 if d % 2 else (2 * d) % 10
            prev_dp[contribution] += 1
    for pos in range(1, m):
        current_dp = defaultdict(int)
        for sum_mod, count in prev_dp.items():
            for d in range(0, 10):
                new_contrib = d if d % 2 else (2 * d) % 10
                new_sum = (sum_mod + new_contrib) % 10
                current_dp[new_sum] += count
        prev_dp = current_dp
    total = sum(sum_mod * count for sum_mod, count in prev_dp.items())
    return total

def compute_sum_tight(X_str):
    n = len(X_str)
    X = list(map(int, X_str))
    dp = [defaultdict(int) for _ in range(2)]
    if n == 1:
        max_d = X[0]
        for d in range(0, max_d + 1):
            contrib = d if d % 2 else (2 * d) % 10
            tight = d == max_d
            dp[tight][contrib % 10] += 1
    else:
        max_d = X[0]
        for d in range(1, max_d + 1):
            contrib = d if d % 2 else (2 * d) % 10
            tight = d == max_d
            dp[tight][contrib % 10] += 1
    for pos in range(1, n):
        next_dp = [defaultdict(int) for _ in range(2)]
        for tight in [0, 1]:
            if not dp[tight]:
                continue
            current_max = X[pos] if tight else 9
            for sum_mod, count in dp[tight].items():
                for d in range(0, current_max + 1):
                    new_tight = tight and (d == current_max)
                    contrib = d if d % 2 else (2 * d) % 10
                    new_sum = (sum_mod + contrib) % 10
                    next_dp[new_tight][new_sum] += count
        dp = next_dp
    total = 0
    for tight_dp in dp:
        for sum_mod, count in tight_dp.items():
            total += sum_mod * count
    return total

def compute_sum(X):
    if X < 0:
        return 0
    X_str = str(X)
    n = len(X_str)
    sum_less = sum(compute_sum_m_digits(m) for m in range(1, n))
    sum_eq = compute_sum_tight(X_str)
