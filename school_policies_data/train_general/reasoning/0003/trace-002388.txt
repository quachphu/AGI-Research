To solve this problem, we need to simulate the movement of ants on a chessboard and determine the order in which they fall off the board, considering their interactions when they collide. The key challenge is efficiently handling collisions and direction changes without explicitly simulating every step due to the large possible dimensions of the board.

### Approach
1. **Initial Setup**: Read the input values and store each ant's position, direction, and initial exit time.
2. **Collision Detection**: For each possible collision between ants moving east (E) and south (S) in the same group (defined by the sum of their coordinates), determine if the collision occurs on a white square (even sum) and compute the collision time.
3. **Swap Handling**: If a collision on a white square is valid (happens before either ant falls off), swap the ants' numbers to simulate the direction change effect.
4. **Sorting**: Sort the ants by their exit times and their numbers to determine the final order in which they fall off the board.

### Solution Code
```python
def main():
    import sys
    input = sys.stdin.read().split()
    idx = 0
    W = int(input[idx]); idx += 1
    H = int(input[idx]); idx += 1
    N = int(input[idx]); idx += 1

    ants = []
    for i in range(N):
        x = int(input[idx]); idx += 1
        y = int(input[idx]); idx += 1
        d = input[idx]; idx += 1
        s = x + y
        if d == 'E':
            exit_time = (W - x) + 1
        else:
            exit_time = (H - y) + 1
        ants.append((exit_time, i + 1, x, y, d, s))  # Include s for grouping

    # Group ants by s where s is even
    from collections import defaultdict
    groups = defaultdict(lambda: {'E': [], 'S': []})
    for ant in ants:
        exit_time, num, x, y, d, s = ant
        if s % 2 == 0:
            if d == 'E':
                groups[s]['E'].append((y, exit_time, num))
            else:
                groups[s]['S'].append((y, exit_time, num))

    # Now process each group to find possible swaps
    swapped = list(range(N + 1))  # swapped[i] is the current number of ant i (1-based)
    for i in range(1, N + 1):
        swapped[i] = i

    for s in groups:
        east_list = sorted(groups[s]['E'], key=lambda x: x[0])  # sorted by y
        south_list = sorted(groups[s]['S'], key=lambda x: x[0])  # sorted by y

        # For each pair (east, south) where east.y >= south.y and valid t
        for (y_a, exit_a, num_a) in east_list:
            for (y_b, exit_b, num_b) in south_list:
                if y_a < y_b:
                    continue  # t would be negative
                t = y_a - y_b
                if t < exit_a and t < exit_b:
                    # Swap the numbers
                    swapped[num_a], swapped[num_b] = swapped[num_b], swapped[num_a]

    # Now, create a list of ants with their swapped numbers and exit times
    final_ants = []
    for ant in ants:
        exit_time, num, x, y, d, s = ant
        final_ants.append((exit_time, swapped[num]))

    # Sort by exit_time, then by number
    final_ants.sort(key=lambda x: (x[0], x[1]))

    for exit_time, num in final_ants:
        print(num)

if __name__ == '__main__':
    main()
```
