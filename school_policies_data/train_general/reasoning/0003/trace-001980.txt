class LCA:
    def __init__(self, tree, root):
        self.n = len(tree)
        self.root = root
        self.log = 0
        while (1 << self.log) < self.n:
            self.log +=1
        self.up = [[-1]*(self.n+1) for _ in range(self.log+1)]
        self.depth = [0]*(self.n+1)
        # Build using BFS
        q = deque()
        q.append(root)
        self.up[0][root] = -1
        self.depth[root] =0
        while q:
            u = q.popleft()
            for v in tree[u]:
                if self.up[0][v] == -1 and v != root:
                    self.up[0][v] = u
                    self.depth[v] = self.depth[u]+1
                    q.append(v)
        # Precompute up table
        for k in range(1, self.log+1):
            for v in range(1, self.n+1):
                if self.up[k-1][v] != -1:
                    self.up[k][v] = self.up[k-1][self.up[k-1][v]]
    def lca(self, u, v):
        if self.depth[u] < self.depth[v]:
            u, v = v, u
        # Bring u to the same depth as v
        for k in range(self.log, -1, -1):
            if self.depth[u] - (1 <<k) >= self.depth[v]:
                u = self.up[k][u]
        if u ==v:
            return u
        for k in range(self.log, -1, -1):
            if self.up[k][u] != -1 and self.up[k][u] != self.up[k][v]:
                u = self.up[k][u]
                v = self.up[k][v]
        return self.up[0][u]

To solve this problem, we need to determine the number of important roads for each merchant. An important road for a merchant is one whose removal would disconnect the merchant's warehouse from their shop. 

### Approach
1. **Identify Bridges**: Use Tarjan's algorithm to find all bridges in the graph. Bridges are edges whose removal increases the number of connected components.
2. **2-Edge-Connected Components**: Group cities into components such that any two cities within a component are connected by at least two distinct paths. This is done using BFS/DFS, considering only non-bridge edges.
3. **Construct Bridge Tree**: Build a tree where each node represents a 2-edge-connected component, and edges represent bridges between these components.
4. **LCA Preprocessing**: Preprocess the bridge tree using binary lifting to efficiently compute the Lowest Common Ancestor (LCA) and the distance between nodes in the tree.
5. **Query Processing**: For each merchant, determine the number of bridges on the path between their warehouse and shop using the preprocessed LCA information.

### Solution Code
```python
import sys
from collections import deque

def main():
    sys.setrecursionlimit(1 << 25)
    n, m = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Step 1: Find all bridges using Tarjan's algorithm iteratively
    bridges = set()
    disc = [0] * (n + 1)
    low = [0] * (n + 1)
    time_counter = 1
    visited = [False] * (n + 1)
    parent = [-1] * (n + 1)
    
    for start in range(1, n + 1):
        if not visited[start]:
            stack = []
            stack.append((start, False, -1))
            while stack:
                node, processed, prev = stack.pop()
                if processed:
                    for neighbor in adj[node]:
                        if neighbor == prev:
                            continue
                        if disc[neighbor] > disc[node]:  # Tree edge, part of DFS tree
                            low[node] = min(low[node], low[neighbor])
                            if low[neighbor] > disc[node]:
                                bridges.add((node, neighbor))
                                bridges.add((neighbor, node))
                        else:  # Back edge or forward edge
                            low[node] = min(low[node], disc[neighbor])
                else:
                    if visited[node]:
                        continue
                    visited[node] = True
                    disc[node] = low[node] = time_counter
                    time_counter += 1
                    stack.append((node, True, prev))
                    # Push children
                    children = []
                    for neighbor in adj[node]:
                        if not visited[neighbor] and neighbor != prev:
                            parent[neighbor] = node
                            children.append(neighbor)
                    for child in reversed(children):
                        stack.append((child, False, node))
    
    # Step 2: Assign components using BFS with non-bridge edges
    component = [0] * (n + 1)
    comp_id = 0
    for u in range(1, n + 1):
        if component[u] == 0:
            comp_id += 1
            q = deque()
            q.append(u)
            component[u] = comp_id
            while q:
                current = q.popleft()
                for v in adj[current]:
                    if (current, v) not in bridges and component[v] == 0:
                        component[v] = comp_id
                        q.append(v)
    
    # Step 3: Build the bridge tree
    bridge_tree = [[] for _ in range(comp_id + 1)]
    bridge_edges = set()
    for u in range(1, n + 1):
        for v in adj[u]:
            if (u, v) in bridges:
                c1 = component[u]
                c2 = component[v]
                if c1 != c2:
                    if (c1, c2) not in bridge_edges and (c2, c1) not in bridge_edges:
                        bridge_edges.add((c1, c2))
                        bridge_tree[c1].append(c2)
                        bridge_tree[c2].append(c1)
    
    # Step 4: Preprocess LCA for the bridge tree
    if comp_id == 0:
        root = 0
    else:
        root = component[1]
    
    class LCA:
        def __init__(self, tree, root):
            self.n = len(tree) - 1  # Assuming tree is 1-based
            self.root = root
            self.log = 0
            while (1 << self.log) <= self.n:
                self.log += 1
            self.up = [[-1] * (self.n + 1) for _ in range(self.log)]
            self.depth = [0] * (self.n + 1)
            # BFS to set up parent and depth
            q = deque()
            q.append(root)
            self.up[0][root] = -1
            visited_lca = [False] * (self.n + 1)
            visited_lca[root] = True
            while q:
                u = q.popleft()
                for v in tree[u]:
                    if not visited_lca[v] and v != self.root:
                        self.up[0][v] = u
                        self.depth[v] = self.depth[u] + 1
                        visited_lca[v] = True
                        q.append(v)
            # Fill the rest of the up table
            for k in range(1, self.log):
                for v in range(1, self.n + 1):
                    if self.up[k-1][v] != -1:
                        self.up[k][v] = self.up[k-1][self.up[k-1][v]]
                    else:
                        self.up[k][v] = -1
        
        def lca(self, u, v):
            if u == v:
                return u
            if self.depth[u] < self.depth[v]:
                u, v = v, u
            # Bring u to the depth of v
            for k in range(self.log-1, -1, -1):
                if self.up[k][u] != -1 and self.depth[self.up[k][u]] >= self.depth[v]:
                    u = self.up[k][u]
            if u == v:
                return u
            for k in range(self.log-1, -1, -1):
                if self.up[k][u] != -1 and self.up[k][u] != self.up[k][v]:
                    u = self.up[k][u]
                    v = self.up[k][v]
            return self.up[0][u]
    
    if comp_id == 0:
        # Only one component, no bridges
        lca_instance = None
    else:
        lca_instance = LCA(bridge_tree, root)
    
    # Process queries
    k = int(sys.stdin.readline())
    for _ in range(k):
        s, l = map(int, sys.stdin.readline().split())
        cs = component[s]
        cl = component[l]
        if cs == cl:
            print(0)
        else:
            lca_node = lca_instance.lca(cs, cl)
            distance = lca_instance.depth[cs] + lca_instance.depth[cl] - 2 * lca_instance.depth[lca_node]
            print(distance)

if __name__ == "__main__":
    main()
```

### Explanation
1. **Finding Bridges**: Tarjan's algorithm is used to identify bridges in the graph efficiently.
2. **Component Assignment**: Nodes are grouped into 2-edge-connected components using BFS, considering only non-bridge edges.
3. **Bridge Tree Construction**: The bridge tree is built where each node represents a component and edges represent bridges.
4. **LCA Preprocessing**: The bridge tree is preprocessed to allow efficient LCA queries, which help in determining the number of bridges on the path between two components.
5. **Query Processing**: For each query, the number of bridges (important roads) is determined by the path length in the bridge tree using LCA.

This approach ensures that we efficiently handle up to the maximum constraints, providing an optimal solution for large input sizes.
