```python
import sys
from collections import deque

def main():
    # Precompute the minimum steps from all possible states to the target using BFS
    target = (0, 1, 2, 3, 4, 5, 6, 7)
    visited = {target: 0}
    queue = deque([target])

    while queue:
        current = queue.popleft()
        current_steps = visited[current]
        zero_pos = current.index(0)

        # Determine all possible adjacent positions for the zero tile
        adjacent = []
        # Check up (if zero is in the second row)
        if zero_pos >= 4:
            adjacent.append(zero_pos - 4)
        # Check down (if zero is in the first row)
        if zero_pos < 4:
            adjacent.append(zero_pos + 4)
        # Check left (if not in the first column)
        if zero_pos % 4 != 0:
            adjacent.append(zero_pos - 1)
        # Check right (if not in the last column)
        if (zero_pos + 1) % 4 != 0:
            adjacent.append(zero_pos + 1)

        # Generate new states by swapping zero with adjacent tiles
        for pos in adjacent:
            new_state = list(current)
            new_state[zero_pos], new_state[pos] = new_state[pos], new_state[zero_pos]
            new_state = tuple(new_state)
            if new_state not in visited:
                visited[new_state] = current_steps + 1
                queue.append(new_state)

    # Process each input line and output the precomputed result
    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        state = tuple(map(int, line.split()))
        print(visited.get(state, 0))

if __name__ == '__main__':
    main()
```
