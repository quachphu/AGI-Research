The solution involves grouping Pokémon types by their count across all gyms and then computing permutations within these groups.

### Approach

2. **Hashable Key for Signatures**: For each type, create a hashable key representing its count signature. This key is generated from the sorted list of (gym index, count) pairs where the count is non-zero.
3. **Factorial Calculation**: Precompute factorials modulo \(10^9 + 7\) up to the maximum number of types to efficiently compute the result.

import sys
from collections import defaultdict, Counter

MOD = 10**9 + 7

def main():
    n, m = map(int, sys.stdin.readline().split())
    type_to_gyms = defaultdict(list)
    
    for gym_idx in range(n):
        parts = list(map(int, sys.stdin.readline().split()))
        gi = parts[0]
        types = parts[1:]
        counter = Counter(types)
        for x, cnt in counter.items():
            type_to_gyms[x].append((gym_idx, cnt))
    
    group_counts = defaultdict(int)
    
    for x in range(1, m + 1):
        if x in type_to_gyms:
            sorted_list = sorted(type_to_gyms[x], key=lambda t: t[0])
            key = tuple(sorted_list)
        else:
            key = tuple()
        group_counts[key] += 1
    
    max_fact = m
    fact = [1] * (max_fact + 1)
    for i in range(2, max_fact + 1):
        fact[i] = fact[i-1] * i % MOD
    
    result = 1
    for k in group_counts.values():
        result = result * fact[k] % MOD
    
    print(result)

if __name__ == "__main__":
    main()
