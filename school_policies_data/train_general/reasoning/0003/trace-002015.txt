The solution is to precompute a convolution of sum_hr, sum_min, sum_sec.

To solve this problem, we need to count how many valid date-time combinations on a digital clock have exactly N glowing segments, considering that some segments might be broken and never light up. The solution involves systematically checking each possible date and time, computing the number of glowing segments, and summing up valid combinations.

### Approach
1. **Precompute Segment Information**: Determine the number of glowing segments for each digit (0-9) at each position (0-13) considering broken segments.
2. **Precompute Time Components**: Calculate the number of glowing segments for each possible hour, minute, and second, and store these counts in dictionaries.
3. **Convolution of Time Components**: Combine the precomputed hour, minute, and second counts to determine the number of valid time combinations for each possible total of glowing segments.
4. **Iterate Over All Dates**: For each valid date (year, month, day), compute the total glowing segments from the date components and check if the remaining segments needed to reach N can be achieved with valid times.

### Solution Code
```python
import sys
from collections import defaultdict

def main():
    predefined_segments = [
        {0, 1, 2, 3, 4, 5},   # Digit 0
        {1, 2},                # Digit 1
        {0, 1, 3, 4, 6},      # Digit 2
        {0, 1, 2, 3, 6},      # Digit 3
        {1, 2, 5, 6},         # Digit 4
        {0, 2, 3, 5, 6},      # Digit 5
        {0, 2, 3, 4, 5, 6},   # Digit 6
        {0, 1, 2},            # Digit 7
        {0, 1, 2, 3, 4, 5, 6},# Digit 8
        {0, 1, 2, 3, 5, 6},   # Digit 9
    ]

    N = int(sys.stdin.readline())
    K = int(sys.stdin.readline())
    broken = [set() for _ in range(14)]
    for _ in range(K):
        p, q = map(int, sys.stdin.readline().split())
        broken[p].add(q)
    
    # Precompute the number of lit segments for each digit and position
    seg_counts = [[0] * 10 for _ in range(14)]
    for p in range(14):
        for d in range(10):
            available = predefined_segments[d].difference(broken[p])
            seg_counts[p][d] = len(available)
    
    # Precompute sum counts for hours, minutes, seconds
    hr_counts = defaultdict(int)
    for hour in range(24):
        h_str = f"{hour:02d}"
        h1, h2 = int(h_str[0]), int(h_str[1])
        sum_hr = seg_counts[8][h1] + seg_counts[9][h2]
        hr_counts[sum_hr] += 1
    
    min_counts = defaultdict(int)
    for minute in range(60):
        m_str = f"{minute:02d}"
        m1, m2 = int(m_str[0]), int(m_str[1])
        sum_min = seg_counts[10][m1] + seg_counts[11][m2]
        min_counts[sum_min] += 1
    
    sec_counts = defaultdict(int)
    for second in range(60):
        s_str = f"{second:02d}"
        s1, s2 = int(s_str[0]), int(s_str[1])
        sum_sec = seg_counts[12][s1] + seg_counts[13][s2]
        sec_counts[sum_sec] += 1
    
    # Compute time_counts convolution
    time_counts = defaultdict(int)
    for sum_hr, cnt_hr in hr_counts.items():
        for sum_min, cnt_min in min_counts.items():
            for sum_sec, cnt_sec in sec_counts.items():
                total = sum_hr + sum_min + sum_sec
                time_counts[total] += cnt_hr * cnt_min * cnt_sec
    
    total = 0
    # Iterate over all possible years and months
    for year in range(10000):
        y_str = f"{year:04d}"
        sum_yr = sum(seg_counts[i][int(y_str[i])] for i in range(4))
        
        for month in range(1, 13):
            m_str = f"{month:02d}"
            m1, m2 = int(m_str[0]), int(m_str[1])
            sum_mo = seg_counts[4][m1] + seg_counts[5][m2]
            
            # Determine max_day for the current month and year
            if month in [1, 3, 5, 7, 8, 10, 12]:
                max_day = 31
            elif month in [4, 6, 9, 11]:
                max_day = 30
            else:
                if (year % 400 == 0) or (year % 100 != 0 and year % 4 == 0):
                    max_day = 29
                else:
                    max_day = 28
            
            # Calculate sum_day counts for valid days
            day_sum_counts = defaultdict(int)
            for day in range(1, max_day + 1):
                d_str = f"{day:02d}"
                d1, d2 = int(d_str[0]), int(d_str[1])
                sum_day = seg_counts[6][d1] + seg_counts[7][d2]
                day_sum_counts[sum_day] += 1
            
            # Check remaining sum and accumulate total
            for sum_day, cnt_day in day_sum_counts.items():
                remaining = N - (sum_yr + sum_mo + sum_day)
                if remaining >= 0:
                    total += cnt_day * time_counts.get(remaining, 0)
    
    print(total)

if __name__ == "__main__":
    main()
```

### Explanation
1. **Precompute Segment Information**: For each digit (0-9) and each position (0-13), calculate how many segments would be lit considering any broken segments.
2. **Time Components**: For each hour, minute, and second, compute the number of glowing segments and store these counts.
3. **Convolution**: Combine the counts for hours, minutes, and seconds to determine how many valid times exist for each possible total of glowing segments.
4. **Date Iteration**: For each valid date, compute the sum of glowing segments from the date components and check if the remaining segments can be achieved with valid times, summing up valid combinations.
