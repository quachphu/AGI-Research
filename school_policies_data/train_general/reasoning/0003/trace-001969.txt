import bisect
import sys

def main():

    fib = [1, 1]  # fib[0] = F(1), fib[1] = F(2), ...

        next_val = fib[-1] + fib[-2]
        if next_val > 2 * 10**18:

    input_data = sys.stdin.read().split()

        target = N + 1  # We need fib[i] >= target
