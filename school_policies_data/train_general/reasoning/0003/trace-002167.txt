def compute_sum_digits(n):
    if n == 0:
        return 0
    sum_d = 0
    d = 1
    while True:
        lower = 10 ** (d - 1)
        if lower > n:
            break
        upper = 10 ** d - 1
        current_upper = min(upper, n)
        a = lower
        b = current_upper
        total = b - a + 1
        div3 = (b // 3) - ((a - 1) // 3)
        div5 = (b // 5) - ((a - 1) // 5)
        div15 = (b // 15) - ((a - 1) // 15)
        count = total - (div3 + div5 - div15)
        sum_d += count * d
        d += 1
    return sum_d

def compute_sum_others(n):
