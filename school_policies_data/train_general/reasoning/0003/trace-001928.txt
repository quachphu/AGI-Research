### Approach

2. **Grouping Candidates**: Group these candidates by the required substitutions (positions and characters) needed for the substring to match t_i. Each group's contribution is the sum of the costs of all candidates in that group.

def main():

            t_chars = [pair[1] for pair in required_pairs]
            if len(t_chars) != len(set(t_chars)):

            if required_pairs:
                key = frozenset(required_pairs)
                group_dict[key] += c

    # Build pair_to_groups: maps each (p, c) to list of (group_key, weight)
    pair_to_groups = defaultdict(list)
    for key, weight in group_dict.items():
        for (p, c) in key:
            pair_to_groups[(p, c)].append((key, weight))

    max_total = -float('inf')

    # Backtracking to assign characters to q_positions
    def backtrack(index, current_sum, used_chars, assignment):
        nonlocal max_total
        if index == m:
            total = base_sum + current_sum
            if total > max_total:
