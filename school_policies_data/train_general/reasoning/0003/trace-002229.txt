    alpha = 1.0  # reflection coefficient
    gamma = 2.0  # expansion coefficient
    rho = 0.5    # contraction coefficient
    sigma = 0.5  # shrink coefficient

    # Create initial simplex
    simplex = [ (initial[0], initial[1]) ]
    delta = 1.0
    simplex.append( (initial[0] + delta, initial[1]) )
    simplex.append( (initial[0], initial[1] + delta) )

    for it in range(max_iter):
        # Evaluate all points in the simplex
        values = [objective(x, y) for x, y in simplex]
        sorted_simplex = sorted(zip(values, simplex), key=lambda x: x[0])
        # Extract best, next worst, worst
        best_val, best = sorted_simplex[0]
        next_worst_val, next_worst = sorted_simplex[1]
        worst_val, worst = sorted_simplex[2]

        # Compute centroid of the best two points
        centroid_x = (best[0] + next_worst[0]) / 2
        centroid_y = (best[1] + next_worst[1]) / 2

        # Reflect
        xr = centroid_x + alpha * (centroid_x - worst[0])
        yr = centroid_y + alpha * (centroid_y - worst[1])
        val_r = objective(xr, yr)

        if val_r < best_val:
            # Expand
            xe = centroid_x + gamma * (xr - centroid_x)
            ye = centroid_y + gamma * (yr - centroid_y)
            val_e = objective(xe, ye)
            if val_e < val_r:
                # Replace worst with expanded
                simplex.remove(worst)
                simplex.append( (xe, ye) )
            else:
                # Replace worst with reflected
                simplex.remove(worst)
                simplex.append( (xr, yr) )
        elif val_r < next_worst_val:
            # Replace worst with reflected
            simplex.remove(worst)
            simplex.append( (xr, yr) )
        else:
            # Contract
            if val_r < worst_val:
                # Outside contraction
                xc = centroid_x + rho * (xr - centroid_x)
                yc = centroid_y + rho * (yr - centroid_y)
                val_c = objective(xc, yc)
                if val_c < val_r:
                    simplex.remove(worst)
                    simplex.append( (xc, yc) )
                else:
                    # Shrink
                    new_simplex = [best]
                    for point in simplex[1:]:
                        x = best[0] + sigma * (point[0] - best[0])
                        y = best[1] + sigma * (point[1] - best[1])
                        new_simplex.append( (x, y) )
                    simplex = new_simplex
            else:
                # Inside contraction
                xc = centroid_x - rho * (centroid_x - worst[0])
                yc = centroid_y - rho * (centroid_y - worst[1])
                val_c = objective(xc, yc)

                        y = best[1] + sigma * (point[1] - best[1])

                    simplex = new_simplex

        # Check for convergence

        max_val = max(current_values)
        min_val = min(current_values)

1. **Initialization**: Start with the centroid of all base positions as the initial guess for the headquarters location.
2. **Gradient Descent**: Use numerical gradient estimation to determine the direction of steepest descent and adjust the headquarters location iteratively.
3. **Adaptive Learning Rate**: Adjust the learning rate dynamically to ensure convergence while avoiding overshooting the minimum.
4. **Multiple Restarts**: Perturb the initial guess multiple times to avoid getting stuck in local minima and refine the solution with a smaller learning rate.
