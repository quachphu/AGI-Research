Implementing the segment tree:

The segment tree needs to support:

- Range updates (add a delta to a range)

- Query for the maximum in the entire tree.

Each node in the segment tree will store the current maximum for its range, and a lazy delta for pending updates.

The structure for the segment tree is as follows:

Each node contains:

- max_val: the maximum value in the range.

- lazy: the pending delta to add to all elements in the range.

When a range update is applied, the lazy is propagated to children nodes when necessary.

The implementation is manageable.

class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size <<=1
        self.max_tree = [0] * (2 * self.size)
        self.lazy = [0] * (2 * self.size)
        # Initialize leaves
        for i in range(self.n):
            self.max_tree[self.size + i] = data[i]
        # Build the tree
        for i in range(self.size-1, 0, -1):
            self.max_tree[i] = max(self.max_tree[2*i], self.max_tree[2*i+1])

    def push(self, node, l, r):
        if self.lazy[node] != 0 and node < self.size:
            # Propagate to children
            self.max_tree[2*node] += self.lazy[node]
            self.max_tree[2*node+1] += self.lazy[node]
            self.lazy[2*node] += self.lazy[node]
            self.lazy[2*node+1] += self.lazy[node]
            self.lazy[node] = 0

    def range_add(self, a, b, val, node=1, l=0, r=None):
        if r is None:
            r = self.size -1
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.max_tree[node] += val
            self.lazy[node] += val
            return
        self.push(node, l, r)
        mid = (l + r) // 2
        self.range_add(a, b, val, 2*node, l, mid)
        self.range_add(a, b, val, 2*node+1, mid+1, r)
        self.max_tree[node] = max(self.max_tree[2*node], self.max_tree[2*node+1])

    def query_max(self, a=0, b=None, node=1, l=0, r=None):
        if b is None:
            b = self.size -1
        if r is None:
            r = self.size -1
        if a > r or b < l:
            return -float('inf')
        if a <= l and r <= b:
            return self.max_tree[node]
        self.push(node, l, r)
        mid = (l + r) // 2
        left = self.query_max(a, b, 2*node, l, mid)
        right = self.query_max(a, b, 2*node+1, mid+1, r)
        return max(left, right)

n, m, p = map(int, input().split())

weapons = []
for _ in range(n):
    a, ca = map(int, input().split())
    weapons.append( (a, ca) )
weapons.sort()

armors = []
sorted_b = []
sorted_cb = []
for _ in range(m):
    b, cb = map(int, input().split())
    armors.append( (b, cb) )
armors.sort()
sorted_b = [b for b, cb in armors]
sorted_cb = [cb for b, cb in armors]

monsters = []
for _ in range(p):
    x, y, z = map(int, input().split())
    monsters.append( (x, y, z) )
monsters.sort()

# Initialize the segment tree with -cb_j
data = [-cb for cb in sorted_cb]
st = SegmentTree(data)

max_profit = -float('inf')
monster_ptr = 0

for a_i, ca in weapons:
    # Add all monsters with x < a_i
    while monster_ptr < p and monsters[monster_ptr][0] < a_i:
        x, y, z = monsters[monster_ptr]
        # Find the first armor j with b_j > y
        idx = bisect.bisect_right(sorted_b, y)
        if idx < m:
            st.range_add(idx, m-1, z)
        monster_ptr +=1
    current_max = st.query_max(0, m-1)
    profit = current_max - ca
    if profit > max_profit:
        max_profit = profit

To solve this problem, we need to determine the maximum profit Roma can achieve by selecting one weapon and one armor set, and then defeating as many monsters as possible. The profit is calculated as the total coins obtained from defeated monsters minus the cost of the selected weapon and armor.

### Approach
1. **Sort Weapons and Monsters**: Sort weapons by their attack modifier and monsters by their defense to efficiently process which monsters can be defeated by each weapon.
2. **Sort Armors**: Sort armors by their defense modifier to efficiently determine which monsters can be defeated by each armor set.
3. **Segment Tree for Range Updates and Queries**: Use a segment tree to maintain and query the maximum value of the profit formula (sum of coins from defeated monsters minus armor cost) dynamically as we process each weapon.

import bisect

class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.max_tree = [-float('inf')] * (2 * self.size)
        self.lazy = [0] * (2 * self.size)
        for i in range(self.n):
            self.max_tree[self.size + i] = data[i]
        for i in range(self.n, self.size):
            self.max_tree[self.size + i] = -float('inf')
        for i in range(self.size - 1, 0, -1):
            self.max_tree[i] = max(self.max_tree[2*i], self.max_tree[2*i+1])

    def push(self, node, l, r):
        if self.lazy[node] != 0 and node < self.size:
            self.max_tree[2*node] += self.lazy[node]
            self.max_tree[2*node+1] += self.lazy[node]
            self.lazy[2*node] += self.lazy[node]
            self.lazy[2*node+1] += self.lazy[node]
            self.lazy[node] = 0

    def range_add(self, a, b, val, node=1, l=0, r=None):
        if r is None:
            r = self.size - 1
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.max_tree[node] += val
            self.lazy[node] += val
            return
        self.push(node, l, r)
        mid = (l + r) // 2
        self.range_add(a, b, val, 2*node, l, mid)
        self.range_add(a, b, val, 2*node+1, mid+1, r)
        self.max_tree[node] = max(self.max_tree[2*node], self.max_tree[2*node+1])

    def query_max(self, a=0, b=None, node=1, l=0, r=None):
        if b is None:
            b = self.size - 1
        if r is None:
            r = self.size - 1
        if a > r or b < l:
            return -float('inf')
        if a <= l and r <= b:
            return self.max_tree[node]
        self.push(node, l, r)
        mid = (l + r) // 2
        left = self.query_max(a, b, 2*node, l, mid)
        right = self.query_max(a, b, 2*node+1, mid+1, r)
        return max(left, right)
