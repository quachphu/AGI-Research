class FenwickTree:

    def __init__(self, size):

        self.n = size

        self.tree = [0]*(self.n +1)

    def update(self, idx, delta):

        while idx <= self.n:

            self.tree[idx] += delta

            idx += idx & -idx

    def query(self, idx):

        res =0

        while idx >0:

            res += self.tree[idx]

            idx -= idx & -idx

        return res

### Approach
1. **Problem Analysis**: The problem can be modeled using dynamic programming (DP) where `dp[i]` represents the number of ways to reach stop `i`. Each bus route contributes to the number of ways to reach its end stop by summing the ways to reach all possible start stops of that bus.

2. **Coordinate Compression**: Given the constraints (especially `n` up to \(10^9\)), we use coordinate compression to handle the large range of stops efficiently. We collect all relevant stops (start and end points of buses, home, and school) and map them to a compressed index.

3. **Fenwick Tree (Binary Indexed Tree)**: This data structure is used to efficiently compute prefix sums and update values, which helps in calculating the number of ways to reach each stop dynamically. The Fenwick Tree allows us to quickly compute the sum of values over a range and update values in logarithmic time.

4. **Sorting and Processing Buses**: Buses are processed in increasing order of their end stops to ensure that we correctly accumulate the number of ways to reach each stop in sequence.

import bisect

MOD = 10**9 + 7

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1
    buses = []
    points = set()
    points.add(0)
    points.add(n)
    for _ in range(m):
        s = int(input[ptr])
        ptr += 1
        t = int(input[ptr])
        ptr += 1
        buses.append((s, t))
        points.add(s)
        points.add(t)
    
    sorted_points = sorted(points)
    buses.sort(key=lambda x: x[1])
    
    class FenwickTree:
        def __init__(self, size):
            self.n = size
            self.tree = [0] * (self.n + 2)
        
        def update(self, idx, delta):
            while idx <= self.n:
                self.tree[idx] = (self.tree[idx] + delta) % MOD
                idx += idx & -idx
        
        def query(self, idx):
            res = 0
            while idx > 0:
                res = (res + self.tree[idx]) % MOD
                idx -= idx & -idx
            return res
    
    ft_size = len(sorted_points)
    ft = FenwickTree(ft_size)
    
    idx_0 = bisect.bisect_left(sorted_points, 0)
    if idx_0 < len(sorted_points) and sorted_points[idx_0] == 0:
        ft.update(idx_0 + 1, 1)
    else:
        print(0)
        return
    
    for s, t in buses:
        t_minus_1 = t - 1
        i_upper = bisect.bisect_right(sorted_points, t_minus_1) - 1
        sum_upper = ft.query(i_upper + 1)
        
        s_minus_1 = s - 1
        i_lower = bisect.bisect_right(sorted_points, s_minus_1) - 1
        sum_lower = ft.query(i_lower + 1)
        
        sum_value = (sum_upper - sum_lower) % MOD
        if sum_value < 0:
            sum_value += MOD
        
        if sum_value > 0:
            i_t = bisect.bisect_left(sorted_points, t)
            if i_t < len(sorted_points) and sorted_points[i_t] == t:
                ft.update(i_t + 1, sum_value)
    
    if n not in sorted_points:
        print(0)
        return
    
    i_n = bisect.bisect_left(sorted_points, n)
    if i_n >= len(sorted_points) or sorted_points[i_n] != n:
        print(0)
        return
    
    sum_n = ft.query(i_n + 1)
    if i_n == 0:
        sum_prev = 0
    else:
        sum_prev = ft.query(i_n)
    
    answer = (sum_n - sum_prev) % MOD
    if answer < 0:
        answer += MOD
    print(answer)

if __name__ == "__main__":
    main()
