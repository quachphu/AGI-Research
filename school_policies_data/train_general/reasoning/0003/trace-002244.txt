```python
MOD = 998244353

def main():
    import sys
    input = sys.stdin.read().split()
    n = int(input[0])
    m = int(input[1])
    
    if n < 3:
        print(0)
        return
    
    k = n - 1
    if m < k:
        print(0)
        return
    
    # Precompute factorial and inverse factorial up to m
    max_fact = m
    fact = [1] * (max_fact + 1)
    for i in range(2, max_fact + 1):
        fact[i] = fact[i-1] * i % MOD
    
    inv_fact = [1] * (max_fact + 1)
    inv_fact[max_fact] = pow(fact[max_fact], MOD-2, MOD)
    for i in range(max_fact - 1, 0, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD
    
    # Compute combination C(m, k)
    c = fact[m] * inv_fact[k] % MOD
    c = c * inv_fact[m - k] % MOD
    
    # Compute pow(2, n-3)
    power = pow(2, n-3, MOD) if n >=3 else 1
    
    res = power * (n-2) % MOD
    res = res * c % MOD
    print(res)

if __name__ == "__main__":
    main()
```
