To solve this problem, we need to determine if a sequence of semicircles drawn between consecutive points on the x-axis will intersect each other. Each semicircle is drawn above the x-axis, and we need to check if any two non-consecutive semicircles intersect at a point other than their endpoints.

### Approach
1. **Input Reading**: Read the number of points and their coordinates.
2. **Iterate Through Pairs**: For each pair of non-consecutive semicircles, check if they intersect.
3. **Geometric Calculations**:
   - **Circle Properties**: Calculate the center and radius of each semicircle.
   - **Intersection Check**: Determine if the circles formed by the semicircles intersect. If they do, compute the intersection points and check if they lie within the x-axis intervals of both semicircles and above the x-axis.

### Solution Code
```python
import math
import sys

def main():
    input = sys.stdin.read().split()
    n = int(input[0])
    if n < 2:
        print("no")
        return
    x = list(map(int, input[1:n+1]))
    epsilon = 1e-9

    for i in range(n-1):
        for j in range(i + 2, n-1):
            xi, xi1 = x[i], x[i+1]
            xj, xj1 = x[j], x[j+1]

            a1, b1 = min(xi, xi1), max(xi, xi1)
            a2, b2 = min(xj, xj1), max(xj, xj1)

            h1 = (xi + xi1) / 2.0
            r1 = abs(xi1 - xi) / 2.0
            h2 = (xj + xj1) / 2.0
            r2 = abs(xj1 - xj) / 2.0

            d = abs(h1 - h2)
            sum_r = r1 + r2
            diff_r = abs(r1 - r2)

            if (sum_r < d - epsilon) or (diff_r > d + epsilon):
                continue

            if math.isclose(h1, h2, rel_tol=1e-9, abs_tol=epsilon):
                if math.isclose(r1, r2, rel_tol=1e-9, abs_tol=epsilon):
                    if a1 < b2 + epsilon and a2 < b1 + epsilon:
                        print("yes")
                        return
                continue
            else:
                denominator = 2 * (h2 - h1)
                numerator = (r1**2 - r2**2) + (h2**2 - h1**2)
                x_val = numerator / denominator

                y_sq = r1**2 - (x_val - h1)**2
                if y_sq < -epsilon:
                    continue

                if (x_val >= a1 - epsilon and x_val <= b1 + epsilon) and \
                   (x_val >= a2 - epsilon and x_val <= b2 + epsilon):
                    print("yes")
                    return

    print("no")

if __name__ == "__main__":
    main()
```
