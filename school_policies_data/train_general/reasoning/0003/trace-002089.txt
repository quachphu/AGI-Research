import sys
import bisect

class DSU:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return False
        if self.rank[x_root] < self.rank[y_root]:
            self.parent[x_root] = y_root
        else:
            self.parent[y_root] = x_root
            if self.rank[x_root] == self.rank[y_root]:
                self.rank[x_root] += 1
        return True

def main():
    input = sys.stdin.read().split()
    ptr = 0
    N = int(input[ptr])
    ptr += 1
    M = int(input[ptr])
    ptr += 1
    G = int(input[ptr])
    ptr += 1
    S = int(input[ptr])
    ptr += 1
    
    edges = []
    for _ in range(M):
        x = int(input[ptr])
        ptr += 1
        y = int(input[ptr])
        ptr += 1
        g = int(input[ptr])
        ptr += 1
        s = int(input[ptr])
        ptr += 1
        edges.append((g, s, x-1, y-1))  # Convert to 0-based
    
    edges.sort()  # Sort by gi, then si
    
    sorted_by_si = []
    min_cost = float('inf')
    
    for i in range(M):
        g_i, s_i, x_i, y_i = edges[i]
        # Insert into sorted_by_si in the correct position
        # Using bisect to find the insertion point
        pos = bisect.bisect_left(sorted_by_si, (s_i, g_i, x_i, y_i))
        sorted_by_si.insert(pos, (s_i, g_i, x_i, y_i))
        
        dsu = DSU(N)
        max_g = 0
        max_s = 0
        count = 0
        
        for e in sorted_by_si:
            s, g, x, y = e
            if x == y:
                continue
            if dsu.find(x) != dsu.find(y):
                dsu.union(x, y)
                if g > max_g:
                    max_g = g
                if s > max_s:
                    max_s = s
