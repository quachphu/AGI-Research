To solve this problem, we need to determine the maximum number of points that can be enclosed by a circle of radius 1 when moved across the xy-plane. The solution involves checking each point and each pair of points to find the optimal circle position that encloses the maximum number of points.

### Approach
1. **Single Points Check**: For each point, consider a circle centered at that point and count how many other points lie within or on the circle (distance ≤ 1).
2. **Pairs of Points Check**: For each pair of points, if their distance is less than 2 (ensured by problem constraints), compute the two possible centers of circles that pass through both points. For each center, count how many points lie within or on the circle.

import math

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    while True:
        if input[ptr] == '0':
            break
        N = int(input[ptr])
        ptr += 1
        points = []
        for _ in range(N):
            x = float(input[ptr])
            y = float(input[ptr + 1])
            ptr += 2
            points.append((x, y))
        
        max_count = 1
        
        # Check each individual point as center
        for (x, y) in points:
            count = 0
            for (px, py) in points:
                dx = px - x
                dy = py - y
                if dx * dx + dy * dy <= 1.0 + 1e-8:
                    count += 1
            if count > max_count:
                max_count = count
        
        # Check all pairs of points for possible centers
        for i in range(N):
            x1, y1 = points[i]
            for j in range(i + 1, N):
                x2, y2 = points[j]
                dx = x2 - x1
                dy = y2 - y1
                d_sq = dx * dx + dy * dy
                if d_sq >= 4.0:
                    continue
                d = math.sqrt(d_sq)
                mid_x = (x1 + x2) / 2.0
                mid_y = (y1 + y2) / 2.0
                try:
                    h = math.sqrt(1.0 - (d_sq / 4.0))
                except:
                    continue
                unit_perp_x = -dy / d
                unit_perp_y = dx / d
                # First center
                c1x = mid_x + h * unit_perp_x
                c1y = mid_y + h * unit_perp_y
                # Second center
                c2x = mid_x - h * unit_perp_x
                c2y = mid_y - h * unit_perp_y
                
                # Check first center
                count1 = 0
                for (px, py) in points:
                    dx_p = px - c1x
                    dy_p = py - c1y
                    dist_sq = dx_p * dx_p + dy_p * dy_p
                    if dist_sq <= 1.0 + 1e-8:
                        count1 += 1
                if count1 > max_count:
                    max_count = count1
                
                # Check second center
                count2 = 0
                for (px, py) in points:
                    dx_p = px - c2x
                    dy_p = py - c2y
                    dist_sq = dx_p * dx_p + dy_p * dy_p
                    if dist_sq <= 1.0 + 1e-8:
                        count2 += 1
                if count2 > max_count:
                    max_count = count2
        
        print(max_count)
