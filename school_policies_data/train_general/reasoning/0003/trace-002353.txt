### Approach

2. **Special Case Handling**: If the lucky digit (b) is zero, we need to count substrings where all digits are zero. For non-zero b, we need to count substrings where the sum of digits modulo (k-1) equals b, adjusted for cases where the sum is non-zero.
3. **Prefix Sum Modulo Technique**: Use prefix sums modulo (k-1) to efficiently count substrings that meet the required congruence conditions. This allows us to avoid explicitly checking all possible substrings, which would be computationally expensive.
4. **Efficient Counting**: Utilize a hash map to track the frequency of prefix sums modulo (k-1) to quickly determine the number of valid substrings ending at each position.

from collections import defaultdict

def main():
    input = sys.stdin.read().split()
    ptr = 0
    k = int(input[ptr])
    ptr += 1
    b = int(input[ptr])
    ptr += 1
    n = int(input[ptr])
    ptr += 1
    s = list(map(int, input[ptr:ptr + n]))
    
    if b == 0:
        current_zero_run = 0
        zero_count = 0
        for d in s:
            if d == 0:
                current_zero_run += 1
            else:
                zero_count += current_zero_run * (current_zero_run + 1) // 2
                current_zero_run = 0
        zero_count += current_zero_run * (current_zero_run + 1) // 2
        print(zero_count)
        return
    
    m = k - 1
    target_mod = 0 if b == m else b
    
    prefix_counts = defaultdict(int)
    prefix_counts[0] = 1
    current_prefix = 0
    total = 0
    
    for digit in s:
        current_prefix = (current_prefix + digit) % m
        required = (current_prefix - target_mod) % m
        total += prefix_counts.get(required, 0)
        prefix_counts[current_prefix] += 1
    
    if target_mod == 0:
        current_zero_run = 0
        zero_count = 0
        for d in s:
            if d == 0:
                current_zero_run += 1

                current_zero_run = 0
