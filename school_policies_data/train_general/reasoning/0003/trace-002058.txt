        edges[a].append(b)
        edges[b].append(a)

    size = [0] * (n + 1)
    visited = [False] * (n + 1)

        u, processed = stack.pop()
        if processed:
            size[u] = 1
            for v in edges[u]:
                if v != parent[u]:
                    size[u] += size[v]
