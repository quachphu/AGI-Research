2. **Binary Search**: For each shield, use binary search to determine the number of monsters with strength greater than or equal to the shield's defense rating.

```python
import bisect

MOD = 998244353

n, m = map(int, input().split())
d = list(map(int, input().split()))
d.sort()

prefix_sum = [0] * (n + 1)
for i in range(n):
    prefix_sum[i + 1] = prefix_sum[i] + d[i]
total_sum = prefix_sum[n]

for _ in range(m):
    a_i, b_i = map(int, input().split())
    idx = bisect.bisect_left(d, b_i)
    k = n - idx
    
    if a_i > k:
        print(0)
    elif a_i == k:
        sum_t = total_sum - (prefix_sum[n] - prefix_sum[idx])
        denominator = k + 1
        inv_den = pow(denominator, MOD - 2, MOD)
        result = (sum_t % MOD) * inv_den % MOD
        print(result)
    else:
        sum_s = prefix_sum[n] - prefix_sum[idx]
        sum_t = total_sum - sum_s
        
        term1_num = sum_s * (k - a_i)
        term1_den = k
        inv_den1 = pow(term1_den, MOD - 2, MOD)
        term1 = (term1_num % MOD) * inv_den1 % MOD
        
        term2_num = sum_t * (k - a_i + 1)
        term2_den = k + 1
        inv_den2 = pow(term2_den, MOD - 2, MOD)
        term2 = (term2_num % MOD) * inv_den2 % MOD
        
        total = (term1 + term2) % MOD
        print(total)
```
