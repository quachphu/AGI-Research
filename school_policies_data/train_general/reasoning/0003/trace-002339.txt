To solve this problem, we need to determine if a supersonic rocket is safe by checking if the power sources from both engines can be transformed such that destroying any power source does not change the generated power field. The key insight is that the power field is determined by the convex hull of the combined power sources, and each vertex of this convex hull must be present in both engines after transformation.

### Approach
1. **Convex Hull Calculation**: Compute the convex hull for both engines. The convex hull is the smallest convex polygon that contains all the points.
2. **Congruence Check**: Check if the convex hulls of both engines are congruent, meaning they can be rotated and/or translated to match each other.
3. **Point Inclusion Check**: Ensure that all points of each engine lie within their respective convex hulls. This ensures that all points are redundant in forming the power field.

import sys
from math import sqrt
from itertools import permutations

class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __sub__(self, other):
        return Point(self.x - other.x, self.y - other.y)

    def cross(self, other):
        return self.x * other.y - self.y * other.x

    def dot(self, other):
        return self.x * other.x + self.y * other.y

    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

    def __repr__(self):
        return f"({self.x}, {self.y})"

def convex_hull(points):
    points = sorted(points, key=lambda p: (p.x, p.y))
    lower = []
    for p in points:
        while len(lower) >= 2 and (lower[-1] - lower[-2]).cross(p - lower[-1]) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(points):
        while len(upper) >= 2 and (upper[-1] - upper[-2]).cross(p - upper[-1]) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]

def get_dist_sq(a, b):
    dx = a.x - b.x
    dy = a.y - b.y
    return dx*dx + dy*dy

def get_edge_lengths_and_angles(hull):
    n = len(hull)
    edges = []
    angles = []
    for i in range(n):
        j = (i+1) % n
        k = (i+2) % n
        a = hull[i]
        b = hull[j]
        c = hull[k]
        edges.append(get_dist_sq(a, b))
        ba = a - b
        bc = c - b
        dot = ba.dot(bc)
        cross = ba.cross(bc)
        angles.append((dot, cross))
    return edges, angles

def congruent(h1, h2):
    if len(h1) != len(h2):
        return False
    n = len(h1)
    edges1, angles1 = get_edge_lengths_and_angles(h1)
    edges2, angles2 = get_edge_lengths_and_angles(h2)
    for shift in range(n):
        for rev in [False, True]:
            match = True
            for i in range(n):
                j = (i + shift) % n
                if rev:
                    j = (shift - i) % n
                if edges1[i] != edges2[j]:
                    match = False
                    break
                if angles1[i] != angles2[j]:
                    match = False
                    break
            if match:
                return True
    return False

def point_in_convex_hull(p, hull):
    n = len(hull)
    for i in range(n):
        a = hull[i]
        b = hull[(i+1) % n]
        cross = (b - a).cross(p - a)
        if cross < 0:
            return False
