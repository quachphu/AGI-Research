2. **Iterate Over Elements**: For each element, update the DP dictionary to include new products formed by multiplying the current element with existing products in the DP.
3. **Track Best Subsequence**: For each product, keep track of the shortest subsequence and the lexicographically smallest sequence.
4. **Determine Optimal Solution**: Compare the maximum product found in the DP with the empty subsequence (product 1) and select the optimal solution based on the problem constraints.

def main():
    import sys
    input = sys.stdin.read().split()
    n = int(input[0])
    a = list(map(int, input[1:n+1]))

    dp = {1: (0, [])}

    for idx in range(n):
        current = a[idx]
        new_dp = {}
        for product in dp:
            length, seq = dp[product]
            # Option 1: do not take this element
            if product in new_dp:
                existing_length, existing_seq = new_dp[product]
                if (length < existing_length) or (length == existing_length and seq < existing_seq):
                    new_dp[product] = (length, seq)
            else:
                new_dp[product] = (length, seq)
            # Option 2: take this element
            new_product = product * current
            new_length = length + 1
            new_seq = seq + [idx + 1]
            if new_product in new_dp:
                existing_length, existing_seq = new_dp[new_product]
                if (new_length < existing_length) or (new_length == existing_length and new_seq < existing_seq):
                    new_dp[new_product] = (new_length, new_seq)
            else:
                new_dp[new_product] = (new_length, new_seq)
        dp = new_dp

    max_candidate = max(dp.keys()) if dp else 1
    best_product = 1
    if max_candidate > 1:
        best_product = max_candidate
    elif max_candidate == 1:
        best_product = 1
    else:
        if 1 > max_candidate:
            best_product = 1
        else:
            best_product = max_candidate

    if best_product == 1:
        print(0)
        return
    else:
        candidates = []
        for prod in dp:
            if prod == best_product:
                candidates.append(dp[prod])
        if not candidates:
            print(0)
            return
        min_length = min(c[0] for c in candidates)
        filtered = [c for c in candidates if c[0] == min_length]
        filtered.sort(key=lambda x: x[1])
        best_seq = filtered[0][1]

    print(len(best_seq))
    for num in best_seq:
