import bisect
from functools import lru_cache

def main():
    import sys
    n = int(sys.stdin.readline())
    s = sys.stdin.readline().strip()
    
    V = []
    K = []
    O = []
    for i, c in enumerate(s):
        if c == 'V':
            V.append(i)
        elif c == 'K':
            K.append(i)
        else:
            O.append(i)
    
    V.sort()
    K.sort()
    O.sort()
    total_V = len(V)
    total_K = len(K)
    total_O = len(O)
    
    @lru_cache(maxsize=None)
    def dp(v_rem, k_rem, o_rem, last_char):
        if v_rem == 0 and k_rem == 0 and o_rem == 0:
            return 0
        
        min_cost = float('inf')
        
        # Try placing V
        if v_rem > 0:
            current_v = total_V - v_rem
            pos_v = V[current_v]
            # Count remaining K's before pos_v
            k_start = total_K - k_rem
            k_before = bisect.bisect_left(K, pos_v, k_start, total_K) - k_start
            # Count remaining O's before pos_v
            o_start = total_O - o_rem
            o_before = bisect.bisect_left(O, pos_v, o_start, total_O) - o_start
            cost = k_before + o_before
            if last_char != 'V':
                new_last = 'V'
                new_v = v_rem - 1
                new_k = k_rem
                new_o = o_rem
                total_cost = cost + dp(new_v, new_k, new_o, new_last)
                if total_cost < min_cost:
                    min_cost = total_cost
            else:
                # V cannot be followed by K, but placing V is allowed if last is V
                new_last = 'V'
                new_v = v_rem - 1
                new_k = k_rem
                new_o = o_rem
                total_cost = cost + dp(new_v, new_k, new_o, new_last)
                if total_cost < min_cost:
                    min_cost = total_cost
        
        # Try placing K
        if k_rem > 0 and last_char != 'V':
            current_k = total_K - k_rem
            pos_k = K[current_k]
            # Count remaining V's before pos_k
            v_start = total_V - v_rem
            v_before = bisect.bisect_left(V, pos_k, v_start, total_V) - v_start
            # Count remaining O's before pos_k
            o_start = total_O - o_rem
            o_before = bisect.bisect_left(O, pos_k, o_start, total_O) - o_start
            cost = v_before + o_before
            new_last = 'K'
            new_v = v_rem
            new_k = k_rem - 1
            new_o = o_rem
            total_cost = cost + dp(new_v, new_k, new_o, new_last)
            if total_cost < min_cost:
                min_cost = total_cost
        
        # Try placing O
        if o_rem > 0:
            current_o = total_O - o_rem
            pos_o = O[current_o]
            # Count remaining V's before pos_o
            v_start = total_V - v_rem
            v_before = bisect.bisect_left(V, pos_o, v_start, total_V) - v_start
            # Count remaining K's before pos_o
            k_start = total_K - k_rem

            cost = v_before + k_before
            new_last = 'O'
            new_v = v_rem
            new_k = k_rem
            new_o = o_rem - 1
            total_cost = cost + dp(new_v, new_k, new_o, new_last)
            if total_cost < min_cost:
                min_cost = total_cost
        
        return min_cost
    
    result = dp(total_V, total_K, total_O, None)
