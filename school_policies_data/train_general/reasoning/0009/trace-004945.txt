        node1.next_main = None

        # Remove node1 from freq[x]
        flist.remove_node(node1)
        if flist.size >= 1:
            # Push x back to heap even if already present
            heapq.heappush(heap, x)

        # Now handle node2
        # Remove node2 from freq[x]
        if x in freq and freq[x].size > 0:
            # Check if node2 is still in the freq[x] list
            # This is a simplified check; in reality, since we just processed node1 and node2 was next, it should be there
            current_flist = freq[x]
            current_flist.remove_node(node2)
            if current_flist.size >= 1:
                heapq.heappush(heap, x)
        else:
            # This case shouldn't happen as per the problem logic
            pass

        # Update node2's value and add to new frequency list
        new_val = 2 * x
        node2.value = new_val

        if new_val not in freq:
            freq[new_val] = FreqList()
        new_flist = freq[new_val]
        new_flist.add_node(node2)
        if new_flist.size == 2:
            heapq.heappush(heap, new_val)

    result = []
