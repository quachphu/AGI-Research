3. **Initial Candidates**: Start with the positions immediately adjacent to each tree, as these are the closest possible positions.
4. **Expand Neighbors**: For each position extracted from the heap, generate its neighboring positions (left and right), compute their minimal distances, and add them to the heap if they haven't been processed yet.

import bisect
import heapq

def main():
    import sys
    input = sys.stdin.read().split()
    idx = 0
    n = int(input[idx])
    idx += 1
    m = int(input[idx])
    idx += 1
    x = list(map(int, input[idx:idx+n]))
    idx += n
    
    x.sort()
    x_set = set(x)
    sorted_x = x
    
    visited = set()
    heap = []
    
    # Initialize heap with adjacent positions of each tree
    for xi in sorted_x:
        for delta in (-1, 1):
            pos = xi + delta
            if pos not in x_set and pos not in visited:
                idx_pos = bisect.bisect_left(sorted_x, pos)
                if idx_pos == 0:
                    closest = sorted_x[0]
                elif idx_pos == len(sorted_x):
                    closest = sorted_x[-1]
                else:
                    left = sorted_x[idx_pos - 1]
                    right = sorted_x[idx_pos]
                    if pos - left <= right - pos:
                        closest = left
                    else:
                        closest = right
                distance = abs(pos - closest)
                heapq.heappush(heap, (distance, pos))
                visited.add(pos)
    
    selected = []
    selected_set = set()
    sum_d = 0
    
    while len(selected) < m and heap:
        distance, pos = heapq.heappop(heap)
        if pos in x_set or pos in selected_set:
            continue
        
        selected.append(pos)
        sum_d += distance
        selected_set.add(pos)
        
        # Generate neighbors
        for delta in (-1, 1):
            neighbor = pos + delta
            if neighbor in x_set or neighbor in selected_set or neighbor in visited:
                continue
            
            # Compute minimal distance for neighbor
            idx_neighbor = bisect.bisect_left(sorted_x, neighbor)
            if idx_neighbor == 0:
                closest = sorted_x[0]
            elif idx_neighbor == len(sorted_x):
                closest = sorted_x[-1]
            else:
                left = sorted_x[idx_neighbor - 1]
                right = sorted_x[idx_neighbor]
                if neighbor - left <= right - neighbor:
                    closest = left
                else:
                    closest = right
