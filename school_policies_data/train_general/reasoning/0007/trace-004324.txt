To solve this problem, we need to determine the probability that a specific subsequence of film endings remains unchanged after a hypothetical rearrangement. The solution involves efficiently computing the product of permutations for each query, considering the constraints on different values of k and large input sizes.

### Approach

1. **Precompute Factorials and Inverses**: Compute factorials and their modular inverses up to a large prime modulus (998244353) to handle permutations efficiently.

2. **Mo's Algorithm**: Use Mo's algorithm to process multiple range queries efficiently. This algorithm helps in maintaining the current window of elements and dynamically updating the product of permutations as the window changes.

3. **Handling Multiple k Values**: Precompute permutation products for each possible value of k to avoid redundant calculations. This is crucial since the problem guarantees a limited number of different k values (up to 100).

4. **Efficient Query Processing**: For each query, compute the product of permutations for the given range and multiply it by the permutations of the remaining elements, leveraging precomputed factorials for quick calculations.

import sys
import math
from collections import defaultdict, deque

MOD = 998244353

def main():
    input = sys.stdin.read().split()
    ptr = 0
    n, m, q = map(int, input[ptr:ptr+3])
    ptr +=3
    e = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    queries = []
    unique_ks = set()
    for _ in range(q):
        l = int(input[ptr])
        r = int(input[ptr+1])
        k = int(input[ptr+2])
        queries.append( (l-1, r-1, k) )
        unique_ks.add(k)
        ptr +=3
    
    max_fact = MOD
    fact = [1]*(max_fact)
    for i in range(1, max_fact):
        fact[i] = fact[i-1] * i % MOD
    inv_fact = [1]*(max_fact)
    inv_fact[max_fact-1] = pow(fact[max_fact-1], MOD-2, MOD)
    for i in range(max_fact-2, -1, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD

    original_count = defaultdict(int)
    pos = defaultdict(list)
    for idx, val in enumerate(e):
        original_count[val] +=1
        pos[val].append(idx)
    
    e_array = e

    mo_blocksize = int(math.sqrt(n)) + 1

    queries_grouped = defaultdict(list)
    for idx, (l, r, k) in enumerate(queries):
        queries_grouped[k].append( (l, r, idx) )
    
    answer = [0]*q

    for k in unique_ks:
        fact_e = defaultdict(list)
        inv_fact_e = defaultdict(list)
        for val in pos.keys():
            oc = original_count[val]
            a_e = k + oc
            max_x = oc
            fact_val = [1]*(max_x+1)
            for x in range(1, max_x+1):
                if a_e - (x-1) <0:
                    fact_val[x] = 0
                else:
                    fact_val[x] = fact_val[x-1] * (a_e - (x-1)) % MOD
            fact_e[val] = fact_val
            inv_fact_val = [1]*(max_x+1)
            for x in range(0, max_x+1):
                if fact_val[x] ==0:
                    inv_fact_val[x] = 0
                else:
                    inv_fact_val[x] = pow(fact_val[x], MOD-2, MOD)
            inv_fact_e[val] = inv_fact_val
        
        q_list = queries_grouped[k]
        for i in range(len(q_list)):
            l, r, idx = q_list[i]
            q_list[i] = (l, r, idx, i)
        
        q_list.sort(key=lambda x: (x[0]//mo_blocksize, x[1] if (x[0]//mo_blocksize)%2 ==0 else -x[1]))
        
        current_l = 0
        current_r = -1
        freq = defaultdict(int)
        current_product = 1
        
        for qq in q_list:
            l, r, idx, _ = qq
            while current_r < r:
                current_r +=1
                val = e_array[current_r]
                old = freq[val]
                current_product = current_product * inv_fact_e[val][old] % MOD
                freq[val] +=1
                new = freq[val]
                if new >= len(fact_e[val]):
                    current_product =0
                else:
                    current_product = current_product * fact_e[val][new] % MOD
            while current_r > r:
                val = e_array[current_r]
                old = freq[val]
                current_product = current_product * inv_fact_e[val][old] % MOD
                freq[val] -=1
                new = freq[val]
                if new >= len(fact_e[val]):
                    current_product =0
                else:
                    current_product = current_product * fact_e[val][new] % MOD
                current_r -=1
            while current_l < l:
                val = e_array[current_l]
                old = freq[val]
                current_product = current_product * inv_fact_e[val][old] % MOD
                freq[val] -=1
                new = freq[val]
                if new >= len(fact_e[val]):
                    current_product =0
                else:
                    current_product = current_product * fact_e[val][new] % MOD
                current_l +=1
            while current_l > l:
                current_l -=1
                val = e_array[current_l]
                old = freq[val]
                current_product = current_product * inv_fact_e[val][old] % MOD
                freq[val] +=1
                new = freq[val]
                if new >= len(fact_e[val]):
                    current_product =0
                else:
                    current_product = current_product * fact_e[val][new] % MOD
            
            len_ = r - l +1
            s_total = m * k + n
            s = s_total - len_
            t = n - len_
            if t <0:
                product_rest =0
            else:
                r_mod = s % MOD
                if r_mod < t:
                    product_rest =0
                else:
                    if r_mod -t <0:
                        product_rest =0
                    else:
                        product_rest = fact[r_mod] * inv_fact[r_mod -t] % MOD
            
            res = current_product * product_rest % MOD
            answer[idx] = res
        
    for ans in answer:
        print(ans)

if __name__ == "__main__":
    main()
