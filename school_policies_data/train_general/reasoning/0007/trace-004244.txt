import sys
import heapq

def dijkstra(n, adj, start):
    d = [float('inf')] * (n+1)
    cnt = [0] * (n+1)
    d[start] = 0
    cnt[start] = 1
    heap = []
    heapq.heappush(heap, (0, start))
    while heap:
        dist_u, u = heapq.heappop(heap)
        if dist_u > d[u]:
            continue
        for v, w in adj[u]:
            if d[v] > d[u] + w:
                d[v] = d[u] + w
                cnt[v] = cnt[u]
                heapq.heappush(heap, (d[v], v))
            elif d[v] == d[u] + w:
                cnt[v] += cnt[u]
    return d, cnt

2. **Count Shortest Paths**: For each city, count the number of shortest paths from `s` to the city (`cnt_s`) and from the city to `t` (`cnt_t`).
3. **Check Each Road**:
   - If the road is part of some shortest path, check if it is part of all shortest paths by comparing the product of counts `cnt_s[a] * cnt_t[b]` with the total number of shortest paths from `s` to `t`.
   - If the road is not part of any shortest path, determine if it can be repaired by adjusting its travel time such that it becomes part of all shortest paths.

import sys
import heapq

def dijkstra(n, adj, start):
    d = [float('inf')] * (n + 1)
    cnt = [0] * (n + 1)
    d[start] = 0
    cnt[start] = 1
    heap = []
    heapq.heappush(heap, (0, start))
    
    while heap:
        dist_u, u = heapq.heappop(heap)
        if dist_u > d[u]:
            continue
        for v, w in adj[u]:
            if d[v] > d[u] + w:
                d[v] = d[u] + w
                cnt[v] = cnt[u]
                heapq.heappush(heap, (d[v], v))
            elif d[v] == d[u] + w:
                cnt[v] += cnt[u]
    return d, cnt

def main():
    n, m, s, t = map(int, sys.stdin.readline().split())
    edges = []
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        a, b, l = map(int, sys.stdin.readline().split())
        edges.append((a, b, l))
        adj[a].append((b, l))
        rev_adj[b].append((a, l))
    
    # Compute shortest paths and counts from s
    d_s, cnt_s = dijkstra(n, adj, s)
    
    # Compute shortest paths and counts to t (using reversed graph)
    d_t_rev, cnt_t_rev = dijkstra(n, rev_adj, t)
    d_t = d_t_rev
    cnt_t = cnt_t_rev
    
    original_d = d_s[t]
    
    for a, b, l in edges:
        if d_s[a] + l + d_t[b] == original_d:
            if cnt_s[a] * cnt_t[b] == cnt_s[t]:

                if new_w >= 1 and new_w < l:
                    cost = l - new_w
                    print(f"CAN {cost}")

                    if new_sum < original_d and 1 < l:
