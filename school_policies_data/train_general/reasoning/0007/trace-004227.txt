import sys
import math

def main():

    lb, rb, tb = map(int, sys.stdin.readline().split())
    
    d = math.gcd(ta, tb)
    max_overlap = 0
    
    for c in range(d):
        # Compute a_min for Alice
        rem_alice = la % d
        delta_alice = (c - rem_alice) % d
        a_min = la + delta_alice
        if a_min > ra:

        current_min = min(len_a, len_b)
        if current_min > max_overlap:
