To solve this problem, we need to calculate the number of unique cells painted by Vika on an infinite sheet of squared paper. The cells can be covered by horizontal or vertical segments, and overlapping cells should be counted only once.

### Approach
1. **Identify Segments**: Separate the input segments into horizontal and vertical segments.
2. **Merge Intervals**: For each row (horizontal segments) and column (vertical segments), merge overlapping intervals to avoid counting the same cell multiple times.
3. **Calculate Coverage**: Compute the total number of cells covered by horizontal and vertical segments separately by summing the lengths of all merged intervals.
4. **Find Overlaps**: Determine the number of cells that are covered by both horizontal and vertical segments by checking intersections between merged intervals of horizontal and vertical segments.

### Solution Code

import bisect

def main():
    import sys
    input = sys.stdin.read().split()
    idx = 0
    n = int(input[idx])
    idx += 1
    
    horizontal = {}
    vertical = {}
    
    for _ in range(n):
        x1 = int(input[idx])
        y1 = int(input[idx+1])
        x2 = int(input[idx+2])
        y2 = int(input[idx+3])
        idx +=4
        
        if y1 == y2:
            # horizontal
            y = y1
            x_start = min(x1, x2)
            x_end = max(x1, x2)
            if y not in horizontal:
                horizontal[y] = []
            horizontal[y].append((x_start, x_end))
        else:
            # vertical
            x = x1
            y_start = min(y1, y2)
            y_end = max(y1, y2)
            if x not in vertical:
                vertical[x] = []
            vertical[x].append((y_start, y_end))
    
    # Merge horizontal intervals
    merged_horizontal = {}
    for y in horizontal:
        intervals = horizontal[y]
        intervals.sort()
        merged = []
        for start, end in intervals:
            if not merged:
                merged.append([start, end])
            else:
                last = merged[-1]
                if start <= last[1] + 1:
                    # merge
                    new_start = last[0]
                    new_end = max(last[1], end)
                    merged[-1] = [new_start, new_end]
                else:
                    merged.append([start, end])
        merged_horizontal[y] = merged
    
    # Merge vertical intervals
    merged_vertical = {}
    for x in vertical:
        intervals = vertical[x]
        intervals.sort()
        merged = []
        for start, end in intervals:
            if not merged:
                merged.append([start, end])
            else:
                last = merged[-1]
                if start <= last[1] + 1:
                    new_start = last[0]
                    new_end = max(last[1], end)
                    merged[-1] = [new_start, new_end]
                else:
                    merged.append([start, end])
        merged_vertical[x] = merged
    
    # Compute horizontal_cells
    horizontal_cells = 0
    for y in merged_horizontal:
        for start, end in merged_horizontal[y]:
            horizontal_cells += end - start + 1
    
    # Compute vertical_cells
    vertical_cells = 0
    for x in merged_vertical:
        for start, end in merged_vertical[x]:
            vertical_cells += end - start + 1
    
    # Preprocess merged_vertical for overlap_cells
    sorted_xv = sorted(merged_vertical.keys())
    xv_list = []
    xv_to_intervals = {}
    for x in sorted_xv:
        xv_list.append(x)
        xv_to_intervals[x] = merged_vertical[x]
    
    overlap_cells = 0
    
    # Process each merged horizontal interval
    for y in merged_horizontal:
        for x_start, x_end in merged_horizontal[y]:
            left = bisect.bisect_left(xv_list, x_start)
            right_idx = bisect.bisect_right(xv_list, x_end)
            
            for i in range(left, right_idx):
                xv = xv_list[i]
                y_intervals = xv_to_intervals[xv]
                # Check if y is in any of the y_intervals
                low = 0
                high = len(y_intervals) - 1
                found = False
                while low <= high:
                    mid = (low + high) // 2
                    s, e = y_intervals[mid]
                    if s <= y <= e:
                        found = True
                        break
                    elif y < s:
                        high = mid - 1
                    else:
                        low = mid + 1
                if found:
                    overlap_cells += 1
    
    total = horizontal_cells + vertical_cells - overlap_cells
    print(total)

if __name__ == "__main__":
    main()
