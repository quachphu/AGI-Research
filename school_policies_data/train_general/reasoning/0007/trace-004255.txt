import math

def count_lazy_numbers():
    import sys
    input = sys.stdin.read
    data = input().split()
    t = int(data[0])
    index = 1
    for _ in range(t):
        l = int(data[index])
        r = int(data[index + 1])
        index += 2
        a = math.isqrt(r)
        b = math.isqrt(l - 1)
        print(a - b)

- **Reading Input Efficiently**: The code reads all input at once for efficiency, especially given the constraints of up to 100,000 test cases.
