    prefix = [0] * (n +1)

    for i in range(n):

        prefix[i+1] = prefix[i] + diffs[i]

    # Compute sliding window minimum for the doubled array (prefix[1..n] + prefix[1..n])

    doubled = prefix[1:] + prefix[1:]

    min_in_window = [0]*(n)

    dq = deque()

    for i in range(len(doubled)):

        # Maintain deque for window of size n

        # Remove elements out of the window [i-n+1, i]

        while dq and dq[0] < i -n +1:

            dq.popleft()

        # Remove elements larger than current

        while dq and doubled[dq[-1]] >= doubled[i]:

            dq.pop()

        dq.append(i)

        # For windows starting at s =0..n-1, the window starts at i-n+1

        # For i >=n-1:

        if i >=n-1:

            s = i -n +1

            if s <n:

import sys
from collections import deque

def find_valid_starts(diffs):
    n = len(diffs)
    if n == 0:
        return []
    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + diffs[i]
    doubled = prefix[1:] + prefix[1:]
    min_in_window = [0] * n
    dq = deque()
    for i in range(len(doubled)):
        while dq and dq[0] < i - n + 1:
            dq.popleft()
        while dq and doubled[dq[-1]] >= doubled[i]:
            dq.pop()
        dq.append(i)
        if i >= n - 1:
            s = i - n + 1
            if s < n:
                min_in_window[s] = doubled[dq[0]]
    valid = []
    for s in range(n):
        if min_in_window[s] - prefix[s] >= 0:
            valid.append(s)
    return valid

def main():
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    b = list(map(int, sys.stdin.readline().split()))
    
    # Clockwise
    cw_diff = [a[i] - b[i] for i in range(n)]
    valid_cw = find_valid_starts(cw_diff)
    
    # Counter-clockwise
    ccw_diff = [a[i] - b[(i - 1) % n] for i in range(n)]
    rev_ccw_diff = ccw_diff[::-1]
    valid_rev_ccw = find_valid_starts(rev_ccw_diff)
    valid_ccw = [(n - s - 1) % n for s in valid_rev_ccw]
    
    # Combine results
    valid_all = set(valid_cw) | set(valid_ccw)
    valid_all_1based = sorted([x + 1 for x in valid_all])
    print(len(valid_all_1based))
    print(' '.join(map(str, valid_all_1based)))

if __name__ == "__main__":
    main()

### Explanation
1. **Reading Input**: The input is read and parsed into the number of post offices, fuel available at each station, and distances between consecutive stations.
2. **Clockwise Check**: Calculate the difference between fuel and distance for each station. Use a sliding window minimum technique to find valid starting points.
3. **Counter-clockwise Check**: Reverse the order of stations and distances, then apply the same algorithm. Convert the valid starting points back to the original indices.
4. **Combining Results**: Merge the valid starting points from both directions, convert them to 1-based indexing, and print the results.

This approach efficiently checks both directions using a sliding window technique, ensuring optimal performance even for large input sizes.
