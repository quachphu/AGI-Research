import sys
from sys import stdin
from collections import deque

def main():
    sys.setrecursionlimit(1 << 25)
    n = int(stdin.readline())
    edges = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, stdin.readline().split())
        edges[u].append(v)
        edges[v].append(u)
    colors = list(map(int, stdin.readline().split()))
    # Check if all colors are the same
    all_same = True
    first_color = colors[0]
    for c in colors:
        if c != first_color:
            all_same = False
            break
    if all_same:
        print("YES")
        print(1)
        return

    # Find all critical edges (u, v where colors are different)
    critical_edges = []
    for u in range(1, n+1):
        for v in edges[u]:
            if v > u:  # Avoid duplicate edges
                if colors[u-1] != colors[v-1]:
                    critical_edges.append((u, v))
    if not critical_edges:
        print("YES")
        print(1)
        return

    # Collect all candidate nodes (common endpoints of all critical edges)
    candidates = set(critical_edges[0])
    for u, v in critical_edges[1:]:
        current = {u, v}
        candidates.intersection_update(current)
        if not candidates:
            print("NO")
            return

    # Check each candidate
    for candidate in candidates:
        valid = True
        # Get all neighbors
        for neighbor in edges[candidate]:
            # Check if the subtree starting at 'neighbor' (excluding 'candidate') has all same color
            visited = {candidate, neighbor}
            stack = [neighbor]
            expected_color = colors[neighbor-1]
            is_mono = True
            while stack:
                node = stack.pop()
                for adj_node in edges[node]:
                    if adj_node in visited:

                        break

                if not is_mono:
                    break
            if not is_mono:
                valid = False
                break
        if valid:
            print("YES")
            print(candidate)
            return
