def poly_mul(p, q, mod):
    # Multiply two polynomials p and q, with coefficients modulo 'mod'.
    m = len(p)
    n = len(q)
    res = [0]*(m + n - 1)
    for i in range(m):
        for j in range(n):
            res[i+j] = (res[i+j] + p[i] * q[j]) % mod
    # Trim trailing zeros

def poly_mod(a, b, mod):
    # Compute a mod b, where b is a polynomial.
    # Returns the remainder of a divided by b.
    a = [x % mod for x in a]
    b = [x % mod for x in b]
    while len(a) >= len(b):
        if len(b) == 0:
            return [0]
        degree_diff = len(a) - len(b)
        if degree_diff < 0:
            break
        # Compute leading coefficient of a and b.
        lc_a = a[-1]
        lc_b = b[-1]
        if lc_b == 0:
            # Can't divide by zero.
            break
        # Compute the multiplicative inverse of lc_b mod mod.
        # Since mod is a prime, and lc_b != 0 mod mod.
        inv_lc_b = pow(lc_b, mod-2, mod)
        factor = (lc_a * inv_lc_b) % mod
        # Subtract factor * b * x^degree_diff from a.
        shifted_b = [0]*degree_diff + [ (x * factor) % mod for x in b ]
        # Pad a to the same length as shifted_b.
        max_len = max(len(a), len(shifted_b))
        a_padded = a + [0]*(max_len - len(a))
        shifted_b_padded = shifted_b + [0]*(max_len - len(shifted_b))
        # Subtract
        a = [ (a_padded[i] - shifted_b_padded[i]) % mod for i in range(max_len) ]
        # Trim trailing zeros
        while len(a) > 0 and a[-1] == 0:
            a.pop()
    return a if len(a) > 0 else [0]

Polynomial gcd:

def poly_gcd(a, b, mod):
    # Compute the GCD of a and b.
    while len(b) > 0 and b[-1] == 0:
        b.pop()
    if len(b) == 0:
        return a
    remainder = poly_mod(a, b, mod)
    return poly_gcd(b, remainder, mod)
