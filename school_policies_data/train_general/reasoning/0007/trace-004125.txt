To solve this problem, we need to compute the probability that a randomly drawn hand of L cards from a deck matches a given hand pattern. The pattern can include wildcards (*) and variables (a, b, c) with optional pluses (+), which denote ranks relative to the variable's base rank.

### Approach
1. **Parse the Input**: Read the dataset parameters (N, M, L) and the hand pattern.
2. **Extract Variables and Plus Counts**: Identify variables and their maximum plus counts from the hand pattern.
3. **Check Validity of Variables**: Ensure variables can be assigned valid base ranks within the given constraints.
4. **Generate Variable Assignments**: Enumerate all possible base rank assignments for the variables.
5. **Calculate Required Counts**: For each assignment, determine the number of cards required for each rank based on the variables and their plus counts.
6. **Check Validity of Required Counts**: Ensure the required counts do not exceed the available cards per rank (N).
7. **Compute Combinations for Required Counts**: Calculate the number of ways to choose the required cards.
8. **Dynamic Programming for Wildcards**: Use dynamic programming to compute the number of ways to choose the remaining cards (wildcards) from the available ranks.
9. **Sum Valid Combinations**: Sum the valid combinations across all assignments and compute the probability.

import math
from itertools import product
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read().split()
    idx = 0
    while True:
        N = int(input[idx])
        M = int(input[idx+1])
        L = int(input[idx+2])
        idx +=3
        if N ==0 and M ==0 and L ==0:
            break
        hand_pattern = input[idx:idx+L]
        idx +=L
        
        variables = set()
        var_max_plus = defaultdict(int)
        for cp in hand_pattern:
            if cp == '*':
                continue
            var = cp[0]
            plus_count = len(cp) -1
            variables.add(var)
            if plus_count > var_max_plus[var]:
                var_max_plus[var] = plus_count
        variables = sorted(list(variables))
        
        valid = True
        for var in variables:
            max_p = var_max_plus[var]
            if M - max_p <1:
                valid = False
                break
        if not valid:
            print("0.0000000000")
            continue
        
        if not variables:
            total_hands = math.comb(N*M, L)
            print("{0:.10f}".format(1.0))
            continue
        
        ranges = []
        for var in variables:
            max_p = var_max_plus[var]
            start = 1
            end = M - max_p
            ranges.append(range(start, end+1))
        
        total =0
        nm_total = N * M
        for assignment in product(*ranges):
            var_to_base = {var: base for var, base in zip(variables, assignment)}
            required_counts = defaultdict(int)
            for cp in hand_pattern:
                if cp == '*':
                    continue
                var = cp[0]
                plus_count = len(cp) -1
                base = var_to_base[var]
                rank = base + plus_count
                required_counts[rank] +=1
            
            invalid = False
            for r, cnt in required_counts.items():
                if cnt > N:
                    invalid = True
                    break
            if invalid:
                continue
            
            available = {}
            for r in range(1, M+1):
                available[r] = N - required_counts.get(r,0)
                if available[r] <0:
                    invalid = True
                    break
            if invalid:
                continue
            
            s = L - sum(required_counts.values())
            if s <0:
                continue
            
            product_comb = 1
            for r, cnt in required_counts.items():
                product_comb *= math.comb(N, cnt)
            
            if s ==0:
                ways_for_s = 1
            else:
                dp = [0]*(s+1)
                dp[0] =1
                for r in range(1, M+1):
                    current_available = available[r]
                    max_take = min(current_available, s)
                    temp_dp = [0]*(s+1)
                    for take in range(0, max_take +1):
                        if take > current_available:
                            continue
                        ways = math.comb(current_available, take)
                        for i in range(take, s+1):
                            temp_dp[i] += dp[i - take] * ways
                    dp = temp_dp
                ways_for_s = dp[s]
            
            total += product_comb * ways_for_s
        
        if nm_total < L:
            print("0.0000000000")
            continue
        total_hands = math.comb(nm_total, L)
        if total_hands ==0:
            print("0.0000000000")
        else:
            prob = total / total_hands
            print("{0:.10f}".format(prob))
            
if __name__ == '__main__':
    main()
