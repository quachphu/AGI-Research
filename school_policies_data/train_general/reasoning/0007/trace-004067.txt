import sys

def process_amidakuji():
    while True:
        line = sys.stdin.readline()
        if not line:
            break
        line = line.strip()
        if not line:
            continue
        n, m, a = map(int, line.split())
        if n == 0 and m == 0 and a == 0:
            break
        horizontal_lines = []
        for _ in range(m):
            h, p, q = map(int, sys.stdin.readline().split())
            horizontal_lines.append((h, p, q))
        # Sort horizontal lines in descending order of height
        horizontal_lines.sort(reverse=True, key=lambda x: x[0])
        current = a
        for h, p, q in horizontal_lines:
            if current == p:
                current = q
            elif current == q:
                current = p
        print(current)

3. **Tracing the Path**: Starting from the initial vertical line (a), each horizontal line is checked. If the current vertical line matches either end of the horizontal line, the position is updated to the connected vertical line. This process continues until all horizontal lines are processed, resulting in the final vertical line position.
