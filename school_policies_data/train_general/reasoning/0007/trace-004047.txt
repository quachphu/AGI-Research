    key = tuple(exponents)
    if key in memo:
        return memo[key]
    if not exponents:
        return 0
    # Determine possible k values
    max_e = exponents[-1]

    for k in range(1, max_e + 1):
        # Check if there is at least one exponent >=k
        if exponents[-1] >=k:

        for e in exponents:

                if new_e >0:

        s.add(compute_grundy(tuple(new_exponents)))
    mex = 0
    while mex in s:

    return mex

### Approach
1. **Prime Factorization**: For each number in the list, factorize it into its prime factors and their respective exponents.
2. **Grundy Number Calculation**: For each prime factor, compute the Grundy number based on the exponents of that prime in the list. This involves recursively determining the minimum excludant (mex) value for all possible moves.
3. **XOR of Grundy Numbers**: The overall game state is determined by XORing the Grundy numbers of all primes. If the result is non-zero, Mojtaba (the first player) wins; otherwise, Arpa wins.

import sys
from sys import stdin
from collections import defaultdict
import math

def factorize(x):
    factors = defaultdict(int)
    if x == 1:
        return factors
    while x % 2 == 0:
        factors[2] += 1
        x //= 2
    i = 3
    while i * i <= x:
        while x % i == 0:
            factors[i] += 1
            x //= i
        i += 2
    if x > 1:
        factors[x] += 1
    return factors

def compute_grundy(exponents):
    key = tuple(exponents)
    if key in memo:
        return memo[key]
    if not exponents:
        return 0

    s = set()

                new_e = e - k
                if new_e > 0:
                    new_exponents.append(new_e)
            else:
                new_exponents.append(e)
        new_exponents.sort()
        new_key = tuple(new_exponents)
        s.add(compute_grundy(new_key))
    mex = 0
    while mex in s:
        mex += 1
    memo[key] = mex
    return mex

        key = tuple(exponents)
