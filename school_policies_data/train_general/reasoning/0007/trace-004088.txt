from collections import defaultdict

def main():

    caps = [tuple(map(int, sys.stdin.readline().split())) for _ in range(m)]

    marker_dict = defaultdict(lambda: {'total':0, 'colors': defaultdict(int)})

    for x, y in markers:

        d = y

        marker_dict[d]['total'] +=1

    cap_dict = defaultdict(lambda: {'total':0, 'colors': defaultdict(int)})

    for a, b in caps:

        d = b

        cap_dict[d]['total'] +=1

        cap_dict[d]['colors'][a] +=1

    all_diameters = set(marker_dict.keys()).union(cap_dict.keys())

    u = 0

    v = 0

    for d in all_diameters:

        m_total = marker_dict[d]['total']

        c_total = cap_dict[d]['total']

        if possible ==0:

        for color, mc in marker_dict[d]['colors'].items():

            cc = cap_dict[d]['colors'].get(color, 0)

            sum_v += min(mc, cc)

        v += sum_v

    print(u, v)

To solve this problem, we need to determine the maximum number of markers that can be closed using available caps, and among those, the maximum number of markers that can be beautifully closed (where both the color and diameter match). The solution involves efficiently grouping markers and caps by their diameters and colors to compute the required values.

### Approach
1. **Group Markers and Caps by Diameter**: First, we group markers and caps by their diameters. For each diameter, we also track the count of each color present.
2. **Calculate Possible Closures**: For each diameter, compute the maximum number of markers that can be closed using the available caps. This is determined by the minimum of the count of markers and caps for that diameter.
3. **Calculate Beautiful Closures**: For each diameter, compute the number of beautifully closed markers by considering the minimum count of each color present in both markers and caps for that diameter.

### Solution Code
```python
import sys
from collections import defaultdict

def main():
    n, m = map(int, sys.stdin.readline().split())
    markers = [tuple(map(int, sys.stdin.readline().split())) for _ in range(n)]
    caps = [tuple(map(int, sys.stdin.readline().split())) for _ in range(m)]
    
    # Group markers by diameter and color
    marker_dict = defaultdict(lambda: {'total': 0, 'colors': defaultdict(int)})
    for x, y in markers:
        d = y
        marker_dict[d]['total'] += 1
        marker_dict[d]['colors'][x] += 1
    
    # Group caps by diameter and color
    cap_dict = defaultdict(lambda: {'total': 0, 'colors': defaultdict(int)})
    for a, b in caps:
        d = b
        cap_dict[d]['total'] += 1
        cap_dict[d]['colors'][a] += 1
    
    all_diameters = set(marker_dict.keys()).union(cap_dict.keys())
    u = 0
    v = 0
    
    for d in all_diameters:
        m_total = marker_dict[d]['total']
        c_total = cap_dict[d]['total']
        possible = min(m_total, c_total)
        if possible == 0:
            continue
        u += possible
        
        sum_v = 0
        for color, mc in marker_dict[d]['colors'].items():
            cc = cap_dict[d]['colors'].get(color, 0)
            sum_v += min(mc, cc)
        v += sum_v
    
    print(u, v)

if __name__ == "__main__":
    main()
```

### Explanation
1. **Grouping by Diameter and Color**: We use dictionaries to group markers and caps by their diameters. Each entry in the dictionary tracks the total count of markers/caps for that diameter and the count of each color within that diameter.
2. **Processing Each Diameter**: For each diameter, we determine the maximum number of markers that can be closed (possible closures) and the number of beautifully closed markers by comparing the counts of each color in both markers and caps.
3. **Efficiency**: The use of dictionaries allows efficient grouping and counting, ensuring the solution handles the upper constraint limits effectively.

This approach ensures that we maximize the number of closed markers first and then maximize the beautifully closed markers within those constraints, providing an optimal solution in O(n + m) time complexity.
