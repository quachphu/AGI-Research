        ptr += 3
        if X == 0 or Y == 0 or Z == 0:
            continue
        # Read B matrix
        B = []
        for i in range(X):
            B.append([])
            for j in range(Y):
                line = list(map(int, input[ptr:ptr+Z]))
                ptr += Z
                B[i].append(line)
        # Compute A matrix
        A = [[[0]*Z for _ in range(Y)] for __ in range(X)]
        for i in range(X):
            for j in range(Y):
                for k in range(Z):
                    sum_val = B[i][j][k]
                    if i > 0:
                        sum_val -= B[i-1][j][k]
                    if j > 0:
                        sum_val -= B[i][j-1][k]
                    if k > 0:
                        sum_val -= B[i][j][k-1]
                    if i > 0 and j > 0:
                        sum_val += B[i-1][j-1][k]
                    if i > 0 and k > 0:
                        sum_val += B[i-1][j][k-1]
                    if j > 0 and k > 0:
                        sum_val += B[i][j-1][k-1]
                    if i > 0 and j > 0 and k > 0:
                        sum_val -= B[i-1][j-1][k-1]
                    A[i][j][k] = sum_val
        # Output A matrix
        for i in range(X):
            for j in range(Y):
                line = ' '.join(map(str, A[i][j])) + ' '
                print(line)
