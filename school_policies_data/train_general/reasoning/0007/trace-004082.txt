import sys
from collections import deque

MOD = 10**9 + 7

def main():
    sys.setrecursionlimit(1 << 25)
    n = int(sys.stdin.readline())
    lr = []
    for _ in range(n):
        l, r = map(int, sys.stdin.readline().split())
        lr.append((l, r))
    
    edges = [[] for _ in range(n)]
    for _ in range(n-1):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges[u].append(v)
        edges[v].append(u)
    
    # Compute parent and depth using BFS
    parent = [-1] * n
    depth = [0] * n
    q = deque([0])
    visited = [False] * n
    visited[0] = True
    while q:
        u = q.popleft()
        for v in edges[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                depth[v] = depth[u] + 1
                q.append(v)
    
    # Precompute LCA binary lifting table (simple but not optimal for large n)
    LOG = 20
    up = [[-1]*n for _ in range(LOG)]
    up[0] = parent[:]
    for k in range(1, LOG):
        for v in range(n):
            if up[k-1][v] != -1:
                up[k][v] = up[k-1][up[k-1][v]]
    
    def lca(u, v):
        if depth[u] < depth[v]:
            u, v = v, u
        # Bring u to the same depth as v
        for k in range(LOG-1, -1, -1):
            if depth[u] - (1 << k) >= depth[v]:
                u = up[k][u]
        if u == v:
            return u
        for k in range(LOG-1, -1, -1):
            if up[k][u] != up[k][v]:
                u = up[k][u]
                v = up[k][v]
        return parent[u]
    
    def distance(u, v):
        ancestor = lca(u, v)
        return depth[u] + depth[v] - 2 * depth[ancestor]
    
    # Precompute A_i and inv_A[i]
    product_A = 1
    A = []
    inv_A = []
    for l, r in lr:
        a = r - l + 1
        A.append(a)
        product_A = product_A * a % MOD
        inv_a = pow(a, MOD-2, MOD)
        inv_A.append(inv_a)
    
    # Precompute s_ij for all pairs i, j
    total = 0
    for i in range(n):
        for j in range(i+1, n):
            li, ri = lr[i]
            lj, rj = lr[j]
            a = max(li, lj)
            b = min(ri, rj)
            if a > b:
                s = 0
            else:
                s = b - a + 1
            d = distance(i, j)
            term = s * inv_A[i] % MOD
            term = term * inv_A[j] % MOD
            term = term * d % MOD
            total = (total + term) % MOD
    
    # Multiply by product_A
    ans = total * product_A % MOD
