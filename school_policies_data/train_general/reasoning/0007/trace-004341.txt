import sys
from sys import stdin
from collections import defaultdict

def main():
    sys.setrecursionlimit(1 << 25)
    n, m, k, q = map(int, stdin.readline().split())
    edges = []
    for _ in range(m):
        a, b = map(int, stdin.readline().split())
        edges.append((a, b))
    
    max_color = k
    # Initialize DSU structures for each color (1-based)
    parent = [ [i for i in range(n+1)] for _ in range(max_color+1)]  # parent[c][u]
    rank = [ [1]*(n+1) for _ in range(max_color+1) ]
    parity = [ [0]*(n+1) for _ in range(max_color+1) ]
    # To track the union operations for each edge in each color
    edge_unions = [ defaultdict(list) for _ in range(max_color+1) ]  # edge_unions[c][ei] = list of (root, old_parent, old_rank)
    current_color = [0] * (m + 1)  # 1-based
    
    def find(c, u):
        original_u = u
        p = u
        p_parity = 0
        while parent[c][p] != p:
            p_parity ^= parity[c][p]
            p = parent[c][p]
        return (p, p_parity)
    
    for _ in range(q):
        parts = stdin.readline().split()
        while len(parts) < 2:
            parts += stdin.readline().split()
        ei, ci = map(int, parts)
        ei_ = ei
        old_col = current_color[ei_]
        if old_col == ci:
            print("YES")
            continue
        u, v = edges[ei_ - 1]

        valid = True

        rv, pv = find(ci, v)
        if ru == rv:
            if (pu + pv) % 2 == 0:

        else:
            valid = True
        
        if not valid:
            print("NO")
            continue

        # Remove from old_col if necessary
        if old_col != 0:
            # Rollback the unions for this edge in old_col
            if ei_ in edge_unions[old_col]:

                for record in reversed(edge_unions[old_col][ei_]):
                    r, old_p, old_r = record
                    parent[old_col][r] = old_p
                    rank[old_col][r] = old_r
