class SegmentTree:
    def __init__(self, size):
        self.n = 1
        while self.n < size:
            self.n <<=1
        self.size = self.n
        self.tree = [0]*(2*self.n)
        self.lazy = [0]*(2*self.n)

    def push(self, node, l, r):
        if self.lazy[node] !=0:
            self.tree[node] += self.lazy[node]
            if l != r:
                self.lazy[2*node] += self.lazy[node]
                self.lazy[2*node+1] += self.lazy[node]
            self.lazy[node] =0

    def range_add(self, a, b, val, node=1, l=0, r=None):
        if r is None:
            r = self.n -1
        self.push(node, l, r)
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.lazy[node] += val
            self.push(node, l, r)
            return
        mid = (l+r)//2
        self.range_add(a, b, val, 2*node, l, mid)
        self.range_add(a, b, val, 2*node+1, mid+1, r)
        self.tree[node] = max(self.tree[2*node], self.tree[2*node+1])

    def point_update(self, pos, val, node=1, l=0, r=None):
        if r is None:
            r = self.n-1
        self.push(node, l, r)
        if l == r:
            self.tree[node] = val
            return
        mid = (l +r)//2
        if pos <= mid:
            self.point_update(pos, val, 2*node, l, mid)
        else:
            self.point_update(pos, val, 2*node+1, mid+1, r)
        self.tree[node] = max(self.tree[2*node], self.tree[2*node+1])

    def query_max(self, a=0, b=None, node=1, l=0, r=None):
        if b is None:
            b = self.n-1
        if r is None:
            r = self.n-1
        self.push(node, l, r)
        if a > r or b < l:
            return -float('inf')
        if a <= l and r <= b:
            return self.tree[node]
        mid = (l + r)//2
        left = self.query_max(a, b, 2*node, l, mid)
        right = self.query_max(a, b, 2*node+1, mid+1, r)
        return max(left, right)

To solve this problem, we need to determine the maximum profit that can be achieved by repairing roads in Linear Kingdom such that the total money from races held on these roads minus the repair costs is maximized. The solution involves dynamic programming and a segment tree to efficiently manage range updates and queries.

### Approach
1. **Problem Analysis**: The goal is to select a subset of roads to repair such that the profit (sum of earnings from races minus repair costs) is maximized. Each race can only be held if all roads in its interval are repaired.
2. **Dynamic Programming with Segment Tree**: We use dynamic programming (DP) to keep track of the maximum profit up to each road. A segment tree is used to efficiently manage range updates and queries for maximum values, which helps in quickly determining the optimal intervals for repairing roads.
3. **Prefix Sum Array**: Compute prefix sums of repair costs to quickly calculate the cost of any interval.
4. **Sorting and Processing Races**: Races are sorted by their upper bound (ub) to process them in increasing order, allowing us to efficiently update the segment tree as we iterate through each road.

### Solution Code
```python
import sys

class SegmentTree:
    def __init__(self, max_pos):
        self.n = 1
        while self.n < max_pos:
            self.n <<= 1
        self.size = self.n
        self.tree = [-float('inf')] * (2 * self.n)
        self.lazy = [0] * (2 * self.n)

    def push(self, node, l, r):
        if self.lazy[node] != 0:
            self.tree[node] += self.lazy[node]
            if l != r:
                self.lazy[2*node] += self.lazy[node]
                self.lazy[2*node + 1] += self.lazy[node]
            self.lazy[node] = 0

    def range_add(self, a, b, val):
        a -= 1  # convert to 0-based index
        b -= 1
        self._range_add(a, b, val, 1, 0, self.n - 1)

    def _range_add(self, a, b, val, node, l, r):
        self.push(node, l, r)
        if a > r or b < l:
            return
        if a <= l and r <= b:
            self.lazy[node] += val
            self.push(node, l, r)
            return
        mid = (l + r) // 2
        self._range_add(a, b, val, 2*node, l, mid)
        self._range_add(a, b, val, 2*node + 1, mid + 1, r)
        self.tree[node] = max(self.tree[2*node], self.tree[2*node + 1])

    def point_update(self, pos, val):
        pos -= 1  # convert to 0-based index
        self._point_update(pos, val, 1, 0, self.n - 1)

    def _point_update(self, pos, val, node, l, r):
        self.push(node, l, r)
        if l == r:
            self.tree[node] = val
            return
        mid = (l + r) // 2
        if pos <= mid:
            self._point_update(pos, val, 2*node, l, mid)
        else:
            self._point_update(pos, val, 2*node + 1, mid + 1, r)
        self.tree[node] = max(self.tree[2*node], self.tree[2*node + 1])

    def query_max(self):
        return self._query_max(0, self.n - 1, 1, 0, self.n - 1)

    def _query_max(self, a, b, node, l, r):
        self.push(node, l, r)
        if a > r or b < l:
            return -float('inf')
        if a <= l and r <= b:
            return self.tree[node]
        mid = (l + r) // 2
        left = self._query_max(a, b, 2*node, l, mid)
        right = self._query_max(a, b, 2*node + 1, mid + 1, r)
        return max(left, right)

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr +=1
    m = int(input[ptr])
    ptr +=1

    costs = []
    for _ in range(n):
        costs.append(int(input[ptr]))
        ptr +=1

    # Compute prefix sums C
    C = [0]*(n+1)
    for i in range(1, n+1):
        C[i] = C[i-1] + costs[i-1]

    # Read races and group by ub
    races = [[] for _ in range(n+2)]  # races[ub] contains (lb, p)
    for _ in range(m):
        lb = int(input[ptr])
        ptr +=1
        ub = int(input[ptr])
        ptr +=1
        p = int(input[ptr])
        ptr +=1
        races[ub].append( (lb, p) )

    max_pos = n +1
    st = SegmentTree(max_pos)
    st.point_update(1, 0)  # initial position 1 is 0 (dp[0] + C[0])

    dp = [0]*(n+2)
    dp[0] =0

    for i in range(1, n+1):
        # Process all races with ub =i
        for (lb, p) in races[i]:
            st.range_add(1, lb, p)

        # Query the maximum value
        max_val = st.query_max()

        # Compute candidate
        candidate = max_val - C[i]

        # Update dp[i]
        dp[i] = max(dp[i-1], candidate)

        # Update position i+1 in the segment tree
        if i+1 <= max_pos:
            st.point_update(i+1, dp[i] + C[i])

    print(max(dp[n], 0))

if __name__ == "__main__":
    main()
```

### Explanation
1. **Segment Tree Initialization**: The segment tree is initialized to handle range updates and point queries efficiently. This helps in managing the dynamic programming transitions where we need to compute the maximum profit for intervals.
2. **Prefix Sum Calculation**: The prefix sum array `C` is used to quickly compute the repair costs for any interval.
3. **Processing Races**: Races are grouped by their upper bound (ub) and processed in increasing order. For each road `i`, we update the segment tree with the profits of races ending at `i` and then compute the maximum profit for intervals ending at `i`.
4. **Dynamic Programming**: The DP array `dp` keeps track of the maximum profit up to each road. The segment tree helps in efficiently finding the optimal intervals to repair, ensuring the solution runs in O(n log n + m log n) time complexity.
