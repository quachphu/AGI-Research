4. **Distance Calculation**: For each pair in the triplet, compute the minimal distance in both clockwise and counter-clockwise directions in the sorted list. Check if all three pairs have the same minimal distance.
5. **Count Valid Triplets**: Count the number of valid triplets where all pairs have the same minimal distance.

import sys
from itertools import combinations

def main():
    n = int(sys.stdin.readline())
    lines = []
    for _ in range(n):
        a, b = map(int, sys.stdin.readline().split())
        if a > b:
            a, b = b, a
        lines.append((a, b))
    
    count = 0
    # Iterate over all possible triplets of lines
    for triplet in combinations(lines, 3):
        endpoints = set()
        for a, b in triplet:
            endpoints.add(a)
            endpoints.add(b)
        if len(endpoints) != 6:
            continue  # duplicate endpoints
        
        # Sort the six endpoints
        sorted_eps = sorted(endpoints)
        # Create a mapping from endpoint to its index in the sorted list
        pos = {ep: idx for idx, ep in enumerate(sorted_eps)}
        distances = []
        valid = True
        for a, b in triplet:
            i = pos[a]
            j = pos[b]
            if i > j:
                i, j = j, i
            # Compute clockwise steps and counter-clockwise steps
            steps_cw = j - i
            steps_ccw = 6 - steps_cw
            min_steps = min(steps_cw, steps_ccw)
            distances.append(min_steps)
        # Check if all distances are the same
