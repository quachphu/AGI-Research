To solve this problem, we need to efficiently manage the orders of customers at a restaurant where each customer orders a specific type of food. If the ordered type is unavailable, the customer is served the cheapest available alternative. The challenge is to handle these orders efficiently given the constraints.

### Approach
1. **Initialization**: Read the input values and initialize the remaining dishes for each food type.
2. **Segment Tree**: Use a segment tree to keep track of the minimum cost food type available. This allows efficient querying and updating of the available food types.
3. **Customer Order Processing**:
   - **Primary Food Type**: First, try to serve the customer from their ordered food type.
   - **Alternative Food Types**: If the ordered type is insufficient, use the segment tree to find the cheapest available alternative and serve the remaining dishes from there.
4. **Updates**: After serving each customer, update the segment tree if any food type runs out of dishes.

import sys

class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.min_tree = [(float('inf'), -1)] * (2 * self.size)
        for i in range(self.n):
            self.min_tree[self.size + i] = data[i]
        for i in range(self.size - 1, 0, -1):
            self.min_tree[i] = min(self.min_tree[2*i], self.min_tree[2*i + 1])
    
    def update(self, pos, value):
        pos += self.size
        self.min_tree[pos] = value
        pos >>= 1
        while pos >= 1:
            left = self.min_tree[2 * pos]
            right = self.min_tree[2 * pos + 1]
            new_val = min(left, right)
            if self.min_tree[pos] == new_val:
                break
            self.min_tree[pos] = new_val
            pos >>= 1
    
    def get_min(self):
        return self.min_tree[1]

def main():
    input = sys.stdin.read().split()
    ptr = 0
    n, m = int(input[ptr]), int(input[ptr+1])
    ptr +=2
    a = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    c = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    
    # Initialize segment tree data
    st_data = []
    for i in range(n):
        st_data.append( (c[i], i+1) )  # types are 1-based
    
    st = SegmentTree(st_data)
    r = a.copy()
    
    for _ in range(m):
        t_j = int(input[ptr]) -1  # convert to 0-based index
        d_j = int(input[ptr+1])
        ptr +=2
        
        cost = 0
        # Process the preferred type
        if r[t_j] > 0:
            take = min(d_j, r[t_j])
            cost += take * c[t_j]
            d_j -= take
            r[t_j] -= take
            if r[t_j] == 0:
                st.update(t_j, (float('inf'), -1))
        if d_j == 0:
            print(cost)
            continue
        
        # Process remaining d_j using the segment tree
        while d_j > 0:
            current_min, current_idx = st.get_min()
            if current_min == float('inf'):
                cost = 0
                break
            idx = current_idx -1  # convert to 0-based
            available = min(d_j, r[idx])
            cost += available * current_min
            d_j -= available
            r[idx] -= available
            if r[idx] == 0:
                st.update(idx, (float('inf'), -1))
        print(cost)

    main()
