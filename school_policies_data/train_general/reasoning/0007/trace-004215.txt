def compute_area(points):
    n = len(points)
    area = 0.0
    for i in range(n):
        x1, y1 = points[i]
        x2, y2 = points[(i+1)%n]
        area += (x1 * y2) - (x2 * y1)
    return abs(area) / 2.0

def compute_left_area(points, dx, dy, d):
    n = len(points)
    intersections_with_edges = []
    left_vertices = []
    for i in range(n):
        x1, y1 = points[i]
        x2, y2 = points[(i+1)%n]
        proj1 = x1*dx + y1*dy
        proj2 = x2*dx + y2*dy

        if proj1 <= d:
            left_vertices.append((x1, y1))

        if (proj1 - d) * (proj2 - d) < 0:
            t = (d - proj1) / (proj2 - proj1) if (proj2 - proj1) != 0 else 0.0
            ix = x1 + t * (x2 - x1)
            iy = y1 + t * (y2 - y1)
            intersections_with_edges.append((i, (ix, iy)))

    if len(intersections_with_edges) != 2:
        return 0.0

    edge1_idx, intersection1 = intersections_with_edges[0]
    edge2_idx, intersection2 = intersections_with_edges[1]

    start = (edge1_idx + 1) % n
    end = edge2_idx
    left_vertices_between = []

        for i in range(start, end + 1):
            left_vertices_between.append(points[i])

            left_vertices_between.append(points[i])
        for i in range(0, end + 1):
            left_vertices_between.append(points[i])

    return compute_area(left_polygon)
