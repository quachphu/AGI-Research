import bisect

class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.min_count = [ (float('inf'), -1) ] * (2 * self.size)
        for i in range(self.n):
            self.min_count[self.size + i] = (data[i], i)
        for i in range(self.size -1, 0, -1):
            left = self.min_count[2 * i]
            right = self.min_count[2 * i +1]
            if left[0] < right[0]:
                self.min_count[i] = left
            elif right[0] < left[0]:
                self.min_count[i] = right
            else:
                # if counts are equal, choose the leftmost position
                if left[1] < right[1]:
                    self.min_count[i] = left
                else:
                    self.min_count[i] = right

    def update_val(self, pos, value):
        pos += self.size
        self.min_count[pos] = (value, pos - self.size)
        pos >>= 1
        while pos >= 1:
            left = self.min_count[2 * pos]
            right = self.min_count[2 * pos +1]
            if left[0] < right[0]:
                self.min_count[pos] = left
            elif right[0] < left[0]:
                self.min_count[pos] = right
            else:
                if left[1] < right[1]:
                    self.min_count[pos] = left
                else:
                    self.min_count[pos] = right
            pos >>=1

    def query(self, l, r):
        res = (float('inf'), -1)
        l += self.size
        r += self.size
        while l <= r:
            if l % 2 ==1:
                current = self.min_count[l]
                if current[0] < res[0] or (current[0] == res[0] and current[1] < res[1]):
                    res = current
                l +=1
            if r %2 ==0:
                current = self.min_count[r]
                if current[0] < res[0] or (current[0] == res[0] and current[1] < res[1]):
                    res = current
                r -=1
            l >>=1
            r >>=1
        return res

To solve this problem, we need to determine the minimum time required to transport all boxes using the available trucks, where each truck can carry one box at a time and must return to the starting point after each delivery except for the final box. The goal is to minimize the total time by optimally assigning boxes to trucks based on their capacities.

### Approach
1. **Sort Boxes and Trucks**: First, sort the boxes in descending order of weight and the trucks in ascending order of their capacity. This helps in efficiently assigning the heaviest boxes to the most capable trucks first.
2. **Binary Search for Eligible Trucks**: For each box, use binary search to find the leftmost truck that can carry the box. This ensures that we only consider trucks with sufficient capacity.
3. **Segment Tree for Efficient Queries**: Use a segment tree to efficiently query the minimum count of boxes assigned to any eligible truck and update the count after each assignment. This helps in balancing the load across trucks to minimize the maximum time taken by any truck.

### Solution Code
```python
import bisect

class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.min_count = [ (float('inf'), -1) ] * (2 * self.size)
        for i in range(self.n):
            self.min_count[self.size + i] = (data[i], i)
        for i in range(self.size - 1, 0, -1):
            left = self.min_count[2 * i]
            right = self.min_count[2 * i + 1]
            if left[0] < right[0]:
                self.min_count[i] = left
            elif right[0] < left[0]:
                self.min_count[i] = right
            else:
                if left[1] < right[1]:
                    self.min_count[i] = left
                else:
                    self.min_count[i] = right

    def update_val(self, pos, value):
        pos += self.size
        self.min_count[pos] = (value, pos - self.size)
        pos >>= 1
        while pos >= 1:
            left = self.min_count[2 * pos]
            right = self.min_count[2 * pos + 1]
            if left[0] < right[0]:
                self.min_count[pos] = left
            elif right[0] < left[0]:
                self.min_count[pos] = right
            else:
                if left[1] < right[1]:
                    self.min_count[pos] = left
                else:
                    self.min_count[pos] = right
            pos >>= 1

    def query(self, l, r):
        res = (float('inf'), -1)
        l += self.size
        r += self.size
        while l <= r:
            if l % 2 == 1:
                current = self.min_count[l]
                if current[0] < res[0] or (current[0] == res[0] and current[1] < res[1]):
                    res = current
                l += 1
            if r % 2 == 0:
                current = self.min_count[r]
                if current[0] < res[0] or (current[0] == res[0] and current[1] < res[1]):
                    res = current
                r -= 1
            l >>= 1
            r >>= 1
        return res

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1
    boxes = list(map(int, input[ptr:ptr + n]))
    ptr += n
    trucks = list(map(int, input[ptr:ptr + m]))
    ptr += m
    
    boxes.sort(reverse=True)
    trucks_sorted = sorted(trucks)
    counts = [0] * m
    
    st = SegmentTree(counts)
    
    for box in boxes:
        left = bisect.bisect_left(trucks_sorted, box)
        if left >= m:
            continue
        min_count, pos = st.query(left, m-1)
        counts[pos] += 1
        st.update_val(pos, counts[pos])
    
    max_time = max(2 * k - 1 for k in counts)
    print(max_time)

if __name__ == "__main__":
    main()
```

### Explanation
1. **Sorting**: The boxes are sorted in descending order to handle the heaviest boxes first, and the trucks are sorted in ascending order to facilitate efficient binary search for eligible trucks.
2. **Binary Search**: For each box, we determine the leftmost truck that can carry the box using binary search. This ensures we only consider trucks with sufficient capacity.
3. **Segment Tree**: The segment tree is used to maintain and query the minimum number of boxes assigned to any eligible truck efficiently. This helps in balancing the load across trucks to minimize the maximum time taken by any truck, which is calculated as \(2k - 1\) for a truck carrying \(k\) boxes. The maximum of these values across all trucks gives the minimum time required to transport all boxes.
