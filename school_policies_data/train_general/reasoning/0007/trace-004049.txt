   - Reverse the main string to simulate the journey from B to A. Then, check if s1 appears in this reversed string followed by s2.

3. **Determine Result**:
   - Based on the results of the forward and backward checks, output the appropriate result: "forward", "backward", "both", or "fantasy".

### Solution Code
```python
def is_possible(main_str, s1, s2):
    len_s1 = len(s1)
    start = 0
    indices = []
    # Find all starting indices of s1 in main_str
    while True:
        idx = main_str.find(s1, start)
        if idx == -1:
            break
        indices.append(idx)
        start = idx + 1
    # Check each occurrence of s1 for s2 in the remaining part
    for idx in indices:
        remaining_start = idx + len_s1
        remaining = main_str[remaining_start:]
        if s2 in remaining:
            return True
    return False

def main():
    import sys
    main_str = sys.stdin.readline().strip()
    s1 = sys.stdin.readline().strip()
    s2 = sys.stdin.readline().strip()
    
    forward = is_possible(main_str, s1, s2)
    reversed_str = main_str[::-1]
    backward = is_possible(reversed_str, s1, s2)
    
    if forward and backward:
```
