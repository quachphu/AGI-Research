1. **Understanding Connected Permutations**: A connected permutation is a sequence of vertices where each consecutive pair is connected by an edge. This problem is equivalent to counting Hamiltonian paths in the given graph.

2. **Inclusion-Exclusion Principle**: We use the inclusion-exclusion principle to count permutations that avoid all forbidden edges. For each subset of forbidden edges, we compute the number of permutations that include these edges as consecutive pairs, adjusting the count by alternating inclusion and exclusion.

3. **Graph Validity Check**: For each subset of forbidden edges, we check if the edges form valid paths. A valid path component is a tree where each node has a degree of 1 or 2, with exactly two nodes of degree 1 (the endpoints of the path).

4. **Factorial Precomputation**: Precompute factorials up to the maximum possible value (100,000) modulo \(10^9 + 7\) to efficiently compute permutations.

5. **Subset Processing**: For each subset of forbidden edges, determine the number of valid permutations using the inclusion-exclusion principle and adjust the result accordingly.

```python
import sys
from itertools import combinations
from collections import defaultdict, deque

MOD = 10**9 + 7

def main():
    input = sys.stdin.read().split()
    ptr = 0
    T = int(input[ptr])
    ptr += 1
    for _ in range(T):
        N = int(input[ptr])
        M = int(input[ptr + 1])
        ptr += 2
        edges = []
        for __ in range(M):
            u = int(input[ptr])
            v = int(input[ptr + 1])
            ptr += 2
            if u > v:
                u, v = v, u
            edges.append((u, v))
        
        # Precompute factorial up to N
        fact = [1] * (N + 1)
        for i in range(1, N + 1):
            fact[i] = fact[i - 1] * i % MOD
        
        ans = 0
        # Generate all subsets of the M edges (by indices)
        for k in range(0, M + 1):
            for subset in combinations(range(M), k):
                size = len(subset)
                edges_in_S = [edges[i] for i in subset]
                
                # Build adjacency list and find all nodes involved
                adj = defaultdict(list)
                nodes = set()
                for u, v in edges_in_S:
                    adj[u].append(v)
                    adj[v].append(u)
                    nodes.add(u)
                    nodes.add(v)
                if not nodes:
                    # S is empty, all components are single nodes
                    sign = (-1) ** size
                    term = sign * 1 * fact[N] % MOD
                    ans = (ans + term) % MOD
                    continue
                
                # Find connected components
                visited = set()
                components = []
                for node in nodes:
                    if node not in visited:
                        q = deque([node])
                        visited.add(node)
                        component = []
                        while q:
                            u = q.popleft()
                            component.append(u)
                            for v in adj[u]:
                                if v not in visited:
                                    visited.add(v)
                                    q.append(v)
                        components.append(component)
                
                valid = True
                d = 0  # number of components with edges (valid paths)
                for comp in components:
                    comp_nodes = set(comp)
                    # Count edges in this component
                    m = 0
                    for u, v in edges_in_S:
                        if u in comp_nodes and v in comp_nodes:
                            m += 1
                    if m == 0:
                        continue  # no edges in this component
                    n_nodes = len(comp)
                    if m != n_nodes - 1:
                        valid = False
                        break
                    # Check degrees
                    degree = defaultdict(int)
                    for u, v in edges_in_S:
                        if u in comp_nodes and v in comp_nodes:
                            degree[u] += 1
                            degree[v] += 1
                    # Check all degrees <= 2
                    for node in comp:
                        if degree[node] > 2:
                            valid = False
                            break
                    if not valid:
                        break
                    # Count nodes with degree 1
                    cnt = 0
                    for node in comp:
                        if degree[node] == 1:
                            cnt += 1
                    if cnt != 2:
                        valid = False
                        break
                    d += 1
                if valid:
                    sign = (-1) ** size
                    power = pow(2, d, MOD)
                    rem = N - size
                    if rem < 0:
                        term = 0
                    else:
                        term = (sign * power * fact[rem]) % MOD
                    ans = (ans + term) % MOD
        print(ans % MOD)

if __name__ == "__main__":
    main()
```
