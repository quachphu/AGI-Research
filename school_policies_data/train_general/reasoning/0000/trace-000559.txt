MOD = 10**9 + 7

def main():
    import sys
    input = sys.stdin.read().split()
    idx = 0
    N = int(input[idx])
    idx += 1
    K = int(input[idx])
    idx += 1
    a_list = list(map(int, input[idx:idx+K]))
    idx += K

    max_fact = N

    # Precompute factorial and inverse factorial up to max_fact
    fact = [1] * (max_fact + 1)
    for i in range(1, max_fact + 1):
        fact[i] = fact[i-1] * i % MOD

    inv_fact = [1] * (max_fact + 1)
    inv_fact[max_fact] = pow(fact[max_fact], MOD-2, MOD)
    for i in range(max_fact - 1, -1, -1):
        inv_fact[i] = inv_fact[i+1] * (i + 1) % MOD

    product = [1]  # Initial polynomial is 1 (x^0 term)

    for a in a_list:
        s_d = N - a
        # Generate polynomial for this day
        poly = []
        for t in range(0, a + 1):
            nt = N - t
            if nt < s_d:
                c = 0
            else:
                c = fact[nt] * inv_fact[s_d] % MOD
                c = c * inv_fact[nt - s_d] % MOD
            inv_t = inv_fact[t] if t <= max_fact else 0
            coeff = c * inv_t % MOD
            poly.append(coeff)
        # Multiply with current product
        new_product = [0] * (len(product) + len(poly) - 1)
        for i in range(len(product)):
            for j in range(len(poly)):
                new_product[i + j] = (new_product[i + j] + product[i] * poly[j]) % MOD
        product = new_product

    # Find coefficient of x^N
    if N >= len(product):
        ans = 0
    else:
        ans = product[N]
    ans = ans * fact[N] % MOD
    print(ans)

if __name__ == "__main__":
