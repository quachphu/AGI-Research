   - For `k>1`, the growth is exponential. We use logarithms to determine the minimum number of seconds required for the second experiment to exceed the first experiment's bacteria count. This avoids directly computing extremely large numbers.

### Solution Code

import math

def main():
    import sys
    k, b, n, t = map(int, sys.stdin.readline().split())
    
    if k == 1:
        z = 1 + n * b
        if t >= z:
            print(0)
        else:
            needed = z - t
            m = (needed + b - 1) // b
            print(m)
    else:
        # Compute parameters for the formula-based approach
        numerator = (k - 1 + b)
        denominator = t * (k - 1) + b
        R = numerator / denominator
        log_R = math.log(R, k)
        m_float = n + log_R
        m_candidate = math.ceil(m_float)
        m = max(0, m_candidate)
        print(m)

if __name__ == "__main__":
    main()
