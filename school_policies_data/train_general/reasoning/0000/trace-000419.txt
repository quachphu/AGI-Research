import sys
import bisect

def main():
    input = sys.stdin.read().split()
    ptr = 0
    N, M = int(input[ptr]), int(input[ptr+1])
    ptr +=2
    a = list(map(int, input[ptr:ptr+N]))
    ptr +=N
    b = list(map(int, input[ptr:ptr+M]))
    ptr +=M
    Q = int(input[ptr])
    ptr +=1
    d_list = list(map(int, input[ptr:ptr+Q]))
    ptr +=Q
    
    # Precompute len_a and len_b
    len_a = [len(str(num)) for num in a]
    len_b = [len(str(num)) for num in b]
    
    # Precompute groups for b
    from collections import defaultdict
    group_b = defaultdict(list)  # key is d_b, value is sorted list of b_j
    len_b_counts = defaultdict(list)
    for idx, num in enumerate(b):
        d_b = len_b[idx]
        group_b[d_b].append(num)
    for d_b in group_b:
        group_b[d_b].sort()
        len_b_counts[d_b] = group_b[d_b]
    
    # Precompute sum_row for each row
    sum_row = []
    for i in range(N):
        ai = a[i]
        d_a = len_a[i]
        total =0
        # Iterate over all possible d_b in group_b
        for d_b in group_b:
            b_list = group_b[d_b]
            B_min = 10**(d_b -1)
            B_max = (10**d_b) -1
            T = 10**(d_a + d_b -1)
            x = (T + ai -1) // ai  # ceil(T/ai)
            x = max(x, B_min)
            if x > B_max:
                cnt_high =0
            else:
                idx = bisect.bisect_left(b_list, x)
                cnt_high = len(b_list) - idx
            cnt_low = len(b_list) - cnt_high
            sum_digits = cnt_high * (d_a + d_b) + cnt_low * (d_a + d_b -1)
            total += sum_digits
        sum_row.append(total)
    
    # Compute prefix sum for rows
    prefix = [0]*(N+1)
    for i in range(N):
        prefix[i+1] = prefix[i] + sum_row[i]
    total_digits = prefix[-1]

    len_b_array = len_b

    result = []

            result.append('x')

        low, high = 1, N

            if prefix[mid] < d:
