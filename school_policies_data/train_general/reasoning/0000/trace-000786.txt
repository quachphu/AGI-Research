class DSU:

    def __init__(self, n):

        self.parent = list(range(n+1))  # 1-based

        self.size = [1] * (n+1)

    def find(self, x):

        if self.parent[x] != x:

            self.parent[x] = self.find(self.parent[x])

        return self.parent[x]

    def union(self, x, y):

        x_root = self.find(x)

        y_root = self.find(y)

        if x_root == y_root:

            return 0

        if self.size[x_root] < self.size[y_root]:

            x_root, y_root = y_root, x_root

        self.parent[y_root] = x_root

        s = self.size[x_root] * self.size[y_root]

        self.size[x_root] += self.size[y_root]

        return s

To solve this problem, we need to calculate the inconvenience after each bridge collapses. The inconvenience is defined as the number of pairs of islands that are no longer connected. Given the constraints, a direct approach of removing bridges one by one and checking connectivity is infeasible. Instead, we use a reverse approach by starting from all bridges removed and adding them back in reverse order while maintaining the connected components using a Disjoint Set Union (DSU) data structure.

### Approach
1. **Reverse Process**: Instead of removing bridges from the initial set, we start from an empty set of bridges and add them back in reverse order. This allows us to efficiently track connected components using DSU.
2. **Union-Find Data Structure**: We use DSU to dynamically track connected components and their sizes. This helps in efficiently merging components and calculating the reduction in inconvenience when a bridge is added.
3. **Inconvenience Calculation**: The initial inconvenience when all bridges are removed is the total number of pairs of islands, which is \( \frac{N(N-1)}{2} \). As we add each bridge, we compute the reduction in inconvenience based on the sizes of the components being merged.

class DSU:
    def __init__(self, n):
        self.parent = list(range(n + 1))  # 1-based indexing
        self.size = [1] * (n + 1)
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return 0  # No merge
        # Union by size
        if self.size[x_root] < self.size[y_root]:
            x_root, y_root = y_root, x_root
        self.parent[y_root] = x_root
        merged_size = self.size[x_root] * self.size[y_root]
        self.size[x_root] += self.size[y_root]
        return merged_size

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    N = int(input[ptr])
    ptr += 1
    M = int(input[ptr])
    ptr += 1
    bridges = []
    for _ in range(M):
        a = int(input[ptr])
        ptr += 1
        b = int(input[ptr])
        ptr += 1
        bridges.append((a, b))
    
    # Process bridges in reverse order
    reversed_bridges = reversed(bridges)
    dsu = DSU(N)
    total_pairs = N * (N - 1) // 2
    res = [total_pairs]
