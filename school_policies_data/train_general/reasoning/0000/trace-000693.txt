import sys
import math

def convert(j, base):
    if j == 0:
        return '0', 1
    digits = []
    while j > 0:
        rem = j % base
        if rem < 10:
            digits.append(str(rem))
        else:
            digits.append(chr(ord('A') + rem - 10))
        j = j // base
    s = ''.join(reversed(digits))
    return s, len(s)

def main():

    ptr = 0

        recipes.append((d, s, c))

    prime_max = 100
    primes = [p for p in primes if p <= prime_max]
    primes.sort()
    
    zeros = {p: set() for p in primes}
    for j in range(n):

            if aj % p == 0:

            s_repr[j][base] = (s_str, s_len)
