from bisect import bisect_left, bisect_right

def solve(s):
    from collections import defaultdict

    positions = defaultdict(list)
    for idx, c in enumerate(s):
        positions[c].append(idx)
    sorted_chars = sorted(positions.keys(), reverse=True)
    available = sorted_chars.copy()
    result = []
    current_start = 0

    while available:
        found = False
        for i, c in enumerate(available):
            # Check if c has any occurrence >= current_start
            idxs = positions[c]
            pos = bisect_left(idxs, current_start)
            if pos >= len(idxs):
                continue
            pos = idxs[pos]
            # Now, check if all other chars in available have occurrences after pos
            valid = True
            for d in available:
                if d == c:
                    continue
                # check if d has any occurrence > pos
                d_idxs = positions[d]
                if not d_idxs:
                    valid = False
                    break
                p = bisect_right(d_idxs, pos)
                if p >= len(d_idxs):
                    valid = False
                    break
            if valid:
                result.append(c)
                current_start = pos + 1
                available.pop(i)
                found = True
                break
        if not found:
            # This should not happen as per problem statement
            break
    return ''.join(result)

To solve this problem, we need to find the lexicographically maximum string that can be obtained by deleting characters from the given string until all characters are unique. The solution involves a greedy approach to select the largest possible characters while ensuring that each character is included exactly once and in the correct order.

### Approach
1. **Precompute Positions**: For each character in the string, precompute the list of indices where it appears.
2. **Sort Characters**: Sort the distinct characters in descending order to prioritize larger characters.
3. **Greedy Selection**: Iterate through the sorted characters and select the largest possible character at each step, ensuring that there are enough remaining characters to form the rest of the string.
4. **Check Validity**: For each candidate character, check if there are sufficient remaining characters after its selected position to form the rest of the string.

### Solution Code
```python
import bisect

def solve():
    import sys
    input = sys.stdin.read().split()
    t = int(input[0])
    cases = input[1:t+1]
    for s in cases:
        from collections import defaultdict
        pos = defaultdict(list)
        for i, c in enumerate(s):
            pos[c].append(i)
        sorted_chars = sorted(pos.keys(), reverse=True)
        available = sorted_chars.copy()
        result = []
        current_start = 0
        while available:
            found = False
            for i in range(len(available)):
                c = available[i]
                indices = pos[c]
                idx = bisect.bisect_left(indices, current_start)
                if idx >= len(indices):
                    continue
                current_pos = indices[idx]
                valid = True
                for d in available:
                    if d == c:
                        continue
                    d_indices = pos[d]
                    d_idx = bisect.bisect_right(d_indices, current_pos)
                    if d_idx >= len(d_indices):
                        valid = False
                        break
                if valid:
                    result.append(c)
                    current_start = current_pos + 1
                    available.pop(i)
                    found = True
                    break
            if not found:
                break
        print(''.join(result))
```

1. **Precompute Positions**: We use a dictionary to store the indices of each character in the input string. This helps in quickly locating the positions of each character during the selection process.
2. **Sort Characters**: By sorting the distinct characters in descending order, we prioritize larger characters, which helps in forming the lexicographically maximum string.
3. **Greedy Selection**: For each character in the sorted list, we check if it can be included in the result by ensuring that there are enough remaining characters after its position to form the rest of the string. This is done using binary search to find the first valid position and checking subsequent positions for other required characters.
4. **Check Validity**: For each candidate character, we ensure that all other required characters have valid positions after the current position. This ensures that once a character is selected, the remaining characters can still form the rest of the string.

This approach efficiently selects the largest possible characters while maintaining the constraints, resulting in the lexicographically maximum string with all unique characters.
