2. **Prefix Sums**: For each quality group, we compute the prefix sums of the sorted prices. This allows us to quickly determine the maximum number of t-shirts a customer can buy within their remaining budget using binary search.
3. **Processing Customers**: For each customer, we iterate through the quality groups in descending order. For each group, we use binary search on the prefix sums to find the maximum number of t-shirts they can buy without exceeding their budget. We then update the remaining budget and continue to the next group until the budget is exhausted.

from collections import defaultdict

def main():
    n = int(sys.stdin.readline())
    shirts = []
    for _ in range(n):
        c, q = map(int, sys.stdin.readline().split())
        shirts.append((q, c))
    
    # Group by quality and sort each group's prices in ascending order
    groups_dict = defaultdict(list)
    for q, c in shirts:
        groups_dict[q].append(c)
    
    # For each group, sort the prices and compute prefix sums
    groups = []
    for q in groups_dict:
        prices = sorted(groups_dict[q])
        prefix = [0]
        s = 0
        for p in prices:
            s += p
            prefix.append(s)
        groups.append((q, prefix))
    
    # Sort the groups in descending order of quality
    groups.sort(reverse=True, key=lambda x: x[0])
    # Extract the prefix sums in order
    group_prefixes = [g[1] for g in groups]
    
    k = int(sys.stdin.readline())
    budgets = list(map(int, sys.stdin.readline().split()))
    
    results = []
    for b in budgets:
        remaining = b
        count = 0
        for prefix in group_prefixes:
            if remaining <= 0:
                break
            # Binary search for the largest x where prefix[x] <= remaining
            low = 0
            high = len(prefix) - 1
