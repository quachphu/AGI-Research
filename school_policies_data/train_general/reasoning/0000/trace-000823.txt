import sys
from functools import lru_cache

def main():
    MOD = 10**9 + 7
    a, b, c = map(int, sys.stdin.readline().split())
    
    if a == 1 and b == 1 and c == 1:
        print("inf")
        return
    
    if a == 1:
        # Handle a=1 case
        if c < b:
            print(0)
            return
        max_i = 0
        current = 1
        while current * b <= c:
            current *= b
            max_i += 1
        powers = [1] * (max_i + 1)
        for i in range(1, max_i + 1):
            powers[i] = powers[i-1] * b
        
        @lru_cache(maxsize=None)
        def dfs_a1(i, rem_b, rem_c):
            if i < 0:
                return 1 if rem_b == 0 and rem_c == 0 else 0
            res = 0
            power = powers[i]
            max_k = min(rem_b, rem_c // power)
            for k in range(max_k + 1):
                new_rem_b = rem_b - k
                new_rem_c = rem_c - k * power
                if new_rem_b < 0 or new_rem_c < 0:
                    continue
                res += dfs_a1(i-1, new_rem_b, new_rem_c)
            return res
        
        result = dfs_a1(max_i, b, c)
        print(result % MOD)
    else:
        # Handle a >1 case
        max_i = 0
        current = 1
        while current * a <= b:
            current *= a
            max_i += 1
        powers_a = [1] * (max_i + 1)
        for i in range(1, max_i + 1):
            powers_a[i] = powers_a[i-1] * a
        powers_b = [1] * (max_i + 1)
        for i in range(1, max_i + 1):

        @lru_cache(maxsize=None)
        def dfs(i, rem_b, rem_c):
            if i < 0:
