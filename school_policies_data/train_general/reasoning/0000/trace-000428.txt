class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size <<=1
        self.min_tree = [float('inf')] * (2 * self.size)
        # Initialize leaves
        for i in range(self.n):
            self.min_tree[self.size + i] = data[i]
        # Build the tree
        for i in range(self.size -1, 0, -1):
            self.min_tree[i] = min(self.min_tree[2*i], self.min_tree[2*i+1])
    
    def update(self, pos, value):
        pos += self.size
        self.min_tree[pos] = value
        while pos >1:
            pos >>=1
            new_val = min(self.min_tree[2*pos], self.min_tree[2*pos+1])
            if self.min_tree[pos] == new_val:
                break
            self.min_tree[pos] = new_val
    
    def query_min(self, l, r):
        res = float('inf')
        l += self.size
        r += self.size
        while l <= r:
            if l %2 ==1:
                res = min(res, self.min_tree[l])
                l +=1
            if r %2 ==0:
                res = min(res, self.min_tree[r])
                r -=1
            l >>=1
            r >>=1
        return res

### Approach
1. **Understanding Lexicographical Order**: The lexicographical order is determined by the first position where two arrays differ. Thus, to minimize the array `c`, we need to ensure that each element `c_i` is as small as possible starting from the first element.

2. **Greedy Strategy with Efficient Data Structures**: For each element `a_i` in array `a`, we need to find the smallest possible `(a_i + b_j) % n` using the remaining elements of `b`. This can be efficiently done using a segment tree to keep track of the available elements in `b` and their minimum values in specific ranges.

3. **Segment Tree for Range Queries**: The segment tree helps in efficiently finding the minimum value in a range. This allows us to quickly determine the smallest possible `b_j` for each `a_i` in two scenarios:
   - When `b_j` is small enough such that `(a_i + b_j) < n`.
   - When `b_j` is large enough such that `(a_i + b_j) % n` is minimized.

import sys

class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size <<= 1
        self.min_tree = [float('inf')] * (2 * self.size)
        # Initialize leaves
        for i in range(self.n):
            self.min_tree[self.size + i] = data[i]
        # Fill the rest of the leaves with inf
        for i in range(self.n, self.size):
            self.min_tree[self.size + i] = float('inf')
        # Build the tree
        for i in range(self.size - 1, 0, -1):
            self.min_tree[i] = min(self.min_tree[2 * i], self.min_tree[2 * i + 1])
    
    def update(self, pos):
        pos += self.size
        self.min_tree[pos] = float('inf')
        pos >>= 1
        while pos >= 1:
            new_val = min(self.min_tree[2 * pos], self.min_tree[2 * pos + 1])
            if self.min_tree[pos] == new_val:
                break
            self.min_tree[pos] = new_val
            pos >>= 1
    
    def query_min(self, l, r):
        res = float('inf')
        l += self.size
        r += self.size
        while l <= r:
            if l % 2 == 1:
                res = min(res, self.min_tree[l])
                l += 1
            if r % 2 == 0:
                res = min(res, self.min_tree[r])
                r -= 1
            l >>= 1
            r >>= 1
        return res

def main():
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    b = list(map(int, sys.stdin.readline().split()))
    
    freq = [0] * n
    for num in b:
        freq[num] += 1
    
    data = []
    for s in range(n):
        if freq[s] > 0:
            data.append(s)
        else:
            data.append(float('inf'))
    
    st = SegmentTree(data)
    result = []
    
    for ai in a:
        max_s_case1 = n - 1 - ai
        s1 = st.query_min(0, max_s_case1)
        case1_val = (ai + s1) % n if s1 != float('inf') else float('inf')
        
        min_s_case2 = (n - ai) % n
        s2 = st.query_min(min_s_case2, n-1)
        case2_val = (ai + s2) % n if s2 != float('inf') else float('inf')
        
        if case1_val <= case2_val:
            chosen_s = s1
            c_i = case1_val
        else:
            chosen_s = s2
            c_i = case2_val
        
        freq[chosen_s] -= 1
        if freq[chosen_s] == 0:
            st.update(chosen_s)
        result.append(c_i)
    
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
