    return 2 if x == 2 and y == 2 else min(x, y)

def or_op(x, y):
    return 2 if x == 2 or y == 2 else max(x, y)

def not_op(x):
    return 2 - x

def parse_formula(s, idx):
    c = s[idx[0]]
    if c in '012':
        val = int(c)
        idx[0] += 1
        return lambda p, q, r: val
    elif c in 'PQR':
        var = c
        idx[0] += 1
        if var == 'P':
            return lambda p, q, r: p
        elif var == 'Q':
            return lambda p, q, r: q
        else:
            return lambda p, q, r: r
    elif c == '-':
        idx[0] += 1
        child = parse_formula(s, idx)
        return lambda p, q, r: not_op(child(p, q, r))
    elif c == '(':
        idx[0] += 1
        left = parse_formula(s, idx)
        op = s[idx[0]]
        idx[0] += 1
        right = parse_formula(s, idx)
        assert s[idx[0]] == ')', "Expected closing parenthesis"
        idx[0] += 1
        if op == '*':
            return lambda p, q, r: and_op(left(p, q, r), right(p, q, r))
        elif op == '+':
            return lambda p, q, r: or_op(left(p, q, r), right(p, q, r))
        else:
            raise ValueError(f"Invalid operator: {op}")
    else:
        raise ValueError(f"Unexpected character: {c}")
