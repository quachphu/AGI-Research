To solve this problem, we need to determine the original string `s` such that each of the given `k` strings can be formed by swapping exactly two characters in `s`. If no such string exists, we should return `-1`.

### Approach
1. **Check Multiset Consistency**: First, ensure all input strings have the same multiset of characters. If they do not, output `-1` immediately.
2. **Handle Single String Case (k=1)**: If there's only one input string, check if it has duplicate characters. If it does, the string itself is the answer. Otherwise, swap the first two characters and return the result.
3. **Check Each Input String as Candidate**: For each input string, check if it can be the original string `s` by verifying if all other strings can be derived from it by exactly one swap.
4. **Generate Candidates from First String**: If no input string is valid, generate all possible candidates by swapping every pair of characters in the first input string and check each candidate against all input strings.

import sys
from collections import defaultdict

def main():
    k, n = map(int, sys.stdin.readline().split())
    strings = [sys.stdin.readline().strip() for _ in range(k)]
    
    # Check if all strings have the same multiset of characters
    def get_multiset(s):
        count = defaultdict(int)
        for c in s:
            count[c] += 1
        return count
    
    multiset = get_multiset(strings[0])
    for s in strings[1:]:
        if get_multiset(s) != multiset:
            print(-1)
            return
    
    if k == 1:
        s = strings[0]
        # Check if there's at least two same characters
        if len(set(s)) < n:
            print(s)
        else:
            # Swap first two characters
            if n < 2:
                print(-1)
            else:
                lst = list(s)
                lst[0], lst[1] = lst[1], lst[0]
                print(''.join(lst))
        return
    
    # Check each input string as a candidate
    for candidate in strings:
        valid = True
        for t in strings:
            diff = []
            for i in range(n):
                if candidate[i] != t[i]:
                    diff.append(i)
                    if len(diff) > 2:
                        valid = False
                        break
            if not valid:
                break
            if len(diff) == 0:
                # Check if candidate has duplicates
                if len(set(candidate)) == n:
                    valid = False
                    break
            elif len(diff) == 2:
                i, j = diff
                if candidate[i] != t[j] or candidate[j] != t[i]:
                    valid = False
                    break
            else:
                valid = False
                break
        if valid:
            print(candidate)
            return
    
    # Generate all possible swaps from the first string
    s1 = list(strings[0])
    # Check s1 itself if it has duplicates
    has_duplicates = len(set(s1)) < n
    if has_duplicates:
        valid = True
        for t in strings:
            diff = []
            for i in range(n):
                if s1[i] != t[i]:
                    diff.append(i)
                    if len(diff) > 2:
                        valid = False
                        break
            if not valid:
                break
            if len(diff) == 0:
                continue
            elif len(diff) == 2:
                i, j = diff
                if s1[i] != t[j] or s1[j] != t[i]:
                    valid = False
                    break
            else:
                valid = False
                break
        if valid:
            print(''.join(s1))
            return
    
    # Generate all swaps of two distinct characters in s1
    for i in range(n):
        for j in range(i + 1, n):
            if s1[i] == s1[j]:
                continue  # no need to check same character swaps here since s1 was already checked
            # Swap i and j
            s_candidate = s1.copy()
            s_candidate[i], s_candidate[j] = s_candidate[j], s_candidate[i]

                        break
                else:
                    valid = False
                    break
            if valid:
                print(s_candidate_str)
                return
