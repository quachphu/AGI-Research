import sys

def sieve(max_num):
    sieve = [True] * (max_num + 1)
    sieve[0] = sieve[1] = False
    for i in range(2, int(max_num ** 0.5) + 1):
        if sieve[i]:
            sieve[i*i : max_num + 1 : i] = [False] * len(sieve[i*i : max_num + 1 : i])
    return sieve

max_x = 10000
sieve_list = sieve(max_x)

# Precompute minimal steps for K 1-5 and x 1-10000
minimal_steps = [None] * 6  # indices 0-5, K=1 uses index 1.

for K in range(1, 6):
    steps = [-1] * (max_x + 1)
    for x in range(1, max_x + 1):
        current = x
        m = 0
        while current <= max_x:
            if sieve_list[current]:
                steps[x] = m
                break
            m += 1
            current = x + m * K
    minimal_steps[K] = steps

def main():
    input = sys.stdin.read().split()
    ptr = 0
    N = int(input[ptr])
    ptr += 1
    K_val = int(input[ptr])
    ptr += 1
    matrix = []

        row = list(map(int, input[ptr:ptr+N]))
        ptr += N
        matrix.append(row)
    
    if N < 3:

        return
    
    min_ops = float('inf')

    best_j = None
    
    for i in range(2, N):  # 1-based i ranges from 2 to N-1 inclusive
