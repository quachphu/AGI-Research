import sys

def solve():

    t = int(sys.stdin.readline())

    for _ in range(t):

        n, m = map(int, sys.stdin.readline().split())

        grid = []

        for _ in range(n):

            line = sys.stdin.readline().strip()

            grid.append(list(line))

        possible = {1, 2}

        for i in range(n):

            if not possible:

                break

            for j in range(m):

                c = grid[i][j]

                if c == '.':

                    continue

                sum_ = i + j

                # pattern 1: even R, odd W

                p1 = 'R' if (sum_ % 2 == 0) else 'W'

                # pattern 2: even W, odd R

                p2 = 'W' if (sum_ % 2 ==0) else 'R'

                compat = set()

                if c == p1:

                    compat.add(1)

                if c == p2:

                    compat.add(2)

                possible &= compat

                if not possible:

                    break

        if not possible:

            print("NO")

        else:

            # choose pattern

            pattern = next(iter(possible))

            # fill the grid

            for i in range(n):

                for j in range(m):

                    if grid[i][j] == '.':

                        sum_ = i + j

                        if pattern == 1:

                            grid[i][j] = 'R' if (sum_ %2 ==0) else 'W'

                        else:

                            grid[i][j] = 'W' if (sum_ %2 ==0) else 'R'

            print("YES")

            for row in grid:

                print(''.join(row))

To solve this problem, we need to determine if we can color a grid such that every red cell ('R') has only white neighbors ('W') and every white cell has only red neighbors. The existing cells cannot be recolored, and we need to fill the blank cells ('.') appropriately.

### Approach
1. **Check Compatibility with Patterns**: The grid must follow a checkerboard pattern, which can be of two types:
   - **Pattern 1**: Cells with even sums of their indices (i + j) are 'R', and those with odd sums are 'W'.
   - **Pattern 2**: Cells with even sums of their indices are 'W', and those with odd sums are 'R'.

2. **Determine Valid Pattern**: For each cell that is already colored ('R' or 'W'), check if it is compatible with either pattern. If a cell is incompatible with both patterns, the answer is "NO".

3. **Fill the Grid**: If a valid pattern is found, fill the blank cells according to that pattern and output the grid. If no valid pattern exists, output "NO".

### Solution Code
```python
import sys

def solve():
    t = int(sys.stdin.readline())
    for _ in range(t):
        n, m = map(int, sys.stdin.readline().split())
        grid = []
        for _ in range(n):
            line = sys.stdin.readline().strip()
            grid.append(list(line))
        
        possible = {1, 2}
        for i in range(n):
            if not possible:
                break
            for j in range(m):
                c = grid[i][j]
                if c == '.':
                    continue
                sum_ij = i + j
                # Check compatibility with pattern 1 and 2
                p1_color = 'R' if sum_ij % 2 == 0 else 'W'
                p2_color = 'W' if sum_ij % 2 == 0 else 'R'
                compat = set()
                if c == p1_color:
                    compat.add(1)
                if c == p2_color:
                    compat.add(2)
                possible &= compat
                if not possible:
                    break
        
        if not possible:
            print("NO")
        else:
            pattern = next(iter(possible))
            for i in range(n):
                for j in range(m):
                    if grid[i][j] == '.':
                        sum_ij = i + j
                        if pattern == 1:
                            grid[i][j] = 'R' if sum_ij % 2 == 0 else 'W'
                        else:
                            grid[i][j] = 'W' if sum_ij % 2 == 0 else 'R'
            print("YES")
            for row in grid:
                print(''.join(row))

solve()
```
