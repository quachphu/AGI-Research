class DSU:

    def __init__(self, n):

        self.parent = list(range(n+1))  # 1-based

        self.rank = [1]*(n+1)

    def find(self, x):

        if self.parent[x] != x:

            self.parent[x] = self.find(self.parent[x])

        return self.parent[x]

    def union(self, x, y):

        x_root = self.find(x)

        y_root = self.find(y)

        if x_root == y_root:

            return

        if self.rank[x_root] < self.rank[y_root]:

            self.parent[x_root] = y_root

        else:

            self.parent[y_root] = x_root

            if self.rank[x_root] == self.rank[y_root]:

                self.rank[x_root] +=1

To solve this problem, we need to determine the minimum amount of money required by the GFA to ensure that football teams from all planets can travel to any other planet using existing spaceways and teleports. The solution involves analyzing the connected components of planets and using a minimum spanning tree (MST) approach to minimize the cost of additional teleports.

### Approach
1. **Input Reading and Initialization**: Read the number of planets (N) and mutual agreements (M). Use a Disjoint Set Union (DSU) data structure to manage the connected components formed by the mutual agreements.
2. **Union-Find Operations**: Process each mutual agreement to union the respective planets, forming connected components.
3. **Cost Evaluation**: For each planet, read the associated cost (tax or sponsorship). Track the minimum tax cost for each connected component.
4. **Component Validation**: Ensure each connected component has at least one planet with a non-negative tax value (type 1 government). If any component lacks such a planet, output `-1`.
5. **Minimum Spanning Tree Calculation**: If valid, compute the minimum cost to connect all components using teleports. This is done by summing the minimum tax values of each component and adjusting using the smallest tax value to form the MST.

import sys
from sys import stdin

class DSU:
    def __init__(self, n):
        self.parent = list(range(n + 1))  # 1-based indexing
        self.rank = [1] * (n + 1)
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return
        if self.rank[x_root] < self.rank[y_root]:
            self.parent[x_root] = y_root
        else:
            self.parent[y_root] = x_root
            if self.rank[x_root] == self.rank[y_root]:
                self.rank[x_root] += 1

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    dsu = DSU(n)
    for _ in range(m):
        a, b = map(int, input().split())
        dsu.union(a, b)
    
    # Read C values for each planet (1-based)
    C = [0] * (n + 1)  # C[1] to C[n]
    for i in range(1, n + 1):
        C[i] = int(sys.stdin.readline())
    
    min_c = dict()  # key: root, value: minimal C in component
    for i in range(1, n + 1):
        if C[i] >= 0:
            root = dsu.find(i)
            if root not in min_c or C[i] < min_c[root]:
                min_c[root] = C[i]
    
    # Check all components have at least one type 1
    roots = set()
    for i in range(1, n + 1):
        roots.add(dsu.find(i))
    
    for r in roots:
        if r not in min_c:
            print(-1)
            return
    
    minima = list(min_c.values())
    k = len(minima)
    if k == 1:
        print(0)
    else:
        sum_m = sum(minima)
        min_m = min(minima)
        total = sum_m + (k - 2) * min_m
        print(total)

if __name__ == "__main__":
    main()
