To solve this problem, we need to calculate the expected number of inversions in a permutation where some elements are replaced by -1. The solution involves computing the contributions from three different parts: known-known pairs, known-unknown pairs, and unknown-unknown pairs. 

### Approach
1. **Known-Known Pairs**: Directly count the inversions among the known elements using a Fenwick Tree (Binary Indexed Tree) for efficient range queries and updates.
2. **Known-Unknown Pairs**: For each known element, compute the expected contributions from pairs involving unknown elements using prefix sums and binary search.
3. **Unknown-Unknown Pairs**: Use combinatorial analysis to determine the expected inversions among the unknown elements, which simplifies to a formula involving the number of unknown elements.

MOD = 998244353

def main():
    import bisect

    n = int(input())
    p = list(map(int, input().split()))

    # Collect known elements and compute the missing elements S
    known = set()
    for x in p:
        if x != -1:
            known.add(x)
    S = []
    for x in range(1, n + 1):
        if x not in known:
            S.append(x)
    m = len(S)

    # Compute prefix sum array for unknowns
    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + (1 if p[i] == -1 else 0)
    total_unknowns = prefix[n]

    # Fenwick Tree to compute inversions among known elements (E_known_known)
    class FenwickTree:
        def __init__(self, size):
            self.n = size
            self.tree = [0] * (self.n + 2)  # +2 to avoid issues

        def update(self, idx, delta):
            while idx <= self.n:
                self.tree[idx] += delta
                idx += idx & -idx

        def query(self, idx):
            res = 0
            while idx > 0:
                res += self.tree[idx]
                idx -= idx & -idx
            return res

    max_val = n
    ft = FenwickTree(max_val)
    E_known_known = 0
    # Process elements from right to left
    for i in reversed(range(n)):
        if p[i] != -1:
            a = p[i]
            count = ft.query(a - 1)
            E_known_known = (E_known_known + count) % MOD
            ft.update(a, 1)

    # Compute sum_contribution for known-unknown pairs
    sum_contribution = 0
    for i in range(n):
        if p[i] == -1:
            continue
        a = p[i]
        left_unknowns = prefix[i]
        right_unknowns = total_unknowns - prefix[i + 1]

        # Find the count of elements in S greater than a
        cnt_g = len(S) - bisect.bisect_right(S, a)
        # Find the count of elements in S less than a
        cnt_l = bisect.bisect_left(S, a)

        sum_contribution = (sum_contribution + left_unknowns * cnt_g + right_unknowns * cnt_l) % MOD

    # Compute E_known_unknown
    if m != 0:
        inv_m = pow(m, MOD - 2, MOD)
        E_known_unknown = (sum_contribution * inv_m) % MOD
    else:
        E_known_unknown = 0

    # Compute E_unknown_unknown
    E_unknown_unknown = 0
    if m >= 2:
        inv_4 = pow(4, MOD - 2, MOD)
        term = (m % MOD) * ((m - 1) % MOD) % MOD
        E_unknown_unknown = term * inv_4 % MOD

    # Total expected value
    total = (E_known_known + E_known_unknown + E_unknown_unknown) % MOD
    print(total)
