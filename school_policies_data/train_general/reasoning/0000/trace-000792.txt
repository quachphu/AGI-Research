To solve this problem, we need to simulate the movement of ants in a tunnel with narrow sections where ants cannot pass each other. The goal is to determine the last ant to leave the tunnel and the time it exits.

### Approach
1. **Problem Analysis**:
   - Ants move at 1 cm/s and can either move left (L) or right (R).
   - When ants meet at a narrow section (any integer position between 1 and l-1), they turn around. In wide sections, they pass each other.
   - The task is to determine the last ant to exit the tunnel and the time it exits.

2. **Key Insights**:
   - **Narrow Sections**: All integer positions between 1 and l-1 are considered narrow sections.
   - **Collision Handling**: Ants reverse direction if they meet at a narrow section; otherwise, they pass each other.
   - **Event Simulation**: Use an event-driven approach to process exits and collisions in chronological order.

3. **Algorithm**:
   - **Event Simulation**: Track exit times and collision times. Process the earliest events first.
   - **Collision Detection**: Calculate potential collision times and positions for pairs of ants moving towards each other.
   - **Direction Reversal**: Check if collisions occur at narrow sections and reverse directions accordingly.

### Solution Code
```python
import sys

def main():
    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        parts = line.split()
        n, l = map(int, parts)
        if n == 0 and l == 0:
            break
        ants = []
        for _ in range(n):
            di, pi = sys.stdin.readline().split()
            pi = int(pi)
            ants.append({'number': _+1, 'dir': di, 'pos': float(pi), 'active': True})
        
        current_time = 0.0
        exit_records = []
        active_ants = [{'number': ant['number'], 'dir': ant['dir'], 'pos': ant['pos'], 'active': True} for ant in ants]
        
        while True:
            active_ants = [ant for ant in active_ants if ant['active']]
            if not active_ants:
                break
            
            exit_events = []
            collision_events = []
            
            for ant in active_ants:
                if ant['dir'] == 'R':
                    exit_time = current_time + (l - ant['pos'])
                else:
                    exit_time = current_time + ant['pos']
                exit_events.append((exit_time, 'exit', ant))
            
            for i in range(len(active_ants)):
                ant_i = active_ants[i]
                if not ant_i['active'] or ant_i['dir'] != 'R':
                    continue
                for j in range(i+1, len(active_ants)):
                    ant_j = active_ants[j]
                    if not ant_j['active'] or ant_j['dir'] != 'L':
                        continue
                    if ant_i['pos'] >= ant_j['pos']:
                        continue
                    dx = ant_j['pos'] - ant_i['pos']
                    collision_time = current_time + dx / 2
                    collision_pos = ant_i['pos'] + dx / 2
                    if collision_time <= current_time:
                        continue
                    collision_events.append((collision_time, 'collision', ant_i, ant_j, collision_pos))
            
            all_events = exit_events + collision_events
            if not all_events:
                break
            
            earliest_time = min(event[0] for event in all_events)
            current_events = [event for event in all_events if event[0] == earliest_time]
            exit_events_at_time = []
            collision_events_at_time = []
            for event in current_events:
                if event[1] == 'exit':
                    exit_events_at_time.append(event)
                else:
                    collision_events_at_time.append(event)
            
            dt = earliest_time - current_time
            current_time = earliest_time
            for ant in active_ants:
                if ant['dir'] == 'R':
                    ant['pos'] += dt
                else:
                    ant['pos'] -= dt
            
            for event in exit_events_at_time:
                exit_time, _, ant = event
                if ant['active']:
                    if (ant['dir'] == 'R' and ant['pos'] >= l - 1e-9) or (ant['dir'] == 'L' and ant['pos'] <= 1e-9):
                        ant['active'] = False
                        exit_side = 'L' if ant['dir'] == 'L' else 'R'
                        exit_records.append((exit_time, ant['number'], exit_side))
            
            for event in collision_events_at_time:
                _, _, ant_i, ant_j, collision_pos = event
                if not ant_i['active'] or not ant_j['active']:
                    continue
                rounded_pos = round(collision_pos)
                if (abs(rounded_pos - collision_pos) < 1e-9) and (1 <= rounded_pos <= l - 1):
                    ant_i['dir'] = 'L' if ant_i['dir'] == 'R' else 'R'
                    ant_j['dir'] = 'L' if ant_j['dir'] == 'R' else 'R']
        
        if not exit_records:
            print("0 0")
            continue
        
        max_time = max(exit_records, key=lambda x: (x[0], -1 if x[2] == 'L' else 1))[0]
        candidates = []
        for record in exit_records:
            if record[0] == max_time:
                candidates.append((record[1], record[2]))
        
        candidates.sort(key=lambda x: (x[1] != 'L'))
        last_ant = candidates[0][0]
        
        if max_time.is_integer():
            print(f"{int(max_time)} {last_ant}")
        else:
            print(f"{max_time} {last_ant}")

if __name__ == "__main__":
    main()
```
