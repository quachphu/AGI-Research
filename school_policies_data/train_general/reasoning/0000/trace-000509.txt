def max_matching(graph, num_left, num_right):
    match_U = [-1] * num_left
    match_V = [-1] * num_right
    dist = [0] * num_left
    result = 0
    
    while bfs(graph, match_U, match_V, dist):
        for u in range(num_left):
            if match_U[u] == -1:
                if dfs(u, graph, match_U, match_V, dist):
                    result += 1
    return result

def bfs(graph, match_U, match_V, dist):
    queue = deque()
    for u in range(len(match_U)):
        if match_U[u] == -1:
            dist[u] = 0
            queue.append(u)
        else:
            dist[u] = float('inf')
    dist_null = float('inf')
    while queue:
        u = queue.popleft()
        if dist[u] < dist_null:
            for v in graph[u]:
                if match_V[v] == -1:
                    dist_null = dist[u] + 1
                elif dist[match_V[v]] == float('inf'):
                    dist[match_V[v]] = dist[u] + 1
                    queue.append(match_V[v])

def dfs(u, graph, match_U, match_V, dist):
    for v in graph[u]:
        if match_V[v] == -1 or (dist[match_V[v]] == dist[u] + 1 and dfs(match_V[v], graph, match_U, match_V, dist)):
            match_U[u] = v
            match_V[v] = u
            return True
    dist[u] = float('inf')
    return False
