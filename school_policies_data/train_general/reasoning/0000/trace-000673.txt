from collections import deque

# Directions: up, right, down, left
DIR = [(-1,0), (0,1), (1,0), (0,-1)]
REV = [2, 3, 0, 1]

# Switch directions for L and R types, rotated 0-3 times (clockwise)
SWITCH_DIRS = {
    'L': [
        {'B': 3, 'S': 1, 'C': 0},
        {'B': 0, 'S': 2, 'C': 1},
        {'B': 1, 'S': 3, 'C': 2},
        {'B': 2, 'S': 0, 'C': 3},
    ],
    'R': [
        {'B': 3, 'S': 1, 'C': 2},
        {'B': 0, 'S': 2, 'C': 3},
        {'B': 1, 'S': 3, 'C': 0},
        {'B': 2, 'S': 0, 'C': 1},
    ]
}

def get_exits(rail_type, rotation):
    exits = {}
    r = rotation % 4
    if rail_type == 'S':
        exits.update({ (3-r)%4: (1-r)%4, (1-r)%4: (3-r)%4 })
    elif rail_type == 'C':
        exits.update({ (3-r)%4: (0-r)%4 %4, (0-r)%4 %4: (3-r)%4 })
        exits[(3-r)%4] = (0 - r) %4
        exits[(0 - r) %4] = (3 - r) %4
    return exits

def is_valid(grid, switches, w, h):
    graph = {}
    for i in range(h):
        for j in range(w):
            cell = grid[i][j]
            t, r = cell
            if t in ['L', 'R']:
                dirs = SWITCH_DIRS[t][r]
                for end in ['B', 'S', 'C']:
                    d = dirs[end]
                    ni, nj = i + DIR[d][0], j + DIR[d][1]
                    if 0 <= ni < h and 0 <= nj < w:
                        rev_dir = REV[d]
                        next_cell = (ni, nj, rev_dir)
                        key = (i, j, d)
                        if key not in graph:
                            graph[key] = []
                        graph[key].append(next_cell)
            else:
                exits = get_exits(t, r)
                for d_in, d_out in exits.items():
                    ni, nj = i + DIR[d_out][0], j + DIR[d_out][1]
                    if 0 <= ni < h and 0 <= nj < w:
                        rev_dir = REV[d_out]
                        key = (i, j, d_in)
                        if key not in graph:
                            graph[key] = []
                        graph[key].append((ni, nj, rev_dir))
    switch_ends = []
    for (i, j, t, r) in switches:
        dirs = SWITCH_DIRS[t][r]
        for end in ['B', 'S', 'C']:
            d = dirs[end]
            switch_ends.append((i, j, d))
    visited = set()
    if not switch_ends:
        return False
    start = switch_ends[0]
    q = deque([start])
    visited.add(start)
    while q:
        node = q.popleft()
        if node in graph:
            for neighbor in graph[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    q.append(neighbor)
    for end in switch_ends:
        if end not in visited:
            return False
    return True

def max_fun_time(grid, switches, w, h):
    max_time = 0
    switch_indices = {(i, j): idx for idx, (i, j, t, r) in enumerate(switches)}
    for i in range(h):
        for j in range(w):
            cell = grid[i][j]
            t, r = cell
            if t not in ['L', 'R']:
                continue
            exits = get_exits(t, r)
            for d_in in exits:
                start_pos = (i, j, d_in)
                state = 0
                visited = {}
                path = []
                current = (start_pos, state)
                step = 0
                while current not in visited:
                    visited[current] = step
                    (i_cur, j_cur, d_cur), state_cur = current
                    cell_cur = grid[i_cur][j_cur]
                    t_cur, r_cur = cell_cur
                    if t_cur in ['L', 'R']:
                        dirs = SWITCH_DIRS[t_cur][r_cur]
                        incoming_dir = d_cur
                        end = None
                        for e in ['B', 'S', 'C']:
                            if dirs[e] == incoming_dir:
                                end = e
                                break
                        if end == 'B':
                            mask = 1 << switch_indices[(i_cur, j_cur)]
                            if (state_cur & mask) == 0:
                                exit_dir = dirs['S']
                                new_state = state_cur | mask
                            else:
                                exit_dir = dirs['C']
                                new_state = state_cur & ~mask
                        else:
                            exit_dir = dirs['B']
                            new_state = state_cur
                        ni = i_cur + DIR[exit_dir][0]
                        nj = j_cur + DIR[exit_dir][1]
                        if ni < 0 or ni >= h or nj < 0 or nj >= w:
                            break

                            break

                        next_pos = (ni, nj, nd)

                        if any(t in ['L', 'R'] for (i, j, d), _ in [current]):
