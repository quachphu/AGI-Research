import sys

mod = 1001
max_v = 1000

# Precompute Fibonacci numbers modulo 1001 for i from 1 to 1000
fib = []
prev_prev = 1  # Represents f(-1)
prev = 1       # Represents f(0)
for i in range(1, max_v + 1):
    current = (prev_prev + prev) % mod
    fib.append(current)
    prev_prev, prev = prev, current

class DSU:
    def __init__(self, size):
        self.parent = list(range(size))
        self.rank = [1] * size

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return
        if self.rank[x_root] < self.rank[y_root]:
            self.parent[x_root] = y_root
        else:
            self.parent[y_root] = x_root
            if self.rank[x_root] == self.rank[y_root]:
                self.rank[x_root] += 1

def process_test_case(V, d):
    dsu = DSU(V)
    for i in range(V):
        for j in range(i + 1, V):
            if abs(fib[i] - fib[j]) < d:
                dsu.union(i, j)
    roots = set()
    for i in range(V):
        roots.add(dsu.find(i))
    return len(roots)

def main():
    input = sys.stdin.read().split()
    idx = 0
    while idx < len(input):
        V = int(input[idx])
        d = int(input[idx + 1])
        idx += 2
        print(process_test_case(V, d))

if __name__ == "__main__":
    main()
