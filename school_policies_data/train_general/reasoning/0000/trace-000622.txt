2. **Check Validity of Each Part**: For each generated part, we check:
   - The part must not be empty.
   - If the part starts with '0', it must be exactly "0".

def count_valid_ips(s):
    n = len(s)
    count = 0
    for a in range(1, min(3, n - 3) + 1):
        max_b = min(a + 3, n - 2)
        for b in range(a + 1, max_b + 1):
            start_c = max(b + 1, n - 3)
            end_c = min(b + 3, n - 1)
            if start_c > end_c:
                continue
            for c in range(start_c, end_c + 1):
                parts = [
                    s[0:a],
                    s[a:b],
                    s[b:c],
                    s[c:]
                ]
                valid = True
                for part in parts:
                    if len(part) > 1 and part[0] == '0':
                        valid = False
                        break
                    num = int(part)
                    if num < 0 or num > 255:
                        valid = False
                        break
                if valid:
                    count += 1
    return count
