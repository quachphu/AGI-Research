def main():
    import sys
    N, x = map(int, sys.stdin.readline().split())
    a = list(map(int, sys.stdin.readline().split()))
    N_val = N
    min_win = [[0] * (N_val + 1) for _ in range(N_val)]
    
    for s in range(N_val):
        current_min = a[s]
        min_win[s][1] = current_min
        for l in range(2, N_val + 1):
            next_pos = (s + l - 1) % N_val
            current_min = min(current_min, a[next_pos])
            min_win[s][l] = current_min
    
    min_total = float('inf')
    for k in range(N_val):
        total = k * x
        for j in range(N_val):
