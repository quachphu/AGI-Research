            continue
        if stripped_line.startswith('try'):
            stack.append(line_num)
        elif stripped_line.startswith('catch'):
            match = re.match(r'catch\s*\(\s*([A-Za-z]+)\s*,\s*"([^"]*)"\s*\)', stripped_line)
            if match:
                exception_type = match.group(1)
                message = match.group(2)
                if stack:
                    try_line = stack.pop()
                    blocks.append({
                        'try_line': try_line,
                        'catch_line': line_num,
                        'exception_type': exception_type,
                        'message': message
                    })
        elif stripped_line.startswith('throw'):
            match = re.match(r'throw\s*\(\s*([A-Za-z]+)\s*\)', stripped_line)
            if match:
                throw_type = match.group(1)
                throw_line = line_num

    candidates = []
    for block in blocks:
        if (block['try_line'] < throw_line and
            block['catch_line'] > throw_line and
            block['exception_type'] == throw_type):
            candidates.append(block)

    if candidates:
        selected = min(candidates, key=lambda x: x['catch_line'])
        print(selected['message'])
    else:
        print("Unhandled Exception")

if __name__ == "__main__":
    main()
