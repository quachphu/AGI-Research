### Approach
1. **Input Handling**: Read the dimensions of the chocolate bar (M, N), the maximum allowed difference K, and the grid itself.
2. **Collect U Cells**: Identify all cells labeled 'U' as these can be assigned to either piece.
3. **Bitmask Iteration**: Iterate over all possible assignments of 'U' cells to Tapan's piece using bitmasks.
4. **Check Validity**: For each bitmask, check:
   - **Non-empty Regions**: Both pieces must be non-empty.

def is_connected(grid, mask, u_cells, primary_char, forbidden_char, is_s_region):
    M = len(grid)
    N = len(grid[0]) if M >0 else 0
    cells = set()
    for i in range(M):
        for j in range(N):
            cell = grid[i][j]
            if cell == forbidden_char:
                continue
            if cell == primary_char:
                cells.add( (i,j) )
            elif cell == 'U':
                idx = u_cells.index( (i,j) )
                in_region = (mask & (1 << idx)) !=0
                if is_s_region and in_region:
                    cells.add( (i,j) )
                elif not is_s_region and not in_region:
                    cells.add( (i,j) )
    if not cells:
        return False
    
    start = next(iter(cells))
    visited = set()
    queue = deque([start])
    visited.add(start)
    directions = [ (-1,0), (1,0), (0,-1), (0,1) ]
    
    while queue:
        x, y = queue.popleft()
        for dx, dy in directions:
            nx = x + dx
            ny = y + dy
            if 0 <= nx < M and 0 <= ny < N:
                if (nx, ny) in cells and (nx, ny) not in visited:
                    visited.add( (nx, ny) )
                    queue.append( (nx, ny) )
    
    return len(visited) == len(cells)

def has_invalid_squares(grid, mask, u_cells, is_s_region):
    M = len(grid)
    N = len(grid[0]) if M else 0
    for i in range(M-1):
        for j in range(N-1):
            all_in_region = True

                            in_current = (mask & (1 << idx)) !=0
                            if is_s_region:
                                if not in_current:
                                    all_in_region = False
                            else:
                                if in_current:
                                    all_in_region = False
                    if not all_in_region:
                        break
                if not all_in_region:
                    break
            if all_in_region:
                if len(set(labels)) == 1:
                    return True
    return False
