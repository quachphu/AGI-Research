The operation involves repeatedly taking the remainder of the number divided by the count of its '1' bits (popcount) until the number becomes zero.

### Approach

2. **Modular Arithmetic**: Precompute the original number modulo (popcount ± 1) to efficiently determine the remainder after flipping each bit.
3. **Precompute Steps**: Precompute the number of steps required to reduce each possible value to zero using dynamic programming.
4. **Efficient Calculation**: For each bit flip, compute the new remainder using modular arithmetic and use the precomputed steps to determine the result efficiently.

def main():
    import sys
    input = sys.stdin.read().split()
    N = int(input[0])
    X = input[1].strip()
    C = X.count('1')
    
    if C == 0:
        for _ in range(N):
            print(1)
        return
    
    # Precompute X mod (C+1) and X mod (C-1) if applicable
    def compute_mod(s, m):
        mod = 0
        for c in s:
            mod = (mod * 2 + (c == '1')) % m
        return mod
    
    m1 = C + 1
    x_mod_m1 = compute_mod(X, m1)
    x_mod_m2 = 0
    if C >= 2:
        m2 = C - 1
        x_mod_m2 = compute_mod(X, m2)
    
    # Precompute f(n) for n up to max needed
    max_f = max(m1, (C-1) if C >=2 else 0)
    # Extend to cover up to 2e5+2 to avoid index issues
    max_f = max(max_f, 200000 + 2)
    f = [0] * (max_f + 1)
    for n in range(1, max_f + 1):
        count = 0
        current = n
        while current != 0:
            pop = bin(current).count('1')
            current %= pop
            count += 1
        f[n] = count
    
    # Process each bit
    results = []
    for i in range(N):
        current_bit = X[i]
        if current_bit == '0':
            m = m1
            e = (N - 1) - i
            delta = pow(2, e, m)
            rem = (x_mod_m1 + delta) % m
            ans = 1 + f[rem]
        else:
            if C == 1:
                ans = 0
            else:
                m = C - 1
                e = (N - 1) - i
                delta = pow(2, e, m)
                rem = (x_mod_m2 - delta) % m
                ans = 1 + f[rem]
        results.append(ans)
    
    print('\n'.join(map(str, results)))

if __name__ == "__main__":
