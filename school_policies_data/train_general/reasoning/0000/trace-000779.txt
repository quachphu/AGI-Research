def is_palindrome(n):

    return s == s[::-1]

    j = 0  # j represents the offset from i
    while True:
        term = (i + j) ** 2
        current_sum += term
        if current_sum > b:
            break

            valid_numbers.add(current_sum)
        j += 1

print(sum(valid_numbers))
