The problem thus reduces to finding the closest pair of points in this plane.

The optimal solution involves using a divide-and-conquer algorithm to find the closest pair of points efficiently. This algorithm works by recursively splitting the set of points, finding the closest pairs in each subset, and then checking for closer pairs that straddle the subsets.

import sys
import math

def readints():
    return list(map(int, sys.stdin.readline().split()))

def distance(p1, p2):
    return (p1[0] - p2[0])**2 + (p1[1] - p2[1])**2

def closest_pair(points):
    points.sort()
    n = len(points)
    return _closest_pair(points, 0, n)

def _closest_pair(points, l, r):
    if r - l <= 3:
        min_dist = float('inf')
        for i in range(l, r):
            for j in range(i+1, r):
                d = distance(points[i], points[j])
                if d < min_dist:
                    min_dist = d
        return min_dist
    
    mid = (l + r) // 2
    mid_x = points[mid][0]
    
    dl = _closest_pair(points, l, mid)
    dr = _closest_pair(points, mid, r)
    d = min(dl, dr)
    
    strip = []
    for i in range(l, r):
        if (points[i][0] - mid_x)**2 < d:
            strip.append(points[i])
    
    strip.sort(key=lambda x: x[1])

        for j in range(i+1, min(i+6, len(strip))):
            current_dist = distance(strip[i], strip[j])
            if current_dist < d:
                d = current_dist
    return d
