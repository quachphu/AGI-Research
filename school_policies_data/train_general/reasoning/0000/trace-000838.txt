import heapq

def dijkstra(n, adj, start):
    INF = float('inf')
    dist = [INF] * (n + 1)
    dist[start] = 0
    heap = [(0, start)]
    visited = [False] * (n + 1)
    
    while heap:
        current_dist, u = heapq.heappop(heap)
        if visited[u]:
            continue
        visited[u] = True
        for v, c in adj[u]:
            if dist[v] > current_dist + c:
                dist[v] = current_dist + c
                heapq.heappush(heap, (dist[v], v))
    return dist
