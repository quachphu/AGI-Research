        self.parity = [0]*(size+1)  # parity from node to parent

    def find(self, u):

            # Save the parent of u before recursion

            orig_parent = self.parent[u]

            # Recursively find the root of the parent

            root, p_orig_parent = self.find(orig_parent)

            # Update the parent and parity of u

            self.parent[u] = root

            self.parity[u] = (self.parity[u] + p_orig_parent) % 2

        return (self.parent[u], self.parity[u])

        # Required_parity is x_u + x_v = required_parity

        root_u, p_u = self.find(u)

        root_v, p_v = self.find(v)
