2. **Calculate Remaining Weight**: For each value of r or b, compute the remaining weight and determine the maximum possible count of the other candy type that can be consumed without exceeding the weight limit.
3. **Track Maximum Joy**: Keep track of the maximum joy units obtained from all valid combinations of red and blue candies.

### Solution Code

def main():
    import sys
    C, Hr, Hb, Wr, Wb = map(int, sys.stdin.readline().split())
    max_joy = 0

    # Function to check ranges for red candies and compute corresponding blue
    def check_red_range(r_start, r_end):
        nonlocal max_joy
        for r in range(r_start, r_end + 1):
            remaining = C - r * Wr
            if remaining < 0:
                continue
            b = remaining // Wb
            current_joy = Hr * r + Hb * b
            if current_joy > max_joy:
                max_joy = current_joy

    r_max = C // Wr
    check_red_range(0, min(r_max, 10**6))
    check_red_range(max(0, r_max - 10**6), r_max)

    # Function to check ranges for blue candies and compute corresponding red
    def check_blue_range(b_start, b_end):
        nonlocal max_joy
        for b in range(b_start, b_end + 1):
            remaining = C - b * Wb
            if remaining < 0:
                continue
            r = remaining // Wr
            current_joy = Hr * r + Hb * b
            if current_joy > max_joy:
                max_joy = current_joy

    b_max = C // Wb
    check_blue_range(0, min(b_max, 10**6))
    check_blue_range(max(0, b_max - 10**6), b_max)
