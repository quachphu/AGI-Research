def hungarian(matrix):
    n = len(matrix)
    m = len(matrix[0])
    assert n <= m
    u = [0] * (n+1)
    v = [0] * (m+1)
    p = [0] * (m+1)
    way = [0] * (m+1)
    for i in range(1, n+1):
        p[0] = i
        minv = [inf] * (m+1)
        used = [False] * (m+1)
        j0 = 0
        i0 = i
        delta = 0
        while True:
            used[j0] = True
            i0 = p[j0]
            delta = inf
            j1 = 0
            for j in range(1, m+1):
                if not used[j]:
                    cur = matrix[i0-1][j-1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            for j in range(m+1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta

            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break
    result = sum(matrix[i][p[i+1]-1] for i in range(n))
    return result, p[1:m+1]

The key insight is that we need to match each T-shirt size from the previous year to a size in the current year such that the sum of character changes is minimized.

2. **Graph Representation**: Represent the problem as a bipartite graph where each node on one side represents a T-shirt size from last year, and each node on the other side represents a size from this year. The edge weights between nodes represent the number of character changes required to convert one size to another.

3. **Hungarian Algorithm**: Use the Hungarian algorithm to find the minimum weight matching in the bipartite graph. This algorithm efficiently solves the assignment problem in O(n^3) time, which is feasible for the given constraint (n ≤ 100).

def hungarian(matrix):
    n = len(matrix)
    u = [0] * (n + 1)
    v = [0] * (n + 1)
    p = [0] * (n + 1)
    way = [0] * (n + 1)
    
    for i in range(1, n + 1):
        p[0] = i
        minv = [float('inf')] * (n + 1)
        used = [False] * (n + 1)
        j0 = 0
        i0 = i
        while True:
            used[j0] = True
            i0 = p[j0]
            delta = float('inf')
            j1 = 0
            for j in range(1, n + 1):
                if not used[j]:
                    cur = matrix[i0 - 1][j - 1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            if delta == float('inf'):
                break
            for j in range(n + 1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            j0 = j1
            if p[j0] == 0:
                break
        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break
    return -v[0]
