4. **Efficient Calculation**: For large values of N (up to 1 billion), we need an efficient way to determine the new room number. This involves determining the number of digits in the new number and then converting each digit step-by-step using the valid digit mappings.

def old_to_new(n):
    if n == 0:
        return "0"
    # Determine the number of digits m
    m = 1
    while (8 ** m - 1) < n:
        m += 1
    remaining = n - (8 ** (m - 1) - 1)
    allowed_first = ['1', '2', '3', '5', '7', '8', '9']
    allowed_rest = ['0', '1', '2', '3', '5', '7', '8', '9']
    result = []
    for pos in range(m):
        if pos == 0:
            allowed = allowed_first
        else:
            allowed = allowed_rest
        digits_left = m - pos - 1
        options = 8 ** digits_left
        index = (remaining - 1) // options
        result.append(allowed[index])
        remaining = (remaining - 1) % options + 1
    return ''.join(result)
