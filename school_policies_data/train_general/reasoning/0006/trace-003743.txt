Implementing Hopcroft-Karp in Python can be tricky, but for N=1e4, it's manageable if optimized.

The algorithm uses BFS to compute layers of the residual graph, then uses DFS to find augmenting paths. The BFS and DFS are repeated until no more augmenting paths are found.

The standard implementation uses adjacency lists, distance arrays, and matching arrays.

def hopcroft_karp(graph, N_left, N_right):
    pair_U = [0] * (N_left + 1)  # for left nodes (airports), 1-based?
    pair_V = [0] * (N_right + 1)  # for right nodes (factories)
    dist = [0] * (N_left + 1)  # distance for BFS layers

    result = 0

    while True:
        # BFS to find all free nodes in U and compute layers
        queue = deque()
        for u in range(1, N_left+1):
            if pair_U[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')

        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in graph[u]:
                    if dist[pair_V[v]] == float('inf'):
                        dist[pair_V[v]] = dist[u] + 1
                        queue.append(pair_V[v])

        # If no augmenting path found, break
        if dist[0] == float('inf'):
            break

        # Find augmenting paths using DFS
        for u in range(1, N_left+1):
            if pair_U[u] == 0:
                if dfs(u, graph, pair_U, pair_V, dist):
                    result += 1

    return result

def dfs(u, graph, pair_U, pair_V, dist):
    if u != 0:
        for v in graph[u]:
            if dist[pair_V[v]] == dist[u] + 1:
                if dfs(pair_V[v], graph, pair_U, pair_V, dist):
                    pair_U[u] = v
                    pair_V[v] = u
                    return True
        dist[u] = float('inf')
        return False
    return True

def hopcroft_karp(adj, d_candidate, N_left, N_right):
    pair_U = [0] * (N_left + 1)  # airports are 1-based
    pair_V = [0] * (N_right + 1)  # factories are 1-based
    dist = [0] * (N_left + 1)
    result = 0

    while True:
        queue = deque()
        for u in range(1, N_left+1):
            if pair_U[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')

        # BFS to build layers
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                # Iterate through edges of u, sorted by d
                for d, v in adj.get(u, []):
                    if d > d_candidate:
                        break  # edges are sorted, so break early
                    # Check if this v can be used to extend the BFS layers
                    if dist[pair_V[v]] == float('inf'):
                        dist[pair_V[v]] = dist[u] + 1
                        queue.append(pair_V[v])

        # If no augmenting path found, break
        if dist[0] == float('inf'):
            break

        # Run DFS to find augmenting paths
        for u in range(1, N_left+1):
            if pair_U[u] == 0:
                if dfs(u, adj, d_candidate, pair_U, pair_V, dist):
                    result += 1

    return result

def dfs(u, adj, d_candidate, pair_U, pair_V, dist):
    if u != 0:
        for d, v in adj.get(u, []):
            if d > d_candidate:
                break
            if dist[pair_V[v]] == dist[u] + 1:
                if dfs(pair_V[v], adj, d_candidate, pair_U, pair_V, dist):
                    pair_U[u] = v
                    pair_V[v] = u
                    return True
        dist[u] = float('inf')
        return False
    return True

The solution involves using a binary search approach combined with the Hopcroft-Karp algorithm to efficiently find the maximum bipartite matching.

### Approach
1. **Problem Analysis**: The problem can be modeled as a bipartite graph matching problem where we need to find a perfect matching (each node in both partitions is matched) with the minimal maximum edge weight. The edge weight represents the number of days required to build the road.

2. **Binary Search on Days**: We perform a binary search on the possible number of days (edge weights) to determine the minimal maximum days required. For each candidate number of days, we check if a perfect bipartite matching exists using only edges with weights less than or equal to the candidate.

3. **Hopcroft-Karp Algorithm**: This algorithm is used to efficiently find the maximum bipartite matching in O(E√V) time, where E is the number of edges and V is the number of vertices. This helps in checking the existence of a perfect matching efficiently for each candidate during the binary search.

### Solution Code
```python
import sys
from bisect import bisect_right
from collections import deque, defaultdict

def main():
    input = sys.stdin.read().split()
    ptr = 0
    N = int(input[ptr])
    ptr += 1
    M = int(input[ptr])
    ptr += 1

    if M < N:
        print(-1)
        return

    adj = defaultdict(list)
    all_ds = set()

    for _ in range(M):
        u = int(input[ptr])
        ptr += 1
        v = int(input[ptr])
        ptr += 1
        d = int(input[ptr])
        ptr += 1
        adj[u].append((d, v))
        all_ds.add(d)

    # Sort adjacency lists for each airport by d
    for u in adj:
        adj[u].sort()

    # Create sorted list of unique d's
    unique_ds = sorted(all_ds)
    if not unique_ds:
        print(-1)
        return

    # Binary search over unique_ds
    low = 0
    high = len(unique_ds) - 1
    answer = -1

    while low <= high:
        mid = (low + high) // 2
        current_d = unique_ds[mid]

        pair_U = [0] * (N + 1)
        pair_V = [0] * (N + 1)
        dist = [0] * (N + 1)
        result = 0

        while True:
            queue = deque()
            for u in range(1, N + 1):
                if pair_U[u] == 0:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = float('inf')
            dist[0] = float('inf')

            # BFS to build layers
            while queue:
                u = queue.popleft()
                if dist[u] < dist[0]:
                    for d_edge, v in adj.get(u, []):
                        if d_edge > current_d:
                            break
                        if dist[pair_V[v]] == float('inf'):
                            dist[pair_V[v]] = dist[u] + 1
                            queue.append(pair_V[v])

            if dist[0] == float('inf'):
                break

            # DFS to find augmenting paths iteratively
            for u_start in range(1, N + 1):
                if pair_U[u_start] == 0:
                    visited = {}
                    path = []
                    stack = [(u_start, iter(adj.get(u_start, [])))]
                    visited[u_start] = True
                    found = False
                    while stack:
                        node, edges_iter = stack[-1]
                        if node == 0:
                            found = True
                            break
                        try:
                            while True:
                                d_edge, v = next(edges_iter)
                                if d_edge > current_d:
                                    continue
                                next_node = pair_V[v]
                                if next_node not in visited:
                                    visited[next_node] = True
                                    path.append((node, v))
                                    stack.append((next_node, iter(adj.get(next_node, []))))
                                    break
                        except StopIteration:
                            stack.pop()
                            if path:
                                path.pop()
                            visited.pop(node, None)
                            dist[node] = float('inf')
                    if found:
                        for u_match, v_match in path:
                            pair_U[u_match] = v_match
                            pair_V[v_match] = u_match
                        result += 1

        if result == N:
            answer = current_d
            high = mid - 1
        else:
            low = mid + 1

    print(answer if answer != -1 else -1)

if __name__ == "__main__":
    main()
```

1. **Reading Input**: The input is read and parsed into a list of edges, where each edge connects an airport to a factory with a given number of days.

2. **Preprocessing**: The edges are sorted by the number of days for each airport, which allows efficient filtering during the binary search.

3. **Binary Search**: We perform a binary search on the sorted unique days. For each candidate day, we check if a perfect bipartite matching exists using edges with days less than or equal to the candidate.

4. **Hopcroft-Karp Algorithm**: This algorithm is used to find the maximum bipartite matching efficiently. The BFS and DFS steps are modified to dynamically consider only edges up to the current candidate day, ensuring optimal performance.

5. **Result**: The minimal maximum days required to build all roads such that every factory is paired with an airport is determined and printed. If no valid pairing exists, -1 is printed.
