import math

def count_pyramids(n):
    count = 0
    while True:
        if n < 2:
            break
        s = math.isqrt(24 * n + 1)
        h_max = (s - 1) // 6
        if h_max < 1:
            break
        cards = h_max * (3 * h_max + 1) // 2
        if cards > n:
            break
        count += 1
        n -= cards
    return count

t = int(input())
