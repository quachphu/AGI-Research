import math

def get_divisors(s):
    divisors = set()
    for i in range(1, int(math.isqrt(s)) + 1):
        if s % i == 0:
            divisors.add(i)
            divisors.add(s // i)
    return divisors

def generate_ab_pairs(d):
    pairs = set()
    max_a = int(math.isqrt(d))
    for a in range(0, max_a + 1):
        a_sq = a * a
        if a_sq > d:
            break
        b_sq = d - a_sq
        b = int(math.isqrt(b_sq))
        if b * b != b_sq:
            continue
        signs_a = [1, -1] if a != 0 else [1]
        signs_b = [1, -1] if b != 0 else [1]
        for sa in signs_a:
            for sb in signs_b:
                x = sa * a
                y = sb * b
                pairs.add((x, y))
                if a != b:
                    pairs.add((y, x))
    return pairs

def is_ginkgo_prime(m, n):
    S = m * m + n * n
    if S <= 1:
        return False

    divisors = get_divisors(S)
    std_divisors = [
        (1, 0),
        (0, 1),
        (-1, 0),
        (0, -1),
        (m, n),
        (-n, m),
        (-m, -n),
        (n, -m)
    ]
    std_set = set(std_divisors)
    generated = set()

    for d in divisors:
        ab_pairs = generate_ab_pairs(d)
        for a, b in ab_pairs:
            denominator = d
            if denominator == 0:
                continue
            numerator1 = a * m + b * n
            numerator2 = a * n - b * m
            if numerator1 % denominator != 0 or numerator2 % denominator != 0:
                continue
            generated.add((a, b))

    for pair in generated:
        if pair not in std_set:
            return False
    return len(generated) == 8

        else:
