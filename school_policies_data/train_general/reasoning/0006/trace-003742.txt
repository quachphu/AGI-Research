To solve this problem, we need to determine the maximum number of weapons (swords and war axes) that you and your follower can carry from a blacksmith shop, considering both your individual carrying capacities. The solution involves considering various strategies for splitting the weapons between you and your follower to maximize the total count.

### Approach
1. **Check Total Capacity**: First, check if the combined capacity of you and your follower is enough to carry all available weapons. If so, the answer is simply the sum of all weapons.
2. **Generate Candidates**: For each test case, generate several candidate strategies for how you can carry weapons. These strategies include:
   - Taking as many swords as possible first, then axes.
   - Taking as many axes as possible first, then swords.
   - Balancing the number of swords and axes based on the combined weight.
3. **Evaluate Candidates**: For each candidate strategy, compute the remaining weapons and determine the maximum number of weapons your follower can carry using similar strategies.
4. **Maximize Total**: Track the maximum number of weapons carried by both you and your follower across all candidate strategies.

### Solution Code
```python
import sys

def main():
    input = sys.stdin.read().split()
    idx = 0
    t = int(input[idx])
    idx += 1
    for _ in range(t):
        p = int(input[idx])
        f = int(input[idx+1])
        idx +=2
        cnt_s = int(input[idx])
        cnt_w = int(input[idx+1])
        idx +=2
        s = int(input[idx])
        w = int(input[idx+1])
        idx +=2
        
        total_weight = cnt_s * s + cnt_w * w
        if total_weight <= p + f:
            print(cnt_s + cnt_w)
            continue
        
        max_total = 0
        
        # Generate candidates for (s_you, a_you)
        candidates = set()
        
        # Swords first
        s_you = min(cnt_s, p // s)
        rem_p = p - s_you * s
        a_you = min(cnt_w, rem_p // w)
        candidates.add((s_you, a_you))
        
        # Axes first
        a_you = min(cnt_w, p // w)
        rem_p = p - a_you * w
        s_you = min(cnt_s, rem_p // s)
        candidates.add((s_you, a_you))
        
        # Optimal s_you based on (p/(s + w))
        s_you_opt = p // (s + w)
        s_you_opt = min(cnt_s, s_you_opt)
        rem_p = p - s_you_opt * s
        a_you_opt = min(cnt_w, rem_p // w)
        candidates.add((s_you_opt, a_you_opt))
        
        # s_you_opt + 1
        s_opt_plus = min(cnt_s, s_you_opt + 1)
        rem_p = p - s_opt_plus * s
        a_opt_plus = min(cnt_w, rem_p // w)
        candidates.add((s_opt_plus, a_opt_plus))
        
        # s_you_opt -1
        s_opt_minus = max(0, s_you_opt - 1)
        rem_p = p - s_opt_minus * s
        a_opt_minus = min(cnt_w, rem_p // w)
        candidates.add((s_opt_minus, a_opt_minus))
        
        # Optimal a_you based on (p/(s + w))
        a_you_opt = p // (s + w)
        a_you_opt = min(cnt_w, a_you_opt)
        rem_p = p - a_you_opt * w
        s_you_a_opt = min(cnt_s, rem_p // s)
        candidates.add((s_you_a_opt, a_you_opt))
        
        # a_you_opt +1
        a_opt_plus = min(cnt_w, a_you_opt + 1)
        rem_p = p - a_opt_plus * w
        s_you_a_plus = min(cnt_s, rem_p // s)
        candidates.add((s_you_a_plus, a_opt_plus))
        
        # a_you_opt -1
        a_opt_minus = max(0, a_you_opt - 1)
        rem_p = p - a_opt_minus * w
        s_you_a_minus = min(cnt_s, rem_p // s)
        candidates.add((s_you_a_minus, a_opt_minus))
        
        # s_you = 0
        s_you_zero = 0
        a_you_zero = min(cnt_w, p // w)
        candidates.add((s_you_zero, a_you_zero))
        
        # a_you = 0
        a_you_zero = 0
        s_you_zero = min(cnt_s, p // s)
        candidates.add((s_you_zero, a_you_zero))
        
        # Iterate through all candidates
        for s_you, a_you in candidates:
            if s_you < 0 or a_you < 0:
                continue
            if s_you > cnt_s or a_you > cnt_w:
                continue
            if s_you * s + a_you * w > p:
                continue
            
            rem_s = cnt_s - s_you
            rem_a = cnt_w - a_you
            
            # Follower's turn: swords first
            s_f = min(rem_s, f // s)
            rem_f = f - s_f * s
            a_f = min(rem_a, rem_f // w)
            total1 = s_you + a_you + s_f + a_f
            
            # Follower's turn: axes first
            a_f = min(rem_a, f // w)
            rem_f = f - a_f * w
            s_f = min(rem_s, rem_f // s)
            total2 = s_you + a_you + s_f + a_f
            
            max_total = max(max_total, total1, total2)
        
        print(max_total)

if __name__ == "__main__":
    main()
```
