def gauss(matrix, n_vars, mod):
    rank = 0
    n_rows = len(matrix)
    # For each column (variable), find a pivot
    for col in range(n_vars):
        # Find the first row with a non-zero entry in this column
        pivot = None
        for r in range(rank, n_rows):
            if matrix[r][col] % mod != 0:
                pivot = r
                break
        if pivot is None:
            continue  # Move to the next column
        # Swap the pivot row with the current rank row
        matrix[rank], matrix[pivot] = matrix[pivot], matrix[rank]
        # Multiply the pivot row by the inverse of the pivot element to make it 1
        val = matrix[rank][col]
        inv_val = inv[val % mod] if val % mod != 0 else 0
        if inv_val == 0:
            # This is a problem, but val mod mod is non-zero, so this shouldn't happen
            continue
        for c in range(col, n_vars + 1):
            matrix[rank][c] = (matrix[rank][c] * inv_val) % mod
        # Eliminate this column from all other rows
        for r in range(n_rows):
            if r != rank and matrix[r][col] != 0:
                factor = matrix[r][col]
                for c in range(col, n_vars + 1):
                    matrix[r][c] = (matrix[r][c] - factor * matrix[rank][c]) % mod
        rank += 1
    # Check for inconsistency
    for r in range(rank, n_rows):
        if matrix[r][-1] % mod != 0:
            return None  # Inconsistent system
    # Determine the pivot columns and free variables
    pivot_cols = []
    for r in range(rank):
        for c in range(n_vars):
            if matrix[r][c] == 1:
                pivot_cols.append(c)
                break
    free_vars = [c for c in range(n_vars) if c not in pivot_cols]
    # Now, for each pivot row, express the pivot variable in terms of free variables
    # We need to build a system where each pivot variable is expressed in terms of free variables
    # Create a list of equations: for each pivot variable, the equation is:

    equations = []

        # Find the pivot column
        pivot_col = None
        for c in range(n_vars):

            if c == pivot_col:
                continue
            coeff = row[c]
            if coeff != 0:
                equation['coefficients'][c] = coeff
        equations.append(equation)
    return {
        'pivot_vars': pivot_cols,
        'free_vars': free_vars,
        'equations': equations,

MOD = 10**9 + 7

def gauss(matrix, n_vars, mod):
    rank = 0
    n_rows = len(matrix)
    for col in range(n_vars):
        pivot = None
        for r in range(rank, n_rows):
            if matrix[r][col] % mod != 0:
                pivot = r
                break
        if pivot is None:
            continue
        matrix[rank], matrix[pivot] = matrix[pivot], matrix[rank]
        val = matrix[rank][col]
        inv_val = pow(val, -1, mod) if val % mod != 0 else 0
        if inv_val == 0:
            continue
        for c in range(col, n_vars + 1):
            matrix[rank][c] = (matrix[rank][c] * inv_val) % mod
        for r in range(n_rows):
            if r != rank and matrix[r][col] != 0:
                factor = matrix[r][col]
                for c in range(col, n_vars + 1):
                    matrix[r][c] = (matrix[r][c] - factor * matrix[rank][c]) % mod
        rank += 1
    for r in range(rank, n_rows):
        if matrix[r][-1] % mod != 0:
            return None
    pivot_cols = []
    for r in range(rank):
        for c in range(n_vars):
            if matrix[r][c] == 1:
                pivot_cols.append(c)
                break
    free_vars = [c for c in range(n_vars) if c not in pivot_cols]
    return {'free_vars': free_vars}
