MOD = 998244353

def main():
    import sys
    n, k = map(int, sys.stdin.readline().split())
    sa = list(map(int, sys.stdin.readline().split()))
    
    rank = [0] * n
    for i in range(n):
        rank[sa[i]] = i
    
    m = 0
    for i in range(n-1):
        x = sa[i]
        y = sa[i+1]
        x_plus_1 = x + 1
        y_plus_1 = y + 1
        
        if x_plus_1 < n and y_plus_1 < n:
            if rank[x_plus_1] > rank[y_plus_1]:
                m += 1
        else:
            if y_plus_1 >= n:
                m += 1
    
    if k < m:
        print(0)
        return
    
    upper = n + k - m - 1
    if upper < n or upper < 0:
        print(0)
        return
    
    max_n = upper
    fact = [1] * (max_n + 1)
    for i in range(1, max_n + 1):
        fact[i] = fact[i-1] * i % MOD
    
    inv_fact = [1] * (max_n + 1)
    inv_fact[max_n] = pow(fact[max_n], MOD-2, MOD)
    for i in range(max_n-1, -1, -1):
