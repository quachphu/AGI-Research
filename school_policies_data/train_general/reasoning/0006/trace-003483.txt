import sys

from sys import stdin

class DSU:

    def __init__(self, size):

        self.parent = list(range(size+1))  # 1-based

        self.size = [1]*(size+1)

    def find(self, x):

        if self.parent[x] != x:

            self.parent[x] = self.find(self.parent[x])

        return self.parent[x]

    def union(self, x, y):

        x_root = self.find(x)

        y_root = self.find(y)

        if x_root == y_root:

            return

        if self.size[x_root] < self.size[y_root]:

            x_root, y_root = y_root, x_root

        self.parent[y_root] = x_root

        self.size[x_root] += self.size[y_root]

def is_lucky(w):

    s = str(w)

    for c in s:

        if c not in {'4', '7'}:

            return False

    return True

### Approach

2. **Union-Find Data Structure**: We use a Disjoint Set Union (DSU) to efficiently group nodes into components based on non-lucky edges. Each component represents a set of nodes that can be reached from each other without crossing a lucky edge.
3. **Component Size Calculation**: For each node, determine the size of its component in the DSU. The valid nodes j and k for a node i must be outside this component, as their paths from i must include at least one lucky edge.
4. **Triple Count Calculation**: For each node i, compute the number of valid pairs (j, k) such that both j and k are outside the component of i. This is calculated as \( s_i \times (s_i - 1) \) where \( s_i \) is the number of nodes outside the component of i.

import sys
from sys import stdin

class DSU:
    def __init__(self, size):
        self.parent = list(range(size + 1))  # 1-based indexing
        self.size = [1] * (size + 1)
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        x_root = self.find(x)
        y_root = self.find(y)
        if x_root == y_root:
            return
        if self.size[x_root] < self.size[y_root]:
            x_root, y_root = y_root, x_root
        self.parent[y_root] = x_root
        self.size[x_root] += self.size[y_root]

def is_lucky(w):
    s = str(w)
    for c in s:
        if c not in {'4', '7'}:
            return False
    return True

def main():
    n = int(stdin.readline())
    if n < 3:
        print(0)
        return
    dsu = DSU(n)
    for _ in range(n - 1):
        u, v, w = map(int, stdin.readline().split())
        if not is_lucky(w):
            dsu.union(u, v)
    total = 0
    for i in range(1, n + 1):
        root = dsu.find(i)
        component_size = dsu.size[root]
        s_i = n - component_size
        total += s_i * (s_i - 1)
    print(total)

if __name__ == "__main__":
    main()
