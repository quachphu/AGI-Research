To solve this problem, we need to determine the minimum time required to deliver a specified number of shovels to a customer in a given city, ensuring the total cost does not exceed a given budget. The solution involves efficiently leveraging the network of cities and stores using graph traversal and greedy algorithms.

### Approach
1. **Graph Representation**: Represent the cities and roads using an adjacency list.
2. **BFS for Shortest Path**: For each query, compute the shortest path from the customer's city to all other cities using Breadth-First Search (BFS).
3. **Store Sorting**: For each query, collect and sort stores based on their distance from the customer's city. If multiple stores are at the same distance, sort them by price.
4. **Priority Queue for Greedy Selection**: Use a min-heap (priority queue) to always select the cheapest available shovels first. This ensures that we maximize the number of shovels bought within the budget.
5. **Iterative Processing**: Process each store in the sorted order, adding them to the priority queue and checking if the required number of shovels can be delivered within the budget. The first valid solution found gives the minimum time.

### Solution Code
```python
import sys
from collections import deque
import heapq

def main():
    input = sys.stdin.read().split()
    ptr = 0

    n = int(input[ptr])
    ptr += 1
    m = int(input[ptr])
    ptr += 1

    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(input[ptr])
        ptr += 1
        v = int(input[ptr])
        ptr += 1
        adj[u].append(v)
        adj[v].append(u)

    w = int(input[ptr])
    ptr += 1

    stores = []
    for _ in range(w):
        ci = int(input[ptr])
        ptr += 1
        ki = int(input[ptr])
        ptr += 1
        pi = int(input[ptr])
        ptr += 1
        stores.append((ci, ki, pi))

    q = int(input[ptr])
    ptr += 1

    for _ in range(q):
        gj = int(input[ptr])
        ptr += 1
        rj = int(input[ptr])
        ptr += 1
        aj = int(input[ptr])
        ptr += 1

        # BFS to compute distances from gj
        dist = [-1] * (n + 1)
        dist[gj] = 0
        q_bfs = deque([gj])
        while q_bfs:
            u = q_bfs.popleft()
            for v in adj[u]:
                if dist[v] == -1:
                    dist[v] = dist[u] + 1
                    q_bfs.append(v)

        # Collect stores with their distance
        store_list = []
        for ci, ki, pi in stores:
            d = dist[ci]
            if d == -1:
                continue  # unreachable
            store_list.append((d, pi, ki))

        # Sort by distance, then price
        store_list.sort(key=lambda x: (x[0], x[1]))

        # Process each store in order
        heap = []
        sum_shovels = 0
        sum_cost = 0
        answer = -1

        for d, p, k in store_list:
            heapq.heappush(heap, (p, k))
            current_max_d = d

            while heap and sum_shovels < rj and sum_cost < aj:
                price, available = heapq.heappop(heap)
                if available <= 0:
                    continue

                remaining_cost = aj - sum_cost
                if remaining_cost <= 0:
                    heapq.heappush(heap, (price, available))
                    break

                max_take_cost = remaining_cost // price
                if max_take_cost <= 0:
                    heapq.heappush(heap, (price, available))
                    continue

                remaining_shovels = rj - sum_shovels
                if remaining_shovels <= 0:
                    heapq.heappush(heap, (price, available))
                    break

                max_take = min(available, max_take_cost, remaining_shovels)
                if max_take <= 0:
                    heapq.heappush(heap, (price, available))
                    continue

                sum_shovels += max_take
                sum_cost += price * max_take
                available -= max_take

                if available > 0:
                    heapq.heappush(heap, (price, available))

                if sum_shovels >= rj and sum_cost <= aj:
                    answer = current_max_d
                    break  # Exit while loop

            if answer != -1:
                break  # Exit for loop

        if sum_shovels >= rj and sum_cost <= aj:
            print(answer)
        else:
            print(-1)
```
