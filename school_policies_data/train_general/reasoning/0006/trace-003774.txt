            idx += 1
            goal = []
            for _ in range(3):
                row = lines[idx].strip().split()
                goal.append(row)
                idx += 1
            datasets.append((x, y, goal))
        return datasets

    def get_new_top(current_top, direction):
        if current_top == 'W':
            return 'R' if direction in ('north', 'south') else 'B'
        elif current_top == 'R':
            return 'B' if direction in ('north', 'south') else 'W'
        elif current_top == 'B':
            return 'W' if direction in ('north', 'south') else 'R'
        return current_top

    datasets = parse_input(sys.stdin.read().splitlines())

    for x, y, goal in datasets:
        goal_grid = tuple(tuple(row) for row in goal)
        initial_grid = []
        for i in range(3):
            row = []
            for j in range(3):
                if (j + 1) == x and (i + 1) == y:
                    row.append('E')
                else:
                    row.append('W')
            initial_grid.append(tuple(row))
        initial_grid = tuple(initial_grid)

        if initial_grid == goal_grid:
            print(0)
            continue

        visited = set()
        queue = deque()
        initial_state = (x, y, initial_grid)
        queue.append((x, y, initial_grid, 0))
        visited.add(initial_state)
        found = False
        answer = -1

        while queue:
            curr_x, curr_y, curr_grid, steps = queue.popleft()
            if steps >= 30:
                continue

            for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                new_empty_x = curr_x + dx
                new_empty_y = curr_y + dy
                if not (1 <= new_empty_x <= 3 and 1 <= new_empty_y <= 3):
                    continue

                cube_x, cube_y = new_empty_x, new_empty_y
                cube_row = cube_y - 1
                cube_col = cube_x - 1
                if curr_grid[cube_row][cube_col] == 'E':
                    continue

                dir_x = curr_x - cube_x
                dir_y = curr_y - cube_y
                if dir_y == 1:
                    direction = 'south'
                elif dir_y == -1:
                    direction = 'north'
                elif dir_x == 1:
                    direction = 'east'
                elif dir_x == -1:
                    direction = 'west'
                else:
                    continue

                current_top = curr_grid[cube_row][cube_col]
                new_top = get_new_top(current_top, direction)

                new_grid = [list(row) for row in curr_grid]
                new_grid[cube_row][cube_col] = 'E'
                old_empty_row = curr_y - 1
                old_empty_col = curr_x - 1
                new_grid[old_empty_row][old_empty_col] = new_top
                new_grid = tuple(tuple(row) for row in new_grid)

                new_state = (new_empty_x, new_empty_y, new_grid)
                if new_state not in visited:
                    if new_grid == goal_grid:
                        answer = steps + 1
                        found = True
                        break
                    visited.add(new_state)
                    queue.append((new_empty_x, new_empty_y, new_grid, steps + 1))
            if found:
                break

        print(answer if found else -1)

if __name__ == "__main__":
    main()

- **Initial State Setup**: The initial state is constructed with all cubes having their top face as White (W) except the empty cell.
- **BFS Implementation**: Using BFS, we explore all possible moves from the initial state. Each move generates a new state by rolling a cube into the empty cell, updating the cube's top face color based on the direction of the roll.
- **State Transition**: The new top face color is determined using the `get_new_top` function, which encapsulates the transition rules based on the current top color and roll direction.
- **Visited States**: To avoid cycles, we track visited states using a set, ensuring each state is processed only once.

This approach efficiently explores the state space to find the shortest path to the target configuration within the allowed 30 steps.
