import sys
from sys import stdin
from collections import defaultdict

def compose(a, b):
    return tuple(a[x - 1] for x in b)

def inverse(a):
    k = len(a)
    a_inv = [0] * k
    for i in range(k):
        a_inv[a[i] - 1] = i + 1
    return tuple(a_inv)

def compute_closure(g, p):
    if p in g:
        return g
    elements = set(g)
    queue = [p]
    elements.add(p)
    while queue:
        current = queue.pop()
        for perm in list(elements):
            new_perm = compose(current, perm)
            if new_perm not in elements:
                elements.add(new_perm)
                queue.append(new_perm)
            new_perm = compose(perm, current)
            if new_perm not in elements:
                elements.add(new_perm)
                queue.append(new_perm)
        inv = inverse(current)
        if inv not in elements:
            elements.add(inv)
            queue.append(inv)
    return frozenset(elements)

def main():
    n, k = map(int, stdin.readline().split())
    permutations = []
    for _ in range(n):
        perm = tuple(map(int, stdin.readline().split()))
        permutations.append(perm)
    
    total_sum = 0
    prev_groups = defaultdict(int)
    
    for idx in range(n):
        p = permutations[idx]
        current_groups = defaultdict(int)
        
        group_p = compute_closure(frozenset(), p)
        current_groups[group_p] += 1

            count = prev_groups[g]

            current_groups[new_g] += count
