To solve this problem, we need to find all prime numbers \( p \) that divide the polynomial \( f(x) \) for every integer \( x \). The solution involves two main cases: primes that divide all coefficients of the polynomial and primes for which the polynomial evaluates to zero for all residues modulo \( p \).

### Approach
1. **Compute the GCD of Coefficients**: The primes that divide all coefficients of the polynomial can be found by computing the greatest common divisor (GCD) of all coefficients and then factorizing it to get the prime factors.
2. **Check Small Primes**: For primes \( p \) that are less than or equal to the degree \( N \) of the polynomial, check if the polynomial evaluates to zero for all residues modulo \( p \). This is done by reducing the polynomial modulo \( p \) and checking if all coefficients of the reduced polynomial are zero.

import sys
from math import gcd
from collections import defaultdict

def main():
    n = int(sys.stdin.readline())
    a_list = [int(sys.stdin.readline()) for _ in range(n+1)]
    
    # Compute GCD of all coefficients
    current_gcd = 0
    for coeff in a_list:
        current_gcd = gcd(current_gcd, coeff)
    
    # Factorize the GCD to get case1 primes
    def factor(g):
        factors = set()
        if g == 0:
            return factors
        g = abs(g)
        if g == 0:
            return factors
        i = 2
        while i * i <= g:
            if g % i == 0:
                factors.add(i)
                while g % i == 0:
                    g //= i
            i += 1
        if g > 1:
            factors.add(g)
        return factors
    
    case1_primes = factor(current_gcd)
    
    # Generate primes up to n using sieve
    def sieve(n_max):
        if n_max < 2:
            return []
        sieve_list = [True] * (n_max + 1)
        sieve_list[0], sieve_list[1] = False, False
        for i in range(2, int(n_max ** 0.5) + 1):
            if sieve_list[i]:
                sieve_list[i*i : n_max+1 : i] = [False] * len(sieve_list[i*i : n_max+1 : i])
        return [i for i, val in enumerate(sieve_list) if val]
    
    sieve_primes = sieve(n)
    
    case2_primes = []
    for p in sieve_primes:
        if p in case1_primes:
            continue
        m = p - 1
        coeff_dict = defaultdict(int)
        for i in range(len(a_list)):
            exponent = n - i  # exponent for term a_{N-i} x^{exponent}
            ai = a_list[i]
            if exponent == 0:
                e = 0
            else:
                e = ((exponent - 1) % m) + 1
            ai_mod = ai % p
            coeff_dict[e] = (coeff_dict[e] + ai_mod) % p
        
        # Check if all coefficients are zero modulo p
        valid = True
        for e in coeff_dict:
            if coeff_dict[e] % p != 0:
                valid = False
                break
        if valid:
            case2_primes.append(p)
    
    # Combine and sort the results
    result = sorted(case1_primes.union(case2_primes))
    for p in result:
        print(p)
