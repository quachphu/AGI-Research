To solve this problem, we need to determine the minimal amount of fuel that can flow from the first node to the last node in a directed acyclic graph (DAG) while adhering to constraints on minimum and maximum fuel flow through each pipe. Additionally, among all valid configurations that achieve this minimal fuel flow, we need to find the one with the maximum possible cost, where the cost is determined by the activation cost and the square of the fuel flow through each active pipe.

### Approach
1. **Graph Representation**: Represent the graph using a dictionary to store the edges and their attributes (minimum flow, maximum flow, activation cost).
2. **Generate Combinations**: For each possible total fuel flow (starting from 0 upwards), generate all valid combinations of fuel flows through the edges from the first node that sum to the current total fuel flow.
3. **Recursive Backtracking**: For each valid combination of flows from the first node, recursively process subsequent nodes to determine valid flows through their outgoing edges, ensuring flow conservation and constraints.
4. **Cost Calculation**: For each valid configuration of flows that reaches the last node with the minimal fuel flow, calculate the total cost and track the maximum cost.

import sys
import itertools

def generate_combinations(constraints, index, current_combo, current_sum, target, results):
    if index == len(constraints):
        if current_sum == target:
            results.append([(constraints[i][0], c) for i, c in enumerate(current_combo)])
        return
    j, l, h, a = constraints[index]
    for c in range(l, h + 1):
        if current_sum + c > target:
            continue
        current_combo.append(c)
        generate_combinations(constraints, index + 1, current_combo, current_sum + c, target, results)
        current_combo.pop()

def process_node(i, assignments, sum_n, F, edges_dict, outgoing_edges, n):
    if i == n:
        return -1  # processed all nodes except n-1, check sum_n in node n-1 processing
    if i == n - 1:
        # after processing node n-1, check sum_n
        if sum_n == F:
            cost = 0
            for (s, f) in assignments:
                c = assignments[(s, f)]
                if c > 0:
                    a = edges_dict[(s, f)][2]
                    cost += a + c ** 2
            return cost
        else:
            return -1
    # Compute incoming_flow for node i
    incoming_flow = 0
    for s in range(1, i):
        if (s, i) in assignments:
            incoming_flow += assignments[(s, i)]
    # Get outgoing edges for node i
    outgoing = outgoing_edges[i]
    if not outgoing:
        # No outgoing edges, so incoming_flow must be zero
        if incoming_flow == 0:
            return process_node(i + 1, assignments, sum_n, F, edges_dict, outgoing_edges, n)
        else:
            return -1
    # Generate all possible combinations of outgoing edges
    constraints = [(j, l, h, a) for (j, l, h, a) in outgoing]
    combinations = []
    generate_combinations(constraints, 0, [], 0, incoming_flow, combinations)
    max_cost = -1
    for combo in combinations:
        new_assignments = assignments.copy()
        new_sum_n = sum_n
        for (j, c) in combo:
            new_assignments[(i, j)] = c
            if j == n:
                new_sum_n += c
        # Proceed to next node
        cost = process_node(i + 1, new_assignments, new_sum_n, F, edges_dict, outgoing_edges, n)
        if cost > max_cost:
            max_cost = cost
    return max_cost

def main():
    n = int(sys.stdin.readline())
    edges = {}
    outgoing = [[] for _ in range(n + 1)]  # outgoing[i] contains edges (j, l, h, a)
    for _ in range(n * (n - 1) // 2):
        s, f, l, h, a = map(int, sys.stdin.readline().split())
        edges[(s, f)] = (l, h, a)
        outgoing[s].append((f, l, h, a))
    # Pre-sort outgoing edges for each node (based on j)
    for i in range(n + 1):
        outgoing[i].sort()
    # Iterate F from 0 upwards
    for F in itertools.count(0):
        # Generate all possible combinations for node 1's outgoing edges
        node1_outgoing = outgoing[1]
        if not node1_outgoing and F == 0 and n == 2:
            # Special case where node1 has no outgoing edges but must reach node2 (n=2)
            # But impossible, since no edges from 1 to 2
            pass
        constraints_node1 = [(j, l, h, a) for (j, l, h, a) in node1_outgoing]
        combinations_node1 = []
        generate_combinations(constraints_node1, 0, [], 0, F, combinations_node1)
        # Check if there are any possible combinations for node1
        if not combinations_node1:
            # Check if F can be zero and no edges from node1 (only possible if n=1, but n >=2)
            if F == 0 and not node1_outgoing and n == 2:
                # But node1 must have an edge to node2 as per input constraints
                # So impossible
                continue
            else:
                continue
        current_max_cost = -1
        for combo in combinations_node1:
            assignments = {}
            sum_n = 0
            valid = True
            for (j, c) in combo:
                s = 1
                l, h, a = edges[(s, j)]
                if not (l <= c <= h):
                    valid = False
                    break
                assignments[(s, j)] = c
                if j == n:
                    sum_n += c
            if not valid:
                continue
            # Proceed to process node 2
            cost = process_node(2, assignments, sum_n, F, edges, outgoing, n)
            if cost > current_max_cost:
                current_max_cost = cost
        if current_max_cost != -1:
            print(F, current_max_cost)
            return
    print("-1 -1")
