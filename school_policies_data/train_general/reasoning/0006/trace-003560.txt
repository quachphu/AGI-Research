import sys
import heapq

def main():
    x, k, n, q = map(int, sys.stdin.readline().split())
    c = list(map(int, sys.stdin.readline().split()))
    special = {}
    for _ in range(q):
        p, wp = map(int, sys.stdin.readline().split())
        special[p] = wp

    # Initial gaps: each pollywog is consecutive
    initial_gaps = tuple([1]*(x-1))
    initial_left = 1
    target_left = n - x + 1

    # Priority queue: (total_cost, current_left, gaps)
    heap = []
    heapq.heappush(heap, (0, initial_left, initial_gaps))

    # Visited dictionary: (left, gaps) -> cost
    visited = {}
    visited[(initial_left, initial_gaps)] = 0

    found = False

    while heap:
        current_cost, left, gaps = heapq.heappop(heap)

        if left > target_left:
            continue
        if left == target_left:
            print(current_cost)
            found = True
            break

        # Calculate current positions
        current_positions = [left]
        s = left
        for gap in gaps:
            s += gap
            current_positions.append(s)
        # The leftmost is current_positions[0]
        # When it jumps, the new left is current_positions[1]
        # The new gaps will be gaps[1:] + [new_gap]

        for jump in range(1, k+1):
            new_pos = current_positions[0] + jump
            # Check if new_pos is not occupied by any other pollywog
            valid = True
            for pos in current_positions[1:]:
                if new_pos == pos:
                    valid = False
                    break
            if not valid:
                continue

            new_left = current_positions[1]
            new_gaps = list(gaps[1:])
            sum_current_gaps = sum(gaps)
            new_gap = new_pos - current_positions[-1]
            if new_gap < 1:
                continue
            new_gaps.append(new_gap)
            if len(new_gaps) != x-1:
                continue
            new_gaps_tuple = tuple(new_gaps)

            # Compute the cost of this jump
            jump_cost = c[jump-1]

            # Check if new_pos is a special stone
            if new_pos in special:
                jump_cost += special[new_pos]

            # Compute new_left and check if new_left is valid
            # The new state's left is new_left, gaps are new_gaps_tuple
            new_total_cost = current_cost + jump_cost

            key = (new_left, new_gaps_tuple)
            if key not in visited or new_total_cost < visited[key]:
                visited[key] = new_total_cost
                heapq.heappush(heap, (new_total_cost, new_left, new_gaps_tuple))

    if not found:
        print(-1)

if __name__ == "__main__":
    main()
