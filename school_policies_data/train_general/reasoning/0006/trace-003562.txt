### Approach
1. **Parse Input**: Read the coordinates of the vertices for both squares.
2. **Axis-Aligned Square Representation**: Determine the minimum and maximum x and y coordinates to form the bounding box of the axis-aligned square.
3. **Rotated Square Representation**: Calculate the center (a, b) and the radius (r) of the diamond shape formed by the rotated square.
4. **Vertex Inclusion Check**: Check if any vertex of the axis-aligned square lies within the diamond and vice versa.
5. **Edge Intersection Check**: Check if any edge of the axis-aligned square intersects with any edge of the rotated square using line segment intersection detection.

def main():
    import sys
    input = sys.stdin.read().splitlines()
    
    # Parse axis-aligned square (first line)
    axis_line = list(map(int, input[0].split()))
    axis_x = [axis_line[i] for i in range(0, 8, 2)]
    axis_y = [axis_line[i+1] for i in range(0, 8, 2)]
    min_x = min(axis_x)
    max_x = max(axis_x)
    min_y = min(axis_y)
    max_y = max(axis_y)
    
    # Vertices of the axis-aligned square
    vertices_axis = [
        (min_x, min_y),
        (max_x, min_y),
        (max_x, max_y),
        (min_x, max_y)
    ]
    
    # Parse rotated square (second line)
    rotated_line = list(map(int, input[1].split()))
    rotated_points = []
    for i in range(0, 8, 2):
        x = rotated_line[i]
        y = rotated_line[i+1]
        rotated_points.append((x, y))
    
    # Calculate center (a, b) of the rotated square
    sum_x = sum(x for x, y in rotated_points)
    sum_y = sum(y for x, y in rotated_points)
    a = sum_x / 4.0
    b = sum_y / 4.0
    
    # Calculate r (using the first vertex)
    x0, y0 = rotated_points[0]
    r = abs(x0 - a) + abs(y0 - b)
    
    # Check if any vertex of axis-aligned square is inside the rotated square
    for x, y in vertices_axis:
        if abs(x - a) + abs(y - b) <= r + 1e-9:
            print("YES")
            return
    
    # Check if any vertex of rotated square is inside the axis-aligned square
    for x, y in rotated_points:
        if (min_x <= x <= max_x) and (min_y <= y <= max_y):
            print("YES")
            return
    
    # Define edges for axis-aligned square
    axis_edges = [
        ( (min_x, min_y), (max_x, min_y) ),
        ( (max_x, min_y), (max_x, max_y) ),
        ( (max_x, max_y), (min_x, max_y) ),
        ( (min_x, max_y), (min_x, min_y) )
    ]
    
    # Define edges for rotated square
    rotated_edges = []
    for i in range(4):
        j = (i + 1) % 4
        rotated_edges.append( (rotated_points[i], rotated_points[j]) )
    
    # CCW function to determine orientation
    def ccw(A, B, C):
        return (B[0] - A[0]) * (C[1] - A[1]) - (B[1] - A[1]) * (C[0] - A[0])
    
    # Check if a point is on a segment
    def is_point_on_segment(P, A, B):
        if (min(A[0], B[0]) <= P[0] <= max(A[0], B[0]) and
            min(A[1], B[1]) <= P[1] <= max(A[1], B[1])):
            return ccw(A, B, P) == 0
        return False
    
    # Check if two segments intersect
    def segments_intersect(A, B, C, D):
        ccw1 = ccw(A, B, C)
        ccw2 = ccw(A, B, D)
        ccw3 = ccw(C, D, A)
        ccw4 = ccw(C, D, B)
        
        if (ccw1 * ccw2 < 0) and (ccw3 * ccw4 < 0):
            return True
        
        if (ccw1 == 0 and is_point_on_segment(C, A, B)) or \
           (ccw2 == 0 and is_point_on_segment(D, A, B)) or \
           (ccw3 == 0 and is_point_on_segment(A, C, D)) or \
           (ccw4 == 0 and is_point_on_segment(B, C, D)):
            return True
        
        return False
    
    # Check all edges for intersection
    for a_edge in axis_edges:
        A, B = a_edge
        for r_edge in rotated_edges:
            C, D = r_edge
            if segments_intersect(A, B, C, D):
                print("YES")
                return
    
    print("NO")
