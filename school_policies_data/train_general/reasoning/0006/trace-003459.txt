To solve this problem, we need to determine if it is possible to express a given integer \( N \) as the sum of two gaps between consecutive prime numbers, where each gap is between primes \( \geq 3 \).

### Approach
1. **Generate Primes**: Use the Sieve of Eratosthenes to generate all prime numbers up to \( 10^7 \).
2. **Filter Primes**: Consider only primes \( \geq 3 \) since the problem constraints require all primes to be at least 3.
3. **Compute Gaps**: Calculate the gaps between consecutive primes from the filtered list.
4. **Track Gap Frequencies**: Use a frequency counter to track how many times each gap occurs.
5. **Precompute Possible Sums**: Create a boolean array to check if each possible sum of two gaps (up to twice the maximum gap) can be formed using the gaps encountered.

### Solution Code
```python
import sys
from collections import defaultdict

def sieve(n):
    sieve = bytearray([1]) * (n + 1)
    sieve[0] = sieve[1] = 0
    for i in range(2, int(n ** 0.5) + 1):
        if sieve[i]:
            sieve[i*i::i] = b'\x00' * len(sieve[i*i::i])
    primes = []
    for i in range(2, n + 1):
        if sieve[i]:
            primes.append(i)
    return primes

def main():
    primes = sieve(10**7)
    primes_above_3 = [p for p in primes if p >= 3]
    possible = set()
    S_max = 0

    if len(primes_above_3) >= 2:
        gaps = []
        prev = primes_above_3[0]
        for p in primes_above_3[1:]:
            gaps.append(p - prev)
            prev = p

        if gaps:
            freq = defaultdict(int)
            for g in gaps:
                freq[g] += 1

            unique_gaps = list(freq.keys())
            max_gap = max(unique_gaps)
            S_max = 2 * max_gap
            possible_arr = [False] * (S_max + 1)

            n_unique = len(unique_gaps)
            for i in range(n_unique):
                g1 = unique_gaps[i]
                for j in range(i, n_unique):
                    g2 = unique_gaps[j]
                    s = g1 + g2
                    if s > S_max:
                        continue
                    if i == j:
                        if freq[g1] >= 2:
                            possible_arr[s] = True
                    else:
                        if freq[g1] >= 1 and freq[g2] >= 1:
                            possible_arr[s] = True
            possible = possible_arr
        else:
            possible = set()

    input_data = sys.stdin.read().split()
    T = int(input_data[0])
    results = []
    for i in range(1, T + 1):
        N = int(input_data[i])
        if len(primes_above_3) < 2:
            results.append("NO")
        else:
            if not gaps:
                results.append("NO")
                continue
            if N > S_max:
                results.append("NO")
            else:
                results.append("YES" if possible[N] else "NO")
    print('\n'.join(results))
```
