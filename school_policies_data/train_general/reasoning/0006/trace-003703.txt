Additionally, among all possible ways to achieve the maximum number of occurrences, we need to find the one with the minimum number of replacements.

1. **Precompute Validity and Cost for Each Window**:
   - Determine if each possible starting position `i` in `s` can form a valid occurrence of `t` after replacements.

    else:

q_even = [0] * (n + 1)
q_odd = [0] * (n + 1)

    if pos % 2 == 0 and s[pos] == '?':

        q_odd[x] = q_odd[x-1]

# Precompute validity and cost for each window
window_cost = [None] * (n - m + 1)
for i in range(n - m + 1):
    L = i + 1  # 1-based start
    R = i + m  # 1-based end (i+0 ... i+m-1 is 1-based L to R)
    p = i % 2
    # Check even parts (same parity as i)
    if p == 0:
        invalid_even = a_even_prefix[R] - a_even_prefix[L-1]
    else:
        invalid_even = a_odd_prefix[R] - a_odd_prefix[L-1]
    # Check odd parts (opposite parity)
    if p == 0:
        invalid_odd = b_odd_prefix[R] - b_odd_prefix[L-1]
    else:
        invalid_odd = b_even_prefix[R] - b_even_prefix[L-1]
    if invalid_even == 0 and invalid_odd == 0:
        # Compute cost
        if p == 0:
            count_even = q_even[R] - q_even[L-1]
            count_odd = q_odd[R] - q_odd[L-1]
        else:
            count_even = q_odd[R] - q_odd[L-1]
            count_odd = q_even[R] - q_even[L-1]
        cost = count_even + count_odd
        window_cost[i] = cost
    else:
        window_cost[i] = None

# Dynamic programming to find maximum count with minimal cost
dp = [ (0, 0) ] * (n)
for j in range(n):
    # Carry over the previous best
    current_count, current_cost = dp[j-1] if j > 0 else (0, 0)
    # Check if there's a window ending at j
    if j >= m - 1:
        i_start = j - (m - 1)
        if i_start >= 0 and i_start < len(window_cost):

            if cost_i is not None:

                prev_j = i_start - 1

                new_cost = prev_cost + cost_i
                if new_count > current_count:
                    current_count, current_cost = new_count, new_cost
                elif new_count == current_count and new_cost < current_cost:
                    current_cost = new_cost
    dp[j] = (current_count, current_cost)
