To solve this problem, we need to transform the binary digits of nodes in a tree from their initial state to a target state using the minimum possible cost. Each operation allows shuffling the digits in a subtree, incurring a cost based on the subtree's root node. The solution involves checking feasibility and then using a systematic approach to minimize the cost.

### Approach
1. **Feasibility Check**: First, check if the total number of 0s and 1s in the initial and target states are the same. If not, it's impossible to achieve the target.
2. **Minimum Cost Calculation**: For each node, compute the minimum cost along the path from the root to that node. This helps in determining the optimal node to use for operations.
3. **Post-order Traversal**: Process each node in post-order to accumulate discrepancies (nodes needing changes) and handle them at the lowest possible cost using the precomputed minimum costs.

### Solution Code

import sys
from collections import deque

def main():
    n = int(sys.stdin.readline())
    a = [0] * (n + 1)
    b = [0] * (n + 1)
    c = [0] * (n + 1)
    for i in range(1, n+1):
        ai, bi, ci = map(int, sys.stdin.readline().split())
        a[i] = ai
        b[i] = bi
        c[i] = ci

    edges = [[] for _ in range(n+1)]
    for _ in range(n-1):
        u, v = map(int, sys.stdin.readline().split())
        edges[u].append(v)
        edges[v].append(u)

    sum_b = sum(b[i] for i in range(1, n+1))
    sum_c = sum(c[i] for i in range(1, n+1))
    if sum_b != sum_c:
        print(-1)
        return

    # Compute min_a for each node using BFS
    min_a = [0] * (n + 1)
    visited = [False] * (n + 1)
    q = deque()
    q.append(1)
    visited[1] = True
    min_a[1] = a[1]

    while q:
        u = q.popleft()
        for v in edges[u]:
            if not visited[v]:
                visited[v] = True
                min_a[v] = min(min_a[u], a[v])
                q.append(v)

    # Iterative post-order traversal
    res_up = {}
    res_down = {}
    stack = [(1, None, False)]
    cost = 0

    while stack:
        u, parent, visited_flag = stack.pop()
        if not visited_flag:
            stack.append((u, parent, True))
            # Collect children (excluding parent)
            children = []
            for v in edges[u]:
                if v != parent:
                    children.append(v)
            # Push children in reverse order to process left to right
            for v in reversed(children):
                stack.append((v, u, False))
        else:
            # Calculate up and down for this node
            up = 0
            down = 0
            if b[u] != c[u]:
                if b[u] == 0:
                    up = 1
                else:
                    down = 1
            # Accumulate from children
            for v in edges[u]:
                if v != parent:
                    up += res_up.get(v, 0)
                    down += res_down.get(v, 0)
            # Compute pairs and add to cost
            pairs = min(up, down)
            cost += pairs * 2 * min_a[u]
            up -= pairs
            down -= pairs
            res_up[u] = up
            res_down[u] = down

    print(cost)

if __name__ == "__main__":
    main()
