def parse_terms(s):

    terms = []

    i = 0

    while i < len(s):

        if i+2 < len(s) and s[i] == '.' and s[i+1] == '.' and s[i+2] == '.':

            terms.append('...')

            i +=3

        elif s[i] in ', ':

            terms.append(s[i])

            i +=1

        else:

            # number: collect all digits

            j = i

            while j < len(s) and s[j].isdigit():

                j +=1

            terms.append(s[i:j])

            i = j

    return terms

            terms.append('...')

            terms.append(s[i])
            i += 1
        else:
            # Number
            j = i

            terms.append(s[i:j])

    parsed = parse_terms(s)
    non_space = []
    space_indicators = []
    previous_was_space = False
    for term in parsed:
        if term == ' ':
            previous_was_space = True
        else:
            if non_space:
                space_indicators.append(previous_was_space)
            non_space.append(term)
            previous_was_space = False
    # Now process non_space and space_indicators
    output = []
    has_space_before_list = [False] + space_indicators
    for i in range(len(non_space)):
        term = non_space[i]
        has_space_before = has_space_before_list[i]
        # Check if we need to add space before
        add_space = False
        if term == '...' and i > 0:
            if not output or output[-1] != ' ':
                add_space = True
        elif i > 0 and is_number(non_space[i-1]) and is_number(term) and has_space_before:
            add_space = True
        if add_space:
            output.append(' ')
        # Add the term
        output.append(term)
        # Check if we need to add space after
        if term == ',' and i != len(non_space) - 1:
            output.append(' ')
    return ''.join(output)
