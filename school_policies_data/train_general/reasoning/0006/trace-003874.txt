### Solution Code
```python
import sys

def main():
    input = sys.stdin.read().split()
    ptr = 0
    N = int(input[ptr])
    ptr += 1
    Q = int(input[ptr])
    ptr += 1
    
    obstacles = []
    for _ in range(N):
        xi = int(input[ptr])
        yi = int(input[ptr+1])
        zi = int(input[ptr+2])
        ri = int(input[ptr+3])
        li = int(input[ptr+4])
        obstacles.append((xi, yi, zi, ri, li))
        ptr +=5
    
    for _ in range(Q):
        sx = int(input[ptr])
        sy = int(input[ptr+1])
        sz = int(input[ptr+2])
        dx = int(input[ptr+3])
        dy = int(input[ptr+4])
        dz = int(input[ptr+5])
        ptr +=6
        
        sum_l = 0
        for (xi, yi, zi, ri, li) in obstacles:
            # S is (sx, sy, sz), D is (dx, dy, dz)

            abx = dx - sx
            aby = dy - sy
            abz = dz - sz

            apx = xi - sx
            apy = yi - sy
            apz = zi - sz
            
            dot = apx * abx + apy * aby + apz * abz
            len_sq = abx**2 + aby**2 + abz**2

            t = dot / len_sq
            t_clamped = max(0.0, min(1.0, t))

if __name__ == "__main__":
```
