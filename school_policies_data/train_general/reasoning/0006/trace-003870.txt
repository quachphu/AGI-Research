A Monday-Saturday prime is a Monday-Saturday number greater than 1 that has no Monday-Saturday divisors other than 1 and itself.

def get_divisors(n):
    divisors = set()
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            divisors.add(i)
            divisors.add(n // i)
    divisors.add(n)
    return sorted(divisors)

def is_ms_prime(d):
    for i in range(2, int(d**0.5) + 1):
        if d % i == 0:
            if i % 7 in {1, 6}:
                return False
            other = d // i
            if other != i and other % 7 in {1, 6}:
                return False
    return True

    ms_divisors = [d for d in divisors if d % 7 in {1, 6}]
    primes = []
    for d in ms_divisors:
        if is_ms_prime(d):
            primes.append(d)
    primes = sorted(primes)
    print(f"{n}: {' '.join(map(str, primes))}")
