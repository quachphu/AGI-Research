Each transformation of a pair of identical characters costs one unit of energy. The goal is to determine the minimum energy required or if it's impossible.

### Approach
1. **Split the String into Runs**: Identify consecutive runs of 'Y' or 'N' characters.
2. **Dynamic Programming (DP) Setup**: Use a DP approach to track the minimum energy required to process each run such that the resulting string has no consecutive identical characters.
3. **Process Each Run**: For each run, determine possible ways to split it into valid segments, considering the previous character to ensure no consecutive duplicates.
4. **Transitions Between Runs**: Track transitions between runs using DP states to ensure valid sequences and compute the minimum energy cost.

def process_test_cases():
    import sys
    input = sys.stdin.read().split()
    idx = 0
    T = int(input[idx])
    idx += 1
    for _ in range(T):
        L = int(input[idx])
        idx += 1
        S = input[idx]
        idx += 1
        
        # Split into runs
        runs = []
        if L == 0:
            print(0)
            continue
        current_char = S[0]
        current_length = 1
        for c in S[1:]:
            if c == current_char:
                current_length += 1
            else:
                runs.append((current_char, current_length))
                current_char = c
                current_length = 1
        runs.append((current_char, current_length))
        
        options_list = []
        for c, m in runs:
            options = []
            if m % 2 == 0:
                # Even length: two options
                cost = m // 2
                other_char = 'Y' if c == 'N' else 'N'
                options.append((c, other_char, cost))
                options.append((other_char, c, cost))
            else:
                # Odd length: one option
                cost = (m - 1) // 2
                options.append((c, c, cost))
            options_list.append(options)
        
        # Dynamic programming
        if not options_list:
            print(0)
            continue
        
        # Initialize for the first run
        dp = {}
        for start, end, cost in options_list[0]:
            if end not in dp or cost < dp[end]:
                dp[end] = cost
        
        # Process subsequent runs
        for i in range(1, len(options_list)):
            current_options = options_list[i]
            new_dp = {}
            for prev_end in dp:
                prev_cost = dp[prev_end]
                for start, end, cost in current_options:
                    if start != prev_end:
                        total_cost = prev_cost + cost
                        if end not in new_dp or total_cost < new_dp.get(end, float('inf')):
                            new_dp[end] = total_cost
            dp = new_dp
