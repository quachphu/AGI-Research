    for i in range(2, int(n ** 0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, n + 1, i):
                is_prime[j] = False
    return is_prime

max_n = 10000
is_prime = sieve(max_n)
primes = [i for i, prime in enumerate(is_prime) if prime]

counts = [0] * (max_n + 1)

for q in primes:
    two_q = 2 * q
    max_p_val = max_n - two_q
    if max_p_val < 2:
        continue
    idx = bisect.bisect_right(primes, max_p_val)
    for p in primes[:idx]:
        n = p + two_q
