    k = n - 2

    return True

def next_permutation(arr):
    n = len(arr)
    k = n - 2
    while k >= 0 and arr[k] >= arr[k + 1]:
        k -= 1
    if k == -1:
        return False
    l = n - 1
    while arr[l] <= arr[k]:
        l -= 1
    arr[k], arr[l] = arr[l], arr[k]
    arr[k+1:] = arr[k+1:][::-1]
    return True

def previous_permutation(arr):
    n = len(arr)
    k = n - 2

        k -= 1

        return False
    l = n - 1
    while arr[l] >= arr[k]:
        l -= 1
    arr[k], arr[l] = arr[l], arr[k]
