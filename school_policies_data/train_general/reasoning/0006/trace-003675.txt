The solution involves using Heavy-Light Decomposition (HLD) to break the tree into chains, allowing efficient range sum queries and point updates using a Fenwick Tree (Binary Indexed Tree). Additionally, we use binary lifting to quickly find the Lowest Common Ancestor (LCA) of two nodes.

### Approach
1. **Tree Representation**: Read the input to construct the tree using an adjacency list.
2. **Depth, Parent, and Size Calculation**: Perform a Depth-First Search (DFS) to compute the depth, parent, and size of each subtree for every node.
3. **Binary Lifting for LCA**: Precompute a table for binary lifting to efficiently find the LCA of any two nodes.
4. **Heavy-Light Decomposition (HLD)**: Decompose the tree into chains to allow efficient range queries. Each node is assigned a position in a linear array.
5. **Fenwick Tree**: Use a Fenwick Tree to handle point updates and range sum queries on the linear array obtained from HLD.
6. **Query Handling**: For each query, update the Fenwick Tree for point updates or compute the sum along the path between two nodes using the LCA and HLD decomposition.

import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    ptr = 0
    N = data[ptr]
    ptr += 1
    cost = [0] * (N + 1)
    for i in range(1, N + 1):
        cost[i] = data[ptr]
        ptr += 1
    adj = [[] for _ in range(N + 1)]
    for _ in range(N - 1):
        a = data[ptr]
        b = data[ptr + 1]
        adj[a].append(b)
        adj[b].append(a)
        ptr += 2

    parent = [0] * (N + 1)
    depth = [0] * (N + 1)
    size = [0] * (N + 1)

    def dfs(u, p):
        parent[u] = p
        size[u] = 1
        for v in adj[u]:
            if v != p:
                depth[v] = depth[u] + 1
                dfs(v, u)
                size[u] += size[v]
    dfs(1, 0)

    log_max = 20
    up = [[0] * (N + 1) for _ in range(log_max)]
    up[0] = parent
    for k in range(1, log_max):
        for u in range(1, N + 1):
            up[k][u] = up[k - 1][up[k - 1][u]]

    def get_lca(u, v):
        if depth[u] < depth[v]:
            u, v = v, u
        for k in range(log_max - 1, -1, -1):
            if depth[u] - (1 << k) >= depth[v]:
                u = up[k][u]
        if u == v:
            return u
        for k in range(log_max - 1, -1, -1):
            if up[k][u] != up[k][v]:
                u = up[k][u]
                v = up[k][v]
        return parent[u]

    head = [0] * (N + 1)
    pos = [0] * (N + 1)
    current_pos = 0

    def decompose(u, h):
        nonlocal current_pos
        head[u] = h
        pos[u] = current_pos
        current_pos += 1
        max_size = -1
        heavy = -1
        for v in adj[u]:
            if v != parent[u] and size[v] > max_size:
                max_size = size[v]
                heavy = v
        if heavy != -1:
            decompose(heavy, h)
            for v in adj[u]:
                if v != parent[u] and v != heavy:
                    decompose(v, v)
    decompose(1, 1)

    class FenwickTree:
        def __init__(self, size):
            self.n = size
            self.tree = [0] * (self.n + 1)
        def add(self, idx, delta):
            idx += 1
            while idx <= self.n:
                self.tree[idx] += delta
                idx += idx & -idx
        def query(self, idx):
            res = 0
            idx += 1
            while idx > 0:
                res += self.tree[idx]
                idx -= idx & -idx
            return res
        def range_query(self, l, r):
            return self.query(r) - self.query(l - 1)
