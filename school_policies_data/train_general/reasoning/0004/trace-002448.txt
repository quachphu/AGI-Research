import sys
from collections import defaultdict

def main():
    n = int(sys.stdin.readline())
    k_list = list(map(int, sys.stdin.readline().split()))
    
    max_k = max(k_list) if k_list else 0
    primes = []
    sieve = [True] * (5001)
    sieve[0] = sieve[1] = False
    for i in range(2, 5001):
        if sieve[i]:
            primes.append(i)
            for j in range(i*i, 5001, i):
                sieve[j] = False
    
    vp = [defaultdict(int) for _ in range(max(k_list, default=0)+1)]
    
    for k in k_list:
        if k == 0:
            continue
        for p in primes:
            if p > k:
                break
            total = 0
            current = k
            while current > 0:
                current //= p
                total += current
            vp[k][p] = total
    
    prime_freq = defaultdict(list)
    for k in k_list:
        if k == 0:
            continue
        factors = []
        for p in primes:
            if p > k:
                break
            cnt = vp[k][p]
            for _ in range(cnt):
                factors.append(p)
        for i in range(len(factors)+1):
            if i == 0:
                prime_freq[()].append(k)
            else:
                prime_freq[tuple(factors[:i])].append(k)
    
    max_sum = 0
    best_len = 0
    current_sum = 0
    current_len = 0
    stack = [()]
    
    while stack:
        path = stack.pop()
        count = len(prime_freq.get(path, []))
        current_total = 2 * count * (len(path) + 1) - n * (len(path) + 1)
        if current_total > max_sum:
            max_sum = current_total
            best_len = len(path) + 1
        if len(path) < 100:
            last_p = path[-1] if path else 0
            for p in primes:
