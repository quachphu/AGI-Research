def main():
    import sys
    N, K = map(int, sys.stdin.readline().split())
    s = sys.stdin.readline().strip()
    
    # DP table: dp[i][l][r] = number of ways after i steps, with painted interval [l, r] (1-based)
    # l ranges from 0 to N+1, r ranges from 0 to N+1. l=0 and r=0 represents no painted area.
    # To handle empty intervals, we use l > r to indicate that the painted area is empty.
    dp = [[[0]*(N+2) for _ in range(N+2)] for __ in range(K+1)]
    dp[0][0][0] = 1  # Initial state: no painted area
    
    for step in range(1, K+1):
        current_char = s[step-1]
        for prev_l in range(N+2):
            for prev_r in range(N+2):
                if dp[step-1][prev_l][prev_r] == 0:
                    continue
                if current_char == 'r':
                    # Paint red: choose any contiguous interval [a, b], which can be empty
                    # Empty interval: new_l = prev_l, new_r = prev_r

                    # Non-empty intervals
                    for a in range(1, N+1):
                        for b in range(a, N+1):
                            if prev_l == 0 and prev_r == 0:
                                new_l, new_r = a, b
                            elif prev_l > prev_r:
                                new_l, new_r = a, b
                            else:
                                new_l = min(prev_l, a)
                                new_r = max(prev_r, b)
                            dp[step][new_l][new_r] = (dp[step][new_l][new_r] + dp[step-1][prev_l][prev_r]) % MOD
                else:

                    if prev_l > prev_r:
