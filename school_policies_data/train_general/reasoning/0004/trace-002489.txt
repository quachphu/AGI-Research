### Approach
1. **Tree Representation**: Represent the tree using an adjacency list to keep track of the neighbors of each vertex.
2. **Priority Handling**: Initially, each vertex's priority is its own label. When an "up" query is processed, the vertex's priority is updated to be 1 more than the current maximum priority.
3. **Simulating the Burning Process**: For each query, simulate the burning process by maintaining a priority queue of current leaves (vertices with degree 1). The leaves are processed in order of their priorities, and their neighbors' degrees are adjusted dynamically as they are removed.

import sys
import heapq
from collections import defaultdict, deque

def main():
    sys.setrecursionlimit(1 << 25)
    n, q = map(int, sys.stdin.readline().split())
    edges = [[] for _ in range(n+1)]
    degree = [0] * (n + 1)
    for _ in range(n-1):
        u, v = map(int, sys.stdin.readline().split())
        edges[u].append(v)
        edges[v].append(u)
        degree[u] += 1
        degree[v] += 1

    current_priorities = list(range(n+1))  # p_v = v initially
    max_priority = n

    for _ in range(q):
        query = sys.stdin.readline().split()
        if not query:
            continue
        if query[0] == 'up':
            v = int(query[1])
            max_priority += 1
            current_priorities[v] = max_priority
        elif query[0] == 'when' or query[0] == 'compare':
            # Simulate the burning process with current priorities
            # This is not feasible for large n and q, but works for small samples
            prio = current_priorities.copy()
            burn_order = []
            deg = degree.copy()
            leaf_heap = []
            for node in range(1, n+1):
                if deg[node] == 1:
                    heapq.heappush(leaf_heap, (prio[node], node))

            remaining = n
            step = 0
            burn_time = [0] * (n + 1)
            while leaf_heap:
                step += 1
                p, u = heapq.heappop(leaf_heap)
                burn_time[u] = step
                burn_order.append(u)
                for neighbor in edges[u]:
                    if burn_time[neighbor] != 0:
                        continue
                    deg[neighbor] -= 1
                    if deg[neighbor] == 1:
                        heapq.heappush(leaf_heap, (prio[neighbor], neighbor))
                remaining -= 1

            # After processing all leaves, the last node is handled
            last_node = None
            for node in range(1, n+1):
                if burn_time[node] == 0:
                    last_node = node
            if last_node is not None:
                burn_time[last_node] = step + 1

            if query[0] == 'when':

                print(burn_time[v])
