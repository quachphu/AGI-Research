### Approach

def generate_candidates(y):
    candidates = set()
    s = str(y)
    original = y
    if 1000 <= original <= 2011:
        candidates.add(original)
    for pos in range(4):
        current_digit = s[pos]
        for new_digit in range(10):
            if str(new_digit) == current_digit:
                continue
            if pos == 0 and new_digit == 0:
                continue
            new_s = s[:pos] + str(new_digit) + s[pos+1:]
            new_num = int(new_s)
            if 1000 <= new_num <= 2011:
                candidates.add(new_num)
    return sorted(candidates)
