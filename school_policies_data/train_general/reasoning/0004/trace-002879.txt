### Approach
1. **Model the Roads as Chords**: Each road connects two cities and can be considered as a chord on the circular ring. We need to check if these chords can be placed inside or outside without intersecting.
2. **Conflict Detection**: Two roads (chords) intersect if their endpoints are interleaved around the circle. This can be checked by determining if exactly one endpoint of each road lies within the interval defined by the other road's endpoints.
3. **Graph Bipartition**: Represent roads as nodes in a graph where edges indicate conflicts (i.e., roads that cannot be placed on the same side). Check if this graph is bipartite, which means it can be colored using two colors (representing inside and outside).

import sys
from collections import deque

def main():
    n, m = map(int, sys.stdin.readline().split())
    roads = []
    for _ in range(m):
        a, b = map(int, sys.stdin.readline().split())
        roads.append((a, b))
    
    processed = []
    for a, b in roads:
        d_cw = (b - a) % n
        d_ccw = (a - b) % n
        if d_cw <= d_ccw:
            s, e = a, b
        else:
            s, e = b, a
        processed.append((s, e))
    
    adj = [[] for _ in range(m)]
    for i in range(m):
        a_i, b_i = roads[i]
        s_i, e_i = processed[i]
        for j in range(i + 1, m):
            a_j, b_j = roads[j]
            s_j, e_j = processed[j]
            # Check if roads share any endpoint
            if {a_i, b_i} & {a_j, b_j}:
                continue
            # Check if exactly one of s_j or e_j is in i's interval
            in_j_s = (s_i <= s_j <= e_i) if s_i <= e_i else (s_j >= s_i or s_j <= e_i)
            in_j_e = (s_i <= e_j <= e_i) if s_i <= e_i else (e_j >= s_i or e_j <= e_i)
            if in_j_s ^ in_j_e:
                adj[i].append(j)
                adj[j].append(i)
    
    color = [-1] * m
    is_bipartite = True
    for i in range(m):
        if color[i] == -1:
            queue = deque([i])
            color[i] = 0
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        queue.append(v)
                    elif color[v] == color[u]:
                        is_bipartite = False
                        break
                if not is_bipartite:
                    break
            if not is_bipartite:
                break

    else:
