```python
n, m, k = map(int, input().split())

for r in range(n + 1):
    if 2 * r == n:
        if r * m == k:
            print("Yes")
            exit()
    else:
        d = n - 2 * r
        numerator = k - r * m
        if d == 0:
            continue  # This case is already handled by the if clause above
        if numerator % d != 0:
            continue
        c = numerator // d
        if 0 <= c <= m:
            print("Yes")
            exit()

print("No")
```
