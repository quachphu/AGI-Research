The solution involves maintaining the sorted order of the coordinates and efficiently computing the necessary values after each insertion or deletion.

### Approach
1. **Sort the Coordinates**: Keep the coordinates sorted to efficiently find the neighbors of any element during insertion or deletion.
2. **Track Maximum and Minimum Values**: Maintain the current maximum and minimum values of the coordinates to compute the total range.
3. **Track Maximum Gap**: The key insight is that the minimum number of moves is determined by the total range of the coordinates minus the largest gap between consecutive elements. This is because the largest gap can be split to reduce the total number of moves needed.

import bisect

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    idx = 0
    n, q = int(data[idx]), int(data[idx+1])
    idx +=2
    p = list(map(int, data[idx:idx+n]))
    idx +=n
    queries = []
    for _ in range(q):
        t = int(data[idx])
        x = int(data[idx+1])
        queries.append((t, x))
        idx +=2
    
    p.sort()
    current_max = p[-1] if n else 0
    current_min = p[0] if n else 0
    max_gap = 0
    if len(p) > 1:
        max_gap = max(b - a for a, b in zip(p, p[1:]))
    
    # Function to compute and print the answer
    def compute_answer():
        if not p:
            return 0
        return current_max - current_min - max_gap
    
    # Initial answer
    output = [str(compute_answer())]
    
    for t, x in queries:
        if t == 1:
            # Insert x
            bisect.insort(p, x)
            # Update current_max and current_min
            if len(p) == 1:
                current_max = current_min = x
            else:
                if x > current_max:
                    current_max = x
                elif x < current_min:
                    current_min = x
        else:
            # Remove x
            pos = bisect.bisect_left(p, x)
            if pos < len(p) and p[pos] == x:
                p.pop(pos)
            # Update current_max and current_min if necessary
            if not p:
                current_max = 0
                current_min = 0
            else:
                current_max = p[-1]

        if len(p) <= 1:
            max_gap = 0
        else:
            max_gap = max(b - a for a, b in zip(p, p[1:]))
        
        output.append(str(compute_answer()))
