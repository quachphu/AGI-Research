def is_leap(year):
    return year % 4 == 0

def is_valid(d, m, y):
    if m < 1 or m > 12:
        return False
    if d < 1:
        return False
    max_days = 31
    if m in [4, 6, 9, 11]:
        max_days = 30
    elif m == 2:
        max_days = 29 if is_leap(y) else 28
    return d <= max_days

def main():

    birth_date = sys.stdin.readline().strip()

    dd_f, mm_f, yy_f = map(int, finals_date.split('.'))
