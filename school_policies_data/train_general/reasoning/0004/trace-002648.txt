import sys

def main():
    t = int(sys.stdin.readline())
    for _ in range(t):
        n = int(sys.stdin.readline())
        variables = {}
        last_var = None
        for __ in range(n):
            line = sys.stdin.readline().strip()
            if ' := ' in line:
                var, val = line.split(' := ')
                count = 0
                for i in range(len(val) - 3 + 1):
                    if val[i:i+4] == 'haha':
                        count += 1
                prefix = val[:3] if len(val) >= 3 else val
                suffix = val[-3:] if len(val) >= 3 else val
                length = len(val)
                variables[var] = (count, prefix, suffix, length)
                last_var = var
            else:
                var_part, rest = line.split(' = ')
                a, b = rest.split(' + ')
                a_data = variables[a]
                b_data = variables[b]
                combined_overlap = a_data[2] + b_data[1]
                overlap = 0
                for i in range(len(combined_overlap) - 3 + 1):
                    if combined_overlap[i:i+4] == 'haha':
                        overlap += 1
                new_count = a_data[0] + b_data[0] + overlap
                new_len = a_data[3] + b_data[3]
                if a_data[3] >= 3:
                    new_prefix = a_data[1]
                else:
                    needed = 3 - a_data[3]
                    added = b_data[1][:needed]
                    new_prefix = (a_data[1] + added)[:3]
                combined_suffix = a_data[2] + b_data[2]
                if len(combined_suffix) >= 3:
                    new_suffix = combined_suffix[-3:]
                else:
                    new_suffix = combined_suffix
                variables[var_part] = (new_count, new_prefix, new_suffix, new_len)
                last_var = var_part
        print(variables[last_var][0])

if __name__ == "__main__":
    main()
