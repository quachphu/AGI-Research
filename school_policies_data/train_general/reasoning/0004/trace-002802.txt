To solve this problem, we need to determine the minimum time Vasya will spend to complete a ski race from point 0 to point L on the X-axis, using optional take-off ramps optimally. The solution involves modeling the problem as a graph and using Dijkstra's algorithm to find the shortest path.

### Approach
1. **Model the Problem as a Graph**: Represent each significant point (start, finish, ramp start points, and landing points) as nodes in a graph.
2. **Edges**: Create edges between consecutive points (movement edges) and edges representing the use of ramps (ramp edges). Movement edges allow bidirectional movement with time equal to the distance between points. Ramp edges are directed edges from the start of the ramp to the landing point with a time cost that includes preparation and flight time.
3. **Dijkstra's Algorithm**: Use Dijkstra's algorithm to find the shortest path from the start (0) to the finish (L). Track the path to determine which ramps are used.

### Solution Code
```python
import heapq

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr += 1
    L = int(input[ptr])
    ptr += 1
    
    ramps = []
    points = {0, L}
    
    for i in range(n):
        xi = int(input[ptr])
        ptr += 1
        di = int(input[ptr])
        ptr += 1
        ti = int(input[ptr])
        ptr += 1
        pi = int(input[ptr])
        ptr += 1
        ramps.append((xi, di, ti, pi))
        s_i = xi - pi
        if s_i >= 0:
            points.add(s_i)
        e_i = xi + di
        points.add(e_i)
    
    sorted_points = sorted(points)
    m = len(sorted_points)
    pos_to_idx = {pos: idx for idx, pos in enumerate(sorted_points)}
    
    # Build adjacency list
    adj = [[] for _ in range(m)]
    # Add movement edges (bidirectional)
    for i in range(m - 1):
        a = sorted_points[i]
        b = sorted_points[i + 1]
        cost = b - a
        adj[i].append((i + 1, cost, None))
        adj[i + 1].append((i, cost, None))
    
    # Add ramp edges (directed)
    for ramp_idx in range(n):
        xi, di, ti, pi = ramps[ramp_idx]
        s_i = xi - pi
        if s_i < 0:
            continue
        if s_i not in pos_to_idx:
            continue  # shouldn't happen as we added all s_i >=0
        e_i = xi + di
        if e_i not in pos_to_idx:
            continue  # also shouldn't happen
        s_idx = pos_to_idx[s_i]
        e_idx = pos_to_idx[e_i]
        cost = pi + ti
        adj[s_idx].append((e_idx, cost, ramp_idx + 1))  # ramp indices start at 1
    
    # Dijkstra's algorithm
    start_idx = pos_to_idx[0]
    end_idx = pos_to_idx[L]
    INF = float('inf')
    distance = [INF] * m
    distance[start_idx] = 0
    predecessor = [None] * m
    ramp_used = [None] * m
    
    heap = []
    heapq.heappush(heap, (0, start_idx))
    
    while heap:
        current_time, u = heapq.heappop(heap)
        if u == end_idx:
            break
        if current_time > distance[u]:
            continue
        for (v, cost, ramp) in adj[u]:
            new_time = current_time + cost
            if new_time < distance[v]:
                distance[v] = new_time
                predecessor[v] = u
                ramp_used[v] = ramp
                heapq.heappush(heap, (new_time, v))
    
    # Backtrack to find ramps used
    ramps_in_path = []
    current = end_idx
    while current != start_idx:
        ramp = ramp_used[current]
        if ramp is not None:
            ramps_in_path.append(ramp)
        prev = predecessor[current]
        if prev is None:
            break  # no path, but problem says input is valid
        current = prev
    ramps_in_path.reverse()
    
    print(distance[end_idx])
    print(len(ramps_in_path))
    if ramps_in_path:
        print(' '.join(map(str, ramps_in_path)))

if __name__ == "__main__":
    main()
```
