### Solution Code
```python
def main():
    import sys
    input = sys.stdin.read().split()
    n = int(input[0])
    a = list(map(int, input[1:n+1]))
    s = input[n+1]
    
    current = a.copy()
    for op in s:
        next_current = []
        for i in range(len(current)-1):
            if op == 'M':
                next_current.append(max(current[i], current[i+1]))
            else:
                next_current.append(min(current[i], current[i+1]))
        current = next_current
    print(current[0])

if __name__ == "__main__":
    main()
```
