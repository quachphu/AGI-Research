        self.n = size
        self.size = 1
        while self.size < self.n:
            self.size <<=1
        self.data = [-1]*(2*self.size)
        self.lazy = [-1]*(2*self.size)

    def push(self, node, l, r):
        if self.lazy[node] == -1:
            return
        if node < self.size:
            self.data[2*node] = max(self.data[2*node], self.lazy[node])
            self.lazy[2*node] = max(self.lazy[2*node], self.lazy[node])
            self.data[2*node+1] = max(self.data[2*node+1], self.lazy[node])
            self.lazy[2*node+1] = max(self.lazy[2*node+1], self.lazy[node])
        self.lazy[node] = -1

    def update_range(self, a, b, val, node=1, l=0, r=None):

            r = self.size -1
        if a > r or b < l:
            return

            self.lazy[node] = max(self.lazy[node], val)
