def roll(d, direction):
    top, front, right, left, back, bottom = d
    if direction == 'N':
        return [front, bottom, right, left, top, back]
    elif direction == 'S':
        return [back, top, right, left, bottom, front]
    elif direction == 'E':
        return [left, front, top, bottom, back, right]
    elif direction == 'W':
        return [right, front, bottom, top, back, left]
    else:
        return d

def generate_all_rotations(start):
    from collections import deque
    visited = set()
    queue = deque([tuple(start)])
    visited.add(tuple(start))
    directions = ['N', 'S', 'E', 'W']
    while queue:
        current = queue.popleft()
        current_list = list(current)
        for d in directions:
            new_state = roll(current_list, d)
            new_t = tuple(new_state)
            if new_t not in visited:
                visited.add(new_t)
                queue.append(new_t)
    return visited
