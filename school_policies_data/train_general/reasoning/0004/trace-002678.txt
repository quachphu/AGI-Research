### Approach
1. **Tree Structure and Subtree Sizes**: We first build the tree using adjacency lists and compute the size of each subtree when the tree is rooted at an initial node (e.g., node 1). This helps us understand the structure and size of each subtree rooted at different nodes.

2. **Initial Product Calculation**: Using the subtree sizes from the initial root, we compute the product of the sizes of all subtrees. This product is used to determine the number of ways to write numbers starting from the initial root.

3. **Rerooting Technique**: Using a breadth-first search (BFS), we dynamically adjust the product of subtree sizes when rerooting the tree from one node to another. This allows us to efficiently compute the product for all nodes in linear time.

4. **Modular Arithmetic**: Since the results need to be modulo \(10^9 + 7\), we use modular inverses to handle division operations correctly under modulo constraints.

import sys
from collections import deque

MOD = 10**9 + 7

def main():
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read().split()
    idx = 0
    N = int(input[idx])
    idx += 1
    edges = [[] for _ in range(N+1)]
    for _ in range(N-1):
        a = int(input[idx])
        b = int(input[idx+1])
        edges[a].append(b)
        edges[b].append(a)
        idx += 2

    root = 1
    parent = [0] * (N+1)
    size_map = dict()
    stack = [(root, 0)]
    post_order = []
    while stack:
        u, p = stack.pop()
        post_order.append(u)
        parent[u] = p
        for v in edges[u]:
            if v != p:
                stack.append((v, u))

    size = [1] * (N+1)
    for u in reversed(post_order):
        for v in edges[u]:
            if v != parent[u]:
                size[u] += size[v]
        if parent[u] != 0:
            size_map[(parent[u], u)] = size[u]

    product_s_edges = 1
    for (u, v) in size_map:
        product_s_edges = product_s_edges * size_map[(u, v)] % MOD

    P = [0] * (N+1)
    P[root] = N * product_s_edges % MOD

    q = deque()
    q.append((root, 0))
    while q:
        u, p = q.popleft()
        for v in edges[u]:
            if v == p:
                continue
            if (u, v) in size_map:
                s_uv = size_map[(u, v)]
            else:
                s_vu = size_map[(v, u)]
                s_uv = N - s_vu
            inv_s_uv = pow(s_uv, MOD-2, MOD)
            P_v = P[u] * ( (N - s_uv) % MOD ) % MOD
            P_v = P_v * inv_s_uv % MOD
            P[v] = P_v
            q.append((v, u))

    fact = 1
    for i in range(1, N+1):
        fact = fact * i % MOD

    for k in range(1, N+1):
        ans = fact * pow(P[k], MOD-2, MOD) % MOD
        print(ans)
