import sys
from collections import Counter

def determine_winner(s):

    odd_count = sum(1 for cnt in counts.values() if cnt % 2 != 0)
    if odd_count <= 1:
        return "JACK"
    else:
        return "JACK" if (odd_count - 1) % 2 == 0 else "DAVY JONES"

t = int(sys.stdin.readline())
