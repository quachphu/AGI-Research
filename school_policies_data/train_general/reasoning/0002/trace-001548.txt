MOD = 10**9 +9

n, a, b, k = map(int, input().split())
s = input().strip()

inv_a = pow(a, MOD-2, MOD)
q = (b * inv_a) % MOD

S = 0
current_q_power = 1  # q^0
for i in range(k):
    sign = 1 if s[i] == '+' else -1
    term = sign * current_q_power
    S = (S + term) % MOD
    current_q_power = (current_q_power * q) % MOD

r = pow(q, k, MOD)
m = (n + 1) // k

if r == 1:
    geometric_sum = m % MOD
else:
