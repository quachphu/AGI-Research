To solve this problem, we need to compute the sum of the shortest distances between all pairs of white squares in a grid, considering that some squares are black. The grid can be very large, but the number of black squares is small (up to 30). The solution involves efficiently calculating the Manhattan distances and adjusting for paths blocked by black squares.

### Approach

1. **Sum of Manhattan Distances**: Calculate the sum of Manhattan distances for all pairs of white squares as if there were no black squares. This involves mathematical formulas to compute the contributions of rows and columns separately.

2. **Adjust for Black Squares**: Subtract the contributions of pairs involving black squares. This includes pairs where one or both squares are black.

3. **Adjust for Blocked Paths**: For each black square, compute the number of pairs of white squares whose shortest path is blocked by the black square, leading to an increased distance. Each such pair contributes an adjustment to the total sum.

### Solution Code

```python
import bisect

MOD = 10**9 + 7

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    H = int(input[ptr])
    ptr += 1
    W = int(input[ptr])
    ptr += 1
    N = int(input[ptr])
    ptr += 1
    black = []
    row_black = {}
    col_black = {}
    for _ in range(N):
        x = int(input[ptr])
        ptr += 1
        y = int(input[ptr])
        ptr += 1
        black.append((x, y))
        if x not in row_black:
            row_black[x] = []
        row_black[x].append(y)
        if y not in col_black:
            col_black[y] = []
        col_black[y].append(x)
    
    for x in row_black:
        row_black[x].sort()
    for y in col_black:
        col_black[y].sort()
    
    row_counts = {x: len(ys) for x, ys in row_black.items()}
    col_counts = {y: len(xs) for y, xs in col_black.items()}

    sum_x_total = (W * W) * (H * (H * H - 1) // 3) % MOD
    sum_y_total = (H * H) * (W * (W * W - 1) // 3) % MOD

    sum_black_black_x = 0
    sum_black_black_y = 0
    for i in range(N):
        x_i, y_i = black[i]
        for j in range(N):
            x_j, y_j = black[j]
            sum_black_black_x += abs(x_i - x_j)
            sum_black_black_y += abs(y_i - y_j)
    sum_black_black_x %= MOD
    sum_black_black_y %= MOD

    white_in_row = [W] * H
    for x in row_black:
        white_in_row[x] = W - row_counts[x]
    
    white_in_col = [H] * W
    for y in col_black:
        white_in_col[y] = H - col_counts[y]

    prefix_sum_row = [0] * H
    prefix_weight_row = [0] * H
    current_sum = 0
    current_weight = 0
    for x in range(H):
        current_sum += white_in_row[x]
        current_weight += x * white_in_row[x]
        prefix_sum_row[x] = current_sum
        prefix_weight_row[x] = current_weight
    
    suffix_sum_row = [0] * H
    suffix_weight_row = [0] * H
    current_sum = 0
    current_weight = 0
    for x in range(H-1, -1, -1):
        current_sum += white_in_row[x]
        current_weight += x * white_in_row[x]
        suffix_sum_row[x] = current_sum
        suffix_weight_row[x] = current_weight
    
    sum_black_white_x = 0
    for x_a, y_a in black:
        left = x_a
        if left >= 0:
            sum1 = x_a * (prefix_sum_row[x_a] if x_a < H else 0) - (prefix_weight_row[x_a] if x_a < H else 0)
        else:
            sum1 = 0
        
        right = H - 1 - x_a
        if right > 0 and x_a + 1 < H:
            sum2 = (suffix_weight_row[x_a + 1] - x_a * suffix_sum_row[x_a + 1])
        else:
            sum2 = 0
        
        total = sum1 + sum2
        sum_black_white_x += total
        sum_black_white_x %= MOD

    prefix_sum_col = [0] * W
    prefix_weight_col = [0] * W
    current_sum = 0
    current_weight = 0
    for y in range(W):
        current_sum += white_in_col[y]
        current_weight += y * white_in_col[y]
        prefix_sum_col[y] = current_sum
        prefix_weight_col[y] = current_weight
    
    suffix_sum_col = [0] * W
    suffix_weight_col = [0] * W
    current_sum = 0
    current_weight = 0
    for y in range(W-1, -1, -1):
        current_sum += white_in_col[y]
        current_weight += y * white_in_col[y]
        suffix_sum_col[y] = current_sum
        suffix_weight_col[y] = current_weight
    
    sum_black_white_y = 0
    for x_a, y_a in black:
        left = y_a
        if left >= 0:
            sum1 = y_a * (prefix_sum_col[y_a] if y_a < W else 0) - (prefix_weight_col[y_a] if y_a < W else 0)
        else:
            sum1 = 0
        
        right = W - 1 - y_a
        if right > 0 and y_a + 1 < W:
            sum2 = (suffix_weight_col[y_a + 1] - y_a * suffix_sum_col[y_a + 1])
        else:
            sum2 = 0
        
        total = sum1 + sum2
        sum_black_white_y += total
        sum_black_white_y %= MOD

    sum_x_black = (sum_black_black_x + 2 * sum_black_white_x) % MOD
    sum_y_black = (sum_black_black_y + 2 * sum_black_white_y) % MOD

    sum_x_part = (sum_x_total - sum_x_black) % MOD
    sum_y_part = (sum_y_total - sum_y_black) % MOD

    sum_all_manhattan = ((sum_x_part + sum_y_part) // 2) % MOD

    sum_adjustments = 0
    for x, y in black:
        if x in row_black:
            row = row_black[x]
            pos = bisect.bisect_left(row, y)
            left = y - pos
            right_columns = len(row) - bisect.bisect_right(row, y)
            right = (W - 1 - y) - right_columns
            sum_adjustments += left * right * 2
        
        if y in col_black:
            col = col_black[y]
            pos = bisect.bisect_left(col, x)
            top = x - pos
            bottom_rows = len(col) - bisect.bisect_right(col, x)
            bottom = (H - 1 - x) - bottom_rows
            sum_adjustments += top * bottom * 2
        
        sum_adjustments %= MOD

    total = (sum_all_manhattan + sum_adjustments) % MOD
    print(total)

if __name__ == "__main__":
    main()
```
