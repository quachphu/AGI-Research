To solve this problem, we need to determine the minimal number of fundamental family relations required to describe the relationship between two people. The solution involves modeling the family tree as a graph and using breadth-first search (BFS) to find the shortest path between two nodes, which represents the minimal distance.

### Approach
1. **Model Family Relationships**: Track each person's father, mother, and gender using dictionaries. Explicit sibling relations (brother/sister) are stored as edges in a graph.
2. **Infer Sibling Relations**: Siblings are inferred from shared parents. For each pair of people with the same father and mother, add an edge between them.
3. **Build the Graph**: Construct a graph where edges represent parent-child relationships and sibling relationships (both explicit and inferred).
4. **Shortest Path Calculation**: For each query, use BFS to find the shortest path between two nodes in the graph, which gives the minimal number of fundamental relations.

from collections import defaultdict, deque

# Initialize dictionaries to track family relations and genders
father = dict()
mother = dict()
gender = dict()
explicit_edges = set()  # Stores tuples (A, B) for brother/sister relations

# Read input for N and R
n, r = map(int, input().split())

# Process each relation
for _ in range(r):
    parts = input().split()
    A = parts[0]
    F = parts[2]
    B = parts[-1]

    if F == 'father':
        gender[A] = 'male'
        father[B] = A
    elif F == 'mother':
        gender[A] = 'female'
        mother[B] = A
    elif F == 'son':
        gender[A] = 'male'
        # B is parent of A, determine if father or mother
        if B in gender:
            if gender[B] == 'male':
                father[A] = B
            else:
                mother[A] = B
        else:
            if A not in father:
                father[A] = B
                gender[B] = 'male'
            else:
                mother[A] = B
                gender[B] = 'female'
    elif F == 'daughter':
        gender[A] = 'female'
        if B in gender:
            if gender[B] == 'male':
                father[A] = B
            else:
                mother[A] = B
        else:
            if A not in father:
                father[A] = B
                gender[B] = 'male'
            else:
                mother[A] = B
                gender[B] = 'female'
    elif F in ['brother', 'sister']:
        explicit_edges.add((A, B))
        explicit_edges.add((B, A))
        if F == 'brother':
            gender[A] = 'male'
        else:
            gender[A] = 'female'

# Collect all unique persons from all sources
all_persons = set()
all_persons.update(father.keys())
all_persons.update(mother.keys())
all_persons.update([a for a, _ in explicit_edges])
all_persons.update([b for _, b in explicit_edges])

# Generate inferred siblings based on parents
parent_pair_to_children = defaultdict(list)
for person in all_persons:
    fp = father.get(person, None)
    mp = mother.get(person, None)
    if fp is not None and mp is not None:
        parent_pair_to_children[(fp, mp)].append(person)

# Add inferred sibling edges
for children in parent_pair_to_children.values():
    m = len(children)
    for i in range(m):
        for j in range(i + 1, m):
            a = children[i]
            b = children[j]
            explicit_edges.add((a, b))
            explicit_edges.add((b, a))

# Build the graph
graph = defaultdict(list)
for person in all_persons:
    fp = father.get(person, None)
    if fp is not None:
        graph[person].append(fp)
        graph[fp].append(person)
    mp = mother.get(person, None)
    if mp is not None:
        graph[person].append(mp)
        graph[mp].append(person)
# Add explicit and inferred sibling edges
for a, b in explicit_edges:
    graph[a].append(b)
    graph[b].append(a)

# Process queries
q = int(input())
for _ in range(q):
    X, Y = input().split()
    if X == Y:
        print(0)
        continue
    # BFS to find the shortest path
    visited = set()
    queue = deque([(X, 0)])
    visited.add(X)
    found = False
    while queue:
        current, dist = queue.popleft()
        if current == Y:
            print(dist)
            found = True
            break
        for neighbor in graph[current]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, dist + 1))
    if not found:
        print(-1)
