1. **Input Reading**: Read the number of test cases. For each test case, read the number of stations (N) and railway lines (k). Then, read the railway lines and construct the adjacency list for the graph.
2. **Dijkstra's Algorithm**: Implement Dijkstra's algorithm to find the shortest path and track predecessors to reconstruct the path.
3. **Path Reconstruction**: Use the predecessors array from Dijkstra's algorithm to reconstruct the path from the start to end node.
4. **Combine Paths**: If both paths (A to B and B to C) exist, combine them to form the full path from A to C via B, ensuring no duplicate nodes at the junction point B.

### Solution Code
```python
import sys
import heapq

def main():
    input = sys.stdin.read().split()
    ptr = 0
    T = int(input[ptr])
    ptr += 1
    for _ in range(T):
        N = int(input[ptr])
        k = int(input[ptr + 1])
        ptr += 2
        adj = [[] for _ in range(N + 1)]
        for __ in range(k):
            a = int(input[ptr])
            b = int(input[ptr + 1])
            d = int(input[ptr + 2])
            adj[a].append((b, d))
            adj[b].append((a, d))
            ptr += 3
        A = int(input[ptr])
        B = int(input[ptr + 1])
        C = int(input[ptr + 2])
        ptr += 3

        # Dijkstra from A to B
        def dijkstra(start, end):
            INF = float('inf')
            dist = [INF] * (N + 1)
            predecessors = [None] * (N + 1)
            dist[start] = 0
            heap = [(0, start)]
            heapq.heapify(heap)
            while heap:
                current_dist, u = heapq.heappop(heap)
                if u == end:
                    break
                if current_dist > dist[u]:
                    continue
                for v, w in adj[u]:
                    if dist[v] > dist[u] + w:
                        dist[v] = dist[u] + w
                        predecessors[v] = u
                        heapq.heappush(heap, (dist[v], v))
            if dist[end] == INF:
                return (None, None)
            else:
                return (dist[end], predecessors)

        dist_ab, pred_ab = dijkstra(A, B)
        if dist_ab is None:
            print("No Train Found.")
            continue

        dist_bc, pred_bc = dijkstra(B, C)
        if dist_bc is None:
            print("No Train Found.")
            continue

        def get_path(pred, start, end):
            path = []
            current = end
            while current != start:
                path.append(current)
                current = pred[current]
                if current is None:
                    return None
            path.append(start)
            path.reverse()
            return path

        path_ab = get_path(pred_ab, A, B)
        path_bc = get_path(pred_bc, B, C)
        if path_ab is None or path_bc is None:
            print("No Train Found.")
        else:
            total_distance = dist_ab + dist_bc
            full_path = path_ab + path_bc[1:]
            print(total_distance)
            print(' '.join(map(str, full_path)))

if __name__ == "__main__":
    main()
```
