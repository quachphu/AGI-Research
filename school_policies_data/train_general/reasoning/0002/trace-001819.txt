import sys

def main():
    n = int(sys.stdin.readline())
    a = list(map(int, sys.stdin.readline().split()))
    a_sorted = sorted(a, reverse=True)
    sum_a = sum(a)
    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i+1] = prefix[i] + a_sorted[i]
    
    valid_ks = []
    
    required_parity = sum_a % 2
    for k in range(0, n + 1):
        if k % 2 != required_parity:
            continue
        
        s = a_sorted.copy()
        low, high = 0, n
        while low < high:
            mid = (low + high) // 2
            if s[mid] >= k:
                low = mid + 1
            else:
                high = mid
        pos = low
        s.insert(pos, k)
        
        valid = True
        m = n + 1
        extended_prefix = [0] * (m + 1)
        for i in range(m):
            extended_prefix[i+1] = extended_prefix[i] + s[i]
        
        for r in range(1, m + 1):
            sum_r = extended_prefix[r]
            min_sum = 0
            for i in range(r, m):
                min_sum += min(s[i], r)
            rhs = r * (r - 1) + min_sum
            if sum_r > rhs:
                valid = False
                break
        if valid:
            valid_ks.append(k)
    
    if not valid_ks:
        print(-1)
    else:
        print(' '.join(map(str, sorted(valid_ks))))

if __name__ == "__main__":
