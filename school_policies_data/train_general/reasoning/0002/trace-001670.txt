import sys
import math

def sieve(max_x):
    is_prime = [True] * (max_x + 1)
    is_prime[0] = is_prime[1] = False
    for i in range(2, int(math.sqrt(max_x)) + 1):
        if is_prime[i]:
            for j in range(i * i, max_x + 1, i):
                is_prime[j] = False
    return is_prime

def main():
    sieve_max = 10**6
    is_prime = sieve(sieve_max)
    
    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        n, x = map(int, line.split())
        if n == 0 and x == 0:
            break
        dish_values = []
        for _ in range(n):
            v = int(sys.stdin.readline().strip())
            dish_values.append(v)
        
        current_gcd = dish_values[0]
        for v in dish_values[1:]:
            current_gcd = math.gcd(current_gcd, v)
            if current_gcd == 1:
                break
        
        if current_gcd == 1:
            has_one = any(v == 1 for v in dish_values)
            if has_one:
                if x < 2:
                    print("NA")
                    continue
                s = x
                while s >= 2:
                    if is_prime[s]:
                        print(s)
                        break
                    s -= 1
                else:
                    print("NA")
            else:
                dp = [False] * (x + 1)
                dp[0] = True
                for v in dish_values:
                    for s in range(v, x + 1):
                        if dp[s - v]:
                            dp[s] = True
                max_prime = -1
                for s in range(x, 1, -1):
                    if dp[s] and is_prime[s]:
                        max_prime = s
                        break
                print(max_prime if max_prime != -1 else "NA")
        else:
            if current_gcd > sieve_max:
                is_gcd_prime = True

                    is_gcd_prime = False

                        if current_gcd % i == 0:

                            break
            else:
                is_gcd_prime = is_prime[current_gcd]
