### Approach
1. **Game Analysis**: The game can be analyzed using Grundy numbers and dynamic programming. The key is to determine if the current player can force a win from any given state.
2. **Dynamic Programming (DP)**: We precompute all possible states for sequences of length 1, 2, and 3. Each state is a sorted tuple of the sequence elements. We check all possible moves from each state to see if it leads to a losing state for the opponent.
3. **State Generation**: For each possible sequence length (1, 2, 3), generate all sorted tuples of elements up to 300. Process these states in increasing order of their element sums to ensure all transitions are precomputed.
4. **Move Simulation**: For each state, check all possible moves (subtracting from a single element or all elements) and determine if any move leads to a losing state for the opponent.

import sys

def main():
    input = sys.stdin.read().split()
    n = int(input[0])
    a = list(map(int, input[1:n+1]))
    a_sorted = tuple(sorted(a))
    
    if n == 1:
        print("BitLGM" if a[0] != 0 else "BitAryo")
        return
    
    if n == 2:
        # Precompute dp for n=2
        dp = {}
        states = []
        for a1 in range(301):
            for a2 in range(a1, 301):
                states.append((a1, a2))
        states.sort(key=lambda x: x[0] + x[1])
        for state in states:
            a1, a2 = state
            if a1 == 0 and a2 == 0:
                dp[state] = False
                continue
            winning = False
            # Subtract from first element
            if a1 > 0:
                for x in range(1, a1 + 1):
                    new_a = a1 - x
                    new_state = tuple(sorted((new_a, a2)))
                    if not dp.get(new_state, False):
                        winning = True
                        break
                if winning:
                    dp[state] = True
                    continue
            # Subtract from second element
            if a2 > 0:
                for x in range(1, a2 + 1):
                    new_b = a2 - x
                    new_state = tuple(sorted((a1, new_b)))
                    if not dp.get(new_state, False):
                        winning = True
                        break
                if winning:
                    dp[state] = True
                    continue
            # Global subtract
            min_val = min(a1, a2)
            if min_val > 0:
                for x in range(1, min_val + 1):
                    new_a = a1 - x
                    new_b = a2 - x
                    new_state = (new_a, new_b)
                    if not dp.get(new_state, False):
                        winning = True
                        break
                if winning:
                    dp[state] = True
                    continue
            dp[state] = False
        print("BitLGM" if dp[a_sorted] else "BitAryo")
    elif n == 3:
        # Precompute dp for n=3
        dp = {}
        states = []
        for a1 in range(301):
            for a2 in range(a1, 301):
                for a3 in range(a2, 301):
                    states.append((a1, a2, a3))
        states.sort(key=lambda x: sum(x))
        for state in states:
            a1, a2, a3 = state
            if a1 == 0 and a2 == 0 and a3 == 0:
                dp[state] = False
                continue
            winning = False
            # Subtract from first element
            if a1 > 0:
                for x in range(1, a1 + 1):

                    new_state = tuple(sorted((a1, a2, new_c)))
                    if not dp.get(new_state, False):
                        winning = True
                        break
                if winning:
                    dp[state] = True
                    continue
            # Global subtract
            min_val = a1
            if min_val > 0:
