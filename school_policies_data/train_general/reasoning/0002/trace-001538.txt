def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr +=1
    s = list(map(int, input[ptr:ptr+n]))
    ptr +=n
    m = int(input[ptr])
    ptr +=1
    actions = []
    for _ in range(m):
        act = input[ptr]
        team = input[ptr+1]
        actions.append( (act, int(team)) )
        ptr +=2

    sorted_s = sorted(s, reverse=True)
    groups = []
    if not sorted_s:
        print(0)
        return

    current_count = 1
    for strength in sorted_s[1:]:
        if strength == current_strength:
            current_count +=1
        else:
            groups.append( (current_strength, current_count) )
            current_strength = strength
            current_count = 1
    groups.append( (current_strength, current_count) )
    groups = tuple(groups)
    
    @lru_cache(maxsize=None)
    def dfs(action_idx, available_groups):
        if action_idx == m:
            return 0
        action, team = actions[action_idx]
        if action == 'p':
            if not available_groups:
                return 0
            current_s, current_cnt = available_groups[0]
            new_groups = list(available_groups)
            if current_cnt == 1:
                new_groups = new_groups[1:]
            else:
                new_groups[0] = (current_s, current_cnt -1)
            new_groups = tuple(new_groups)
            add = current_s if team == 1 else -current_s
            return dfs(action_idx +1, new_groups) + add
        else:

            for i in range(len(available_groups)):

                    new_groups.pop(i)

                current_diff = dfs(action_idx +1, new_groups_tuple)

                        best_diff = current_diff
                else:
                    if best_diff is None or current_diff < best_diff:
                        best_diff = current_diff
            return best_diff if best_diff is not None else 0
    
    result = dfs(0, groups)
