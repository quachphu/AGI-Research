```python
MOD = 10**9 + 7

def main():
    import sys
    input = sys.stdin.read().split()
    Q = int(input[0])
    idx = 1
    for _ in range(Q):
        X = int(input[idx])
        K = int(input[idx + 1])
        idx += 2
        
        # Generate Chefonacci numbers up to X in reverse order
        chef = []
        a, b = 1, 2
        if a <= X:
            chef.append(a)
        if b <= X and b not in chef:
            chef.append(b)
        while True:
            c = a + b
            if c > X:
                break
            chef.append(c)
            a, b = b, c
        chef.sort(reverse=True)
        
        from functools import lru_cache
        
        @lru_cache(maxsize=None)
        def dfs(remaining, k_left, index):
            if remaining == 0 and k_left == 0:
                return 1
            if index >= len(chef) or remaining < 0 or k_left < 0:
                return 0
            res = 0
            f = chef[index]
            max_t = min(k_left, remaining // f)
            for t in range(max_t, -1, -1):
                new_rem = remaining - t * f
                new_k = k_left - t
                res += dfs(new_rem, new_k, index + 1)
                res %= MOD
            return res % MOD
        
        ans = dfs(X, K, 0)
        print(ans % MOD)

if __name__ == "__main__":
    main()
```
