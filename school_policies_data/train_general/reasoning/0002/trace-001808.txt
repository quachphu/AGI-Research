MOD = 10007

def main():

    K = int(input[ptr])
    ptr += 1

        x = int(input[ptr])
        ptr += 1
        y = int(input[ptr])
        ptr += 1
        marked.add((x, y))

    # Check if initial pattern has any bingo line
    def check_bingo():
        # Check all rows
        for i in range(N):
            if all((i, j) in marked for j in range(N)):
                return True
        # Check all columns
        for j in range(N):
            if all((i, j) in marked for i in range(N)):
                return True
        # Check main diagonal
        if all((i, i) in marked for i in range(N)):
            return True
        # Check anti-diagonal
        if all((i, N-1-i) in marked for i in range(N)):
            return True
        return False

    if check_bingo():
        print(0)
        return

    T = N * N - N - K
    M = N * N - K
    if T < 0 or T > M:
        print(0)
        return

    # Precompute combinations modulo MOD using dynamic programming
    max_n = max(M, N)
    comb = [[0] * (max_n + 1) for _ in range(max_n + 1)]
    for n in range(max_n + 1):
        comb[n][0] = 1
        for k in range(1, n+1):
            comb[n][k] = (comb[n-1][k] + comb[n-1][k-1]) % MOD

    # Collect all lines (rows, columns, diagonals) and their unmarked cells
    lines = []

    for i in range(N):

        cells = [(i, j) for i in range(N)]
