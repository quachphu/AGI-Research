def main():

    rl_groups = defaultdict(lambda: {'R': [], 'L': []})

    ru_groups = defaultdict(lambda: {'R': [], 'U': []})
    rd_groups = defaultdict(lambda: {'R': [], 'D': []})
    lu_groups = defaultdict(lambda: {'L': [], 'U': []})
    ld_groups = defaultdict(lambda: {'L': [], 'D': []})

    for _ in range(n):
        x, y, u = sys.stdin.readline().split()
        x = int(x)
        y = int(y)
        if u == 'R':
            # Add to R-L groups (y)
            rl_groups[y]['R'].append(x)
            # Add to RU groups (x+y)
            s = x + y
            ru_groups[s]['R'].append(x)
            # Add to RD groups (x - y)
            d = x - y
            rd_groups[d]['R'].append(x)
        elif u == 'L':
            rl_groups[y]['L'].append(x)
            d = x - y
            lu_groups[d]['L'].append(x)
            s = x + y
            ld_groups[s]['L'].append(x)
        elif u == 'U':
            ud_groups[x]['U'].append(y)
            s = x + y
            ru_groups[s]['U'].append(x)
            d = x - y
            lu_groups[d]['U'].append(x)
        elif u == 'D':
            ud_groups[x]['D'].append(y)
            d = x - y
            rd_groups[d]['D'].append(x)
            s = x + y
            ld_groups[s]['D'].append(x)
        else:
            pass  # invalid, but per problem statement, u is valid.

    min_time = float('inf')

    # Process R-L groups
    for y in rl_groups:
        group = rl_groups[y]
        R = sorted(group['R'])
        L = sorted(group['L'])
        if not R or not L:
            continue
        current_min = float('inf')
        for r_x in R:
            idx = bisect.bisect_left(L, r_x)
            if idx < len(L):
                current_min = min(current_min, L[idx] - r_x)
        if current_min != float('inf'):
            min_time = min(min_time, 5 * current_min)

    # Process U-D groups
    for x in ud_groups:
        group = ud_groups[x]
        U = sorted(group['U'])
        D = sorted(group['D'])
        if not U or not D:
            continue
        current_min = float('inf')
        for u_y in U:
            idx = bisect.bisect_left(D, u_y)
            if idx < len(D):
                current_min = min(current_min, D[idx] - u_y)
        if current_min != float('inf'):
            min_time = min(min_time, 5 * current_min)

    # Process RU groups
    for s in ru_groups:
        group = ru_groups[s]
        R = sorted(group['R'])
        U = sorted(group['U'])
        if not R or not U:
            continue
        current_min = float('inf')
        for r_x in R:
            idx = bisect.bisect_left(U, r_x)
            if idx < len(U):
                current_min = min(current_min, U[idx] - r_x)
        if current_min != float('inf'):
            min_time = min(min_time, 10 * current_min)

    for d in rd_groups:

        R = sorted(group['R'])

        if not R or not D:

        current_min = float('inf')
        for r_x in R:

            if idx < len(D):
                current_min = min(current_min, D[idx] - r_x)
        if current_min != float('inf'):
            min_time = min(min_time, 10 * current_min)

    # Process LU groups

To solve this problem, we need to determine if any pair of airplanes will collide and, if so, find the earliest time of collision. The solution involves efficiently checking potential collision scenarios by grouping airplanes based on their movement directions and coordinates.

### Approach
1. **Grouping by Directions and Coordinates**:
   - **Same Axis Movements**: Group airplanes moving towards each other on the same horizontal or vertical line (e.g., Right-Left on the same y-coordinate, Up-Down on the same x-coordinate).
   - **Perpendicular Movements**: Group airplanes moving in perpendicular directions that could intersect (e.g., Right-Up, Right-Down, Left-Up, Left-Down) based on their summed or differenced coordinates.

2. **Efficient Collision Detection**:
   - For each group, sort coordinates and use binary search to find the minimum time to collision efficiently.
   - Handle each group type separately to ensure all potential collisions are checked.

### Solution Code
```python
import sys
import bisect
from collections import defaultdict

def main():
    n = int(sys.stdin.readline())
    rl_groups = defaultdict(lambda: {'R': [], 'L': []})  # Grouped by y
    ud_groups = defaultdict(lambda: {'U': [], 'D': []})   # Grouped by x
    ru_groups = defaultdict(lambda: {'R': [], 'U': []})   # Grouped by x + y
    rd_groups = defaultdict(lambda: {'R': [], 'D': []})   # Grouped by x - y
    lu_groups = defaultdict(lambda: {'L': [], 'U': []})   # Grouped by x - y
    ld_groups = defaultdict(lambda: {'L': [], 'D': []})   # Grouped by x + y

    for _ in range(n):
        x, y, u = sys.stdin.readline().split()
        x = int(x)
        y = int(y)
        if u == 'R':
            rl_groups[y]['R'].append(x)
            s = x + y
            ru_groups[s]['R'].append(x)
            d = x - y
            rd_groups[d]['R'].append(x)
        elif u == 'L':
            rl_groups[y]['L'].append(x)
            d = x - y
            lu_groups[d]['L'].append(x)
            s = x + y
            ld_groups[s]['L'].append(x)
        elif u == 'U':
            ud_groups[x]['U'].append(y)
            s = x + y
            ru_groups[s]['U'].append(x)
            d = x - y
            lu_groups[d]['U'].append(x)
        elif u == 'D':
            ud_groups[x]['D'].append(y)
            d = x - y
            rd_groups[d]['D'].append(x)
            s = x + y
            ld_groups[s]['D'].append(x)

    min_time = float('inf')

    # Process R-L groups (same y)
    for y in rl_groups:
        group = rl_groups[y]
        R = sorted(group['R'])
        L = sorted(group['L'])
        if not R or not L:
            continue
        current_min = float('inf')
        for r_x in R:
            idx = bisect.bisect_left(L, r_x)
            if idx < len(L):
                current_min = min(current_min, L[idx] - r_x)
        if current_min != float('inf'):
            min_time = min(min_time, 5 * current_min)

    # Process U-D groups (same x)
    for x in ud_groups:
        group = ud_groups[x]
        U = sorted(group['U'])
        D = sorted(group['D'])
        if not U or not D:
            continue
        current_min = float('inf')
        for u_y in U:
            idx = bisect.bisect_left(D, u_y)
            if idx < len(D):
                current_min = min(current_min, D[idx] - u_y)
        if current_min != float('inf'):
            min_time = min(min_time, 5 * current_min)

    # Process RU groups (sum x+y)
    for s in ru_groups:
        group = ru_groups[s]
        R = sorted(group['R'])
        U = sorted(group['U'])
        if not R or not U:
            continue
        current_min = float('inf')
        for r_x in R:
            idx = bisect.bisect_left(U, r_x)
            if idx < len(U):
                current_min = min(current_min, U[idx] - r_x)
        if current_min != float('inf'):
            min_time = min(min_time, 10 * current_min)

    # Process RD groups (diff x-y)
    for d in rd_groups:
        group = rd_groups[d]
        R = sorted(group['R'])
        D = sorted(group['D'])
        if not R or not D:
            continue
        current_min = float('inf')
        for r_x in R:
            idx = bisect.bisect_left(D, r_x)
            if idx < len(D):
                current_min = min(current_min, D[idx] - r_x)
        if current_min != float('inf'):
            min_time = min(min_time, 10 * current_min)

    # Process LU groups (diff x-y)
    for d in lu_groups:
        group = lu_groups[d]
        L = sorted(group['L'])
        U = sorted(group['U'])
        if not L or not U:
            continue
        current_min = float('inf')
        for l_x in L:
            idx = bisect.bisect_right(U, l_x) - 1
            if idx >= 0:
                current_min = min(current_min, l_x - U[idx])
        if current_min != float('inf'):
            min_time = min(min_time, 10 * current_min)

    # Process LD groups (sum x+y)
    for s in ld_groups:
        group = ld_groups[s]
        L = sorted(group['L'])
        D = sorted(group['D'])
        if not L or not D:
            continue
        current_min = float('inf')
        for l_x in L:
            idx = bisect.bisect_right(D, l_x) - 1
            if idx >= 0:
                current_min = min(current_min, l_x - D[idx])
        if current_min != float('inf'):
            min_time = min(min_time, 10 * current_min)

    if min_time == float('inf'):
        print("SAFE")
    else:
        print(int(min_time))

if __name__ == '__main__':
    main()
```

2. **Grouping**: Airplanes are grouped into different categories based on their movement directions and coordinates.
3. **Collision Detection**: For each group, coordinates are sorted and binary search is used to efficiently find the minimum collision time.
4. **Result**: The minimum collision time across all groups is determined and printed, or "SAFE" if no collisions are detected.

This approach ensures that we efficiently check all potential collision scenarios without iterating through all possible pairs, making the solution feasible for large input sizes.
