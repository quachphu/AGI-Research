        m_var = var_pattern.match(line)
        if not m_var:
            continue
        var_type = m_var.group(1)
        var_name = m_var.group(2)
        variables[var_name] = var_type
    
    # Read calls
    k = int(sys.stdin.readline())
    call_pattern = re.compile(r'^\s*([a-z0-9]+)\s*\(\s*([^)]*)\s*\)\s*$', re.IGNORECASE)
    for _ in range(k):
        line = sys.stdin.readline().rstrip('\n')
        m_call = call_pattern.match(line)
        if not m_call:
            print(0)
            continue
        proc_name = m_call.group(1)
        vars_str = m_call.group(2)
        var_names = [v.strip() for v in vars_str.split(',')] if vars_str.strip() else []
        try:
            args_types = [variables[v] for v in var_names]
        except KeyError:
            print(0)
            continue
        count = 0
        if proc_name in procedures:
            for param_list in procedures[proc_name]:
                if len(param_list) != len(args_types):
                    continue
                valid = True
                for p, a in zip(param_list, args_types):
                    if p != 'T' and p != a:
                        valid = False
                        break
                if valid:
                    count += 1
        print(count)
