from itertools import combinations

MOD = 998244353

def cross(o, a, b):
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

def convex_hull(points):
    points = sorted(points)
    if len(points) == 1:
        return points.copy()
    lower = []
    for p in points:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(points):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]

def is_convex_polygon(subset):
    ch = convex_hull(subset)
    return set(ch) == set(subset) and len(ch) >= 3

def point_in_convex_polygon(poly, point):

        a = poly[i]
        b = poly[(i+1) % len(poly)]
        val = cross(a, b, point)
        if val < 0:
            return False

        y = int(input[idx+1])
