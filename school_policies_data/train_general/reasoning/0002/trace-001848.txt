3. **DFS for Time Stamps**: Perform a DFS to compute in-time and out-time for each node. These times help determine subtree relationships and ancestor-descendant relationships efficiently.

import sys
from collections import deque

def main():
    input = sys.stdin.read().split()
    ptr = 0
    N = int(input[ptr])
    ptr += 1
    
    adj = [[] for _ in range(N + 1)]
    for _ in range(N - 1):
        A = int(input[ptr])
        B = int(input[ptr + 1])
        adj[A].append(B)
        adj[B].append(A)
        ptr += 2
    
    parent = [0] * (N + 1)
    children = [[] for _ in range(N + 1)]
    q = deque([1])
    parent[1] = 0
    
    while q:
        u = q.popleft()
        for v in adj[u]:
            if v != parent[u]:
                parent[v] = u
                children[u].append(v)
                q.append(v)
    
    in_time = [0] * (N + 1)
    out_time = [0] * (N + 1)
    time = 0
    stack = [(1, False)]
    
    while stack:
        node, visited = stack.pop()
        if not visited:
            time += 1
            in_time[node] = time
            stack.append((node, True))
            for child in reversed(children[node]):
                stack.append((child, False))
        else:
            time += 1
            out_time[node] = time
    
    Q = int(input[ptr])
    ptr += 1
    
    res = []
    for _ in range(Q):
        typ = int(input[ptr])
        X = int(input[ptr + 1])
        Y = int(input[ptr + 2])
        ptr += 3
        
        if typ == 0:
            if in_time[X] <= in_time[Y] and out_time[X] >= out_time[Y]:
                res.append("YES")
            else:
                res.append("NO")
        else:
            if in_time[Y] <= in_time[X] <= out_time[Y]:
                res.append("YES")
            else:
                res.append("NO")
    
    print('\n'.join(res))

if __name__ == "__main__":
    main()
