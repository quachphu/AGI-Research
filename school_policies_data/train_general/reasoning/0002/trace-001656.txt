### Approach
1. **Preprocess the Array into Runs**: Convert the array into a list of consecutive runs, where each run is represented by a tuple of the value and its length.
2. **Linked List and Priority Queue**: Use a linked list to maintain the runs dynamically as segments are removed and merged. Use a max-heap (priority queue) to efficiently retrieve the longest leftmost segment.
3. **Merge Segments**: When a segment is removed, check if its adjacent segments can be merged (if they have the same value) and update the linked list and heap accordingly.

```python
import heapq

def main():
    import sys
    input = sys.stdin.read().split()
    n = int(input[0])
    arr = list(map(int, input[1:n+1]))
    
    if n == 0:
        print(0)
        return
    
    # Preprocess into runs
    runs = []
    current = arr[0]
    count = 1
    for num in arr[1:]:
        if num == current:
            count += 1
        else:
            runs.append((current, count))
            current = num
            count = 1
    runs.append((current, count))
    
    # Create nodes for each run
    class Node:
        __slots__ = ('value', 'length', 'start', 'prev', 'next', 'active')
        def __init__(self, value, length, start):
            self.value = value
            self.length = length
            self.start = start
            self.prev = None
            self.next = None
            self.active = True
    
    nodes = []
    start_index = 0
    prev_node = None
    for val, cnt in runs:
        node = Node(val, cnt, start_index)
        nodes.append(node)
        if prev_node is not None:
            prev_node.next = node
            node.prev = prev_node
        prev_node = node
        start_index += 1
    
    heap = []
    for node in nodes:
        heapq.heappush(heap, (-node.length, node.start, node))
    
    count = 0
    
    while heap:
        neg_len, start, node = heapq.heappop(heap)
        if not node.active:
            continue
        node.active = False
        count += 1
        
        prev_node = node.prev
        next_node = node.next
        
        # Update prev and next pointers
        if prev_node is not None:
            prev_node.next = next_node
        if next_node is not None:
            next_node.prev = prev_node
        
        # Check if prev and next can be merged
        if prev_node is not None and next_node is not None and prev_node.value == next_node.value:
            # Create new merged node
            new_val = prev_node.value
            new_length = prev_node.length + next_node.length
            new_start = prev_node.start
            new_node = Node(new_val, new_length, new_start)
            
            # Link new node to the linked list
            new_node.prev = prev_node.prev
            new_node.next = next_node.next
            if new_node.prev is not None:
                new_node.prev.next = new_node
            if new_node.next is not None:
                new_node.next.prev = new_node
            
            # Mark the old nodes as inactive
            prev_node.active = False
            next_node.active = False
            
            # Push the new node into the heap
            new_node.active = True
            heapq.heappush(heap, (-new_node.length, new_node.start, new_node))
    
    print(count)

if __name__ == "__main__":
    main()
```

### Explanation
1. **Preprocessing**: Convert the input array into segments (runs) of consecutive elements. Each run is represented by its value and length.
2. **Linked List**: Maintain a linked list of these runs to efficiently handle removals and merges.
3. **Max-Heap**: Use a max-heap to quickly retrieve the longest segment. The heap is ordered by segment length and start index to ensure the leftmost segment is chosen when there are ties in length.
4. **Merging Segments**: When a segment is removed, check its adjacent segments. If they can be merged (same value), create a new segment, update the linked list, and push the new segment into the heap.

This approach efficiently handles the dynamic nature of the problem, ensuring each operation is processed in logarithmic time relative to the number of segments, leading to an overall time complexity of O(n log n), which is suitable for large input sizes.
