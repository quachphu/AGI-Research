def generate_min(n_str):
    digits = list(n_str)
    if len(digits) == 1:
        return n_str
    
    non_zero_digits = [d for d in digits if d != '0']
    min_non_zero = min(non_zero_digits)
    first_idx = digits.index(min_non_zero)
    
    if first_idx != 0:
        digits[0], digits[first_idx] = digits[first_idx], digits[0]
    
    remaining = digits[1:]
    remaining.sort()
    
    return ''.join([digits[0]] + remaining)
