class Node:
    __slots__ = ('children', 'fail', 'max_len')
    def __init__(self):
        self.children = dict()
        self.fail = None
        self.max_len = 0

def main():
    import sys
    from collections import deque

    S = sys.stdin.readline().strip()
    N = int(sys.stdin.readline())
    P = [sys.stdin.readline().strip() for _ in range(N)]

    # Build the trie structure
    root = Node()
    for pattern in P:
        current = root
        for char in pattern:
            if char not in current.children:
                current.children[char] = Node()
            current = current.children[char]
        if len(pattern) > current.max_len:
            current.max_len = len(pattern)

    # Build failure links and propagate max_len
    queue = deque()
    for child in root.children.values():
        child.fail = root
        queue.append(child)

    while queue:
        current_node = queue.popleft()
        for char, child in current_node.children.items():
            # Find fail node for the child
            fail_node = current_node.fail
            while fail_node is not None and char not in fail_node.children:
                fail_node = fail_node.fail
            child.fail = fail_node.children[char] if (fail_node and char in fail_node.children) else root
            # Update the max_len for the child
            child.max_len = max(child.max_len, child.fail.max_len)
            queue.append(child)

    # Process the string S to collect intervals
    current = root
    intervals = []
    for i in range(len(S)):
        char = S[i]
        # Move to the appropriate child or follow failure links
        while current is not None and char not in current.children:
            current = current.fail
        if current is None:
            current = root
        else:
            current = current.children[char]
        # Record the interval if there's a valid max_len
        if current.max_len > 0:
            start = i - current.max_len + 1
            end = i
            if start >= 0:
                intervals.append((start, end))

    # Greedily select points to cover all intervals
    intervals.sort(key=lambda x: x[1])
    count = 0
    last_end = -1
    for start, end in intervals:
        if start > last_end:
            count += 1
            last_end = end

    print(count)

if __name__ == "__main__":
    main()
