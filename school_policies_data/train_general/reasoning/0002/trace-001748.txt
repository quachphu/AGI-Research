    # Compute A = F - D + E
    Ax = F[0] - D[0] + E[0]

    # Compute B = D - E + F
    Bx = D[0] - E[0] + F[0]

    # Sort the vertices
    vertices.sort(key=lambda p: (p[0], p[1]))

    # Format each to 4 decimal places
    for v in vertices:

if __name__ == "__main__":
