import heapq

def get_top_sums(a, b, k):
    if not a or not b or k == 0:
        return []
    heap = []
    visited = set()
    heapq.heappush(heap, (-(a[0] + b[0]), 0, 0))
    visited.add((0, 0))
    result = []
    while len(result) < k and heap:
        current_sum_neg, i, j = heapq.heappop(heap)
        current_sum = -current_sum_neg
        result.append(current_sum)
        if i + 1 < len(a) and (i + 1, j) not in visited:
