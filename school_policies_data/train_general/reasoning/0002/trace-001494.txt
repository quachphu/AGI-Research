To solve this problem, we need to find the maximum integer \( n \) such that the number of prime numbers up to \( n \) (denoted as \( \pi(n) \)) is less than or equal to \( A \) times the number of palindromic numbers up to \( n \) (denoted as \( \text{rub}(n) \)), where \( A \) is given as a fraction \( \frac{p}{q} \).

### Approach
1. **Generate Palindromic Numbers**: We generate all possible palindromic numbers up to a sufficiently large upper limit (2,000,000 in this case) and store them in a sorted list.
2. **Sieve of Eratosthenes**: We use this algorithm to precompute the number of primes up to each number within our upper limit. This allows us to quickly determine \( \pi(n) \) for any \( n \) within the range.
3. **Check Condition**: For each number \( n \) up to our upper limit, we check if \( \pi(n) \leq A \times \text{rub}(n) \). We track the maximum \( n \) that satisfies this condition.

import bisect

def generate_palindromes(max_n):
    palindromes = []
    length = len(str(max_n))
    for l in range(1, length + 1):
        if l == 1:
            for num in range(1, 10):
                if num <= max_n:
                    palindromes.append(num)
        else:
            if l % 2 == 0:
                half = l // 2
                start = 10 ** (half - 1)
                end = 10 ** half - 1
                for first_half in range(start, end + 1):
                    s = str(first_half)
                    palindrome = s + s[::-1]
                    num = int(palindrome)
                    if num <= max_n:
                        palindromes.append(num)
            else:
                k = (l + 1) // 2
                start = 10 ** (k - 1)
                end = 10 ** k - 1
                for first_half in range(start, end + 1):
                    s = str(first_half)
                    palindrome = s + s[:-1][::-1]
                    num = int(palindrome)
                    if num <= max_n:
                        palindromes.append(num)
    palindromes.sort()
    return palindromes

def sieve(max_n):
    is_prime = [True] * (max_n + 1)
    is_prime[0], is_prime[1] = False, False
    for i in range(2, int(max_n ** 0.5) + 1):
        if is_prime[i]:
            for j in range(i*i, max_n + 1, i):
                is_prime[j] = False
    prime_count = [0] * (max_n + 1)
    count = 0
    for i in range(max_n + 1):
        if is_prime[i]:
            count += 1
        prime_count[i] = count
    return prime_count

p, q = map(int, input().split())
A = p / q
max_sieve = 2000000

prime_count = sieve(max_sieve)
palindromes = generate_palindromes(max_sieve)

max_n_found = -1
for n in range(1, max_sieve + 1):
    pi = prime_count[n]
    rub = bisect.bisect_right(palindromes, n)
    if pi <= A * rub:
        max_n_found = n
