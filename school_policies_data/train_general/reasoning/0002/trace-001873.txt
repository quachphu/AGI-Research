MOD = 998244353

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    n = int(input[ptr])
    ptr +=1
    m = int(input[ptr])
    ptr +=1
    
    merc_l = [0] * (n + 1)
    merc_r = [0] * (n + 1)
    for i in range(1, n+1):
        l = int(input[ptr])
        ptr +=1
        r = int(input[ptr])
        ptr +=1
        merc_l[i] = l
        merc_r[i] = r
    
    edges_input = []
    for _ in range(m):
        a = int(input[ptr])
        ptr +=1
        b = int(input[ptr])
        ptr +=1
        edges_input.append((a, b))
    
    # Step 1: Compute cnt_k using difference array
    delta = [0]*(n+2)
    for i in range(1, n+1):
        l = merc_l[i]
        r = merc_r[i]
        delta[l] += 1
        if r +1 <= n:
            delta[r+1] -=1
    
    cnt = [0]*(n+1)
    for k in range(1, n+1):
        cnt[k] = cnt[k-1] + delta[k]
    
    # Precompute factorial and inverse factorial up to max needed value (3e5)
    max_n = 3*10**5
    fact = [1]*(max_n+1)
    for i in range(1, max_n+1):
        fact[i] = fact[i-1] * i % MOD
    inv_fact = [1]*(max_n+1)
    inv_fact[max_n] = pow(fact[max_n], MOD-2, MOD)
    for i in range(max_n-1, -1, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % MOD
    
    # Precompute sum_s for s from 0 to 40
    sum_s = [ [0]*(n+1) for _ in range(41) ]  # sum_s[s][k] is sum up to k
    
    for s in range(0, 41):
        current_sum = 0
        sum_s[s][0] = 0
        for k in range(1, n+1):
            if k < s or cnt[k] < k:
                sum_s[s][k] = current_sum
                continue
            a = cnt[k] - s
            b = k - s
            if a < b or b <0:
                sum_s[s][k] = current_sum
                continue
            numerator = fact[a]
            denominator = inv_fact[b] * inv_fact[a - b] % MOD
            c_val = numerator * denominator % MOD
            current_sum = (current_sum + c_val) % MOD
            sum_s[s][k] = current_sum
    
    ans = 0
    
    # Iterate over all subsets of forbidden edges
    for mask in range(0, 1 << m):
        edges_in_X = []
        for j in range(m):
            if (mask >> j) & 1:
                edges_in_X.append(edges_input[j])
        
        s = 0
        L_X = 1
        R_X = n

            v = set()
            for a, b in edges_in_X:
                v.add(a)
                v.add(b)
            s = len(v)
            # Compute L_X and R_X
            L_X = -1
            R_X = float('inf')
            for a, b in edges_in_X:
                la = merc_l[a]
                lb = merc_l[b]
                current_L = max(la, lb)
                if current_L > L_X:
                    L_X = current_L
                ra = merc_r[a]
                rb = merc_r[b]
                current_R = min(ra, rb)
                if current_R < R_X:
                    R_X = current_R
            # Check if L_X > R_X
            if L_X > R_X:
                continue
        
        a = max(L_X, s)
        b = R_X
        if a > b:
            continue
        if s >40:
            continue
        
        # Get the sum from a to b for s
        if a >n or b >n or a <1 or b <1:
            continue
        
        if a ==0:

        prev = sum_s[s][a-1] if a >=1 else 0
        current = sum_s[s][b]
        contribution = (current - prev) % MOD
        
        # Apply sign
        num_edges = len(edges_in_X)
        sign = (-1)**num_edges
        if sign ==1:
            ans = (ans + contribution) % MOD
        else:
            ans = (ans - contribution) % MOD
    
    # Ensure ans is non-negative
    ans = ans % MOD
    print(ans)
