def longest_increasing_subsequence(arr):
    if not arr:
        return []

    n = len(arr)
    dp = [1] * n  # Initialize the dynamic programming array with all elements set to 1
    prev = [-1] * n  # Initialize the previous index array with all elements set to -1

    for i in range(1, n):
        for j in range(i):
            if arr[i] > arr[j] and dp[i] < dp[j] + 1:
                dp[i] = dp[j] + 1  # Update the length of increasing subsequence ending at index i
                prev[i] = j  # Store the index of the previous element in the subsequence

    max_length = max(dp)  # Find the maximum length of increasing subsequence
    index = dp.index(max_length)  # Get the index of the last element in the longest increasing subsequence

    result = []  # Initialize an empty list to store the longest increasing subsequence
    while index != -1:  # Iterate through the previous index array to build the subsequence
        result.append(arr[index])  # Add the current element to the result list
        index = prev[index]  # Move to the previous element in the subsequence

    return result[::-1]  # Reverse the result list and return it

	"os"

func main() {
	var input string
	var output string
	var level int
	var decompress bool

	flag.StringVar(&input, "i", "", "Input file or directory")
	flag.StringVar(&output, "o", "", "Output file or directory")
	flag.IntVar(&level, "l", gzip.DefaultCompression, "Compression level (0-9)")
	flag.BoolVar(&decompress, "d", false, "Decompress mode")

	flag.Parse()

	if input == "" || output == "" {
		fmt.Println("Please specify both input and output paths.")
		os.Exit(1)
	}

	err := filepath.Walk(input, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}

		if !info.IsDir() {
			outPath := strings.Replace(path, input, output, 1)

			if decompress {
				err = decompressFile(path, outPath)
			} else {
				err = compressFile(path, outPath, level)
			}

			if err != nil {
				return err
			}
		}

		return nil
	})

	if err != nil {
		fmt.Printf("Error: %v\n", err)
		os.Exit(1)
	}
}

func compressFile(src, dest string, level int) error {
	srcFile, err := os.Open(src)
	if err != nil {
		return err
	}
	defer srcFile.Close()

	err = os.MkdirAll(filepath.Dir(dest), 0755)
	if err != nil {
		return err
	}

	destFile, err := os.Create(dest + ".gz")
	if err != nil {
		return err
	}
	defer destFile.Close()

	gw, err := gzip.NewWriterLevel(destFile, level)
	if err != nil {
		return err
	}
	defer gw.Close()

	_, err = io.Copy(gw, srcFile)
	return err
}

	if err != nil {
		return err
	}
	defer srcFile.Close()

	gr, err := gzip.NewReader(srcFile)

		return err
	}

	if err != nil {
		return err
	}
	defer destFile.Close()

	_, err = io.Copy(destFile, gr)
	return err

```python
import math

class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __str__(self):
        return f"({self.x}, {self.y})"

    def distance(self, other_point):
        dx = self.x - other_point.x
        dy = self.y - other_point.y
        return math.sqrt(dx**2 + dy**2)

    def midpoint(self, other_point):
        mx = (self.x + other_point.x) / 2
        my = (self.y + other_point.y) / 2
        return Point(mx, my)


# Example usage:
p1 = Point(0, 0)
p2 = Point(3, 4)
print("Point 1:", p1)
print("Point 2:", p2)
print("Distance:", p1.distance(p2))
print("Midpoint:", p1.midpoint(p2))
```
