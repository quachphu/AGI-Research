function decimalToBinary(decimalNumber) {
    // If the input is not a number or less than 0, return an error message
    if (typeof decimalNumber !== 'number' || decimalNumber < 0) {
        return "Invalid input. Please enter a non-negative number.";
    }

    // Convert the decimal number to binary and return it
