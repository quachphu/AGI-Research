Create a JavaScript/Node.js application that connects to an MQTT broker and subscribes to a specified topic. When a message is received on the subscribed topic, the application should save the message content to a local file named after the current timestamp. Use command line arguments for specifying the MQTT broker's address and the topic to subscribe to.
PLAINFORMAT
</user>
<bot>
const fs = require('fs');
const mqtt = require('mqtt');

// Read command line arguments
const args = process.argv.slice(2);
if (args.length < 2) {
  console.error('Usage: node mqttToFile.js <broker_address> <topic>');
  process.exit(1);
}

const [brokerAddress, topic] = args;

// Connect to the MQTT broker
const client = mqtt.connect(brokerAddress);

client.on('connect', () => {
  console.log(`Connected to ${brokerAddress}`);
  // Subscribe to the specified topic
  client.subscribe(topic, (err) => {
    if (err) {
      console.error(`Failed to subscribe to ${topic}:`, err.message);
      process.exit(1);
    }
    console.log(`Subscribed to ${topic}`);
  });
});

client.on('message', (receivedTopic, message) => {
  if (receivedTopic === topic) {
    const timestamp = Date.now();
    const filename = `${timestamp}.txt`;

    // Save the message content to a local file named after the current timestamp
    fs.writeFile(filename, message.toString(), (err) => {
      if (err) {
        console.error(`Failed to write message to ${filename}:`, err.message);
      } else {
        console.log(`Message saved to ${filename}`);
      }
    });
  }
});

client.on('error', (err) => {
  console.error('Error:', err.message);
  process.exit(1);
});
