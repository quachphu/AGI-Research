Design a C++ class hierarchy for geometric shapes, including circles, rectangles, and triangles. Each shape should have methods to calculate its area and perimeter. Also, create a function that accepts a list of shapes and computes the total area and perimeter of all shapes in the list.
PLAINFORMAT
</user>
<bot>
#include <iostream>
#include <vector>
#include <cmath>

class Shape {
public:
    virtual double area() const = 0;
    virtual double perimeter() const = 0;
};

class Circle : public Shape {
public:
    Circle(double radius) : radius(radius) {}

    double area() const override {
        return M_PI * radius * radius;
    }

    double perimeter() const override {
        return 2 * M_PI * radius;
    }

private:
    double radius;
};

class Rectangle : public Shape {
public:
    Rectangle(double width, double height) : width(width), height(height) {}

    double area() const override {
        return width * height;
    }

    double perimeter() const override {
        return 2 * (width + height);
    }

private:
    double width;
    double height;
};

class Triangle : public Shape {
public:
    Triangle(double a, double b, double c) : a(a), b(b), c(c) {}

    double area() const override {
        double s = (a + b + c) / 2;
        return sqrt(s * (s - a) * (s - b) * (s - c));
    }

    double perimeter() const override {
        return a + b + c;
    }

private:
    double a, b, c;
};

void total_area_and_perimeter(const std::vector<Shape*>& shapes, double& total_area, double& total_perimeter) {
    total_area = 0;
    total_perimeter = 0;

    for (const auto& shape : shapes) {
        total_area += shape->area();
        total_perimeter += shape->perimeter();
    }
}

int main() {
    Circle circle(5);
    Rectangle rectangle(4, 6);
    Triangle triangle(3, 4, 5);

    std::vector<Shape*> shapes = {&circle, &rectangle, &triangle};

    double total_area, total_perimeter;
    total_area_and_perimeter(shapes, total_area, total_perimeter);

    std::cout << "Total area: " << total_area << std::endl;
    std::cout << "Total perimeter: " << total_perimeter << std::endl;

    return 0;
}
