    """get objects defined in enclosing code that are referred to by func

    While the new base vectors are in the new coordinate system basis, any

        characters except the dot used for Python package names are

def reverse_bfs(src):
    """Perform a reverse breadth-first search from src in the residual
    network.
    """
    heights = {src: 0}
    q = deque([(src, 0)])
    while q:
        u, height = q.popleft()
        height += 1
        for v, attr in R_pred[u].items():
            if v not in heights and attr["flow"] < attr["capacity"]:
                heights[v] = height
                q.append((v, height))

    """Turn all the substitution dictionaries in sorted tuples of tuples so
    they are hashable, to detect duplicates so we don't write out redundant

    allSubstitutions = set()
    condSubst = []
